新文件（或覆盖）：quentin/self_programming/engine.py

这个文件里定义三样东西：

PatchCandidate：LLM 生成的候选补丁（我们前面已经在其它地方用到了）

PatchTestReport：一条补丁在测试环境里的实验结果

SelfProgrammingEngine：

把补丁应用到临时 repo 副本

在这个副本里跑 pytest/ruff/mypy

生成结构化测试报告

提供一个手动 apply_patch_to_repo，真正写回主仓库（不会自动做）

⚠️ 默认不会动你的主 repo，只在 .quentin/self_programming/tmp/<patch_id>/ 下搞事情。

# quentin/self_programming/engine.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, List, Optional
import logging
import shutil
import subprocess
import time
import uuid

logger = logging.getLogger(__name__)


# ------------------------------------------------------------
# Dataclasses
# ------------------------------------------------------------

@dataclass
class PatchCandidate:
    """
    由 SkillEvolutionLoop / BlueprintImprover / ResearchLayer 生成的补丁候选。

    target:
      - 至少需要包含 file_path（相对 repo 根目录）
      - 也可以包含 identifier / symbol_name / 其它 meta
    """
    patch_id: str
    target: Dict[str, Any]
    description: str
    patch_text: str
    tests: List[str]
    meta: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class PatchTestReport:
    """
    对单个 patch 在临时 repo 上跑完测试后的结果。
    """
    patch_id: str
    target: Dict[str, Any]
    status: str                     # "passed" | "failed" | "error"
    metrics: Dict[str, Any]
    logs: Dict[str, Any]
    temp_repo_path: str
    error: Optional[str]
    meta: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


# 默认测试套件配置：可以按需改
DEFAULT_TEST_SUITES: Dict[str, Dict[str, Any]] = {
    "unit": {
        "name": "pytest",
        "cmd": ["pytest", "-q"],
        "required": False,    # pytest 不存在就跳过
    },
    "static_analysis": {
        "name": "ruff",
        "cmd": ["ruff", "check", "."],
        "required": False,
    },
    "type_check": {
        "name": "mypy",
        "cmd": ["mypy", "."],
        "required": False,
    },
}


class SelfProgrammingEngine:
    """
    自编程引擎核心实现。

    能力：
      - 将 PatchCandidate 应用到一个临时 repo 副本
      - 在该副本上运行测试命令（pytest / ruff / mypy 等）
      - 生成 PatchTestReport
      - 提供 select_patches_to_promote 选出“值得推广”的 patch
      - 提供 apply_patch_to_repo 手动把 patch 写回主 repo（不会自动做）
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        work_root: Optional[str | Path] = None,
        test_suites_config: Optional[Dict[str, Dict[str, Any]]] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        # 工作目录（存临时 repo、日志等）
        self.work_root = (
            Path(work_root).resolve()
            if work_root is not None
            else (self.repo_root / ".quentin" / "self_programming")
        )
        self.work_root.mkdir(parents=True, exist_ok=True)

        # 测试套件配置
        self.test_suites_config = test_suites_config or DEFAULT_TEST_SUITES

    # ------------------------------------------------------------
    # 对外主接口：跑测试
    # ------------------------------------------------------------
    def run_tests_for_patch(
        self,
        *,
        candidate: PatchCandidate,
        test_suites: List[str],
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> PatchTestReport:
        """
        针对一个 PatchCandidate：
          1. 复制 repo 到临时目录
          2. 在临时目录应用 patch（写入目标文件）
          3. 在临时目录跑指定的测试套件
          4. 返回 PatchTestReport

        注意：
          - 不会修改主仓库
          - 测试失败不会抛异常，而是记录在 report.status 和 logs 里
        """
        ctx = extra_context or {}
        patch_id = candidate.patch_id

        # 1) 准备临时 repo 副本
        temp_repo = self._prepare_temp_repo(patch_id)
        logger.info("SelfProgrammingEngine: temp repo for patch %s at %s", patch_id, temp_repo)

        # 2) 在临时 repo 中应用 patch
        try:
            self._apply_patch_to_directory(candidate, temp_repo)
        except Exception as e:
            logger.exception("SelfProgrammingEngine: failed to apply patch %s", patch_id)
            return PatchTestReport(
                patch_id=patch_id,
                target=candidate.target,
                status="error",
                metrics={},
                logs={},
                temp_repo_path=str(temp_repo),
                error=f"failed_to_apply_patch: {e}",
                meta={"stage": "apply_patch", "extra_context": ctx},
            )

        # 3) 在临时 repo 中运行测试套件
        suite_results = self._run_test_suites_in_dir(temp_repo, test_suites)

        # 4) 汇总总体状态
        overall_status, metrics = self._summarize_suite_results(suite_results)

        report = PatchTestReport(
            patch_id=patch_id,
            target=candidate.target,
            status=overall_status,
            metrics=metrics,
            logs={"suites": suite_results},
            temp_repo_path=str(temp_repo),
            error=None if overall_status == "passed" else None,
            meta={
                "stage": "test",
                "test_suites": test_suites,
                "extra_context": ctx,
            },
        )
        return report

    # ------------------------------------------------------------
    # 选择可推广 patch（简单版）
    # ------------------------------------------------------------
    def select_patches_to_promote(
        self,
        reports: List[PatchTestReport],
        *,
        max_promote: int = 1,
    ) -> List[PatchTestReport]:
        """
        从一组 PatchTestReport 中选出“值得推广”的 patch。

        v1 策略：
          - 只选 status == "passed" 的
          - 按 severity（如果有的话）和测试覆盖情况简单排序
          - 最多返回 max_promote 个

        注意：这里仅仅是“推荐”，不会自动改主仓库。
        """
        passed = [r for r in reports if r.status == "passed"]
        if not passed:
            return []

        # 简单排序：按 metrics 里的 severity（高优先）、再按通过的测试数
        def score(rep: PatchTestReport) -> int:
            severity_map = {"low": 1, "medium": 2, "high": 3}
            sev = rep.metrics.get("severity_score", 0)
            sev_score = severity_map.get(sev, 0) if isinstance(sev, str) else sev
            passed_count = rep.metrics.get("num_suites_passed", 0)
            return sev_score * 10 + passed_count

        passed_sorted = sorted(passed, key=score, reverse=True)
        return passed_sorted[:max_promote]

    # ------------------------------------------------------------
    # 手动应用 patch 到主 repo（需要人工/AI gate）
    # ------------------------------------------------------------
    def apply_patch_to_repo(self, candidate: PatchCandidate) -> Path:
        """
        ⚠️ 高风险操作：把 patch 直接写入主仓库。

        建议：
          - 只在你或“AI 审批层”确认之后调用
          - 可以在调用前先把主 repo commit / 备份

        当前实现：直接覆盖 target.file_path 对应的文件内容。
        """
        file_path = candidate.target.get("file_path")
        if not file_path:
            raise ValueError("PatchCandidate.target.file_path is required to apply patch to repo")

        abs_path = self.repo_root / file_path
        abs_path.parent.mkdir(parents=True, exist_ok=True)
        abs_path.write_text(candidate.patch_text, encoding="utf-8")

        logger.info("SelfProgrammingEngine: patch %s applied to repo file %s", candidate.patch_id, abs_path)
        return abs_path

    # ------------------------------------------------------------
    # 内部：准备临时 repo 副本
    # ------------------------------------------------------------
    def _prepare_temp_repo(self, patch_id: str) -> Path:
        """
        为某个 patch 准备一个干净的 repo 副本：

        tmp/<patch_id> 目录结构：
          - 完整复制 repo_root（排除 .git / __pycache__ 等）
        """
        temp_root = self.work_root / "tmp" / patch_id
        if temp_root.exists():
            shutil.rmtree(temp_root)

        # 复制整个仓库（可以按需优化 / 忽略大目录）
        ignore = shutil.ignore_patterns(
            ".git",
            ".idea",
            ".venv",
            "__pycache__",
            ".mypy_cache",
            ".pytest_cache",
            ".ruff_cache",
        )
        shutil.copytree(self.repo_root, temp_root, ignore=ignore)
        return temp_root

    # ------------------------------------------------------------
    # 内部：在某个目录中应用 patch
    # ------------------------------------------------------------
    def _apply_patch_to_directory(self, candidate: PatchCandidate, root: Path) -> None:
        file_path = candidate.target.get("file_path")
        if not file_path:
            raise ValueError("PatchCandidate.target.file_path is required to apply patch to directory")

        abs_path = root / file_path
        abs_path.parent.mkdir(parents=True, exist_ok=True)
        abs_path.write_text(candidate.patch_text, encoding="utf-8")

        logger.debug(
            "SelfProgrammingEngine: wrote patch %s to %s (temp repo)",
            candidate.patch_id,
            abs_path,
        )

    # ------------------------------------------------------------
    # 内部：在临时目录中运行测试套件
    # ------------------------------------------------------------
    def _run_test_suites_in_dir(
        self,
        temp_repo: Path,
        test_suites: List[str],
    ) -> List[Dict[str, Any]]:
        results: List[Dict[str, Any]] = []

        for suite_name in test_suites:
            cfg = self.test_suites_config.get(suite_name)
            if cfg is None:
                # 未配置的 suite 直接标记为 skipped
                results.append(
                    {
                        "suite": suite_name,
                        "configured": False,
                        "status": "skipped",
                        "reason": "no_config",
                    }
                )
                continue

            cmd = cfg["cmd"]
            required = bool(cfg.get("required", False))
            suite_label = cfg.get("name", suite_name)

            start = time.perf_counter()
            try:
                proc = subprocess.run(
                    cmd,
                    cwd=temp_repo,
                    capture_output=True,
                    text=True,
                    timeout=600,  # 10 分钟上限，按需调
                )
                duration = time.perf_counter() - start
            except FileNotFoundError:
                # 命令不存在，比如没装 pytest / ruff
                results.append(
                    {
                        "suite": suite_name,
                        "label": suite_label,
                        "configured": True,
                        "status": "skipped" if not required else "error",
                        "reason": "command_not_found",
                        "cmd": cmd,
                    }
                )
                continue
            except subprocess.TimeoutExpired:
                duration = time.perf_counter() - start
                results.append(
                    {
                        "suite": suite_name,
                        "label": suite_label,
                        "configured": True,
                        "status": "error",
                        "reason": "timeout",
                        "cmd": cmd,
                        "duration_sec": duration,
                    }
                )
                continue
            except Exception as e:
                duration = time.perf_counter() - start
                logger.exception("SelfProgrammingEngine: error running test suite %s", suite_name)
                results.append(
                    {
                        "suite": suite_name,
                        "label": suite_label,
                        "configured": True,
                        "status": "error",
                        "reason": f"exception: {e}",
                        "cmd": cmd,
                        "duration_sec": duration,
                    }
                )
                continue

            status = "passed" if proc.returncode == 0 else "failed"
            results.append(
                {
                    "suite": suite_name,
                    "label": suite_label,
                    "configured": True,
                    "status": status,
                    "cmd": cmd,
                    "returncode": proc.returncode,
                    "stdout": proc.stdout,
                    "stderr": proc.stderr,
                    "duration_sec": duration,
                }
            )

        return results

    # ------------------------------------------------------------
    # 内部：汇总测试套件结果
    # ------------------------------------------------------------
    def _summarize_suite_results(
        self,
        suite_results: List[Dict[str, Any]],
    ) -> (str, Dict[str, Any]):
        """
        将每个测试套件的结果汇总为一个整体状态。

        规则（简单版）：
          - 只要有 status == "failed" 的 → overall = "failed"
          - 否则只要有 status == "error" 的 → overall = "error"
          - 否则 → overall = "passed"
        """
        num = len(suite_results)
        num_passed = sum(1 for r in suite_results if r.get("status") == "passed")
        num_failed = sum(1 for r in suite_results if r.get("status") == "failed")
        num_error = sum(1 for r in suite_results if r.get("status") == "error")
        num_skipped = sum(1 for r in suite_results if r.get("status") == "skipped")

        if num_failed > 0:
            overall = "failed"
        elif num_error > 0:
            overall = "error"
        else:
            overall = "passed"

        metrics = {
            "num_suites": num,
            "num_suites_passed": num_passed,
            "num_suites_failed": num_failed,
            "num_suites_error": num_error,
            "num_suites_skipped": num_skipped,
            # severity_score 可以由外部注入，这里只保留字段位
        }
        return overall, metrics

# quentin/diagnostics/dashboard.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging

from quentin.self_programming.engine import PatchTestReport

logger = logging.getLogger(__name__)


@dataclass
class DashboardSnapshot:
    """
    系统状态快照，用于“仪表盘”展示。

    - blueprint_summary: 来自 BlueprintGapReport.summary
    - immune_summary: 来自 runtime.run_immune_cycle() 的 aggregated 结果
    - patch_summary: 根据 PatchTestReport 列表汇总
    - notes: 其它你想放进去的轻量信息
    """
    timestamp: str
    blueprint_summary: Dict[str, Any]
    immune_summary: Dict[str, Any]
    patch_summary: Dict[str, Any]
    notes: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


def build_dashboard_snapshot(
    *,
    blueprint_report: Optional[Dict[str, Any]],
    immune_result: Optional[Dict[str, Any]],
    patch_reports: List[PatchTestReport],
    notes: Optional[Dict[str, Any]] = None,
) -> DashboardSnapshot:
    """
    根据当前一轮的蓝皮书报告 / 免疫结果 / patch 测试结果，构造 DashboardSnapshot。
    """
    ts = datetime.utcnow().isoformat()

    # 1) 蓝皮书部分
    if blueprint_report is None:
        bp_summary = {
            "gap_count": 0,
            "missing_files": 0,
            "stub_implementations": 0,
            "spec_version": "unknown",
        }
    else:
        summary = blueprint_report.get("summary") or {}
        bp_summary = {
            "gap_count": summary.get("gap_count", 0),
            "missing_files": summary.get("missing_files", 0),
            "stub_implementations": summary.get("stub_implementations", 0),
            "spec_version": blueprint_report.get("spec_version", "unknown"),
        }

    # 2) 免疫循环部分（这里保持通用，只做简单提取）
    if immune_result is None:
        im_summary = {
            "cases_in_last_scan": 0,
        }
    else:
        im_summary = {
            "cases_in_last_scan": immune_result.get("count", 0),
        }

    # 3) patch 测试部分
    num_patches = len(patch_reports)
    num_passed = sum(1 for r in patch_reports if r.status == "passed")
    num_failed = sum(1 for r in patch_reports if r.status == "failed")
    num_error = sum(1 for r in patch_reports if r.status == "error")

    patch_summary = {
        "num_patches": num_patches,
        "num_passed": num_passed,
        "num_failed": num_failed,
        "num_error": num_error,
    }

    snapshot = DashboardSnapshot(
        timestamp=ts,
        blueprint_summary=bp_summary,
        immune_summary=im_summary,
        patch_summary=patch_summary,
        notes=notes or {},
    )
    return snapshot


def write_dashboard_snapshot(
    *,
    repo_root: str | Path,
    snapshot: DashboardSnapshot,
    filename: str = "data/dashboard/latest_snapshot.json",
) -> Path:
    """
    把 DashboardSnapshot 写成 JSON 文件。

    默认路径：<repo_root>/data/dashboard/latest_snapshot.json
    """
    repo_root = Path(repo_root).resolve()
    out_path = repo_root / filename
    out_path.parent.mkdir(parents=True, exist_ok=True)

    text = json.dumps(snapshot.to_dict(), ensure_ascii=False, indent=2)
    out_path.write_text(text, encoding="utf-8")

    logger.info("Dashboard snapshot written to %s", out_path)
    return out_path
