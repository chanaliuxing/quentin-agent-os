class MetaAgentRuntime:
    def __init__(
        self,
        *,
        model_router,
        memory_fabric,
        tool_manager,
        critic,
        council=None,
        repo_root: str | Path = ".",
        research_layer: Optional[ResearchLayer] = None,
        # 可选：允许外部覆盖 spec / registry / improver
        blueprint_spec: Optional[BlueprintSpec] = None,
        capabilities_registry: Optional[CapabilitiesRegistry] = None,
        blueprint_improver: Optional[BlueprintImprover] = None,
        # ... 其它参数 ...
    ) -> None:
        self.model_router = model_router
        self.memory_fabric = memory_fabric
        self.tool_manager = tool_manager
        self.critic = critic
        self.council = council
        self.repo_root = Path(repo_root).resolve()

        # Research Layer（上一步已经实现）
        self.research_layer = research_layer

        # -------- 蓝皮书 & 能力注册表 --------
        self.blueprint_spec: BlueprintSpec = blueprint_spec or load_default_spec_v3()
        self.capabilities_registry: CapabilitiesRegistry = (
            capabilities_registry or CapabilitiesRegistry(repo_root=self.repo_root)
        )

        # -------- 蓝皮书改进器：优先使用多模型“开会” --------
        if blueprint_improver is not None:
            self.blueprint_improver = blueprint_improver
        else:
            if self.council is not None:
                # 使用 MultiModelCouncil 做多 AI 会议
                self.blueprint_improver = BlueprintImprover(
                    council=self.council,
                    default_domain="blueprint-improvement",
                )
            else:
                # 回退为单 LLM（比如 model_router 中的默认模型）
                self.blueprint_improver = BlueprintImprover(
                    llm=model_router,      # 要求 model_router 实现 complete()/__call__()
                    default_domain="blueprint-improvement",
                )
这样，runtime 内部就有：
blueprint_spec + capabilities_registry + blueprint_improver + research_layer
后面就能一键跑“蓝皮书自迭代”。

二、核心入口：run_blueprint_improvement_cycle
目标：
让 runtime 能够自己完成这一串动作：

BlueprintGapTracker.scan() → 拿到蓝皮书 gap 报告。

BlueprintImprover.propose_improvement_plans(...) → 为缺失/部分实现能力生成实验计划。

ResearchLayer.create_experiment(...) → 注册实验。

ResearchLayer.run_research_cycle(...) → 跑一轮研究周期。

在 MetaAgentRuntime 类里增加一个方法：

    def run_blueprint_improvement_cycle(
        self,
        *,
        max_gaps: int = 10,
        research_request: Optional[ResearchCycleRequest] = None,
    ) -> Dict[str, Any]:
        """
        蓝皮书自迭代的一键入口：

          1. 扫描蓝皮书与当前实现的差距（gap_report）
          2. 针对缺失/部分实现能力，用多模型会议/LLM 生成改进实验计划
          3. 为这些计划创建 experiment（挂到 ResearchLayer）
          4. 调用 ResearchLayer.run_research_cycle(...) 跑一轮研究周期

        返回一个结构化 dict，包含：
          - gap_report（蓝皮书差距概览）
          - improvement_plans（新生成的计划）
          - created_experiments（新建的 experiment_id 列表）
          - research_cycle_result（本轮研究周期的结果）
        """
        if self.research_layer is None:
            raise RuntimeError("ResearchLayer is not attached to MetaAgentRuntime.")

        # 1) 蓝皮书 gap 扫描
        gap_tracker = BlueprintGapTracker(
            spec=self.blueprint_spec,
            registry=self.capabilities_registry,
        )
        gap_report = gap_tracker.scan()
        gap_dict = gap_report.to_dict()

        # 如果没有 missing/partial，就直接返回“没有 gap”
        if not gap_report.missing and not gap_report.partial:
            return {
                "status": "no_gaps",
                "reason": "All blueprint capabilities are marked as ok or unknown.",
                "gap_report": gap_dict,
            }

        # 2) 根据 gap 报告，生成改进计划（多 AI 会议优先）
        improvement_plans: List[GapImprovementPlan] = self.blueprint_improver.propose_improvement_plans(
            gap_report,
            max_gaps=max_gaps,
        )

        if not improvement_plans:
            return {
                "status": "no_plans",
                "reason": "LLM / council failed to propose plans for current gaps.",
                "gap_report": gap_dict,
            }

        # 3) 把计划注册成 experiments（交给 ResearchLayer/ExperimentOrchestrator）
        experiments_created: List[str] = []
        exp_orch = self.research_layer.experiment_orchestrator
        variant_ids = list(exp_orch.variants_runtimes.keys())

        for plan in improvement_plans:
            record = self.research_layer.create_experiment(
                name=plan.experiment_name,
                description=plan.experiment_description,
                variants=variant_ids,
                tasks=plan.tasks,  # 已经是 ExperimentTask 列表
                meta={
                    "domain": "blueprint-improvement",
                    "priority": plan.priority,
                    "capability_id": plan.capability_id,
                },
            )
            experiments_created.append(record.config.experiment_id)

        # 4) 跑一轮研究周期（ResearchLayer 内部会决定跑哪些 experiment）
        req = research_request or ResearchCycleRequest(
            strategy="simple",
            max_experiments=None,  # 用预算控制器默认值
        )
        cycle_result = self.research_layer.run_research_cycle(req)

        return {
            "status": "ok",
            "gap_report": gap_dict,
            "improvement_plans": [p.to_dict() for p in improvement_plans],
            "created_experiments": experiments_created,
            "research_cycle_result": cycle_result.to_dict(),
        }
这就是“蓝皮书自迭代”的总开关：
runtime 不需要知道具体 domain，只围着蓝皮书 spec 去做“研究 + 演化”。

三、实际怎么用？
举个 最小示例（你可以后面写一个 scripts/run_blueprint_cycle.py）：

from quentin.runtime.system import MetaAgentRuntime
from quentin.research.experiments import ExperimentOrchestrator
from quentin.research.layer import ResearchLayer

# 1. 准备 variants_runtimes：不同配置的 MetaAgentRuntime 自己或子 runtime
variants_runtimes = {
    "baseline": some_runtime_baseline,
    "council": some_runtime_with_council,
}

# 2. 构造 ExperimentOrchestrator + ResearchLayer
exp_orch = ExperimentOrchestrator(
    repo_root=".",
    variants_runtimes=variants_runtimes,
)
research_layer = ResearchLayer(
    repo_root=".",
    experiment_orchestrator=exp_orch,
)

# 3. 构造主 MetaAgentRuntime，并注入 ResearchLayer
runtime = MetaAgentRuntime(
    model_router=your_model_router,
    memory_fabric=your_memory_fabric,
    tool_manager=your_tool_manager,
    critic=your_critic,
    council=your_multi_model_council,
    repo_root=".",
    research_layer=research_layer,
)

# 4. 一键跑“蓝皮书自迭代周期”
result = runtime.run_blueprint_improvement_cycle(max_gaps=5)
print(result["status"])
print(result["created_experiments"])
