一、SelfImprovementManager 补齐通用接口
文件：quentin/self_improvement/manager.py
（只需要在原有类上加方法，不必重写整个文件）

假设你内部是用一个 dict 存任务（之前我们在别的模块里已经 list_tasks() 了），可以加：

# quentin/self_improvement/manager.py

from __future__ import annotations

import json
import logging
from dataclasses import asdict
from pathlib import Path
from typing import Dict, Any, List, Optional

from quentin.self_improvement.schemas import ImprovementTask

logger = logging.getLogger(__name__)


class SelfImprovementManager:
    """
    说明：这里只展示和本次改动相关的接口。
    如果你已经有 __init__ / _load / _save / list_tasks 等实现，
    只需要把 get_task / upsert_task 这两个方法加进去即可。
    """

    def __init__(self, path: str | Path = "data/improvement_tasks.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._loaded = False
        self._tasks: Dict[str, ImprovementTask] = {}

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                raw = json.loads(self.path.read_text(encoding="utf-8") or "{}")
                self._tasks = {
                    tid: ImprovementTask.from_dict(d)
                    for tid, d in raw.items()
                }
            except Exception:
                logger.exception("Failed to load improvement tasks from %s", self.path)
                self._tasks = {}
        else:
            self._tasks = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        raw = {tid: asdict(t) for tid, t in self._tasks.items()}
        self.path.write_text(json.dumps(raw, ensure_ascii=False, indent=2), encoding="utf-8")

    # --- 原来就有的接口（示意） ---
    def list_tasks(self) -> List[ImprovementTask]:
        self._load()
        return list(self._tasks.values())

    # --- 新增：按 task_id 取任务 ---
    def get_task(self, task_id: str) -> Optional[ImprovementTask]:
        self._load()
        return self._tasks.get(task_id)

    # --- 新增：插入或更新任务 ---
    def upsert_task(self, task: ImprovementTask) -> None:
        """
        如果 task_id 已存在则覆盖，否则新增。
        """
        self._load()
        self._tasks[task.task_id] = task
        self._save()
如果你的 SelfImprovementManager 已经有类似方法，只要保证：
list_tasks()、get_task(task_id)、upsert_task(task) 这三个名字和语义一致即可。

二、Blueprint → ImprovementTask 同步模块
新文件：quentin/self_improvement/blueprint_sync.py

这个模块做事情：

调 BlueprintGapTracker.scan() 拿到所有 BlueprintGap

把它们变成 / 更新为 ImprovementTask

返回一个简单的同步报告

# quentin/self_improvement/blueprint_sync.py
from __future__ import annotations

import logging
from dataclasses import dataclass, asdict
from typing import Dict, Any, List, Optional

from quentin.blueprint.gap_tracker import BlueprintGapTracker, BlueprintGap
from quentin.self_improvement.manager import SelfImprovementManager
from quentin.self_improvement.schemas import ImprovementTask

logger = logging.getLogger(__name__)


@dataclass
class BlueprintSyncReport:
    created: int = 0
    updated: int = 0
    skipped: int = 0

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class BlueprintImprovementSync:
    """
    把蓝皮书里的每一条 requirement 映射为一个 ImprovementTask。

    约定：
    - task_id = f"bp::{capability_id}::{requirement_id}"
    - capability      = capability_id
    - requirement_id  = requirement_id
    - description     = "[Blueprint] {title}: {description}"
    - overall_status  = spec_status（missing / placeholder / partial / ready）
    """

    def __init__(
        self,
        gap_tracker: Optional[BlueprintGapTracker] = None,
        si_manager: Optional[SelfImprovementManager] = None,
    ) -> None:
        self.gap_tracker = gap_tracker or BlueprintGapTracker()
        self.si_manager = si_manager or SelfImprovementManager()

    # ---------- 映射规则（从 BlueprintGap -> ImprovementTask） ----------

    def _task_id_for_gap(self, gap: BlueprintGap) -> str:
        return f"bp::{gap.capability_id}::{gap.requirement_id}"

    def _build_description(self, gap: BlueprintGap) -> str:
        title = gap.title or gap.requirement_id
        desc = gap.description or ""
        status = gap.spec_status
        return f"[Blueprint][{status}] {title}: {desc}"

    # ---------- 主入口 ----------

    def sync_from_blueprint(
        self,
        *,
        status_filter: Optional[List[str]] = None,
        min_priority: float = 0.0,
    ) -> BlueprintSyncReport:
        """
        从蓝皮书同步 ImprovementTask。

        参数：
        - status_filter: 只同步某些状态（例如 ["missing", "placeholder"]），None 表示全部
        - min_priority : 只同步优先级 >= 该值的 requirement
        """
        gaps = self.gap_tracker.scan()
        existing = {t.task_id: t for t in self.si_manager.list_tasks()}

        report = BlueprintSyncReport()

        # 统一小写过滤
        status_filter_lower = (
            {s.lower() for s in status_filter} if status_filter else None
        )

        for gap in gaps:
            if status_filter_lower and gap.spec_status.lower() not in status_filter_lower:
                report.skipped += 1
                continue
            if gap.priority < min_priority:
                report.skipped += 1
                continue

            task_id = self._task_id_for_gap(gap)
            desc = self._build_description(gap)

            if task_id in existing:
                task = existing[task_id]
                changed = False

                # 对齐 capability / requirement / status / description
                if getattr(task, "capability", None) != gap.capability_id:
                    task.capability = gap.capability_id  # type: ignore[attr-defined]
                    changed = True

                if getattr(task, "requirement_id", None) != gap.requirement_id:
                    task.requirement_id = gap.requirement_id  # type: ignore[attr-defined]
                    changed = True

                if getattr(task, "overall_status", None) != gap.spec_status:
                    task.overall_status = gap.spec_status  # type: ignore[attr-defined]
                    changed = True

                if getattr(task, "description", None) != desc:
                    task.description = desc  # type: ignore[attr-defined]
                    changed = True

                if changed:
                    self.si_manager.upsert_task(task)
                    report.updated += 1
                else:
                    report.skipped += 1
            else:
                # 创建新任务 —— 尽量只用我们确定存在的字段
                task = ImprovementTask(
                    task_id=task_id,
                    capability=gap.capability_id,
                    requirement_id=gap.requirement_id,
                    description=desc,
                    overall_status=gap.spec_status,
                )
                self.si_manager.upsert_task(task)
                report.created += 1

        logger.info(
            "BlueprintImprovementSync finished: %s",
            report.to_dict(),
        )
        return report
这里刻意只用 ImprovementTask 已知字段：
task_id / capability / requirement_id / description / overall_status，
其它字段（如果你在 schemas.py 里有）都走默认值，避免冲突。

def cmd_improvement_sync_blueprint(args: argparse.Namespace) -> None:
    if args.status:
        status_filter = [s.strip() for s in args.status.split(",") if s.strip()]
    else:
        status_filter = None

    syncer = BlueprintImprovementSync()
    report = syncer.sync_from_blueprint(
        status_filter=status_filter,
        min_priority=args.min_priority,
    )
    data = report.to_dict()
    print("Blueprint → ImprovementTask sync result:")
    print(f"- created: {data['created']}")
    print(f"- updated: {data['updated']}")
    print(f"- skipped: {data['skipped']}")
