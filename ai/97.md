# quentin/runtime/config.py
from __future__ import annotations

from dataclasses import dataclass, field, asdict
from typing import Any, Dict


@dataclass
class LoopBudgetConfig:
    """
    控制各个循环的大致资源预算（v1 用于提示 / logging）。
    """
    max_traces_per_sleep: int = 100
    max_traces_per_immune_scan: int = 200


@dataclass
class CouncilPolicyConfig:
    """
    多 AI 开会 / 合议模式的策略。

    这里只是配置，不直接实现逻辑；
    具体什么时候真的开会，由 orchestrator / planner 解释这些字段。
    """
    # 对“任务级别”的控制
    enable_for_critical_tasks: bool = True
    enable_for_long_running_tasks: bool = True

    # 对“阶段”的控制
    use_for_planning: bool = True           # 做计划时启用多模 deliberation
    use_for_reflection: bool = True        # 反思 / 总结时启用合议
    use_for_coding_retries: bool = True    # 编程多次失败时触发合议

    # 简单阈值（交给上层解释）
    coding_retry_council_threshold: int = 2   # 连续失败 >= 2 次就开会


@dataclass
class RuntimeConfig:
    """
    MetaAgentRuntime 的顶层配置。
    """
    loop_budget: LoopBudgetConfig = field(default_factory=LoopBudgetConfig)
    council_policy: CouncilPolicyConfig = field(default_factory=CouncilPolicyConfig)

    # 将来可以在这里挂：
    # - 安全策略（宪法 id）
    # - 日志级别
    # - 默认模型路由策略 id 等
    extra: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        return d
多 AI 开会的具体“什么时候触发”，由 CouncilPolicyConfig 描述；
真正怎么用这些配置（比如 planner 看到“critical 任务 + 做计划阶段”就调用 multi-LLM council），交给 orchestrator 实现。

二、MetaAgentRuntime：quentin/runtime/system.py
这个类把所有核心组件串起来，给外部只暴露几个清晰方法：

run_task(goal, context)：认知循环

run_immune_cycle()：免疫循环

run_research_cycle()：把 ImmuneCase 变成 patch 实验（初版 stub）

# quentin/runtime/system.py
from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Optional, Tuple
from pathlib import Path
import logging
import uuid
from datetime import datetime

from quentin.runtime.config import RuntimeConfig
from quentin.trace.spec import ExecutionTrace, StepTrace
from quentin.trace.store import TraceStore
from quentin.memory.fabric import MemoryFabric, MemoryItem
from quentin.loops.immune import ImmuneLoop, ImmuneCaseStore, ImmuneCase
from quentin.loops.sleep import SleepLoop
from quentin.evolution.skill_loop import SkillEvolutionLoop
from quentin.self_programming.engine import SelfProgrammingEngine
from quentin.research.interface import ResearchInterface

logger = logging.getLogger(__name__)


@dataclass
class TaskRequest:
    """
    对外暴露的任务请求结构。
    """
    task_id: str
    goal: str
    domain: str = "general"
    priority: str = "normal"      # "low" / "normal" / "high" / "critical"
    context: Dict[str, Any] = None

    def to_dict(self) -> Dict[str, Any]:
        return {
            "task_id": self.task_id,
            "goal": self.goal,
            "domain": self.domain,
            "priority": self.priority,
            "context": self.context or {},
        }


@dataclass
class TaskResult:
    """
    run_task 的返回结构。
    """
    task_id: str
    goal: str
    output: Any
    trace: ExecutionTrace
    immune_signals: List[Dict[str, Any]]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "task_id": self.task_id,
            "goal": self.goal,
            "output": self.output,
            "trace": self.trace.to_dict(),
            "immune_signals": self.immune_signals,
        }


class MetaAgentRuntime:
    """
    Meta-Agent v3 的“总控大脑”。

    负责把：
    - Cognitive Loop（执行任务）
    - Immune Loop（异常检测）
    - Sleep Loop（记忆巩固 + 演化）
    - Research / Self-Programming
    串在一起。

    注意：
    - 这里不直接做 LLM 调用 / 具体策略，只负责调用 orchestrator 等组件。
    - 具体的 planner / executor / multi-LLM council 在 orchestrator 里实现。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        config: Optional[RuntimeConfig] = None,
        # 任务执行层（你可以传入自己的 orchestrator）
        orchestrator: Any,
        # 可选：多模型合议会（multi-LLM council）
        council: Optional[Any] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.config = config or RuntimeConfig()

        # —— 核心存储组件 —— #
        self.trace_store = TraceStore(repo_root=self.repo_root)
        self.memory_fabric = MemoryFabric(repo_root=self.repo_root)
        self.immune_case_store = ImmuneCaseStore(repo_root=self.repo_root)

        # —— 自编程 & 研究组件 —— #
        self.self_programming_engine = SelfProgrammingEngine(repo_root=self.repo_root)
        self.skill_evolution_loop = SkillEvolutionLoop(self.self_programming_engine)
        self.sleep_loop = SleepLoop(
            repo_root=self.repo_root,
            memory_fabric=self.memory_fabric,
            trace_store=self.trace_store,
            evolution_loop=self.skill_evolution_loop,
            self_programming_engine=self.self_programming_engine,
        )
        self.immune_loop = ImmuneLoop(
            repo_root=self.repo_root,
            trace_store=self.trace_store,
            case_store=self.immune_case_store,
        )
        self.research = ResearchInterface(repo_root=self.repo_root)

        # —— 执行层入口 —— #
        self.orchestrator = orchestrator
        self.council = council

    # ------------------------------------------------------------------
    # 1. 认知循环：执行单个任务
    # ------------------------------------------------------------------
    def run_task(self, goal: str, context: Optional[Dict[str, Any]] = None) -> TaskResult:
        """
        对外的主要入口：

        1. 构造 TaskRequest（带 domain / priority / council 策略等信息）
        2. 调用 orchestrator.run_task()
        3. 记录 ExecutionTrace
        4. 写入短期记忆（STS）
        5. 把 trace 交给 ImmuneLoop 做初步异常检查
        """
        ctx = dict(context or {})
        task_id = ctx.get("task_id") or uuid.uuid4().hex
        domain = ctx.get("domain", "general")
        priority = ctx.get("priority", "normal")

        request = TaskRequest(
            task_id=task_id,
            goal=goal,
            domain=domain,
            priority=priority,
            context=ctx,
        )

        logger.info("MetaAgentRuntime.run_task: task_id=%s, goal=%s", task_id, goal)

        # 将 RuntimeConfig 和 CouncilPolicy 提供给 orchestrator，
        # 由它决定何时启用多模型合议等高级功能。
        orchestration_context = {
            "runtime_config": self.config.to_dict(),
            "council": self.council,
        }

        # ==== 调用执行层 ====
        # 约定：orchestrator.run_task(...) 返回 (output, trace)
        # 其中 trace 必须是 ExecutionTrace。
        output, trace = self._invoke_orchestrator(request, orchestration_context)

        # ==== 写 Trace ====
        self.trace_store.append(trace)

        # ==== 写 STS 记忆（task 级 episodic） ====
        self._write_task_memory(trace)

        # ==== Immune 初步观测 ====
        immune_signals = self._run_immune_observe(trace)

        return TaskResult(
            task_id=task_id,
            goal=goal,
            output=output,
            trace=trace,
            immune_signals=[s for s in immune_signals],
        )

    def _invoke_orchestrator(
        self,
        request: TaskRequest,
        orchestration_context: Dict[str, Any],
    ) -> Tuple[Any, ExecutionTrace]:
        """
        统一封装对 orchestrator 的调用。

        真实实现中，请确保 orchestrator 提供：
        output, trace = orchestrator.run_task(request=request, context=orchestration_context)
        """
        if not hasattr(self.orchestrator, "run_task"):
            raise RuntimeError("Orchestrator must implement run_task(request, context)")

        result = self.orchestrator.run_task(
            request=request,
            context=orchestration_context,
        )
        if not isinstance(result, tuple) or len(result) != 2:
            raise RuntimeError("orchestrator.run_task must return (output, trace)")

        output, trace = result
        if not isinstance(trace, ExecutionTrace):
            raise RuntimeError("orchestrator.run_task must return ExecutionTrace as second element")

        return output, trace

    def _write_task_memory(self, trace: ExecutionTrace) -> None:
        """
        把单个任务的执行结果写入 STS。

        SleepLoop 也会对 recent traces 做 batch 写入，这里相当于“实时版”。
        将来如果觉得重复，可以在 SleepLoop 里做去重。
        """
        item = MemoryItem(
            memory_id=uuid.uuid4().hex,
            memory_type="episodic",
            content={
                "trace_id": trace.trace_id,
                "task_id": trace.task_id,
                "domain": trace.domain,
                "status": trace.status,
                "execution_summary": trace.execution_summary,
                "lessons_learned": trace.lessons_learned,
                "top_level_metrics": trace.top_level_metrics,
            },
            tier="STS",
            domain=trace.domain,
            created_at=trace.created_at or datetime.utcnow().isoformat(),
            importance=0.7 if trace.status == "success" else 0.4,
            origin={"type": "trace", "task_id": trace.task_id},
        )
        self.memory_fabric.add_memory(item)

    def _run_immune_observe(self, trace: ExecutionTrace) -> List[Dict[str, Any]]:
        """
        对单个任务跑 ImmuneLoop 的快速检测。

        注意：这里只生成 ImmuneSignal，不一定立刻形成 ImmuneCase。
        聚合扫描在 run_immune_cycle() 里做。
        """
        signals = self.immune_loop.observe_trace(trace)
        return [s.to_dict() for s in signals]

    # ------------------------------------------------------------------
    # 2. 免疫循环：周期性扫描异常
    # ------------------------------------------------------------------
    def run_immune_cycle(self, *, limit_traces: Optional[int] = None) -> Dict[str, Any]:
        """
        周期性调用（比如每小时一次）：

        - 从最近 trace 中聚合失败模式
        - 生成 ImmuneCase 列表，写入 data/immune/cases.jsonl
        - 返回本轮生成的 case，用于 logs 或 ResearchLayer 后续处理
        """
        limit = limit_traces or self.config.loop_budget.max_traces_per_immune_scan
        cases: List[ImmuneCase] = self.immune_loop.scan_recent_traces(limit=limit)

        return {
            "generated_cases": [c.to_dict() for c in cases],
            "count": len(cases),
        }

    # ------------------------------------------------------------------
    # 3. 睡眠循环：记忆巩固 + 技能/代码演化
    # ------------------------------------------------------------------
    def run_sleep_cycle(self, *, extra_context: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """
        SleepLoop 的统一入口：

        - 写批量 STS 记忆
        - 简单 MTS 候选 promoción
        - 触发 SkillEvolutionLoop.run_once（从成功 trace 中挖技能）
        """
        ctx = dict(extra_context or {})
        ctx.setdefault("evolution_context", {})
        ctx.setdefault("test_suites", ["unit", "static_analysis"])

        result = self.sleep_loop.run_once(extra_context=ctx)

        return result

    # ------------------------------------------------------------------
    # 4. 研究 / 自编程循环（从 ImmuneCase 到 Patch 实验）
    # ------------------------------------------------------------------
    def run_research_cycle(self, *, max_cases: int = 20) -> Dict[str, Any]:
        """
        初版 Research 循环：

        思路：
        1. 从 ImmuneCaseStore 读取最新若干 case（目前 ImmuneCaseStore 只有 append，可以后续加 load）
           —— v1: 暂时直接调用 immune_loop.scan_recent_traces() 当作 “本轮 case”
        2. 把 case 映射为 issues，丢给 SkillEvolutionLoop.generate_patches_from_issues()
        3. 对这些 patch 跑测试 / 实验（目前 run_tests_for_patch 还是 stub）
        """
        # 1) 获取 case（v1: 现用 scan_recent_traces 代替从 case_store 读取）
        cases: List[ImmuneCase] = self.immune_loop.scan_recent_traces(limit=max_cases)
        issues: List[Dict[str, Any]] = []

        for case in cases:
            issues.append(
                {
                    "issue_id": case.case_id,
                    "source": "immune_case",
                    "target": case.target,
                    "reason": case.summary,
                    "metrics": {
                        "severity": case.severity,
                        "signal_count": len(case.signals),
                    },
                }
            )

        # 2) 生成 PatchCandidate
        patch_candidates = self.skill_evolution_loop.generate_patches_from_issues(issues)

        # 3) 评估 Patch（内部会调用 SelfProgrammingEngine.run_tests_for_patch）
        patch_reports = self.skill_evolution_loop.evaluate_patches(
            patch_candidates,
            test_suites=["unit", "static_analysis"],
            extra_context={"source": "research_cycle"},
        )

        promotable = self.self_programming_engine.select_patches_to_promote(
            patch_reports,
            max_promote=1,
        )

        return {
            "cases": [c.to_dict() for c in cases],
            "patch_candidates": [p.to_dict() for p in patch_candidates],
            "patch_reports": [r.to_dict() for r in patch_reports],
            "promotable_patches": [r.to_dict() for r in promotable],
        }
