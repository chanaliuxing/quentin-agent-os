from dataclasses import dataclass, field, asdict
from datetime import datetime
from typing import Any, Dict, List, Optional


@dataclass
class TaskSpec:
    """
    通用任务描述（域无关）：
    - description: 人类自然语言任务
    - type      : 任务类型标签（"generic" / "analysis" / "coding" / ...）
    - priority  : 简单优先级（数值越大越高）
    """

    task_id: str
    description: str
    type: str = "generic"
    priority: int = 0
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class TaskResult:
    """
    一次任务执行的结果摘要（供上层记忆 / 研究 / UI 使用）。
    """

    task_id: str
    success: bool
    score: float

    final_output: str
    plan_steps: List[str] = field(default_factory=list)
    step_outputs: List[str] = field(default_factory=list)

    retries: int = 0
    started_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    finished_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    critic_reflection: Optional[str] = None
    critic_next_action_hint: Optional[str] = None
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, Optional
import json
import logging
import time

from quentin.llm.types import LLMConfig

logger = logging.getLogger(__name__)


@dataclass
class RouteFeedback:
    mode: str
    task_type: str
    model: str
    provider: str
    latency_ms: float
    success: bool
    score: float = 0.0

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class ModelRouter:
    """
    ModelRouter v1：

    - 针对不同 mode（planner/executor/critic/embedding）和 task_type，
      返回一个 LLMConfig（目前是静态规则）
    - 提供 record_feedback 用来写入 data/model_router_stats.jsonl，
      为以后做 bandit/RL 路由准备数据。

    v1 策略（本地环境假设）：
      - planner / critic  → qwen3-vl:latest
      - executor          → deepseek-r1:1.5b
      - embedding         → bge-m3
    """

    def __init__(
        self,
        *,
        stats_store: str | Path = "data/model_router_stats.jsonl",
    ) -> None:
        self.stats_store = Path(stats_store)
        self.stats_store.parent.mkdir(parents=True, exist_ok=True)

    # ---------- 路由 ----------

    def route(
        self,
        *,
        mode: str,
        task_type: str = "generic",
    ) -> LLMConfig:
        mode = mode.lower()
        task_type = task_type.lower()

        # 可以根据 task_type 做更多细分；v1 先简单写死
        if mode in ("planner", "critic"):
            return LLMConfig(
                provider="ollama",
                model="qwen3-vl:latest",
                role=f"{mode}_{task_type}",
            )
        elif mode == "executor":
            return LLMConfig(
                provider="ollama",
                model="deepseek-r1:1.5b",
                role=f"{mode}_{task_type}",
            )
        elif mode == "embedding":
            return LLMConfig(
                provider="ollama",
                model="bge-m3",
                role=f"{mode}_{task_type}",
            )
        else:
            # fallback：用 qwen
            return LLMConfig(
                provider="ollama",
                model="qwen3-vl:latest",
                role=f"{mode}_{task_type}",
            )

    # ---------- 反馈记录（供未来学习用） ----------

    def record_feedback(self, fb: RouteFeedback) -> None:
        try:
            with self.stats_store.open("a", encoding="utf-8") as f:
                f.write(json.dumps(fb.to_dict(), ensure_ascii=False) + "\n")
        except Exception:
            logger.exception("ModelRouter.record_feedback: failed to append to %s", self.stats_store)

# quentin/core/critic.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, List, Optional
import json
import logging

from quentin.llm.multi_council import MultiAICouncil, CouncilDecision
from quentin.llm.types import LLMConfig

logger = logging.getLogger(__name__)


@dataclass
class CriticReport:
    success: bool
    score: float
    reflection: str
    next_action_hint: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class TaskCritic:
    """
    任务级 Critic：

    - 输入：task_description, plan_steps, step_outputs
    - 输出：成功与否、0~1 评分、反思+下一步建议
    - 内部用 MultiAICouncil 做多模型会议，减少单模型幻觉风险。
    """

    def __init__(self, council: Optional[MultiAICouncil] = None) -> None:
        if council is None:
            members = [
                LLMConfig(
                    provider="ollama",
                    model="qwen3-vl:latest",
                    role="task_judge",
                ),
                LLMConfig(
                    provider="ollama",
                    model="deepseek-r1:1.5b",
                    role="reflection_helper",
                ),
            ]
            council = MultiAICouncil(members=members)
        self.council = council

    def review(
        self,
        *,
        task_description: str,
        plan_steps: List[str],
        step_outputs: List[str],
    ) -> CriticReport:
        ctx_lines: List[str] = [
            "TASK DESCRIPTION:",
            task_description,
            "",
            "PLAN STEPS:",
        ]
        for i, s in enumerate(plan_steps):
            ctx_lines.append(f"{i+1}. {s}")

        ctx_lines.append("")
        ctx_lines.append("STEP OUTPUTS:")
        for i, s in enumerate(step_outputs):
            ctx_lines.append(f"{i+1}. {s}")

        context = "\n".join(ctx_lines)

        question = (
            "You are a CRITIC COUNCIL evaluating a task execution.\n\n"
            "You must:\n"
            "1. Judge whether the task is completed satisfactorily.\n"
            "2. Give a numeric score between 0.0 and 1.0.\n"
            "3. Provide a short reflection (what went wrong or well).\n"
            "4. Provide a short next_action_hint (how to improve in a next attempt).\n\n"
            "Return ONLY ONE JSON object:\n"
            "{\n"
            "  \"success\": true/false,\n"
            "  \"score\": 0.0-1.0,\n"
            "  \"reflection\": \"...\",\n"
            "  \"next_action_hint\": \"...\"\n"
            "}\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="task_critic",
        )
        ans = decision.final_answer or "{}"

        try:
            s = ans.strip()
            first = s.find("{")
            last = s.rfind("}")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)
        except Exception:
            logger.exception("TaskCritic.review: failed to parse council JSON: %r", ans)
            data = {
                "success": False,
                "score": 0.0,
                "reflection": "Failed to parse critic council answer.",
                "next_action_hint": "Retry with clearer constraints and more step-by-step reasoning.",
            }

        success = bool(data.get("success", False))
        try:
            score = float(data.get("score", 0.0))
        except Exception:
            score = 0.0
        reflection = str(data.get("reflection", "") or "")
        next_hint = str(data.get("next_action_hint", "") or "")

        return CriticReport(
            success=success,
            score=score,
            reflection=reflection,
            next_action_hint=next_hint,
        )

# quentin/core/cognitive_loop.py
from __future__ import annotations

from datetime import datetime
from typing import List, Optional
import json
import logging
import time
import uuid

from quentin.core.tasks import TaskSpec, TaskResult
from quentin.core.traces import ExecutionTrace
from quentin.core.critic import TaskCritic, CriticReport
from quentin.core.immune_loop import ImmuneLoop
from quentin.llm.router import ModelRouter, RouteFeedback
from quentin.llm.client import call_llm  # 假设已有的通用 LLM 调用工具
from quentin.memory.fabric import FileBackedMemoryFabric
from quentin.research.experiments import ResearchManager, Experiment

logger = logging.getLogger(__name__)


class CognitiveLoop:
    """
    Cognitive Loop v1：

    - 输入：TaskSpec
    - 步骤：
        1）Planner：LLM 生成结构化 plan steps
        2）Executor：逐步执行 plan（目前仅 LLM 思维，不接外部工具）
        3）Critic：多 AI 会议给出 success/score/reflection
        4）若失败或得分低，基于 reflection 重试一次（简单 self-correction）
        5）写 ExecutionTrace → ImmuneLoop
        6）把经验写入 MemoryFabric
        7）写入 ResearchManager 作为认知循环指标

    v1 不涉及特定领域，完全通用。
    """

    def __init__(
        self,
        *,
        model_router: Optional[ModelRouter] = None,
        critic: Optional[TaskCritic] = None,
        immune_loop: Optional[ImmuneLoop] = None,
        memory_fabric: Optional[FileBackedMemoryFabric] = None,
        research_manager: Optional[ResearchManager] = None,
    ) -> None:
        self.model_router = model_router or ModelRouter()
        self.critic = critic or TaskCritic()
        self.immune_loop = immune_loop or ImmuneLoop()
        self.memory_fabric = memory_fabric or FileBackedMemoryFabric()
        self.research_manager = research_manager or ResearchManager()

    # ---------- 主入口 ----------

    def run_task(self, task: TaskSpec) -> TaskResult:
        started_at = datetime.utcnow().isoformat()

        # 第一次尝试
        plan_steps, step_outputs, critic_report = self._run_once(task, retry_hint=None)
        retries = 0

        # 如果失败或得分太低，则基于 reflection 再试一次
        if not critic_report.success or critic_report.score < 0.7:
            retries = 1
            plan_steps, step_outputs, critic_report = self._run_once(
                task,
                retry_hint=critic_report.next_action_hint or critic_report.reflection,
            )

        final_output = step_outputs[-1] if step_outputs else ""
        finished_at = datetime.utcnow().isoformat()

        result = TaskResult(
            task_id=task.task_id,
            success=critic_report.success,
            score=critic_report.score,
            final_output=final_output,
            plan_steps=plan_steps,
            step_outputs=step_outputs,
            retries=retries,
            started_at=started_at,
            finished_at=finished_at,
            critic_reflection=critic_report.reflection,
            critic_next_action_hint=critic_report.next_action_hint,
        )

        # 写 Trace + ImmuneLoop
        trace = self._build_trace(task, result)
        self.immune_loop.process_trace(trace)

        # 写 Memory（简单版）
        self._store_memory(task, result)

        # 写 Research
        self._record_research(task, result)

        return result

    # ---------- 单次尝试：规划 + 执行 + 评审 ----------

    def _run_once(
        self,
        task: TaskSpec,
        retry_hint: Optional[str],
    ) -> (List[str], List[str], CriticReport):
        # Planner
        plan_steps = self._plan(task, retry_hint=retry_hint)

        # Executor
        step_outputs = self._execute(task, plan_steps)

        # Critic
        critic_report = self.critic.review(
            task_description=task.description,
            plan_steps=plan_steps,
            step_outputs=step_outputs,
        )
        return plan_steps, step_outputs, critic_report

    # ---------- Planner ----------

    def _plan(self, task: TaskSpec, retry_hint: Optional[str]) -> List[str]:
        cfg = self.model_router.route(mode="planner", task_type=task.type)

        system = (
            "You are a PLANNER for a self-evolving agent. "
            "You must break the user task into a small number of clear steps.\n"
            "Return ONLY JSON object like:\n"
            "{ \"steps\": [\"step 1...\", \"step 2...\", ...] }\n"
        )
        if retry_hint:
            system += f"\nPrevious attempt had issues. Improvement hint: {retry_hint}\n"

        user = f"Task: {task.description}"

        t0 = time.time()
        raw = call_llm(cfg, user_prompt=user, system_prompt=system)
        latency = (time.time() - t0) * 1000.0

        # 记录路由反馈（未来可学习）
        fb = RouteFeedback(
            mode="planner",
            task_type=task.type,
            model=cfg.model,
            provider=cfg.provider,
            latency_ms=latency,
            success=True,
        )
        self.model_router.record_feedback(fb)

        # 解析 JSON
        steps: List[str] = []
        try:
            s = raw.strip()
            first = s.find("{")
            last = s.rfind("}")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)
            arr = data.get("steps") or []
            steps = [str(x) for x in arr if str(x).strip()]
        except Exception:
            logger.exception("CognitiveLoop._plan: failed to parse planner JSON: %r", raw)
            # fallback：用整段当作一步
            steps = [f"(planner_output) {raw.strip()}"]

        if not steps:
            steps = [f"Solve the task directly: {task.description}"]

        return steps

    # ---------- Executor ----------

    def _execute(self, task: TaskSpec, plan_steps: List[str]) -> List[str]:
        cfg = self.model_router.route(mode="executor", task_type=task.type)

        system = (
            "You are an EXECUTOR for a self-evolving agent.\n"
            "For each step, you must reason and produce the best possible answer "
            "for that step, given the overall task.\n"
            "You should respond in natural language, no JSON, no code blocks unless necessary.\n"
        )

        outputs: List[str] = []
        for idx, step in enumerate(plan_steps, start=1):
            user = (
                f"Overall task: {task.description}\n"
                f"Current step #{idx}: {step}\n\n"
                "Please execute this step and output the result."
            )
            t0 = time.time()
            out = call_llm(cfg, user_prompt=user, system_prompt=system)
            latency = (time.time() - t0) * 1000.0

            fb = RouteFeedback(
                mode="executor",
                task_type=task.type,
                model=cfg.model,
                provider=cfg.provider,
                latency_ms=latency,
                success=True,
            )
            self.model_router.record_feedback(fb)

            outputs.append(out.strip())

        return outputs

    # ---------- Trace ----------

    def _build_trace(self, task: TaskSpec, result: TaskResult) -> ExecutionTrace:
        trace_id = str(uuid.uuid4())
        metrics = {
            "score": result.score,
            "retries": float(result.retries),
            "num_steps": float(len(result.plan_steps)),
        }
        tags = {
            "task_type": task.type,
            "success": result.success,
        }

        return ExecutionTrace(
            trace_id=trace_id,
            task_id=task.task_id,
            component_id="cognitive_loop.v1",
            blueprint_capability_id="cognitive_loop",
            target_module="quentin.core.cognitive_loop",
            target_symbol="CognitiveLoop.run_task",
            success=result.success,
            error_type=None if result.success else "task_failed",
            error_message=None if result.success else result.critic_reflection,
            start_time=result.started_at,
            end_time=result.finished_at,
            metrics=metrics,
            tags=tags,
        )

    # ---------- Memory ----------

    def _store_memory(self, task: TaskSpec, result: TaskResult) -> None:
        if not result.success:
            # v1：只对成功任务写入记忆
            return
        summary = (
            f"Task '{task.description}' was solved with score {result.score:.2f}. "
            f"Final output: {result.final_output[:400]}"
        )
        self.memory_fabric.add_semantic_lesson(
            tier="STS",
            content=summary,
            domain="generic",
            source="cognitive_loop",
        )

        if result.critic_reflection:
            hint = (
                f"For tasks similar to '{task.description}', "
                f"remember: {result.critic_reflection}"
            )
            self.memory_fabric.add_procedural_hint(
                tier="MTS",
                content=hint,
                component_id="cognitive_loop.v1",
            )

    # ---------- Research ----------

    def _record_research(self, task: TaskSpec, result: TaskResult) -> None:
        rm = self.research_manager
        capability = "cognitive:loop"
        requirement_id = "cognitive_loop_success_rate"

        exps = rm.list_experiments()
        exp: Optional[Experiment] = None
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "cognitive_loop"
            ):
                exp = e
                break

        if exp is None:
            exp = rm.create_experiment(
                name="CognitiveLoop",
                purpose="cognitive_loop",
                capability=capability,
                requirement_id=requirement_id,
                tags=["cognitive", "loop"],
            )

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics={
                "success": 1.0 if result.success else 0.0,
                "score": result.score,
                "retries": float(result.retries),
            },
            context={
                "task_id": task.task_id,
                "task_type": task.type,
            },
        )
