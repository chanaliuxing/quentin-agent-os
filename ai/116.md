# quentin/runtime/system.py
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Any, Dict, Optional

from quentin.traces.core import TraceRecorder, JsonlTraceRecorder
from quentin.runtime.cognitive_loop import CognitiveLoop, TaskRequest, TaskResult
from quentin.runtime.immune_loop import ImmuneLoop, AnomalyDetectorConfig, ImmuneAction
from quentin.runtime.sleep_loop import SleepLoop, SleepLoopConfig
from quentin.skills.evolution import SkillEvolutionLoop, SkillEvolutionStats


@dataclass
class MetaAgentRuntime:
    """
    系统中控（Base Layer + 部分 Meta Layer 的入口）。

    负责：
      - 统一对外暴露“执行任务”的接口（可以走 CognitiveLoop）
      - 对 ResearchLayer 提供：
          * run_blueprint_improvement_cycle(...)
          * auto_update_capabilities_from_repo(...)
      - 挂载四大 Loop：
          * cognitive_loop
          * immune_loop
          * sleep_loop
          * skill_evolution_loop
    """

    model_router: Any
    memory_fabric: Any
    tool_manager: Any
    critic: Any
    council: Optional[Any] = None
    repo_root: str | Path = "."
    research_layer: Optional[Any] = None

    # 新增：观测与循环
    trace_recorder: Optional[TraceRecorder] = None
    cognitive_loop: Optional[CognitiveLoop] = None
    immune_loop: Optional[ImmuneLoop] = None
    sleep_loop: Optional[SleepLoop] = None
    skill_evolution_loop: Optional[SkillEvolutionLoop] = None

    def __post_init__(self) -> None:
        self.repo_root = Path(self.repo_root).resolve()

        # 默认 trace_recorder
        if self.trace_recorder is None:
            self.trace_recorder = JsonlTraceRecorder(repo_root=self.repo_root)

        # Loop 可能在外部构造后再 attach，这里不强制

    # --------------------------------------------------
    # 接 ResearchLayer 的挂载
    # --------------------------------------------------
    def attach_research_layer(self, research_layer: Any) -> None:
        self.research_layer = research_layer

    # --------------------------------------------------
    # 统一任务执行接口（底层执行，不含 self-reflection）
    # 被 CognitiveLoop 调用
    # --------------------------------------------------
    def execute_task(
        self,
        *,
        task_id: str,
        description: str,
        payload: Dict[str, Any],
        context: Dict[str, Any],
        trace_id: str,
        trace_recorder: Optional[TraceRecorder] = None,
    ) -> Dict[str, Any]:
        """
        这是“最底层执行引擎”，目前实现极简版本：
          - 把 description + payload + context 拼成一个 prompt
          - 调用 model_router(prompt)
          - 以后可以在这里挂 planner / executor / 工具调用等复杂逻辑

        注意：
          - CognitiveLoop 会负责 Trace + Critic + Lessons
          - ImmuneLoop 的统计也可以在上层（CognitiveLoop）记录
        """
        tr = trace_recorder or self.trace_recorder

        # 你可以在这里拓展：调用 ToolManager、Council 等
        prompt = (
            "你是一个通用智能体，请按照用户任务完成工作。\n\n"
            f"任务描述:\n{description}\n\n"
            f"任务输入(JSON):\n{payload}\n\n"
            f"上下文(JSON):\n{context}\n\n"
            "请直接给出你的输出结果，使用尽量结构化的形式。"
        )

        # 这里暂时不细分多模型会议，使用主路由即可
        text = self.model_router(prompt)
        return {"raw_output": text}

    # --------------------------------------------------
    # 对外暴露的“run_task”入口（带认知循环）
    # --------------------------------------------------
    def run_task(
        self,
        description: str,
        payload: Dict[str, Any],
        context: Optional[Dict[str, Any]] = None,
    ) -> TaskResult:
        """
        对外推荐使用这个接口执行任务：
          - 会经过 CognitiveLoop
          - 自动写 Trace / Summary / Lessons / Memory
          - ImmuneLoop 的统计也会在 CognitiveLoop 里被记录
        """
        if not self.cognitive_loop:
            raise RuntimeError("CognitiveLoop is not attached to MetaAgentRuntime.")

        return self.cognitive_loop.run_task(description, payload, context or {})

    # --------------------------------------------------
    # 与蓝皮书自迭代相关的接口（你之前已经部分实现）
    # --------------------------------------------------
    def auto_update_capabilities_from_repo(self, repo_root: str | Path | None = None) -> Dict[str, Any]:
        """
        这里假设你已经实现了 CapabilityRegistry + LLMBasedCapabilityProbe 等，
        现在只是保留接口。你可以在之前的基础上加入 trace 记录。
        """
        # TODO: 调用 BlueprintGapTracker + LLMBasedCapabilityProbe
        # 当前先给一个占位返回结构，避免打断其它逻辑。
        return {
            "registry_snapshot": {
                "capabilities": {},
            },
            "probe_log": [],
        }

    def run_blueprint_improvement_cycle(
        self,
        *,
        max_gaps: int,
        research_request: Any,
    ) -> Dict[str, Any]:
        """
        这里也是假设你已经有 BlueprintImprover + ResearchLayer 的逻辑。
        当前先作为占位接口，后续可以逐步填充。
        """
        if not self.research_layer:
            return {
                "created_experiments": [],
                "research_cycle_result": {},
            }

        return self.research_layer.run_blueprint_improvement_cycle(
            runtime=self,
            max_gaps=max_gaps,
            research_request=research_request,
        )

    # --------------------------------------------------
    # 免疫 / 睡眠 / 技能进化的统一入口（给定期 scheduler / ResearchLayer 调用）
    # --------------------------------------------------
    def run_immune_check(self) -> ImmuneAction:
        if not self.immune_loop:
            return ImmuneAction(action_type="none", reason="no_immune_loop", metadata={})
        return self.immune_loop.analyze_and_maybe_act()

    def run_sleep_cycle(self) -> Dict[str, Any]:
        if not self.sleep_loop:
            return {"status": "no_sleep_loop"}
        return self.sleep_loop.run_once()

    def run_skill_evolution_cycle(self) -> SkillEvolutionStats:
        if not self.skill_evolution_loop:
            return SkillEvolutionStats(
                patterns_mined=0,
                candidates_generated=0,
                experiments_created=0,
                candidates_accepted=0,
                candidates_rejected=0,
            )
        return self.skill_evolution_loop.run_once()

def create_runtime_for_blueprint_self_iteration(repo_root: str | Path = ".") -> MetaAgentRuntime:
    """
    最小可运行版本的 MetaAgentRuntime，用本地 Ollama 模型 + 极简 Memory/Tool/Critic + 四大 Loop。
    """
    repo_root = Path(repo_root).resolve()

    # 1) 主 LLM（给 runtime 直接用的 model_router）
    main_llm = OllamaLLMClient(
        OllamaModelConfig(
            name="qwen3-vl:latest",
            base_url="http://localhost:11434",
            temperature=0.3,
            top_p=0.9,
            num_ctx=8192,
            num_predict=1024,
        )
    )

    # 2) 多 AI 会议面板（MultiModelCouncil）
    council = create_default_ollama_council(base_url="http://localhost:11434")

    # 3) Memory / Tool / Critic
    memory_fabric = SimpleMemoryFabric(repo_root=repo_root)
    tool_manager = ToolManager()
    critic = SimpleCritic(llm=main_llm)

    # 4) TraceRecorder（统一给所有 loop 用）
    trace_recorder = JsonlTraceRecorder(repo_root=repo_root)

    # 5) 先构造 baseline & council 两个 runtime（暂时不挂 loops）
    baseline_runtime = MetaAgentRuntime(
        model_router=main_llm,
        memory_fabric=memory_fabric,
        tool_manager=tool_manager,
        critic=critic,
        council=None,
        repo_root=repo_root,
        research_layer=None,
        trace_recorder=trace_recorder,
    )

    council_runtime = MetaAgentRuntime(
        model_router=main_llm,
        memory_fabric=memory_fabric,
        tool_manager=tool_manager,
        critic=critic,
        council=council,
        repo_root=repo_root,
        research_layer=None,
        trace_recorder=trace_recorder,
    )

    variants_runtimes = {
        "baseline": baseline_runtime,
        "council": council_runtime,
    }

    # 6) ExperimentOrchestrator + ResearchLayer
    exp_orch = ExperimentOrchestrator(
        repo_root=repo_root,
        variants_runtimes=variants_runtimes,
    )
    research_layer = ResearchLayer(
        repo_root=repo_root,
        experiment_orchestrator=exp_orch,
    )

    baseline_runtime.attach_research_layer(research_layer)
    council_runtime.attach_research_layer(research_layer)

    # 7) 构造四大 Loop，并挂到“主 runtime”（我们以 council_runtime 为主）
    #    CognitiveLoop 需要 runtime 本身，所以先有 runtime，再 new loop，再挂回去。
    immune_loop = ImmuneLoop(
        config=AnomalyDetectorConfig(
            window_size=100,
            error_rate_threshold=0.3,
            max_latency_ms=None,
        ),
        trace_recorder=trace_recorder,
    )

    sleep_loop = SleepLoop(
        config=SleepLoopConfig(
            max_traces_per_run=500,
            promotion_threshold=3,
            run_mode="nightly",
        ),
        trace_recorder=trace_recorder,
        memory_fabric=memory_fabric,
        repo_root=repo_root,
    )

    skill_evolution_loop = SkillEvolutionLoop(
        trace_recorder=trace_recorder,
        experiment_orchestrator=exp_orch,
        llm=main_llm,  # 也可以换成 council 做多模型生成
    )

    cognitive_loop = CognitiveLoop(
        trace_recorder=trace_recorder,
        memory_fabric=memory_fabric,
        critic=critic,
        runtime=council_runtime,   # 注意：这里传入的是主 runtime 自己
    )

    # 把这些 loop 挂在主 runtime 上
    council_runtime.cognitive_loop = cognitive_loop
    council_runtime.immune_loop = immune_loop
    council_runtime.sleep_loop = sleep_loop
    council_runtime.skill_evolution_loop = skill_evolution_loop

    # baseline_runtime 如果需要，也可以共享这些 loop 或者另外挂一套
    baseline_runtime.cognitive_loop = cognitive_loop
    baseline_runtime.immune_loop = immune_loop
    baseline_runtime.sleep_loop = sleep_loop
    baseline_runtime.skill_evolution_loop = skill_evolution_loop

    # 8) 返回作为“主中控”的 runtime（配有 council + loops）
    return council_runtime

from pathlib import Path
import json
from typing import Dict, Any


class ExperimentOrchestrator:
    def __init__(self, repo_root: str | Path, variants_runtimes: Dict[str, Any]) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.variants_runtimes = variants_runtimes
        self._experiments_dir = self.repo_root / "data" / "experiments"
        self._experiments_dir.mkdir(parents=True, exist_ok=True)

    # ... 你之前已经有的一些方法，如 run_experiment, run_cycle 等 ...

    # 新增：
    def register_skill_experiment(self, skill_candidate: Dict[str, Any]) -> str:
        """
        注册一个“技能评估实验”。

        当前版本非常简单：
          - 生成一个 experiment_id
          - 把 candidate 信息写入 data/experiments/skill_candidates.jsonl
        """
        exp_id = f"skill-exp-{hash(json.dumps(skill_candidate, sort_keys=True)) & 0xFFFFFFFF}"

        record = {
            "experiment_id": exp_id,
            "type": "skill_candidate",
            "skill_candidate": skill_candidate,
        }

        path = self._experiments_dir / "skill_candidates.jsonl"
        tmp = path.with_suffix(".tmp")

        # 追加写入 JSONL
        with tmp.open("a", encoding="utf-8") as f:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")
        tmp.replace(path)

        return exp_id
