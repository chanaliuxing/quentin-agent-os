@dataclass
class ActionRequest:
    """
    一次“重要动作”的申请：

    - action_id     : 全局唯一 ID（可由调用方预先生成，也可留空自动生成）
    - actor         : 谁发起的（如 'cognitive_loop', 'selfprog_engine', 'sleep_loop'）
    - action_type   : 动作类型（'llm_call', 'tool_call', 'code_patch', 'experiment_run', ...）
    - domain        : 领域（'core', 'desktop', 'legal', 'trading', 'cross', ...）
    - environment   : 执行环境（'sandbox', 'staging', 'prod'）
    - risk_level    : 风险等级（'low', 'medium', 'high', 'critical'）

    - estimated_cost_tokens : 预估 token 消耗
    - estimated_cost_usd    : 预估费用（可以为 0，先作为抽象字段）
    - description           : 人类可读描述
    - metadata              : 任何额外信息（如 patch diff 摘要、实验 config 摘要等）
    """

    action_id: str
    actor: str
    action_type: str
    domain: str
    environment: str
    risk_level: str

    estimated_cost_tokens: int = 0
    estimated_cost_usd: float = 0.0
    description: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def new(
        cls,
        *,
        actor: str,
        action_type: str,
        domain: str = "core",
        environment: str = "sandbox",
        risk_level: str = "medium",
        estimated_cost_tokens: int = 0,
        estimated_cost_usd: float = 0.0,
        description: str = "",
        metadata: Optional[Dict[str, Any]] = None,
    ) -> "ActionRequest":
        return cls(
            action_id=str(uuid.uuid4()),
            actor=actor,
            action_type=action_type,
            domain=domain,
            environment=environment,
            risk_level=risk_level,
            estimated_cost_tokens=estimated_cost_tokens,
            estimated_cost_usd=estimated_cost_usd,
            description=description,
            metadata=metadata or {},
        )


@dataclass
class PolicyDecision:
    """
    宪法层对一次动作的决定：

    - allow       : 是否允许
    - reason      : 简短理由（给人或上层日志看）
    - severity    : 信息等级（'info', 'warning', 'error', 'critical'）
    - modifications:
        允许但需要做的小改动，例如：
        {
          "downgrade_environment": "sandbox",
          "max_cost_tokens": 20000,
          "require_canary": true
        }
    """

    allow: bool
    reason: str
    severity: str = "info"
    modifications: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


# ---------- Policy Engine ----------

class PolicyEngine:
    """
    PolicyEngine v1：

    - 从 config/constitution.json 读取“宪法规则”
    - 基于 environment / domain / action_type / risk_level 做快速决策
    - 对于边界情况（如 high risk in prod），可以调用 MultiAICouncil 做 AI 审议

    宪法配置示例（config/constitution.json）：

    {
      "version": "v1",
      "global": {
        "max_risk_in_prod": "medium",
        "allow_self_programming": true
      },
      "environments": {
        "sandbox": {
          "max_risk": "critical"
        },
        "staging": {
          "max_risk": "high"
        },
        "prod": {
          "max_risk": "medium"
        }
      },
      "domains": {
        "trading": {
          "allow_self_programming": false
        },
        "legal": {
          "allow_self_programming": false
        },
        "desktop": {
          "allow_self_programming": true
        }
      },
      "action_types": {
        "code_patch": {
          "require_review": true
        },
        "experiment_run": {
          "max_parallel": 3
        }
      }
    }
    """

    _RISK_LEVEL_ORDER = {
        "low": 0,
        "medium": 1,
        "high": 2,
        "critical": 3,
    }

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        constitution_path: str | Path = "config/constitution.json",
        decision_log_path: str | Path = "data/policy_decisions.jsonl",
        council: Optional[MultiAICouncil] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.constitution_path = self.repo_root / constitution_path
        self.decision_log_path = self.repo_root / decision_log_path
        self.decision_log_path.parent.mkdir(parents=True, exist_ok=True)

        self._constitution: Dict[str, Any] = {}
        self._load_constitution()

        if council is None:
            members = [
                LLMConfig(
                    provider="ollama",
                    model="qwen3-vl:latest",
                    role="constitutional_lawyer",
                ),
                LLMConfig(
                    provider="ollama",
                    model="deepseek-r1:1.5b",
                    role="risk_auditor",
                ),
            ]
            council = MultiAICouncil(members=members)
        self.council = council

    # ---------- 公共接口 ----------

    def evaluate(self, action: ActionRequest) -> PolicyDecision:
        """
        主入口：同步返回决策结果。
        """
        base_decision = self._eval_static_rules(action)

        if not base_decision.allow:
            # 明确违反宪法，直接拒绝
            self._log_decision(action, base_decision)
            return base_decision

        # borderline case：risk 高、在 prod / staging、或者 estimated_cost 特别大
        if self._is_borderline(action):
            ai_decision = self._consult_council(action, base_decision)
            self._log_decision(action, ai_decision)
            return ai_decision

        # 正常通过
        self._log_decision(action, base_decision)
        return base_decision

    # ---------- 内部：静态规则 ----------

    def _eval_static_rules(self, action: ActionRequest) -> PolicyDecision:
        env_cfg = (self._constitution.get("environments") or {}).get(action.environment, {})
        dom_cfg = (self._constitution.get("domains") or {}).get(action.domain, {})
        act_cfg = (self._constitution.get("action_types") or {}).get(action.action_type, {})
        global_cfg = self._constitution.get("global") or {}

        # 1. 环境下的最大 risk
        env_max_risk = env_cfg.get("max_risk") or global_cfg.get("max_risk_in_prod", "medium")
        if not self._risk_allowed(action.risk_level, env_max_risk):
            return PolicyDecision(
                allow=False,
                severity="error",
                reason=(
                    f"Risk level '{action.risk_level}' exceeds environment '{action.environment}' "
                    f"max_risk '{env_max_risk}'."
                ),
            )

        # 2. 自编程限制（按 domain + global）
        if action.action_type == "code_patch":
            allow_self_prog = dom_cfg.get("allow_self_programming")
            if allow_self_prog is None:
                allow_self_prog = global_cfg.get("allow_self_programming", False)
            if not allow_self_prog:
                return PolicyDecision(
                    allow=False,
                    severity="error",
                    reason=f"Self-programming is disabled for domain '{action.domain}'.",
                )

        # 3. 简单 cost 约束（示例：单次动作 token 超出阈值则需要 degrade）
        max_tokens_per_action = int(global_cfg.get("max_tokens_per_action", 200_000))
        if action.estimated_cost_tokens > max_tokens_per_action:
            return PolicyDecision(
                allow=True,
                severity="warning",
                reason=(
                    f"Estimated tokens {action.estimated_cost_tokens} exceed "
                    f"global.soft_limit {max_tokens_per_action}. Allow with downgrade."
                ),
                modifications={
                    "max_cost_tokens": max_tokens_per_action,
                    "require_canary": True,
                },
            )

        # 4. action_types 级别的一些标志（示例：code_patch 必须 require_review）
        if act_cfg.get("require_review") and action.environment == "prod":
            return PolicyDecision(
                allow=True,
                severity="warning",
                reason="Production code_patch requires review. Marking require_review flag.",
                modifications={"require_review": True},
            )

        # 默认允许
        return PolicyDecision(
            allow=True,
            severity="info",
            reason="Allowed by static policy.",
            modifications={},
        )

    # ---------- 内部：AI 审议 ----------

    def _is_borderline(self, action: ActionRequest) -> bool:
        # 简单规则：高风险 + staging/prod；或者 tokens 特别大
        risk_order = self._RISK_LEVEL_ORDER.get(action.risk_level, 1)
        is_high_risk = risk_order >= self._RISK_LEVEL_ORDER["high"]
        is_prod_like = action.environment in ("prod", "staging")
        high_cost = action.estimated_cost_tokens > 150_000

        return (is_high_risk and is_prod_like) or high_cost

    def _consult_council(
        self,
        action: ActionRequest,
        base_decision: PolicyDecision,
    ) -> PolicyDecision:
        """
        用多 AI 会议审议边界情况。

        约定：
        - Council 返回 JSON：
          {
            "allow": true/false,
            "severity": "info|warning|error|critical",
            "reason": "...",
            "modifications": {...}
          }
        """
        context = json.dumps(action.to_dict(), ensure_ascii=False, indent=2)
        question = (
            "You are a CONSTITUTIONAL COUNCIL for a self-evolving AI system.\n"
            "Given the following action request and a baseline static policy decision,\n"
            "you must decide whether to allow it, and if allowed, whether to impose\n"
            "additional constraints.\n\n"
            "Return ONLY one JSON object:\n"
            "{\n"
            "  \"allow\": true/false,\n"
            "  \"severity\": \"info|warning|error|critical\",\n"
            "  \"reason\": \"...\",\n"
            "  \"modifications\": {\"key\": \"value\", ...}\n"
            "}\n"
        )

        merged_context = (
            "=== ACTION REQUEST ===\n"
            f"{context}\n\n"
            "=== STATIC POLICY DECISION ===\n"
            f"{json.dumps(base_decision.to_dict(), ensure_ascii=False, indent=2)}\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=merged_context,
            mode="policy_decision",
        )
        ans = decision.final_answer or "{}"

        try:
            s = ans.strip()
            first = s.find("{")
            last = s.rfind("}")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)
            allow = bool(data.get("allow", base_decision.allow))
            severity = str(data.get("severity", base_decision.severity))
            reason = str(data.get("reason", base_decision.reason))
            mods = data.get("modifications") or {}
            if not isinstance(mods, dict):
                mods = {}
            return PolicyDecision(
                allow=allow,
                severity=severity,
                reason=reason,
                modifications=mods,
            )
        except Exception:
            logger.exception("PolicyEngine._consult_council: parse error: %r", ans)
            # 回退到 static 决策
            return base_decision

    # ---------- 内部工具 ----------

    def _risk_allowed(self, req: str, max_level: str) -> bool:
        return self._RISK_LEVEL_ORDER.get(req, 0) <= self._RISK_LEVEL_ORDER.get(max_level, 1)

    def _load_constitution(self) -> None:
        if not self.constitution_path.exists():
            # 给一个保守默认值
            self._constitution = {
                "version": "v1-default",
                "global": {
                    "max_risk_in_prod": "medium",
                    "allow_self_programming": True,
                    "max_tokens_per_action": 200_000,
                },
                "environments": {},
                "domains": {},
                "action_types": {},
            }
            return
        try:
            raw = json.loads(self.constitution_path.read_text(encoding="utf-8") or "{}")
            self._constitution = raw or {}
        except Exception:
            logger.exception(
                "PolicyEngine._load_constitution: failed to read %s",
                self.constitution_path,
            )
            self._constitution = {}

    def _log_decision(self, action: ActionRequest, decision: PolicyDecision) -> None:
        record = {
            "action": action.to_dict(),
            "decision": decision.to_dict(),
            "logged_at": datetime.utcnow().isoformat(),
        }
        try:
            with self.decision_log_path.open("a", encoding="utf-8") as f:
                f.write(json.dumps(record, ensure_ascii=False) + "\n")
        except Exception:
            logger.exception(
                "PolicyEngine._log_decision: failed to append to %s",
                self.decision_log_path,
            )

from __future__ import annotations

from dataclasses import dataclass, asdict, field
from pathlib import Path
from typing import Any, Dict, Optional
import json
import logging
from datetime import datetime, date

logger = logging.getLogger(__name__)


@dataclass
class BudgetProfile:
    """
    某个预算 profile（比如 'default', 'research', 'prod_trading'）的配额：

    - daily_token_limit / monthly_token_limit
    - daily_usd_limit   / monthly_usd_limit
    - hard_stop: 超出是否直接拒绝（True）还是只告警（False）
    """

    profile_id: str
    daily_token_limit: int
    monthly_token_limit: int

    daily_usd_limit: float
    monthly_usd_limit: float

    hard_stop: bool = True
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class BudgetUsage:
    """
    某 profile 当前的使用状态：
    - day / month 维度的 tokens / usd
    """

    profile_id: str
    day: str           # YYYY-MM-DD
    month: str         # YYYY-MM
    tokens_today: int
    tokens_month: int
    usd_today: float
    usd_month: float

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class BudgetDecision:
    allow: bool
    reason: str
    remaining_tokens_today: int
    remaining_tokens_month: int
    remaining_usd_today: float
    remaining_usd_month: float

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class BudgetController:
    """
    BudgetController v1：

    - 使用 config/budgets.json 定义多个 BudgetProfile
    - 使用 data/budget_state.json 维护当前使用量
    - 提供 check_and_reserve(profile_id, tokens, usd) 进行限额检查 + 预定

    budgets.json 示例：

    {
      "profiles": {
        "default": {
          "daily_token_limit": 200000,
          "monthly_token_limit": 5000000,
          "daily_usd_limit": 5.0,
          "monthly_usd_limit": 100.0,
          "hard_stop": true
        },
        "research": {
          "daily_token_limit": 1000000,
          "monthly_token_limit": 2e7,
          "daily_usd_limit": 20.0,
          "monthly_usd_limit": 400.0,
          "hard_stop": false
        }
      },
      "routing": {
        "environment:prod": "default",
        "environment:sandbox": "research",
        "domain:trading": "default"
      }
    }
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        budgets_config_path: str | Path = "config/budgets.json",
        state_path: str | Path = "data/budget_state.json",
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.budgets_config_path = self.repo_root / budgets_config_path
        self.state_path = self.repo_root / state_path
        self.state_path.parent.mkdir(parents=True, exist_ok=True)

        self._profiles: Dict[str, BudgetProfile] = {}
        self._routing: Dict[str, str] = {}
        self._usage: Dict[str, BudgetUsage] = {}

        self._load_config()
        self._load_state()

    # ---------- 配额检查入口 ----------

    def resolve_profile(self, *, environment: str, domain: str) -> str:
        """
        根据 environment / domain 选择 profile_id。
        优先级：
          1) domain:xxx
          2) environment:xxx
          3) default
        """
        key_dom = f"domain:{domain}"
        key_env = f"environment:{environment}"
        if key_dom in self._routing:
            return self._routing[key_dom]
        if key_env in self._routing:
            return self._routing[key_env]
        return "default"

    def check_and_reserve(
        self,
        *,
        profile_id: str,
        tokens: int,
        usd: float,
    ) -> BudgetDecision:
        today = date.today().strftime("%Y-%m-%d")
        month = date.today().strftime("%Y-%m")

        profile = self._profiles.get(profile_id)
        if not profile:
            # 如果没配置 profile，则视作无限额，但会 Warn
            logger.warning("BudgetController: profile '%s' not found, treating as unlimited.", profile_id)
            profile = BudgetProfile(
                profile_id=profile_id,
                daily_token_limit=10**12,
                monthly_token_limit=10**13,
                daily_usd_limit=10**6,
                monthly_usd_limit=10**7,
                hard_stop=False,
            )

        usage = self._usage.get(profile_id)
        if not usage or usage.day != today or usage.month != month:
            usage = BudgetUsage(
                profile_id=profile_id,
                day=today,
                month=month,
                tokens_today=0,
                tokens_month=0,
                usd_today=0.0,
                usd_month=0.0,
            )

        new_tokens_today = usage.tokens_today + tokens
        new_tokens_month = usage.tokens_month + tokens
        new_usd_today = usage.usd_today + usd
        new_usd_month = usage.usd_month + usd

        over_daily_tokens = new_tokens_today > profile.daily_token_limit
        over_month_tokens = new_tokens_month > profile.monthly_token_limit
        over_daily_usd = new_usd_today > profile.daily_usd_limit
        over_month_usd = new_usd_month > profile.monthly_usd_limit

        any_over = over_daily_tokens or over_month_tokens or over_daily_usd or over_month_usd

        if any_over and profile.hard_stop:
            # 不更新 usage，直接拒绝
            reason = "Budget exceeded: "
            parts = []
            if over_daily_tokens:
                parts.append("daily tokens")
            if over_month_tokens:
                parts.append("monthly tokens")
            if over_daily_usd:
                parts.append("daily USD")
            if over_month_usd:
                parts.append("monthly USD")
            reason += ", ".join(parts)
            return BudgetDecision(
                allow=False,
                reason=reason,
                remaining_tokens_today=max(0, profile.daily_token_limit - usage.tokens_today),
                remaining_tokens_month=max(0, profile.monthly_token_limit - usage.tokens_month),
                remaining_usd_today=max(0.0, profile.daily_usd_limit - usage.usd_today),
                remaining_usd_month=max(0.0, profile.monthly_usd_limit - usage.usd_month),
            )

        # 允许，更新 usage
        usage.tokens_today = new_tokens_today
        usage.tokens_month = new_tokens_month
        usage.usd_today = new_usd_today
        usage.usd_month = new_usd_month
        self._usage[profile_id] = usage
        self._flush_state()

        reason = "Within budget."
        if any_over and not profile.hard_stop:
            reason = "Budget exceeded soft limit, but hard_stop=false."

        return BudgetDecision(
            allow=True,
            reason=reason,
            remaining_tokens_today=max(0, profile.daily_token_limit - usage.tokens_today),
            remaining_tokens_month=max(0, profile.monthly_token_limit - usage.tokens_month),
            remaining_usd_today=max(0.0, profile.daily_usd_limit - usage.usd_today),
            remaining_usd_month=max(0.0, profile.monthly_usd_limit - usage.usd_month),
        )

    # ---------- 配置 & 状态 ----------

    def _load_config(self) -> None:
        if not self.budgets_config_path.exists():
            # 默认一个“无限但 soft 限制”的 default profile
            self._profiles = {
                "default": BudgetProfile(
                    profile_id="default",
                    daily_token_limit=1_000_000,
                    monthly_token_limit=20_000_000,
                    daily_usd_limit=10.0,
                    monthly_usd_limit=200.0,
                    hard_stop=False,
                )
            }
            self._routing = {}
            return
        try:
            raw = json.loads(self.budgets_config_path.read_text(encoding="utf-8") or "{}")
        except Exception:
            logger.exception("BudgetController._load_config: failed to read %s", self.budgets_config_path)
            raw = {}

        profiles_raw = raw.get("profiles") or {}
        routing_raw = raw.get("routing") or {}

        self._profiles = {}
        for pid, pdata in profiles_raw.items():
            self._profiles[pid] = BudgetProfile(
                profile_id=pid,
                daily_token_limit=int(pdata.get("daily_token_limit", 0)),
                monthly_token_limit=int(pdata.get("monthly_token_limit", 0)),
                daily_usd_limit=float(pdata.get("daily_usd_limit", 0.0)),
                monthly_usd_limit=float(pdata.get("monthly_usd_limit", 0.0)),
                hard_stop=bool(pdata.get("hard_stop", True)),
                metadata=pdata.get("metadata") or {},
            )
        self._routing = {str(k): str(v) for k, v in routing_raw.items()}

    def _load_state(self) -> None:
        if not self.state_path.exists():
            self._usage = {}
            return
        try:
            raw = json.loads(self.state_path.read_text(encoding="utf-8") or "{}")
            self._usage = {
                pid: BudgetUsage(
                    profile_id=pid,
                    day=v["day"],
                    month=v["month"],
                    tokens_today=int(v["tokens_today"]),
                    tokens_month=int(v["tokens_month"]),
                    usd_today=float(v["usd_today"]),
                    usd_month=float(v["usd_month"]),
                )
                for pid, v in raw.items()
            }
        except Exception:
            logger.exception("BudgetController._load_state: failed to read %s", self.state_path)
            self._usage = {}

    def _flush_state(self) -> None:
        try:
            raw = {pid: u.to_dict() for pid, u in self._usage.items()}
            self.state_path.write_text(json.dumps(raw, ensure_ascii=False, indent=2), encoding="utf-8")
        except Exception:
            logger.exception("BudgetController._flush_state: failed to write %s", self.state_path)

from __future__ import annotations

from dataclasses import dataclass, asdict

from quentin.kernel.policy import PolicyEngine, ActionRequest, PolicyDecision

@dataclass
class GuardResult:
    allow: bool
    reason: str
    policy_decision: PolicyDecision
    budget_decision: BudgetDecision

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class SafetyKernel:
    """
    SafetyKernel v1：

    - 统一守门口：任何“重要动作”先调用 guard_action
    - 内部组合：
        - PolicyEngine：检查宪法/风险边界
        - BudgetController：检查配额/算力预算

    用法示例（在 SelfProgrammingEngine 里）：

        action = ActionRequest.new(
            actor="selfprog_engine",
            action_type="code_patch",
            domain="core",
            environment="sandbox",
            risk_level="high",
            estimated_cost_tokens=estimated_tokens,
            description="Generate patch for mutable/selfprog/tools.py",
            metadata={"target_path": target_path}
        )
        guard = safety_kernel.guard_action(action)
        if not guard.allow:
            # 直接拒绝，自编程不能执行
            ...

    """

    def __init__(
        self,
        *,
        policy_engine: Optional[PolicyEngine] = None,
        budget_controller: Optional[BudgetController] = None,
    ) -> None:
        self.policy_engine = policy_engine or PolicyEngine()
        self.budget_controller = budget_controller or BudgetController()

    def guard_action(
        self,
        action: ActionRequest,
    ) -> GuardResult:
        # 1) Policy decision
        p_dec = self.policy_engine.evaluate(action)
        if not p_dec.allow:
            # 预算检查可以省略，因为已经拒绝
            dummy_budget = BudgetDecision(
                allow=False,
                reason="Skipped: policy denied.",
                remaining_tokens_today=0,
                remaining_tokens_month=0,
                remaining_usd_today=0.0,
                remaining_usd_month=0.0,
            )
            return GuardResult(
                allow=False,
                reason=p_dec.reason,
                policy_decision=p_dec,
                budget_decision=dummy_budget,
            )

        # 2) 预算 profile 选择
        profile_id = self.budget_controller.resolve_profile(
            environment=action.environment,
            domain=action.domain,
        )

        # 从 policy modifications 里读取强制上限（如果有的话）
        est_tokens = action.estimated_cost_tokens
        est_usd = action.estimated_cost_usd

        max_tokens = p_dec.modifications.get("max_cost_tokens")
        if isinstance(max_tokens, int) and max_tokens > 0:
            est_tokens = min(est_tokens, max_tokens)

        b_dec = self.budget_controller.check_and_reserve(
            profile_id=profile_id,
            tokens=est_tokens,
            usd=est_usd,
        )

        allow = p_dec.allow and b_dec.allow
        reason = p_dec.reason if not b_dec.allow else f"{p_dec.reason} | {b_dec.reason}"

        return GuardResult(
            allow=allow,
            reason=reason,
            policy_decision=p_dec,
            budget_decision=b_dec,
        )

from typing import List

def make_decision_council(
    *,
    router: ModelRouter,
    domain: str,
    mode: str,
) -> MultiAICouncil:
    """
    构造一个带 4 个角色的 MultiAICouncil：

    - Speed-Agent         : 偏向速度（router 决定）
    - Safety-Agent        : 偏向安全性/保守
    - Cost-Agent          : 偏向省 token（可以与 Speed 复用）
    - Domain-Expert-Agent : 偏领域知识（如 legal/trading/domain-specific）

    v1 简化实现：都通过 ModelRouter 拿模型，只是 role 不同，
    以后可以在 router.rules 中对不同 role 做更细粒度配置。
    """
    domain = domain.lower()
    mode = mode.lower()

    # Speed-Agent
    cfg_speed = router.route(mode=mode, task_type=domain)
    speed = LLMConfig(
        provider=cfg_speed.provider,
        model=cfg_speed.model,
        role=f"speed_agent::{mode}::{domain}",
    )

    # Safety-Agent（目前复用 qwen，等以后可以切换安全向模型）
    cfg_safety = router.route(mode="critic", task_type=domain)
    safety = LLMConfig(
        provider=cfg_safety.provider,
        model=cfg_safety.model,
        role=f"safety_agent::{mode}::{domain}",
    )

    # Cost-Agent（暂时和 speed 一样，未来可以切 ultra-small 模型）
    cfg_cost = router.route(mode=mode, task_type=domain)
    cost = LLMConfig(
        provider=cfg_cost.provider,
        model=cfg_cost.model,
        role=f"cost_agent::{mode}::{domain}",
    )

    # Domain-Expert-Agent（同样通过 router，让 router 可以以后绑定专用模型）
    cfg_domain = router.route(mode=mode, task_type=domain)
    expert = LLMConfig(
        provider=cfg_domain.provider,
        model=cfg_domain.model,
        role=f"domain_expert_agent::{mode}::{domain}",
    )

    members: List[LLMConfig] = [speed, safety, cost, expert]
    return MultiAICouncil(members=members)
后续你可以慢慢把之前的 MultiAICouncil(members=[...]) 替换成 make_decision_council(router, domain, mode)，统一多角色 Council 的结构。
