from __future__ import annotations

from dataclasses import dataclass, asdict, field
from datetime import datetime
from typing import Dict, Any, Optional


@dataclass
class TaskSpec:
    """
    认知循环要处理的“任务定义”。

    这是一个通用 schema，不绑定任何具体领域：
    - task_id      : 唯一任务 ID
    - description  : 自然语言描述任务做什么
    - domain       : "generic" / "desktop" / "legal" / "trading" / ...
    - input        : 任意结构化输入（比如 JSON）
    - meta         : 其他辅助信息（优先级、触发来源等）
    """
    task_id: str
    description: str
    domain: str = "generic"

    input: Dict[str, Any] = field(default_factory=dict)
    meta: Dict[str, Any] = field(default_factory=dict)

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class TaskResult:
    """
    任务执行结果的顶层包装：
    - success            : 整体是否成功
    - execution_summary  : 结构化总结（由 CognitiveLoop 生成）
    - lessons_learned    : 反思/经验总结（由 Critic 生成）
    """
    task_id: str
    success: bool
    execution_summary: Dict[str, Any]
    lessons_learned: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

from __future__ import annotations

import json
import logging
from dataclasses import dataclass, asdict, field
from datetime import datetime
from pathlib import Path
from typing import Dict, Any, List, Optional

logger = logging.getLogger(__name__)


@dataclass
class StepTrace:
    """
    认知循环中的单步 Trace 记录。

    - step_id    : 唯一 ID（可用 "task_id::step_index"）
    - task_id    : 所属任务
    - step_index : 第几步（0,1,2,...）
    - phase      : "plan" / "execute" / "critic" / "other"
    - role       : 角色名称，方便以后区分 Planner/Executor/Critic/Agent 名字
    - action     : 本步做了什么的简要描述
    - input      : 这一步的输入（任意 JSON）
    - output     : 这一步的输出（任意 JSON）
    - created_at : 时间戳
    """
    step_id: str
    task_id: str
    step_index: int
    phase: str
    role: str
    action: str

    input: Dict[str, Any] = field(default_factory=dict)
    output: Dict[str, Any] = field(default_factory=dict)
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ExecutionSummary:
    """
    任务完成后的结构化总结（供 Memory / Research 使用）。

    - summary_text : 自然语言总结
    - success      : 是否成功
    - metrics      : 任意指标（耗时、步数等）
    - errors       : 错误列表
    - artefacts    : 衍生 artefacts（如生成的文件路径、代码片段等）
    """
    summary_text: str
    success: bool
    metrics: Dict[str, Any] = field(default_factory=dict)
    errors: List[str] = field(default_factory=list)
    artefacts: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class LessonsLearned:
    """
    Critic 生成的经验/反思：
    - text      : 自然语言
    - tags      : 标记类型，例如["retry_strategy", "error_pattern.io_timeout"]
    - meta      : 任意扩展信息
    """
    text: str
    tags: List[str] = field(default_factory=list)
    meta: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

class FileBackedTraceStore:
    """
    Trace 存储：一个 JSON 文件，结构是：
    {
      "task_id1": {
        "task_id": "...",
        "steps": [StepTraceDict, ...],
        "execution_summary": ExecutionSummaryDict or null,
        "lessons_learned": LessonsLearnedDict or null
      },
      ...
    }
    """

    def __init__(self, path: str | Path = "data/traces.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._cache: Dict[str, Dict[str, Any]] = {}
        self._loaded = False

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                with self.path.open("r", encoding="utf-8") as f:
                    self._cache = json.load(f)
            except Exception:
                logger.exception("Failed to load traces from %s", self.path)
                self._cache = {}
        else:
            self._cache = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self.path.open("w", encoding="utf-8") as f:
            json.dump(self._cache, f, ensure_ascii=False, indent=2)

    # ---------- 操作接口 ----------

    def append_step(self, step: StepTrace) -> None:
        self._load()
        task = self._cache.get(step.task_id)
        if task is None:
            task = {
                "task_id": step.task_id,
                "steps": [],
                "execution_summary": None,
                "lessons_learned": None,
            }
        task["steps"].append(step.to_dict())
        self._cache[step.task_id] = task
        self._save()

    def set_summary_and_lessons(
        self,
        task_id: str,
        summary: ExecutionSummary,
        lessons: LessonsLearned,
    ) -> None:
        self._load()
        task = self._cache.get(task_id)
        if task is None:
            task = {
                "task_id": task_id,
                "steps": [],
                "execution_summary": None,
                "lessons_learned": None,
            }
        task["execution_summary"] = summary.to_dict()
        task["lessons_learned"] = lessons.to_dict()
        self._cache[task_id] = task
        self._save()

    def get_task_trace(self, task_id: str) -> Optional[Dict[str, Any]]:
        self._load()
        return self._cache.get(task_id)

    def list_task_ids(self) -> List[str]:
        self._load()
        return list(self._cache.keys())

import logging
from dataclasses import asdict
from datetime import datetime
from typing import Dict, Any, Optional

from quentin.core.task import TaskSpec, TaskResult
from quentin.core.traces import (
    StepTrace,
    ExecutionSummary,
    LessonsLearned,
    FileBackedTraceStore,
)
from quentin.memory.fabric import MemoryFabric
from quentin.memory.schemas import MemoryUpdatePolicy

logger = logging.getLogger(__name__)


class BasePlanner:
    def plan(self, task: TaskSpec) -> Dict[str, Any]:
        """
        返回一个结构化计划，如：
        {
          "steps": [
            {"id": "step1", "description": "..."},
            ...
          ]
        }
        v1 只要返回一个简单 dict 即可。
        """
        raise NotImplementedError


class BaseExecutor:
    def execute(self, task: TaskSpec, plan: Dict[str, Any]) -> Dict[str, Any]:
        """
        执行计划，返回执行结果（任意结构化数据）。
        """
        raise NotImplementedError


class BaseCritic:
    def evaluate(
        self,
        task: TaskSpec,
        plan: Dict[str, Any],
        exec_output: Dict[str, Any],
    ) -> tuple[ExecutionSummary, LessonsLearned]:
        """
        生成 ExecutionSummary + LessonsLearned。
        """
        raise NotImplementedError
2）简单实现（占位，可被后续自改）
class EchoPlanner(BasePlanner):
    """
    v1 非 LLM 版本：直接把 description 包装成一个单步计划。
    之后可以替换为 LLM 版 Planner。
    """

    def plan(self, task: TaskSpec) -> Dict[str, Any]:
        return {
            "steps": [
                {
                    "id": "step-1",
                    "description": task.description,
                }
            ],
            "meta": {
                "planner": "EchoPlanner",
                "created_at": datetime.utcnow().isoformat(),
            },
        }


class NoopExecutor(BaseExecutor):
    """
    v1 执行器：不做实际动作，只回显输入。
    后续可以按领域接入真正的 Desktop/Legal/Trading 执行。
    """

    def execute(self, task: TaskSpec, plan: Dict[str, Any]) -> Dict[str, Any]:
        return {
            "status": "noop",
            "message": "NoopExecutor executed plan without side effects.",
            "echo": {
                "task": task.to_dict(),
                "plan": plan,
            },
        }


class RuleBasedCritic(BaseCritic):
    """
    v1 Critic：非常简单的规则：
    - 如果执行结果里 status == "error" 则标记失败
    - 否则标记成功
    - lessons_learned 只给出一个模板反思
    """

    def evaluate(
        self,
        task: TaskSpec,
        plan: Dict[str, Any],
        exec_output: Dict[str, Any],
    ) -> tuple[ExecutionSummary, LessonsLearned]:
        status = exec_output.get("status", "ok")
        success = status != "error"

        summary_text = (
            f"Task '{task.description}' executed by NoopExecutor. "
            f"Status: {status}."
        )
        errors = []
        if not success:
            errors.append("Execution status indicates error.")

        summary = ExecutionSummary(
            summary_text=summary_text,
            success=success,
            metrics={
                "status": status,
                "planner": plan.get("meta", {}).get("planner", "unknown"),
            },
            errors=errors,
            artefacts={
                "plan": plan,
                "exec_output": exec_output,
            },
        )

        lessons_text = (
            "This is a placeholder lessons_learned. "
            "Future versions of Critic should analyze plan/exec_output "
            "and summarize concrete improvement points."
        )
        lessons = LessonsLearned(
            text=lessons_text,
            tags=["placeholder"],
            meta={
                "task_domain": task.domain,
            },
        )
        return summary, lessons
3）CognitiveLoop 主体
class CognitiveLoop:
    """
    蓝皮书里的认知闭环 v1 实现：

    TaskSpec ->
      1) Planner.plan()        (写 Trace)
      2) Executor.execute()    (写 Trace)
      3) Critic.evaluate()     (写 Trace)
      4) 写 ExecutionSummary + LessonsLearned 到 TraceStore
      5) 写 Episodic / Semantic Memory 到 MemoryFabric
    """

    def __init__(
        self,
        planner: Optional[BasePlanner] = None,
        executor: Optional[BaseExecutor] = None,
        critic: Optional[BaseCritic] = None,
        trace_store: Optional[FileBackedTraceStore] = None,
        memory_fabric: Optional[MemoryFabric] = None,
    ) -> None:
        self.planner = planner or EchoPlanner()
        self.executor = executor or NoopExecutor()
        self.critic = critic or RuleBasedCritic()
        self.traces = trace_store or FileBackedTraceStore()
        self.memory = memory_fabric or MemoryFabric()

    # ---------- 入口：运行一个任务 ----------

    def run_task(self, task: TaskSpec) -> TaskResult:
        step_index = 0

        # 1) 规划
        plan = self.planner.plan(task)
        step_plan = StepTrace(
            step_id=f"{task.task_id}::step-{step_index}",
            task_id=task.task_id,
            step_index=step_index,
            phase="plan",
            role=self.planner.__class__.__name__,
            action="plan_task",
            input={"task": task.to_dict()},
            output={"plan": plan},
        )
        self.traces.append_step(step_plan)
        step_index += 1

        # 2) 执行
        exec_output = self.executor.execute(task, plan)
        step_exec = StepTrace(
            step_id=f"{task.task_id}::step-{step_index}",
            task_id=task.task_id,
            step_index=step_index,
            phase="execute",
            role=self.executor.__class__.__name__,
            action="execute_plan",
            input={"task": task.to_dict(), "plan": plan},
            output={"exec_output": exec_output},
        )
        self.traces.append_step(step_exec)
        step_index += 1

        # 3) 评估 / 反思
        summary, lessons = self.critic.evaluate(task, plan, exec_output)
        step_critic = StepTrace(
            step_id=f"{task.task_id}::step-{step_index}",
            task_id=task.task_id,
            step_index=step_index,
            phase="critic",
            role=self.critic.__class__.__name__,
            action="evaluate_execution",
            input={
                "task": task.to_dict(),
                "plan": plan,
                "exec_output": exec_output,
            },
            output={
                "execution_summary": summary.to_dict(),
                "lessons_learned": lessons.to_dict(),
            },
        )
        self.traces.append_step(step_critic)

        # 4) 保存 summary & lessons 到 TraceStore
        self.traces.set_summary_and_lessons(
            task_id=task.task_id,
            summary=summary,
            lessons=lessons,
        )

        # 5) 写 Memory Fabric
        self._write_memories(task, summary, lessons)

        result = TaskResult(
            task_id=task.task_id,
            success=summary.success,
            execution_summary=summary.to_dict(),
            lessons_learned=lessons.to_dict(),
        )
        return result

    # ---------- 写记忆 ----------

    def _write_memories(
        self,
        task: TaskSpec,
        summary: ExecutionSummary,
        lessons: LessonsLearned,
    ) -> None:
        """
        v1 策略：
        - 每个任务写一条 episodic memory 到 STS（任务上下文 + summary）
        - 再写一条 semantic memory 到 STS（lessons_learned）
        SleepLoop 会决定以后是否升级到 MTS/LTS 或被遗忘。
        """
        # Episodic memory
        epi_text = (
            f"[Task] {task.description}\n"
            f"[Summary] {summary.summary_text}"
        )
        epi_extra = {
            "task": task.to_dict(),
            "execution_summary": summary.to_dict(),
        }
        epi_policy = MemoryUpdatePolicy(
            promotion_rule="promote_to_MTS_if_used_often",
            decay_rule="drop_if_never_referenced_after_7_days",
            consolidation_batch="nightly",
        )
        self.memory.add_memory(
            tier="STS",
            kind="episodic",
            text=epi_text,
            importance=0.6 if summary.success else 0.4,
            source="task_execution",
            tags=[f"domain:{task.domain}"],
            update_policy=epi_policy,
            extra_content=epi_extra,
        )

        # Semantic memory (lessons)
        sem_text = f"[Lessons] {lessons.text}"
        sem_extra = {
            "task": task.to_dict(),
            "lessons_learned": lessons.to_dict(),
        }
        sem_policy = MemoryUpdatePolicy(
            promotion_rule="promote_to_MTS_if_referenced_in_planning",
            decay_rule="keep_longer_for_high_importance",
            consolidation_batch="nightly",
        )
        self.memory.add_memory(
            tier="STS",
            kind="semantic",
            text=sem_text,
            importance=0.7,
            source="task_lessons",
            tags=["lessons"] + lessons.tags,
            update_policy=sem_policy,
            extra_content=sem_extra,
        )

import uuid

def cmd_task_run_demo(args: argparse.Namespace) -> None:
    task_id = str(uuid.uuid4())
    task = TaskSpec(
        task_id=task_id,
        description=args.description,
        domain=args.domain,
    )

    loop = CognitiveLoop()
    result = loop.run_task(task)

    print(f"Task executed. task_id={task_id}")
    print(f"Success: {result.success}")
    print("Execution summary:")
    print(result.execution_summary.get("summary_text", ""))
    print("\nLessons learned:")
    print(result.lessons_learned.get("text", ""))


def cmd_task_show_trace(args: argparse.Namespace) -> None:
    store = FileBackedTraceStore()
    data = store.get_task_trace(args.task_id)
    if not data:
        print(f"No trace found for task_id={args.task_id}")
        return

    print(f"Trace for task_id={args.task_id}")
    steps = data.get("steps", [])
    for s in steps:
        print(
            f"[{s['step_index']}] phase={s['phase']} role={s['role']} action={s['action']} "
            f"created_at={s['created_at']}"
        )
    summary = data.get("execution_summary")
    if summary:
        print("\nExecution summary:")
        print(summary.get("summary_text", ""))
    lessons = data.get("lessons_learned")
    if lessons:
        print("\nLessons learned:")
        print(lessons.get("text", ""))

和蓝皮书对齐的小结
你现在已经有了一个完整的 认知循环 v1，对蓝皮书的对应关系：

3.1 认知循环（Cognitive Loop）

所有步骤必须写 Trace → StepTrace + FileBackedTraceStore

每次任务结束都生成 execution_summary → ExecutionSummary

lessons_learned 由 Critic 生成 → LessonsLearned + RuleBasedCritic

有价值结果写入 Memory Fabric → _write_memories 写入 STS

2.1 Memory Fabric：三层记忆

认知循环写入 STS，SleepLoop 再决定升/删 → 已打通
