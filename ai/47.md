一、定义自我改进任务结构：ImprovementTask
文件：quentin/self_improvement/schemas.py

# quentin/self_improvement/schemas.py
from __future__ import annotations

from dataclasses import dataclass, field, asdict
from datetime import datetime
from typing import Dict, Any, List, Optional


TASK_STATUS_PENDING = "pending"
TASK_STATUS_IN_PROGRESS = "in_progress"
TASK_STATUS_DONE = "done"
TASK_STATUS_ABANDONED = "abandoned"

# 优先级越小越先做（1 > 2 > 3）
PRIORITY_HIGH = 1      # overall_status = missing
PRIORITY_MEDIUM = 2    # overall_status = placeholder / partial
PRIORITY_LOW = 3       # overall_status = implemented，但仍有优化空间（后面再用）


@dataclass
class ImprovementTask:
    """
    自我改进任务：
    由 BlueprintGapTracker 的结果生成，
    用来驱动 SelfProgrammingEngine / MultiLLMCouncil 等模块做演化。
    """
    task_id: str
    capability: str                # e.g. "memory_fabric"
    requirement_id: Optional[str]  # e.g. "mf.search_semantic"（有些可能是 None）
    overall_status: str            # 能力当前状态：missing / placeholder / partial / implemented
    status: str = TASK_STATUS_PENDING
    priority: int = PRIORITY_MEDIUM
    title: str = ""
    description: str = ""
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    meta: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "ImprovementTask":
        return cls(**data)

    @property
    def key(self) -> str:
        """
        一个稳定的逻辑主键:
        capability + requirement_id (如果有的话)
        """
        if self.requirement_id:
            return f"{self.capability}:{self.requirement_id}"
        return self.capability
说明：

任务以“某个能力 + 某个 requirement”为粒度（例如 memory_fabric + mf.search_semantic）。

task_id 可以等于 key（也可以再加前缀，这里我们后面在 manager 里统一生成）。

以后 SelfProgrammingEngine / ResearchLayer 都可以直接用这个结构。

二、任务管理器 + 文件存储：SelfImprovementManager
这里先用最简单、通用的实现：

用一个 JSON 文件当任务存储（后面你想换成 KVStore / DB 直接把 Store 抽象替换掉就行）。

文件：quentin/self_improvement/manager.py

# quentin/self_improvement/manager.py
from __future__ import annotations

import json
import logging
from dataclasses import asdict
from pathlib import Path
from typing import Dict, Any, List, Optional

from quentin.blueprint.tracker import BlueprintGapTracker, CapabilityStatus
from quentin.self_improvement.schemas import (
    ImprovementTask,
    TASK_STATUS_PENDING,
    PRIORITY_HIGH,
    PRIORITY_MEDIUM,
)

logger = logging.getLogger(__name__)


class FileBackedTaskStore:
    """
    最简单的任务存储：一个 JSON 文件，里面是 {task_id: task_dict}
    以后如果要换成 DB/KV，只需要实现同样的接口。
    """

    def __init__(self, path: str | Path = "data/improvement_tasks.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._cache: Dict[str, Dict[str, Any]] = {}
        self._loaded = False

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                with self.path.open("r", encoding="utf-8") as f:
                    self._cache = json.load(f)
            except Exception as e:
                logger.exception("Failed to load task store from %s", self.path)
                self._cache = {}
        else:
            self._cache = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self.path.open("w", encoding="utf-8") as f:
            json.dump(self._cache, f, ensure_ascii=False, indent=2)

    def get_all(self) -> Dict[str, Dict[str, Any]]:
        self._load()
        return dict(self._cache)

    def get(self, task_id: str) -> Optional[Dict[str, Any]]:
        self._load()
        return self._cache.get(task_id)

    def upsert(self, task: ImprovementTask) -> None:
        self._load()
        self._cache[task.task_id] = task.to_dict()
        self._save()

    def bulk_upsert(self, tasks: List[ImprovementTask]) -> None:
        self._load()
        for t in tasks:
            self._cache[t.task_id] = t.to_dict()
        self._save()


class SelfImprovementManager:
    """
    自我改进总调度（v1）：
    - 从 BlueprintGapTracker 的结果生成/更新 ImprovementTask
    - 提供任务列表给其他模块使用
    """

    def __init__(
        self,
        blueprint_path: str = "quentin/blueprint/meta_agent_v3.yaml",
        store: Optional[FileBackedTaskStore] = None,
    ) -> None:
        self.blueprint_path = blueprint_path
        self.store = store or FileBackedTaskStore()
        self.tracker = BlueprintGapTracker.from_file(blueprint_path)

    # --------- 对外主入口：同步蓝皮书任务 ----------

    def sync_blueprint_tasks(self) -> List[ImprovementTask]:
        """
        1. 使用 BlueprintGapTracker.scan() 获取当前 repo 对照蓝皮书的状态
        2. 把非 implemented 的 capability 转换成任务
        3. 旧任务如果仍然未完成，保持状态；如果蓝皮书已经补齐，后面可以考虑自动标记为 done（现在先不动）
        """
        gap_results = self.tracker.scan()
        existing = self._load_existing_tasks_indexed()

        new_or_updated: List[ImprovementTask] = []

        for cap_name, cap_status in gap_results.items():
            if cap_status.overall_status == "implemented":
                # 未来可以考虑：如果某 cap 之前有任务，现在已经 implemented，可以自动标记为 done
                continue

            # 如果 missing_requirements 为空，就以 capability 为粒度建一个通用任务
            req_ids = cap_status.missing_requirements or [None]

            for req_id in req_ids:
                task_id = self._make_task_id(cap_name, req_id)
                existing_task = existing.get(task_id)

                if existing_task:
                    # 已存在的任务：只更新整体状态/优先级，不改变已手动改过的 status（比如 in_progress）
                    updated = self._update_existing_task_from_cap(
                        existing_task, cap_status, req_id
                    )
                    new_or_updated.append(updated)
                else:
                    # 新任务
                    new_task = self._create_task_from_cap(
                        cap_status, req_id
                    )
                    new_or_updated.append(new_task)

        if new_or_updated:
            self.store.bulk_upsert(new_or_updated)

        return new_or_updated

    # --------- 任务查询接口 ----------

    def list_tasks(self, status: Optional[str] = None) -> List[ImprovementTask]:
        raw = self.store.get_all()
        tasks: List[ImprovementTask] = [
            ImprovementTask.from_dict(d) for d in raw.values()
        ]
        if status:
            tasks = [t for t in tasks if t.status == status]
        # 按 priority 排一下（小的在前）
        tasks.sort(key=lambda t: t.priority)
        return tasks

    # --------- 内部辅助 ----------

    def _load_existing_tasks_indexed(self) -> Dict[str, ImprovementTask]:
        raw = self.store.get_all()
        tasks = {tid: ImprovementTask.from_dict(d) for tid, d in raw.items()}
        return tasks

    @staticmethod
    def _make_task_id(capability: str, requirement_id: Optional[str]) -> str:
        if requirement_id:
            return f"{capability}:{requirement_id}"
        return capability

    @staticmethod
    def _priority_for_cap_status(status: str) -> int:
        """
        简单优先级策略：
        - missing    -> HIGH
        - placeholder/partial -> MEDIUM
        - implemented -> LOW（暂时用不到）
        """
        if status == "missing":
            return PRIORITY_HIGH
        if status in ("placeholder", "partial"):
            return PRIORITY_MEDIUM
        return 99  # implemented 或其他情况，目前不作为自改任务优先处理

    def _create_task_from_cap(
        self,
        cap_status: CapabilityStatus,
        requirement_id: Optional[str],
    ) -> ImprovementTask:
        priority = self._priority_for_cap_status(cap_status.overall_status)
        cap_name = cap_status.name
        task_id = self._make_task_id(cap_name, requirement_id)

        title = f"[{cap_status.overall_status}] capability={cap_name}"
        if requirement_id:
            title += f" / req={requirement_id}"

        description = f"Auto-generated from blueprint gap: capability={cap_name}, status={cap_status.overall_status}"
        if requirement_id:
            description += f", missing requirement={requirement_id}"

        meta: Dict[str, Any] = {
            "overall_status": cap_status.overall_status,
            "components": [
                {
                    "component": c.component,
                    "exists": c.exists,
                    "is_placeholder": c.is_placeholder,
                }
                for c in cap_status.components
            ],
        }

        return ImprovementTask(
            task_id=task_id,
            capability=cap_name,
            requirement_id=requirement_id,
            overall_status=cap_status.overall_status,
            status=TASK_STATUS_PENDING,
            priority=priority,
            title=title,
            description=description,
            meta=meta,
        )

    def _update_existing_task_from_cap(
        self,
        existing_task: ImprovementTask,
        cap_status: CapabilityStatus,
        requirement_id: Optional[str],
    ) -> ImprovementTask:
        # 更新 overall_status 和 priority，其余字段尽量保持不动
        existing_task.overall_status = cap_status.overall_status
        existing_task.priority = self._priority_for_cap_status(cap_status.overall_status)
        # meta 中更新最新的组件状态
        existing_task.meta["overall_status"] = cap_status.overall_status
        existing_task.meta["components"] = [
            {
                "component": c.component,
                "exists": c.exists,
                "is_placeholder": c.is_placeholder,
            }
            for c in cap_status.components
        ]
        return existing_task
说明：

核心函数：sync_blueprint_tasks()

把 BlueprintGapTracker.scan() 的结果 → 转成/更新 ImprovementTask。

overall_status != implemented 的 capability 都会生成任务。

任务 id 按 capability + requirement_id 组合，保证稳定/可重复同步。

任务存到 data/improvement_tasks.json，后面你要换成 DB，把 FileBackedTaskStore 换掉即可。

# quentin/cli/qctl.py
from __future__ import annotations

import argparse
import logging

from quentin.blueprint.tracker import BlueprintGapTracker
from quentin.self_improvement.manager import SelfImprovementManager

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


def cmd_blueprint_scan(args: argparse.Namespace) -> None:
    bp_path = args.blueprint or "quentin/blueprint/meta_agent_v3.yaml"
    tracker = BlueprintGapTracker.from_file(bp_path)
    results = tracker.scan()
    print(BlueprintGapTracker.format_results_as_json(results))


def cmd_improvement_sync(args: argparse.Namespace) -> None:
    bp_path = args.blueprint or "quentin/blueprint/meta_agent_v3.yaml"
    manager = SelfImprovementManager(blueprint_path=bp_path)
    new_tasks = manager.sync_blueprint_tasks()
    logger.info("Synced %d tasks (new or updated).", len(new_tasks))
    # 可以简单打印一下新任务
    for t in new_tasks:
        logger.info(
            "Task: %s | cap=%s | req=%s | status=%s | priority=%s",
            t.task_id,
            t.capability,
            t.requirement_id,
            t.overall_status,
            t.priority,
        )


def cmd_improvement_list(args: argparse.Namespace) -> None:
    status = args.status
    manager = SelfImprovementManager()
    tasks = manager.list_tasks(status=status)
    for t in tasks:
        print(
            f"- [{t.status}] (p={t.priority}) {t.task_id} :: {t.title}"
        )


def main() -> None:
    parser = argparse.ArgumentParser(prog="qctl", description="Project Quentin CLI")
    subparsers = parser.add_subparsers(dest="command", required=True)

    # blueprint-scan
    p_scan = subparsers.add_parser(
        "blueprint-scan", help="Scan current repo against meta_agent_v3 blueprint"
    )
    p_scan.add_argument(
        "--blueprint",
        type=str,
        default="quentin/blueprint/meta_agent_v3.yaml",
        help="Path to blueprint YAML file",
    )
    p_scan.set_defaults(func=cmd_blueprint_scan)

    # improvement-sync
    p_sync = subparsers.add_parser(
        "improvement-sync",
        help="Sync self-improvement tasks from blueprint gaps",
    )
    p_sync.add_argument(
        "--blueprint",
        type=str,
        default="quentin/blueprint/meta_agent_v3.yaml",
        help="Path to blueprint YAML file",
    )
    p_sync.set_defaults(func=cmd_improvement_sync)

    # improvement-list
    p_list = subparsers.add_parser(
        "improvement-list",
        help="List self-improvement tasks",
    )
    p_list.add_argument(
        "--status",
        type=str,
        default=None,
        help="Filter by task status (pending/in_progress/done/abandoned)",
    )
    p_list.set_defaults(func=cmd_improvement_list)

    args = parser.parse_args()
    args.func(args)


if __name__ == "__main__":
    main()
