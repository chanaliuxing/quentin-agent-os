# ============================================
# quentin/tools/schemas.py
# ============================================

from __future__ import annotations

from enum import Enum
from typing import Dict, Any, Optional
from pydantic import BaseModel, Field
from datetime import datetime


class ToolKind(str, Enum):
    """
    工具类型：
    - builtin: 内置纯 Python 工具
    - llm: 通过 LLM 进行“函数调用式”工具（例如 ReAct 工具）
    - external: 需要调用外部系统 / API 的工具
    """
    builtin = "builtin"
    llm = "llm"
    external = "external"


class ToolSpec(BaseModel):
    """
    工具的声明性定义：
    - name: 全局唯一（建议用 domain 前缀，如 "generic.echo"）
    - description: 给 Planner / LLM 使用的自然语言描述
    - kind: ToolKind
    - input_schema / output_schema: OPTIONAL -> 给 LLM / UI 做校验和提示
    """
    name: str
    description: str
    kind: ToolKind = ToolKind.builtin
    domain: str = "generic"

    input_schema: Dict[str, Any] = Field(default_factory=dict)
    output_schema: Dict[str, Any] = Field(default_factory=dict)

    meta: Dict[str, Any] = Field(default_factory=dict)
    created_at: datetime = Field(default_factory=datetime.utcnow)


class ToolCallRequest(BaseModel):
    """
    调用工具时的标准请求结构：
    - tool_name: ToolSpec.name
    - payload: 输入参数（可以是 dict / str / list，尽量统一成 dict）
    """
    tool_name: str
    payload: Dict[str, Any] = Field(default_factory=dict)
    trace_id: Optional[str] = None
    meta: Dict[str, Any] = Field(default_factory=dict)


class ToolCallResult(BaseModel):
    """
    调用工具后的标准返回：
    - success: 是否成功
    - output: 结果（可以任意结构，但尽量是 dict/json friendly）
    - error: 如果失败，这里存错误信息
    """
    tool_name: str
    success: bool
    output: Optional[Dict[str, Any]] = None
    error: Optional[str] = None
    started_at: datetime = Field(default_factory=datetime.utcnow)
    finished_at: Optional[datetime] = None
    meta: Dict[str, Any] = Field(default_factory=dict)
1.2 ToolManager：全局工具注册与调用管理
新建文件：quentin/tools/manager.py

# ============================================
# quentin/tools/manager.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Dict, Callable, Awaitable, Any, List
from time import perf_counter

from quentin.infra.logging import Tracer
from quentin.tools.schemas import ToolSpec, ToolCallRequest, ToolCallResult


ToolHandler = Callable[[ToolCallRequest], Awaitable[ToolCallResult]]


@dataclass
class ToolManager:
    """
    全局工具管理器：
    - 注册 ToolSpec + 对应 handler（异步函数）
    - 提供统一的 call_tool 接口
    - 记录 trace 方便后续研究/可视化
    """
    tracer: Tracer
    specs: Dict[str, ToolSpec] = field(default_factory=dict)
    handlers: Dict[str, ToolHandler] = field(default_factory=dict)

    def register_tool(self, spec: ToolSpec, handler: ToolHandler) -> None:
        if spec.name in self.specs:
            raise ValueError(f"Tool '{spec.name}' already registered.")
        self.specs[spec.name] = spec
        self.handlers[spec.name] = handler

        self.tracer.span(
            "L1",
            "tool_register",
            module="tool_manager",
            input={
                "tool_name": spec.name,
                "domain": spec.domain,
                "kind": spec.kind.value,
            },
        )

    def list_tools(self) -> List[ToolSpec]:
        return list(self.specs.values())

    def list_tools_by_domain(self, domain: str) -> List[ToolSpec]:
        return [s for s in self.specs.values() if s.domain == domain]

    async def call_tool(self, req: ToolCallRequest) -> ToolCallResult:
        if req.tool_name not in self.handlers:
            return ToolCallResult(
                tool_name=req.tool_name,
                success=False,
                error=f"Unknown tool '{req.tool_name}'",
            )

        handler = self.handlers[req.tool_name]
        start = perf_counter()
        self.tracer.span(
            "L1",
            "tool_call_start",
            module="tool_manager",
            input={"tool_name": req.tool_name, "payload": req.payload},
        )
        try:
            result = await handler(req)
        except Exception as e:
            result = ToolCallResult(
                tool_name=req.tool_name,
                success=False,
                error=repr(e),
            )

        result.finished_at = result.finished_at or result.started_at
        latency = perf_counter() - start

        self.tracer.span(
            "L1",
            "tool_call_end",
            module="tool_manager",
            input={"tool_name": req.tool_name},
            output={"success": result.success, "latency_sec": latency},
        )
        return result
以后任何 domain 适配器，只要拿到 ToolManager，就能向里注册自己的工具。

2.1 DomainAdapter & DomainRegistry
新建文件：quentin/domains/base.py

# ============================================
# quentin/domains/base.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Dict, Any, List, Optional, Protocol

from quentin.infra.logging import Tracer
from quentin.tools.manager import ToolManager
from quentin.tools.schemas import ToolSpec, ToolCallRequest, ToolCallResult


@dataclass
class DomainTask:
    """
    领域任务抽象：
    - domain: 领域名称，例如 "generic" / "desktop" / "legal" / "trading"
    - goal: 任务目标（自然语言）
    - input: 任意 payload，由具体 DomainAdapter 自己解释
    """
    domain: str
    goal: str
    input: Dict[str, Any] = field(default_factory=dict)
    meta: Dict[str, Any] = field(default_factory=dict)


class DomainAdapter(Protocol):
    """
    DomainAdapter 协议：
    - name: 领域名称
    - description: 给 Planner / UI 看的介绍
    - register_tools: 在 ToolManager 中注册自己的工具
    - handle_task: 执行一个高层任务（可以用到多个工具）
    """

    name: str
    description: str

    async def register_tools(self, tool_manager: ToolManager) -> None:
        ...

    async def handle_task(
        self,
        task: DomainTask,
        tool_manager: ToolManager,
    ) -> Dict[str, Any]:
        """
        返回一个结果 dict（可以带 log / steps / output 等）
        """
        ...


@dataclass
class DomainRegistry:
    """
    用于管理多个 DomainAdapter：
    - 注册 / 查询 domain
    - 提供 route_task：把 DomainTask 路由到对应 adapter
    """
    tracer: Tracer
    tool_manager: ToolManager
    domains: Dict[str, DomainAdapter] = field(default_factory=dict)

    async def register_domain(self, adapter: DomainAdapter) -> None:
        if adapter.name in self.domains:
            raise ValueError(f"Domain '{adapter.name}' already registered.")

        self.domains[adapter.name] = adapter
        await adapter.register_tools(self.tool_manager)

        self.tracer.span(
            "L1",
            "domain_register",
            module="domain_registry",
            input={"domain": adapter.name},
        )

    def list_domains(self) -> List[Dict[str, Any]]:
        return [
            {
                "name": a.name,
                "description": getattr(a, "description", ""),
            }
            for a in self.domains.values()
        ]

    def get(self, name: str) -> Optional[DomainAdapter]:
        return self.domains.get(name)

    async def route_task(self, task: DomainTask) -> Dict[str, Any]:
        adapter = self.domains.get(task.domain)
        if not adapter:
            raise ValueError(f"Unknown domain '{task.domain}'")
        self.tracer.span(
            "L1",
            "domain_task_route",
            module="domain_registry",
            input={"domain": task.domain, "goal": task.goal},
        )
        return await adapter.handle_task(task, self.tool_manager)
以后任何人想扩展领域，只需要实现一个 DomainAdapter 子类，然后在启动时 register_domain() 即可。

三、内置 GenericDomainAdapter：保证“裸 repo 能用”
我们做一个最简单的通用领域：generic，内置几个示例工具：

generic.echo：原样返回

generic.scratchpad.append：向 KV 里写一条文本

generic.scratchpad.list：列出最近 N 条文本

新建文件：quentin/domains/generic.py

# ============================================
# quentin/domains/generic.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass
from typing import Dict, Any, List

from quentin.infra.logging import Tracer
from quentin.infra.storage.kv_store import KVStore
from quentin.tools.manager import ToolManager
from quentin.tools.schemas import (
    ToolSpec,
    ToolKind,
    ToolCallRequest,
    ToolCallResult,
)
from quentin.domains.base import DomainAdapter, DomainTask


SCRATCHPAD_KEY = "generic:scratchpad:entries"


@dataclass
class GenericDomainAdapter(DomainAdapter):
    """
    内置的 “通用领域”：
    - 不依赖任何外部系统
    - 提供少量示例工具，方便测试 / Demo / Prompt 调用
    """
    tracer: Tracer
    kv: KVStore

    name: str = "generic"
    description: str = "Built-in generic domain with basic echo & scratchpad tools."

    async def register_tools(self, tool_manager: ToolManager) -> None:
        # 1) echo 工具
        echo_spec = ToolSpec(
            name="generic.echo",
            description="Echo back the input text. Useful for testing tool-calls.",
            kind=ToolKind.builtin,
            domain=self.name,
            input_schema={
                "type": "object",
                "properties": {"text": {"type": "string"}},
                "required": ["text"],
            },
            output_schema={
                "type": "object",
                "properties": {"text": {"type": "string"}},
            },
        )

        async def echo_handler(req: ToolCallRequest) -> ToolCallResult:
            txt = str(req.payload.get("text", ""))
            return ToolCallResult(
                tool_name=req.tool_name,
                success=True,
                output={"text": txt},
            )

        tool_manager.register_tool(echo_spec, echo_handler)

        # 2) scratchpad.append
        append_spec = ToolSpec(
            name="generic.scratchpad.append",
            description="Append a line of text into a persistent scratchpad.",
            kind=ToolKind.builtin,
            domain=self.name,
            input_schema={
                "type": "object",
                "properties": {"text": {"type": "string"}},
                "required": ["text"],
            },
        )

        async def append_handler(req: ToolCallRequest) -> ToolCallResult:
            text = str(req.payload.get("text", ""))
            entries: List[str] = await self.kv.get(SCRATCHPAD_KEY) or []
            entries.append(text)
            await self.kv.save(SCRATCHPAD_KEY, entries)
            return ToolCallResult(
                tool_name=req.tool_name,
                success=True,
                output={"count": len(entries)},
            )

        tool_manager.register_tool(append_spec, append_handler)

        # 3) scratchpad.list
        list_spec = ToolSpec(
            name="generic.scratchpad.list",
            description="List recent lines from the scratchpad.",
            kind=ToolKind.builtin,
            domain=self.name,
            input_schema={
                "type": "object",
                "properties": {"limit": {"type": "integer"}},
            },
        )

        async def list_handler(req: ToolCallRequest) -> ToolCallResult:
            limit = int(req.payload.get("limit", 20))
            entries: List[str] = await self.kv.get(SCRATCHPAD_KEY) or []
            entries = entries[-limit:]
            return ToolCallResult(
                tool_name=req.tool_name,
                success=True,
                output={"entries": entries},
            )

        tool_manager.register_tool(list_spec, list_handler)

    async def handle_task(
        self,
        task: DomainTask,
        tool_manager: ToolManager,
    ) -> Dict[str, Any]:
        """
        MVP：直接调用 generic.scratchpad.append 记录一行，然后用 echo 返回。
        将来 Planner 可以把 DomainTask 进一步拆成 Tool 调用序列。
        """
        # 简单 demo：把 goal + input 写入 scratchpad
        payload = {"text": f"{task.goal} | input={task.input}"}
        await tool_manager.call_tool(
            ToolCallRequest(tool_name="generic.scratchpad.append", payload=payload)
        )

        # 再 echo 一下 goal，当作返回结果
        result = await tool_manager.call_tool(
            ToolCallRequest(tool_name="generic.echo", payload={"text": task.goal})
        )
        return {
            "domain": self.name,
            "goal": task.goal,
            "input": task.input,
            "tool_result": result.model_dump(),
        }
