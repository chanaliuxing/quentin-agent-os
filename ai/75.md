from __future__ import annotations

from dataclasses import dataclass, asdict, field
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging
import uuid
import statistics as stats

logger = logging.getLogger(__name__)


# ---------- 数据结构 ----------


@dataclass
class Experiment:
    """
    一个实验的元数据：

    - experiment_id    : 全局唯一 ID
    - name             : 实验名称（人类可读）
    - purpose          : 用途标签（如 'cognitive_loop', 'router', 'immune_loop'）
    - capability       : 对应蓝皮书里的 capability_id
    - requirement_id   : 蓝皮书中对应的 requirement_id
    - tags             : 辅助标签（可用于过滤）
    """

    experiment_id: str
    name: str
    purpose: str
    capability: str
    requirement_id: str
    tags: List[str] = field(default_factory=list)

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "Experiment":
        return cls(
            experiment_id=d["experiment_id"],
            name=d["name"],
            purpose=d["purpose"],
            capability=d["capability"],
            requirement_id=d["requirement_id"],
            tags=d.get("tags", []) or [],
            created_at=d.get("created_at", ""),
            updated_at=d.get("updated_at", ""),
        )


@dataclass
class ExperimentRun:
    """
    一次实验运行的记录：

    - run_id       : 唯一 ID
    - experiment_id: 所属实验
    - variant_id   : 变体 ID（如 'baseline_v1', 'new_router_policy_2'）
    - metrics      : 任意标量指标（如 success_rate, score, latency 等）
    - context      : 任意额外上下文信息
    """

    run_id: str
    experiment_id: str
    variant_id: str
    metrics: Dict[str, float]
    context: Dict[str, Any]

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "ExperimentRun":
        return cls(
            run_id=d["run_id"],
            experiment_id=d["experiment_id"],
            variant_id=d["variant_id"],
            metrics=d.get("metrics", {}) or {},
            context=d.get("context", {}) or {},
            created_at=d.get("created_at", ""),
        )


@dataclass
class VariantSummary:
    """
    某个 variant 在一个实验中的聚合统计：
    - count        : 参与统计的 run 数量
    - mean         : 指标均值
    - std          : 指标标准差
    - min/max      : 极值
    - metric_name  : 指标名
    """

    experiment_id: str
    variant_id: str
    metric_name: str
    count: int
    mean: float
    std: float
    min: float
    max: float

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


# ---------- 管理器 ----------


class ResearchManager:
    """
    ResearchManager v2：

    存储格式：
    - data/experiments.json        : 实验元数据（id -> Experiment）
    - data/experiment_runs.jsonl   : 每行一个 ExperimentRun

    提供能力：
    - create_experiment
    - list_experiments
    - record_run
    - list_runs(experiment_id)
    - summarize_variants(experiment_id, metric_name)
    - compare_variants(experiment_id, metric_name, top_k, higher_is_better)
    - export_results(experiment_id, dest_path)
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        experiments_store: str | Path = "data/experiments.json",
        runs_store: str | Path = "data/experiment_runs.jsonl",
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.experiments_store = self.repo_root / experiments_store
        self.runs_store = self.repo_root / runs_store

        self.experiments_store.parent.mkdir(parents=True, exist_ok=True)
        self.runs_store.parent.mkdir(parents=True, exist_ok=True)

        self._experiments: Dict[str, Experiment] = {}
        self._loaded = False

    # ---------- 实验管理 ----------

    def create_experiment(
        self,
        *,
        name: str,
        purpose: str,
        capability: str,
        requirement_id: str,
        tags: Optional[List[str]] = None,
    ) -> Experiment:
        self._ensure_loaded()
        exp_id = str(uuid.uuid4())
        exp = Experiment(
            experiment_id=exp_id,
            name=name,
            purpose=purpose,
            capability=capability,
            requirement_id=requirement_id,
            tags=tags or [],
        )
        self._experiments[exp_id] = exp
        self._flush_experiments()
        return exp

    def list_experiments(self) -> List[Experiment]:
        self._ensure_loaded()
        return list(self._experiments.values())

    def get_experiment(self, experiment_id: str) -> Optional[Experiment]:
        self._ensure_loaded()
        return self._experiments.get(experiment_id)

    # ---------- Run 记录 ----------

    def record_run(
        self,
        *,
        experiment_id: str,
        variant_id: str,
        metrics: Dict[str, float],
        context: Optional[Dict[str, Any]] = None,
    ) -> ExperimentRun:
        self._ensure_loaded()
        if experiment_id not in self._experiments:
            logger.warning(
                "ResearchManager.record_run: experiment_id=%s not found; "
                "run will still be recorded, but consider creating the experiment first.",
                experiment_id,
            )
        run = ExperimentRun(
            run_id=str(uuid.uuid4()),
            experiment_id=experiment_id,
            variant_id=variant_id,
            metrics=metrics,
            context=context or {},
        )
        self._append_run(run)
        return run

    def list_runs(self, experiment_id: str) -> List[ExperimentRun]:
        runs: List[ExperimentRun] = []
        if not self.runs_store.exists():
            return runs

        try:
            with self.runs_store.open("r", encoding="utf-8") as f:
                for line in f:
                    if not line.strip():
                        continue
                    d = json.loads(line)
                    if d.get("experiment_id") != experiment_id:
                        continue
                    runs.append(ExperimentRun.from_dict(d))
        except Exception:
            logger.exception(
                "ResearchManager.list_runs: failed to read runs from %s",
                self.runs_store,
            )
        return runs

    # ---------- Variant 聚合与对比 ----------

    def summarize_variants(
        self,
        experiment_id: str,
        *,
        metric_name: str,
    ) -> Dict[str, VariantSummary]:
        runs = self.list_runs(experiment_id)
        by_variant: Dict[str, List[float]] = {}
        for r in runs:
            if metric_name not in r.metrics:
                continue
            by_variant.setdefault(r.variant_id, []).append(float(r.metrics[metric_name]))

        summaries: Dict[str, VariantSummary] = {}
        for variant_id, values in by_variant.items():
            if not values:
                continue
            count = len(values)
            mean = float(stats.mean(values))
            std = float(stats.pstdev(values)) if count > 1 else 0.0
            mn = float(min(values))
            mx = float(max(values))
            summaries[variant_id] = VariantSummary(
                experiment_id=experiment_id,
                variant_id=variant_id,
                metric_name=metric_name,
                count=count,
                mean=mean,
                std=std,
                min=mn,
                max=mx,
            )
        return summaries

    def compare_variants(
        self,
        experiment_id: str,
        *,
        metric_name: str,
        top_k: int = 5,
        higher_is_better: bool = True,
    ) -> List[VariantSummary]:
        """
        返回按 metric 均值排序的前 top_k 个变体。
        """
        summaries = self.summarize_variants(experiment_id, metric_name=metric_name)
        items = list(summaries.values())
        if not items:
            return []

        items.sort(
            key=lambda s: s.mean,
            reverse=higher_is_better,
        )
        return items[:top_k]

    # ---------- 导出 ----------

    def export_results(
        self,
        experiment_id: str,
        dest_path: str | Path,
    ) -> None:
        """
        把一个 experiment 的全部信息导出到单个 JSON 文件：
        {
          "experiment": {...},
          "runs": [...],
          "summaries": {
            "<metric>": {
              "<variant>": {...}
            },
            ...
          }
        }
        """
        self._ensure_loaded()
        exp = self.get_experiment(experiment_id)
        if not exp:
            raise ValueError(f"Experiment not found: {experiment_id}")

        runs = self.list_runs(experiment_id)

        # 为常见 metric 做一个 summaries（简单：对 runs 中出现的所有 metric 聚合）
        metric_names = set()
        for r in runs:
            metric_names.update(r.metrics.keys())

        all_summaries: Dict[str, Dict[str, Any]] = {}
        for m in metric_names:
            vs = self.summarize_variants(experiment_id, metric_name=m)
            all_summaries[m] = {vid: s.to_dict() for vid, s in vs.items()}

        out = {
            "experiment": exp.to_dict(),
            "runs": [r.to_dict() for r in runs],
            "summaries": all_summaries,
        }

        dest = Path(dest_path)
        dest.parent.mkdir(parents=True, exist_ok=True)
        dest.write_text(json.dumps(out, ensure_ascii=False, indent=2), encoding="utf-8")

    # ---------- 内部：加载/持久化 ----------

    def _ensure_loaded(self) -> None:
        if self._loaded:
            return
        self._load_experiments()
        self._loaded = True

    def _load_experiments(self) -> None:
        if not self.experiments_store.exists():
            self._experiments = {}
            return
        try:
            raw = json.loads(self.experiments_store.read_text(encoding="utf-8") or "{}")
            self._experiments = {
                eid: Experiment.from_dict(edata) for eid, edata in raw.items()
            }
        except Exception:
            logger.exception(
                "ResearchManager._load_experiments: failed to read %s",
                self.experiments_store,
            )
            self._experiments = {}

    def _flush_experiments(self) -> None:
        try:
            raw = {eid: e.to_dict() for eid, e in self._experiments.items()}
            self.experiments_store.write_text(
                json.dumps(raw, ensure_ascii=False, indent=2),
                encoding="utf-8",
            )
        except Exception:
            logger.exception(
                "ResearchManager._flush_experiments: failed to write %s",
                self.experiments_store,
            )

    def _append_run(self, run: ExperimentRun) -> None:
        try:
            with self.runs_store.open("a", encoding="utf-8") as f:
                f.write(json.dumps(run.to_dict(), ensure_ascii=False) + "\n")
        except Exception:
            logger.exception(
                "ResearchManager._append_run: failed to append to %s",
                self.runs_store,
            )

# quentin/llm/router_optimize.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Dict, List, Tuple
import json
import logging
import statistics as stats

logger = logging.getLogger(__name__)


@dataclass
class RouterStatRecord:
    mode: str
    task_type: str
    model: str
    provider: str
    latency_ms: float
    success: bool
    score: float

    @classmethod
    def from_dict(cls, d: Dict) -> "RouterStatRecord":
        return cls(
            mode=str(d.get("mode", "")).lower(),
            task_type=str(d.get("task_type", "generic")).lower(),
            model=str(d.get("model", "")),
            provider=str(d.get("provider", "")),
            latency_ms=float(d.get("latency_ms", 0.0)),
            success=bool(d.get("success", True)),
            score=float(d.get("score", 0.0)),
        )


@dataclass
class RouterPolicy:
    """
    对于某个 (mode, task_type) 的推荐策略：
    - model / provider : 推荐用哪个模型
    - metric           : 优化使用的指标
    - score            : 聚合后的有效得分（score - λ * latency）
    - raw              : 原始 success_rate / avg_latency / avg_score
    """

    mode: str
    task_type: str
    model: str
    provider: str
    metric: str
    score: float
    success_rate: float
    avg_latency_ms: float
    avg_model_score: float

    def to_dict(self) -> Dict:
        return asdict(self)


class ModelRouterOptimizer:
    """
    从 data/model_router_stats.jsonl 中采样数据，给每个 (mode, task_type) 选出最优模型。

    v1 策略：
    - 按三项统计：
        - success_rate
        - avg_latency_ms
        - avg_model_score  (如果有的话，否则用 success)
    - 综合评价函数：
        effective_score = avg_model_score * success_rate - weight_latency * (avg_latency_ms / 1000)
    - 对每个 (mode, task_type) 选出 effective_score 最大的 model。
    """

    def __init__(
        self,
        *,
        stats_path: str | Path = "data/model_router_stats.jsonl",
        rules_path: str | Path = "config/model_router_rules.json",
        weight_latency: float = 0.1,
    ) -> None:
        self.stats_path = Path(stats_path)
        self.rules_path = Path(rules_path)
        self.weight_latency = weight_latency

    def optimize(self) -> Dict[str, RouterPolicy]:
        records = self._load_records()
        if not records:
            logger.warning("ModelRouterOptimizer: no router stats found; nothing to optimize.")
            return {}

        # 分组：((mode, task_type, model, provider) -> [records])
        by_key: Dict[Tuple[str, str, str, str], List[RouterStatRecord]] = {}
        for r in records:
            key = (r.mode, r.task_type, r.model, r.provider)
            by_key.setdefault(key, []).append(r)

        # 对每个 (mode, task_type) 选模型
        best_policies: Dict[str, RouterPolicy] = {}
        # key2: (mode, task_type) -> [(policy, effective_score)]
        group_by_route: Dict[Tuple[str, str], List[RouterPolicy]] = {}

        for (mode, task_type, model, provider), rs in by_key.items():
            if not rs:
                continue
            success_rate = stats.mean(1.0 if r.success else 0.0 for r in rs)
            latencies = [max(1.0, r.latency_ms) for r in rs]
            avg_latency = stats.mean(latencies)
            if all(r.score == 0.0 for r in rs):
                avg_score = success_rate  # 没有显式 score 就用 success_rate
                metric = "success_rate"
            else:
                avg_score = stats.mean(r.score for r in rs)
                metric = "score"

            effective_score = avg_score * success_rate - self.weight_latency * (avg_latency / 1000.0)

            policy = RouterPolicy(
                mode=mode,
                task_type=task_type,
                model=model,
                provider=provider,
                metric=metric,
                score=effective_score,
                success_rate=success_rate,
                avg_latency_ms=avg_latency,
                avg_model_score=avg_score,
            )

            key_route = (mode, task_type)
            group_by_route.setdefault(key_route, []).append(policy)

        for key_route, policies in group_by_route.items():
            if not policies:
                continue
            # 选择 effective_score 最大的 policy
            policies.sort(key=lambda p: p.score, reverse=True)
            best = policies[0]
            route_key = self._route_key(best.mode, best.task_type)
            best_policies[route_key] = best

        # 写入规则文件
        data = {k: p.to_dict() for k, p in best_policies.items()}
        self.rules_path.parent.mkdir(parents=True, exist_ok=True)
        self.rules_path.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")

        return best_policies

    # ---------- 内部 ----------

    def _load_records(self) -> List[RouterStatRecord]:
        if not self.stats_path.exists():
            return []
        records: List[RouterStatRecord] = []
        try:
            with self.stats_path.open("r", encoding="utf-8") as f:
                for line in f:
                    if not line.strip():
                        continue
                    d = json.loads(line)
                    records.append(RouterStatRecord.from_dict(d))
        except Exception:
            logger.exception("ModelRouterOptimizer._load_records: failed to read %s", self.stats_path)
        return records

    @staticmethod
    def _route_key(mode: str, task_type: str) -> str:
        return f"{mode}::{task_type}"

from pathlib import Path
from typing import Any, Dict, Optional
import json

from quentin.llm.types import LLMConfig

logger = logging.getLogger(__name__)


@dataclass
class RouteFeedback:
    mode: str
    task_type: str
    model: str
    provider: str
    latency_ms: float
    success: bool
    score: float = 0.0

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class ModelRouter:
    """
    ModelRouter v2：

    - 运行时优先读取 config/model_router_rules.json 中的策略：
        key:  "mode::task_type"
        val:  { "model": "...", "provider": "...", ... }
    - 如果没有对应 key，则回退到默认策略（qwen / deepseek / bge-m3）
    - record_feedback 将数据写入 data/model_router_stats.jsonl，供 RouterOptimizer 使用。
    """

    def __init__(
        self,
        *,
        stats_store: str | Path = "data/model_router_stats.jsonl",
        rules_store: str | Path = "config/model_router_rules.json",
    ) -> None:
        self.stats_store = Path(stats_store)
        self.stats_store.parent.mkdir(parents=True, exist_ok=True)

        self.rules_store = Path(rules_store)
        self._rules: Dict[str, Dict[str, Any]] = {}
        self._load_rules()

    # ---------- 路由 ----------

    def route(
        self,
        *,
        mode: str,
        task_type: str = "generic",
    ) -> LLMConfig:
        mode = mode.lower()
        task_type = task_type.lower()
        key = self._route_key(mode, task_type)

        if key in self._rules:
            r = self._rules[key]
            model = r.get("model") or "qwen3-vl:latest"
            provider = r.get("provider") or "ollama"
            role = r.get("role") or f"{mode}_{task_type}"
            return LLMConfig(
                provider=provider,
                model=model,
                role=role,
            )

        # 默认策略（和 v1 一致）
        if mode in ("planner", "critic"):
            return LLMConfig(
                provider="ollama",
                model="qwen3-vl:latest",
                role=f"{mode}_{task_type}",
            )
        elif mode == "executor":
            return LLMConfig(
                provider="ollama",
                model="deepseek-r1:1.5b",
                role=f"{mode}_{task_type}",
            )
        elif mode == "embedding":
            return LLMConfig(
                provider="ollama",
                model="bge-m3",
                role=f"{mode}_{task_type}",
            )
        else:
            return LLMConfig(
                provider="ollama",
                model="qwen3-vl:latest",
                role=f"{mode}_{task_type}",
            )

    # ---------- 反馈 ----------

    def record_feedback(self, fb: RouteFeedback) -> None:
        try:
            with self.stats_store.open("a", encoding="utf-8") as f:
                f.write(json.dumps(fb.to_dict(), ensure_ascii=False) + "\n")
        except Exception:
            logger.exception("ModelRouter.record_feedback: failed to append to %s", self.stats_store)

    # ---------- 内部 ----------

    def _load_rules(self) -> None:
        if not self.rules_store.exists():
            self._rules = {}
            return
        try:
            raw = json.loads(self.rules_store.read_text(encoding="utf-8") or "{}")
            # raw: key -> RouterPolicy dict；我们只关心 model/provider/role 等
            self._rules = raw
        except Exception:
            logger.exception("ModelRouter._load_rules: failed to read %s", self.rules_store)
            self._rules = {}

    @staticmethod
    def _route_key(mode: str, task_type: str) -> str:
        return f"{mode}::{task_type}"

def cmd_router_show(args: argparse.Namespace) -> None:
    router = ModelRouter()

    if not rules:

    print("Current ModelRouter rules:")

        print(f"  provider: {r.get('provider')}")
