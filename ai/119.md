1.1 BlueprintGap 数据结构 & Tracker
# quentin/blueprint/gaps.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, List, Optional

import json


@dataclass
class BlueprintGap:
    """
    一条“蓝皮书 → 实现”之间的缺口（gap）。

    spec 里每个 requirement 建议都有一个唯一 id，
    如果在 capabilities.json 里没有被标记为 implemented，就会被视为 gap。
    """
    requirement_id: str
    category: str          # 例: "loop", "memory", "self_programming", "eval", ...
    description: str
    severity: str          # "low" | "medium" | "high"
    module_hint: Optional[str] = None   # 建议改哪个模块（相对路径）
    symbol_hint: Optional[str] = None   # 建议改哪个函数/类
    meta: Dict[str, Any] = None

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        return d


class BlueprintGapTracker:
    """
    从蓝皮书 spec + capabilities 快照中扫描 gap 的工具。

    约定：
      - spec_path: blueprint/meta_agent_v3_spec.json
        基本结构示例：
          {
            "version": "3.0",
            "requirements": [
              {
                "id": "loop.cognitive.v1",
                "category": "loop",
                "description": "认知循环必须记录 Trace, Summary, Lessons。",
                "priority": "high",
                "module_hint": "quentin/runtime/cognitive_loop.py",
                "symbol_hint": "CognitiveLoop.run_task",
                "meta": {... 任意补充 ...}
              },
              ...
            ]
          }

      - capabilities_path: data/blueprint/capabilities.json
        基本结构示例：
          {
            "implemented_ids": [
              "loop.cognitive.v1",
              "memory.fabric.v1",
              ...
            ],
            "notes": "这一文件可以由系统自迭代或人工维护"
          }
    """

    def __init__(
        self,
        *,
        repo_root: str | Path,
        spec_path: Optional[str | Path] = None,
        capabilities_path: Optional[str | Path] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.spec_path = Path(spec_path) if spec_path else self.repo_root / "blueprint" / "meta_agent_v3_spec.json"
        self.capabilities_path = (
            Path(capabilities_path)
            if capabilities_path
            else self.repo_root / "data" / "blueprint" / "capabilities.json"
        )

    # -----------------------------
    # 公共入口：扫描 gap
    # -----------------------------
    def scan(self, max_gaps: Optional[int] = None) -> List[BlueprintGap]:
        spec = self._load_spec()
        caps = self._load_capabilities()

        implemented_ids = set(caps.get("implemented_ids", []))
        # 兼容另一种结构：{"implemented": {"loop.cognitive.v1": {...}}}
        implemented_ids.update(caps.get("implemented", {}).keys())

        gaps: List[BlueprintGap] = []
        for req in spec.get("requirements", []):
            rid = req.get("id")
            if not rid:
                continue
            if rid in implemented_ids:
                continue

            gap = BlueprintGap(
                requirement_id=rid,
                category=req.get("category", "uncategorized"),
                description=req.get("description", ""),
                severity=req.get("priority", "medium"),
                module_hint=req.get("module_hint"),
                symbol_hint=req.get("symbol_hint"),
                meta=req.get("meta", {}),
            )
            gaps.append(gap)

        # 简单按 severity 排序：high > medium > low
        severity_order = {"high": 0, "medium": 1, "low": 2}
        gaps.sort(key=lambda g: severity_order.get(g.severity, 99))

        if max_gaps is not None:
            gaps = gaps[:max_gaps]

        return gaps

    # -----------------------------
    # 内部：加载 spec & capabilities
    # -----------------------------
    def _load_spec(self) -> Dict[str, Any]:
        if not self.spec_path.exists():
            raise FileNotFoundError(
                f"Blueprint spec not found: {self.spec_path}. "
                "请确保存在 blueprint/meta_agent_v3_spec.json。"
            )
        with self.spec_path.open("r", encoding="utf-8") as f:
            return json.load(f)

    def _load_capabilities(self) -> Dict[str, Any]:
        if not self.capabilities_path.exists():
            # 文件不存在时，视为“一个都没实现”，方便系统从0开始自迭代
            return {"implemented_ids": []}
        with self.capabilities_path.open("r", encoding="utf-8") as f:
            return json.load(f)
你可以先手工写一版非常简单的 meta_agent_v3_spec.json，以后再让系统自己迭代完善。

2. EvalHarness：最小通用评测器 quentin/eval/harness.py
功能：

定义一个简单的 prompt 评测用例 EvalCase

支持在多个 runtime variant（baseline / council / future…）上跑同一批 case

返回原始输出和粗略“匹配度”分数（后面可以再精细化）

# quentin/eval/harness.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, List, Optional


@dataclass
class EvalCase:
    """
    一个最小的评测用例：
      - description: 用例说明
      - input_payload: 直接丢给 runtime.run_task(...) 的 payload
      - expected_keywords: 期望输出中包含的一些关键词（非常粗糙）
    """
    case_id: str
    description: str
    input_payload: Dict[str, Any]
    expected_keywords: List[str]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class EvalResult:
    case_id: str
    variant: str
    raw_output: Any
    score: float
    matched_keywords: List[str]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class EvalHarness:
    """
    一个非常简单的 EvalHarness：

      - 目前默认调用 MetaAgentRuntime.run_task(...)
      - 使用“包含关键字数量 / 总关键字数量”作为粗糙评分
      - 真正的复杂评测（多维指标、LLM-as-judge）可以以后逐步自迭代升级
    """

    def __init__(self, variants_runtimes: Dict[str, Any]) -> None:
        self.variants_runtimes = variants_runtimes

    def run_prompt_eval(
        self,
        cases: List[EvalCase],
        *,
        variants: Optional[List[str]] = None,
    ) -> List[EvalResult]:
        if variants is None:
            variants = list(self.variants_runtimes.keys())

        results: List[EvalResult] = []

        for v in variants:
            runtime = self.variants_runtimes[v]
            for case in cases:
                # 这里我们假设 runtime.run_task(description, payload) 已经实现
                tr = runtime.run_task(
                    description=case.description,
                    payload=case.input_payload,
                    context={},
                )
                # 简单把 output 转成字符串
                text = str(tr.output)

                matched = [kw for kw in case.expected_keywords if kw in text]
                score = len(matched) / max(1, len(case.expected_keywords))

                results.append(
                    EvalResult(
                        case_id=case.case_id,
                        variant=v,
                        raw_output=text,
                        score=score,
                        matched_keywords=matched,
                    )
                )
        return results
现在 EvalHarness 先不强绑定到 patch 实验，只是一个通用评测工具，ExperimentOrchestrator / ResearchLayer 随时可以调用它。

目标：

3.1 扩展 quentin/research/layer.py
# quentin/research/layer.py
from __future__ import annotations

from pathlib import Path
from typing import Any, Dict, List, Optional
import json

from quentin.self_programming.engine import SelfProgrammingEngine
from quentin.self_programming.schema import PatchTarget
from quentin.blueprint.gaps import BlueprintGap, BlueprintGapTracker


class ResearchLayer:
    def __init__(
        self,
        *,
        repo_root: str | Path,
        experiment_orchestrator: Any,
        self_programming_engine: Optional[SelfProgrammingEngine] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.experiment_orchestrator = experiment_orchestrator
        self.self_programming_engine = self_programming_engine

    def attach_self_programming_engine(self, engine: SelfProgrammingEngine) -> None:
        self.self_programming_engine = engine

    # --------------------------------------------------
    # 核心：蓝皮书自我改进循环
    # --------------------------------------------------
    def run_blueprint_improvement_cycle(
        self,
        *,
        runtime: Any,              # MetaAgentRuntime（带 council）
        max_gaps: int = 5,
        research_request: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        v3 蓝皮书里的“Research & Evolution Layer”的一个最小实现：

          1. 用 BlueprintGapTracker 扫描 gap
          2. 对每个 gap 交给多 AI 议会（council）分析，生成 modules_to_improve
          3. 调用 SelfProgrammingEngine 为这些模块生成 patch proposal + 实验
        """
        research_request = research_request or {}

        # 1) 扫描 gap
        gap_tracker = BlueprintGapTracker(repo_root=self.repo_root)
        gaps = gap_tracker.scan(max_gaps=max_gaps)

        if not gaps:
            return {
                "status": "no_gaps_found",
                "gaps": [],
                "modules_to_improve": [],
                "self_programming_summary": None,
            }

        # 2) 用 council 分析每个 gap，得到 PatchTarget 列表
        modules_to_improve: List[PatchTarget] = []

        for gap in gaps:
            suggestions = self._analyze_gap_with_council(runtime, gap)
            for s in suggestions:
                modules_to_improve.append(
                    PatchTarget(
                        kind=s.get("kind", "module"),
                        path=s["path"],
                        symbol=s.get("symbol"),
                        constraints=s.get("constraints", {}),
                    )
                )

        # 3) 自编程引擎：生成 patch proposal + 注册实验
        if not self.self_programming_engine:
            sp_summary = {
                "status": "no_self_programming_engine_attached",
                "num_targets": len(modules_to_improve),
                "num_proposals": 0,
                "experiment_ids": [],
            }
        else:
            sp_summary = self.self_programming_engine.run_cycle(
                modules_to_improve=modules_to_improve,
                context={
                    "problem_description": "Blueprint improvement cycle",
                    "metrics": research_request.get("metrics", {}),
                    "notes": {
                        "selected_gap_ids": [g.requirement_id for g in gaps],
                        "gaps": [g.to_dict() for g in gaps],
                    },
                },
            )

        return {
            "status": "ok",
            "gaps": [g.to_dict() for g in gaps],
            "modules_to_improve": [
                {
                    "kind": t.kind,
                    "path": t.path,
                    "symbol": t.symbol,
                    "constraints": t.constraints,
                }
                for t in modules_to_improve
            ],
            "self_programming_summary": sp_summary,
        }

    # --------------------------------------------------
    # 内部：用多 AI 议会分析 gap -> 生成 PatchTarget 建议
    # --------------------------------------------------
    def _analyze_gap_with_council(
        self,
        runtime: Any,      # MetaAgentRuntime
        gap: BlueprintGap,
    ) -> List[Dict[str, Any]]:
        """
        对每个 gap 调用一次“多 AI 开会”：
          - 优先使用 runtime.council（多模型议会）
          - 如果没有 council，就退化为单模型 main_llm

        返回统一的 modules_to_improve 列表：
          [
            {
              "kind": "module",
              "path": "mutable/xxx.py",
              "symbol": "SomeClass.method",
              "constraints": {...}
            },
            ...
          ]
        """
        council = getattr(runtime, "council", None)
        llm = council or runtime.model_router

        gap_payload = {
            "requirement_id": gap.requirement_id,
            "category": gap.category,
            "description": gap.description,
            "severity": gap.severity,
            "module_hint": gap.module_hint,
            "symbol_hint": gap.symbol_hint,
            "meta": gap.meta or {},
        }

        # 让议会用 JSON 形式给出建议
        prompt = (
            "你们是一个多智能体评审委员会（包含速度、安全、成本、领域专家等视角），"
            "任务是根据下面这条“蓝皮书实现 gap”给出下一步工程落地建议。\n\n"
            "【输入 gap】(JSON)：\n"
            f"{json.dumps(gap_payload, ensure_ascii=False, indent=2)}\n\n"
            "【要求输出】请只输出一个 JSON 对象，不要带任何说明文字，格式如下：\n"
            "{\n"
            '  "modules_to_improve": [\n'
            "    {\n"
            '      "kind": "module",\n'
            '      "path": "mutable/xxx.py",\n'
            '      "symbol": "Optional[具体函数或类名]",\n'
            '      "constraints": {"reason": "为什么要改这里", "notes": "..."}\n'
            "    },\n"
            "    ...\n"
            "  ]\n"
            "}\n\n"
            "如果暂时无法定位到具体模块，请返回 modules_to_improve: []。"
        )

        try:
            raw = llm(prompt)
        except Exception as e:
            # 出错时回退到“基于 module_hint 的默认建议”
            return self._fallback_suggestions_for_gap(gap, f"LLM error: {e!r}")

        try:
            data = json.loads(str(raw))
        except Exception:
            # 如果议会没按 JSON 格式说话，就回退
            return self._fallback_suggestions_for_gap(gap, "Failed to parse council JSON")

        modules = data.get("modules_to_improve", [])
        normalized: List[Dict[str, Any]] = []
        for m in modules:
            if "path" not in m:
                continue
            normalized.append(
                {
                    "kind": m.get("kind", "module"),
                    "path": m["path"],
                    "symbol": m.get("symbol"),
                    "constraints": m.get("constraints", {}),
                }
            )
        if not normalized:
            normalized = self._fallback_suggestions_for_gap(gap, "empty_council_suggestion")

        return normalized

    def _fallback_suggestions_for_gap(
        self,
        gap: BlueprintGap,
        reason: str,
    ) -> List[Dict[str, Any]]:
        """
        council 失效/不给建议时，用一个非常保守的默认策略：
          - 如果有 module_hint，就把它作为 patch target
          - 否则返回空列表
        """
        if not gap.module_hint:
            return []

        # 把蓝皮书里的 module_hint 映射为 mutable 下的一个候选路径（你也可以更细化）
        path = gap.module_hint
        if not path.startswith("mutable/") and not path.startswith("quentin/"):
            # 允许 spec 里直接写相对路径，例如 'quentin/runtime/cognitive_loop.py'
            # 这里不强行改成 mutable，由你自行管理哪些文件可以 auto_updatable
            pass

        return [
            {
                "kind": "module",
                "path": path,
                "symbol": gap.symbol_hint,
                "constraints": {
                    "from": "fallback",
                    "reason": reason,
                },
            }
        ]

# quentin/research/experiments.py
from __future__ import annotations

from pathlib import Path
from typing import Dict, Any, List
import json

from quentin.eval.harness import EvalHarness, EvalCase, EvalResult


class ExperimentOrchestrator:
    def __init__(self, repo_root: str | Path, variants_runtimes: Dict[str, Any]) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.variants_runtimes = variants_runtimes
        self._experiments_dir = self.repo_root / "data" / "experiments"
        self._experiments_dir.mkdir(parents=True, exist_ok=True)

        # 新增：EvalHarness
        self._eval_harness = EvalHarness(variants_runtimes=self.variants_runtimes)

    # ... 之前的 register_skill_experiment / register_patch_experiment 保持不变 ...

    def run_prompt_eval_experiment(
        self,
        *,
        experiment_id: str,
        cases: List[Dict[str, Any]],
        variants: List[str],
    ) -> List[Dict[str, Any]]:
        """
        运行一个简单的“prompt 评测实验”：

          - cases: list[dict]，每个 dict 必须包含：
              - case_id
              - description
              - input_payload
              - expected_keywords
          - variants: 要评测的 runtime 名称列表（例如 ["baseline", "council"]）
        """
        eval_cases = [
            EvalCase(
                case_id=c["case_id"],
                description=c["description"],
                input_payload=c["input_payload"],
                expected_keywords=c.get("expected_keywords", []),
            )
            for c in cases
        ]

        results = self._eval_harness.run_prompt_eval(eval_cases, variants=variants)

        # 写入 data/experiments/prompt_eval_<experiment_id>.jsonl
        path = self._experiments_dir / f"prompt_eval_{experiment_id}.jsonl"
        tmp = path.with_suffix(".tmp")
        with tmp.open("a", encoding="utf-8") as f:
            for r in results:
                f.write(json.dumps(r.to_dict(), ensure_ascii=False) + "\n")
        tmp.replace(path)

        return [r.to_dict() for r in results]
