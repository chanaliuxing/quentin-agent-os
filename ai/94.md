from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional
from datetime import datetime


@dataclass
class SkillStats:
    """
    Skill 的运行指标，由系统自动更新。
    """
    call_count: int = 0
    success_count: int = 0
    last_used_at: Optional[str] = None
    avg_latency_ms: float = 0.0
    avg_reward: float = 0.0  # 任务收益 / 评分等，可选

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class SkillTestCase:
    """
    Skill 的测试用例。
    注意：蓝皮书要求 test_cases 和 safety_level 需要人类创建 / 审核。
    """
    name: str
    description: str
    input_payload: Dict[str, Any] = field(default_factory=dict)
    expected_output: Dict[str, Any] = field(default_factory=dict)
    importance: str = "normal"      # "smoke" / "normal" / "critical"
    created_by: str = "human"
    created_at: str = datetime.utcnow().isoformat()

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class SkillDefinition:
    """
    稳定的 Skill 定义，保存在 LTS/MTS 中。
    """
    skill_id: str                     # 全局唯一，如 "desktop.click_button"
    name: str                         # 人类可读名称
    version: int                      # 整数版本号
    description: str
    domain: str                       # "general" / "desktop" / "legal" / "trading" / ...
    entry_point: str                  # Python 路径，如 "quentin.skills.desktop.click_button:run"
    tags: List[str] = field(default_factory=list)

    created_at: str = datetime.utcnow().isoformat()
    created_by: str = "system"

    safety_level: str = "medium"      # "low" / "medium" / "high"
    status: str = "active"            # "active" / "deprecated" / "candidate" / "disabled"

    supersedes: Optional[str] = None  # 被替换的 skill_id 或 "skill_id@version"
    superseded_by: Optional[str] = None

    test_cases: List[SkillTestCase] = field(default_factory=list)
    performance_stats: SkillStats = field(default_factory=SkillStats)

    meta: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        return d


@dataclass
class SkillCandidate:
    """
    自演化过程中产生的 Skill 候选（还没正式变成 active Skill）。
    """
    candidate_id: str
    base_skill_id: Optional[str]      # 基于哪个现有 skill 改的，如 "desktop.click_button@3"
    proposed_skill: SkillDefinition   # 新 Skill 定义
    origin: Dict[str, Any] = field(default_factory=dict)  # 来源：trace id / immune_case_id / experiment_id 等
    status: str = "pending_eval"      # "pending_eval" / "evaluating" / "accepted" / "rejected"
    created_at: str = datetime.utcnow().isoformat()
    created_by: str = "self_programming_engine"

    notes: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class SkillSupersedesRecord:
    """
    用于记录旧 Skill 与新 Skill 的替换关系。
    """

    new_skill_ref: str                # 如 "desktop.click_button@4"
    reason: str
    created_at: str = datetime.utcnow().isoformat()
    created_by: str = "self_programming_engine"
    meta: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

from __future__ import annotations

from typing import Any, Dict, List, Optional
from pathlib import Path
import json
import logging

from quentin.skills.schema import (
    SkillDefinition,
    SkillCandidate,
    SkillSupersedesRecord,
)

logger = logging.getLogger(__name__)


class SkillStore:
    """
    v1: 基于文件系统的 Skill 存储。
    将来可替换为 DB / 对象存储。

    目录结构：
    - data/skills/definitions/{skill_id}.json
    - data/skills/candidates/{candidate_id}.json
    - data/skills/supersedes.jsonl
    """

    def __init__(self, *, repo_root: str | Path = ".") -> None:
        self.repo_root = Path(repo_root).resolve()
        self.base_dir = self.repo_root / "data" / "skills"
        self.base_dir.mkdir(parents=True, exist_ok=True)

        self.def_dir = self.base_dir / "definitions"
        self.def_dir.mkdir(parents=True, exist_ok=True)

        self.cand_dir = self.base_dir / "candidates"
        self.cand_dir.mkdir(parents=True, exist_ok=True)

        self.supersedes_path = self.base_dir / "supersedes.jsonl"

    # -------- SkillDefinition -------- #
    def save_skill(self, skill: SkillDefinition) -> None:
        path = self.def_dir / f"{skill.skill_id}.json"
        try:
            path.write_text(
                json.dumps(skill.to_dict(), ensure_ascii=False, indent=2),
                encoding="utf-8",
            )
        except Exception:
            logger.exception("SkillStore: failed to save skill %s", path)

    def load_skill(self, skill_id: str) -> Optional[SkillDefinition]:
        path = self.def_dir / f"{skill_id}.json"
        if not path.exists():
            return None
        try:
            obj = json.loads(path.read_text(encoding="utf-8"))
            from quentin.skills.schema import SkillDefinition, SkillStats, SkillTestCase

            stats = SkillStats(**obj.get("performance_stats", {}))
            test_cases = [SkillTestCase(**tc) for tc in obj.get("test_cases", [])]
            return SkillDefinition(
                skill_id=obj["skill_id"],
                name=obj.get("name", ""),
                version=int(obj.get("version", 1)),
                description=obj.get("description", ""),
                domain=obj.get("domain", "general"),
                entry_point=obj.get("entry_point", ""),
                tags=obj.get("tags", []),
                created_at=obj.get("created_at", ""),
                created_by=obj.get("created_by", "system"),
                safety_level=obj.get("safety_level", "medium"),
                status=obj.get("status", "active"),
                supersedes=obj.get("supersedes"),
                superseded_by=obj.get("superseded_by"),
                test_cases=test_cases,
                performance_stats=stats,
                meta=obj.get("meta", {}),
            )
        except Exception:
            logger.exception("SkillStore: failed to load skill %s", path)
            return None

    def list_skills(self) -> List[SkillDefinition]:
        skills: List[SkillDefinition] = []
        for path in self.def_dir.glob("*.json"):
            s = self.load_skill(path.stem)
            if s is not None:
                skills.append(s)
        return skills

    # -------- SkillCandidate -------- #
    def save_candidate(self, cand: SkillCandidate) -> None:
        path = self.cand_dir / f"{cand.candidate_id}.json"
        try:
            path.write_text(
                json.dumps(cand.to_dict(), ensure_ascii=False, indent=2),
                encoding="utf-8",
            )
        except Exception:
            logger.exception("SkillStore: failed to save candidate %s", path)

    def load_candidate(self, candidate_id: str) -> Optional[SkillCandidate]:
        path = self.cand_dir / f"{candidate_id}.json"
        if not path.exists():
            return None
        try:
            obj = json.loads(path.read_text(encoding="utf-8"))
            from quentin.skills.schema import SkillDefinition

            sd = SkillDefinition(**obj["proposed_skill"])
            return SkillCandidate(
                candidate_id=obj["candidate_id"],
                base_skill_id=obj.get("base_skill_id"),
                proposed_skill=sd,
                origin=obj.get("origin", {}),
                status=obj.get("status", "pending_eval"),
                created_at=obj.get("created_at", ""),
                created_by=obj.get("created_by", "self_programming_engine"),
                notes=obj.get("notes", ""),
            )
        except Exception:
            logger.exception("SkillStore: failed to load candidate %s", path)
            return None

    def list_candidates(self, status: Optional[str] = None, limit: int = 100) -> List[SkillCandidate]:
        cands: List[SkillCandidate] = []
        for path in self.cand_dir.glob("*.json"):
            c = self.load_candidate(path.stem)
            if c is None:
                continue
            if status is not None and c.status != status:
                continue
            cands.append(c)
            if len(cands) >= limit:
                break
        return cands

    # -------- Supersedes -------- #
    def append_supersedes_record(self, rec: SkillSupersedesRecord) -> None:
        line = json.dumps(rec.to_dict(), ensure_ascii=False)
        try:
            with self.supersedes_path.open("a", encoding="utf-8") as f:
                f.write(line + "\n")
        except Exception:
            logger.exception("SkillStore: failed to append supersedes record to %s", self.supersedes_path)

from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional
from datetime import datetime


@dataclass
class PatchTargetRef:
    """
    自编程引擎要修改的目标引用。

    target_type:
      - "skill"  : 针对某个 SkillDefinition（通常是 entry_point 里的函数）
      - "module" : 针对某个 Python 模块
    identifier:
      - 对于 "skill"  : skill_id 或 "skill_id@version"
      - 对于 "module": Python 模块路径，如 "quentin/core/model_router.py"
    """
    target_type: str
    identifier: str
    file_path: str
    symbol_name: Optional[str] = None   # 函数/类名，可选
    meta: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class PatchCandidate:
    """
    自编程引擎生成的 Patch 候选。

    patch_diff 使用 unified diff（字符串）或其它形式，v1 不强制。
    """
    patch_id: str
    target: PatchTargetRef
    description: str
    patch_diff: str                     # 统一 diff 或替代形式
    created_at: str = datetime.utcnow().isoformat()
    created_by: str = "self_programming_engine"
    origin: Dict[str, Any] = field(default_factory=dict)   # 如 immune_case_id / experiment_id / trace_id

    status: str = "pending_tests"       # "pending_tests" / "testing" / "tested" / "accepted" / "rejected"

    notes: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class PatchTestResult:
    """
    单个测试的结果。
    """
    test_name: str
    passed: bool
    details: str = ""
    metrics: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class PatchReport:
    """
    Patch 的整体评估结果。
    """
    patch_id: str
    status: str                         # "passed" / "failed" / "error"
    created_at: str = datetime.utcnow().isoformat()

    tests: List[PatchTestResult] = field(default_factory=list)
    coverage: float = 0.0               # 测试覆盖率（占位）
    risk_score: float = 1.0             # 0~1，越低越安全
    metrics_delta: Dict[str, Any] = field(default_factory=dict)  # 如 success_rate 提升多少

    error: Optional[str] = None
    notes: str = ""
    origin: Dict[str, Any] = field(default_factory=dict)        # 如 {"experiment_id": "..."} 等

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

# quentin/self_programming/engine.py
from __future__ import annotations

from typing import Any, Dict, List, Optional, Iterable
from pathlib import Path
import logging
import uuid

from quentin.skills.schema import SkillDefinition, SkillCandidate, SkillSupersedesRecord
from quentin.skills.store import SkillStore
from quentin.self_programming.schema import (
    PatchTargetRef,
    PatchCandidate,
    PatchReport,
    PatchTestResult,
)
from quentin.self_programming.store import PatchStore
from quentin.research.experiments import ExperimentStore, ExperimentConfig, VariantConfig
from quentin.research.orchestrator import ExperimentOrchestrator

logger = logging.getLogger(__name__)


class SelfProgrammingEngine:
    """
    自编程引擎 v1 骨架：

    职责：
    1. 接收“需要改进的模块 / Skill 列表”（来自 ImmuneLoop / ResearchLayer）
    2. 生成 SkillCandidate / PatchCandidate（调用 LLM，v1 留 TODO）
    3. 调度测试 & 评测（对 PatchCandidate / SkillCandidate 跑单测 / 回归 / Eval）
    4. 输出 PatchReport / SupersedesRecord，供 ResearchLayer 决定是否上线

    注意：
    - 不允许修改 kernel / schema / 风控等区域；
    - 仅操作明确标记为“可自动更新”的目标模块。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        skill_store: Optional[SkillStore] = None,
        patch_store: Optional[PatchStore] = None,
        experiment_store: Optional[ExperimentStore] = None,
        experiment_orchestrator: Optional[ExperimentOrchestrator] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.skill_store = skill_store or SkillStore(repo_root=self.repo_root)
        self.patch_store = patch_store or PatchStore(repo_root=self.repo_root)

        self.experiment_store = experiment_store or ExperimentStore(repo_root=self.repo_root)
        self.orchestrator = experiment_orchestrator or ExperimentOrchestrator(
            repo_root=self.repo_root,
            store=self.experiment_store,
        )

        # 维护一个“允许自动更新”的 target 白名单
        self.mutable_targets: List[PatchTargetRef] = []

    # ---------- 边界 & 注册 ---------- #
    def register_mutable_target(self, target: PatchTargetRef) -> None:
        """
        注册一个允许自动修改的目标。
        比如：
        - Desktop 域的工具模块
        - 非关键路径的 helper 函数
        """
        self.mutable_targets.append(target)

    def is_target_allowed(self, target: PatchTargetRef) -> bool:
        """
        核心安全边界：
        - 禁止修改 kernel / schema / 风控 / 交易执行模块
        - 只允许在 mutable 或特定白名单内
        """
        # 1) 简单规则：禁止 kernel、schema、risk
        forbidden_prefixes = [
            "quentin/kernel/",
            "quentin/core/safety_kernel",
            "quentin/core/risk",
            "quentin/schema/",
        ]
        fp = str(target.file_path)
        for pref in forbidden_prefixes:
            if fp.replace("\\", "/").startswith(pref):
                return False

        # 2) 必须在 mutable_targets 或 target_type="skill"
        if target.target_type == "skill":
            return True

        for t in self.mutable_targets:
            if t.target_type == target.target_type and t.identifier == target.identifier:
                return True

        return False

    # ---------- Skill Candidate 生成 ---------- #
    def propose_skill_candidate_from_trace(
        self,
        *,
        base_skill: Optional[SkillDefinition],
        trace_snippet: Dict[str, Any],
        origin: Dict[str, Any],
    ) -> SkillCandidate:
        """
        从成功 Trace 中挖掘一个 Skill 候选（v1: 留 TODO，先返回 stub）。

        未来逻辑：
        - 用 LLM 分析 trace_snippet（多次成功的步骤模式）
        - 归纳为一个通用 Skill
        - 自动生成代码 & 初始测试用例
        """

        candidate_id = uuid.uuid4().hex
        if base_skill is not None:
            skill_id = base_skill.skill_id
            version = base_skill.version + 1
            name = base_skill.name + " (auto v%d)" % version
            domain = base_skill.domain
        else:
            # 完全新 Skill
            skill_id = "auto.skill." + candidate_id[:8]
            version = 1
            name = f"Auto Skill {candidate_id[:8]}"
            domain = trace_snippet.get("domain", "general")

        proposed = SkillDefinition(
            skill_id=skill_id,
            name=name,
            version=version,
            description="Auto-generated skill candidate from trace.",
            domain=domain,
            entry_point="quentin.skills.auto.%s:run" % candidate_id[:8],
            tags=["auto_generated", "from_trace"],
            created_by="self_programming_engine",
            safety_level=base_skill.safety_level if base_skill else "medium",
            status="candidate",
        )

        cand = SkillCandidate(
            candidate_id=candidate_id,
            base_skill_id=f"{base_skill.skill_id}@{base_skill.version}" if base_skill else None,
            proposed_skill=proposed,
            origin=origin,
            status="pending_eval",
            notes="TODO: fill with real code & tests using LLM.",
        )

        self.skill_store.save_candidate(cand)
        return cand

    # ---------- Patch Candidate 生成 ---------- #
    def propose_patch_candidate(
        self,
        *,
        target: PatchTargetRef,
        issue_summary: Dict[str, Any],
        origin: Dict[str, Any],
    ) -> Optional[PatchCandidate]:
        """
        根据 ImmuneCase / metrics 等，为某个目标生成 Patch 候选（stub）。

        v1 仅做占位：生成一个“空 Patch”，真正 diff 由后续 LLM 填充。
        """
        if not self.is_target_allowed(target):
            logger.warning(
                "SelfProgrammingEngine: target not allowed for auto patch: %s",
                target.to_dict(),
            )
            return None

        patch_id = uuid.uuid4().hex
        description = (
            f"Auto patch candidate for {target.identifier}, "
            f"reason={issue_summary.get('reason', 'unknown')}"
        )

        # v1: patch_diff 留空或写 TODO。后续可以让 LLM 根据现有代码 + issue 生成。
        patch = PatchCandidate(
            patch_id=patch_id,
            target=target,
            description=description,
            patch_diff="# TODO: auto-generated patch diff goes here\n",
            origin=origin,
            status="pending_tests",
        )
        self.patch_store.save_candidate(patch)
        return patch

    # ---------- Patch 测试 & 报告 ---------- #
    def run_tests_for_patch(
        self,
        patch: PatchCandidate,
        *,
        test_suites: Optional[List[str]] = None,
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> PatchReport:
        """
        在 Sandbox 环境跑 patch 的测试（v1 stub）。

        未来逻辑：
        - 在临时工作目录中 apply patch_diff
        - 运行 pytest / 静态分析 / fuzz / EvalHarness 等
        - 收集测试结果和指标
        """
        tests = test_suites or ["unit", "static_analysis"]
        results: List[PatchTestResult] = []

        # v1: 全部标记为未执行，提醒后续补
        for tname in tests:
            results.append(
                PatchTestResult(
                    test_name=tname,
                    passed=False,
                    details="Test not implemented; this is a stub.",
                    metrics={},
                )
            )

        report = PatchReport(
            patch_id=patch.patch_id,
            status="failed",   # 因为没有真正测试
            tests=results,
            coverage=0.0,
            risk_score=1.0,
            metrics_delta={},
            error="Tests not implemented (stub).",
            notes="SelfProgrammingEngine.run_tests_for_patch is not yet implemented.",
            origin=extra_context or {},
        )
        self.patch_store.save_report(report)
        return report

    # ---------- 评估 & 选择 ---------- #
    def select_patches_to_promote(
        self,
        reports: Iterable[PatchReport],
        *,
        max_promote: int = 1,
    ) -> List[PatchReport]:
        """
        根据 PatchReport 选择适合进一步进入 Staging / Canary 的 Patch。
        v1 简单规则：status == "passed" 且 risk_score <= 0.4。
        """
        candidates: List[PatchReport] = []
        for r in reports:
            if r.status == "passed" and r.risk_score <= 0.4:
                candidates.append(r)

        # 简单按 risk_score 排序
        candidates.sort(key=lambda r: r.risk_score)
        return candidates[:max_promote]

    # ---------- 应用 Skill Candidate ---------- #
    def accept_skill_candidate(
        self,
        cand: SkillCandidate,
        *,
        reason: str,
    ) -> SkillSupersedesRecord:
        """
        将 SkillCandidate 变成正式 Skill，并记录 supersedes 关系。
        注意：真正的“代码替换”仍需部署系统 / 人类 Gate。
        """
        skill = cand.proposed_skill
        base_ref = cand.base_skill_id

        # 更新 skill 状态并保存
        skill.status = "active"
        self.skill_store.save_skill(skill)

        # 更新 candidate 状态
        cand.status = "accepted"
        self.skill_store.save_candidate(cand)

        # 记录 supersedes
        if base_ref:
            rec = SkillSupersedesRecord(
                old_skill_ref=base_ref,
                new_skill_ref=f"{skill.skill_id}@{skill.version}",
                reason=reason,
                created_by="self_programming_engine",
                meta={"candidate_id": cand.candidate_id},
            )
            self.skill_store.append_supersedes_record(rec)
        else:
            rec = SkillSupersedesRecord(
                old_skill_ref="",
                new_skill_ref=f"{skill.skill_id}@{skill.version}",
                reason=reason,
                created_by="self_programming_engine",
                meta={"candidate_id": cand.candidate_id},
            )
            self.skill_store.append_supersedes_record(rec)

        return rec

# quentin/evolution/skill_loop.py
from __future__ import annotations

from typing import Any, Dict, List, Optional
import logging

from quentin.self_programming.engine import SelfProgrammingEngine
from quentin.self_programming.schema import PatchTargetRef, PatchCandidate, PatchReport
from quentin.skills.schema import SkillDefinition, SkillCandidate

logger = logging.getLogger(__name__)


class SkillEvolutionLoop:
    """
    Skill & Code Evolution Loop v1 骨架（对应蓝皮书 3.4）：

    输入：
    - 高成功率 Trace（来自 CognitiveLoop / TraceStore）
    - ImmuneCase（表现差的模块 / Skill）
    - MetaLayer/Research 提供的“需要重点优化的组件列表”

    输出：
    - 新 SkillCandidate（从成功模式中抽取）
    - PatchCandidate（针对问题模块生成）
    - PatchReport（测试和评估结果）
    - SupersedesRecord（替换旧 Skill）

    v1：只提供结构和关键方法调用，内部逻辑先留 TODO。
    """

    def __init__(self, engine: SelfProgrammingEngine) -> None:
        self.engine = engine

    # ---- 从成功 Trace 中抽取 Skill ---- #
    def mine_skills_from_traces(
        self,
        successful_traces: List[Dict[str, Any]],
    ) -> List[SkillCandidate]:
        """
        根据高成功率 Trace 生成一批 SkillCandidate（stub）。
        真实实现可以使用：
        - pattern mining
        - LLM 对 Trace 的 summary + 抽象
        """
        candidates: List[SkillCandidate] = []

        for tr in successful_traces:
            # v1: 只针对标记为 "skill_seed": true 的 trace，避免乱生成
            if not tr.get("skill_seed"):
                continue
            base_skill: Optional[SkillDefinition] = None
            origin = {
                "type": "trace",
                "trace_id": tr.get("trace_id"),
            }
            cand = self.engine.propose_skill_candidate_from_trace(
                base_skill=base_skill,
                trace_snippet=tr,
                origin=origin,
            )
            candidates.append(cand)

        return candidates

    # ---- 从 ImmuneCase 生成 Patch ---- #
    def generate_patches_from_issues(
        self,
        issue_list: List[Dict[str, Any]],
    ) -> List[PatchCandidate]:
        """
        issue_list 典型来源：
        - ImmuneCase（高失败率模块）
        - ResearchLayer 标记的 "需要优化的工具"
        """
        patches: List[PatchCandidate] = []

        for issue in issue_list:
            target_info = issue.get("target") or {}
            target = PatchTargetRef(
                target_type=target_info.get("target_type", "module"),
                identifier=target_info.get("identifier", ""),
                file_path=target_info.get("file_path", ""),
                symbol_name=target_info.get("symbol_name"),
                meta=target_info.get("meta", {}),
            )
            origin = {
                "type": "issue",
                "issue_id": issue.get("issue_id"),
                "source": issue.get("source", "immune"),
            }
            issue_summary = {
                "reason": issue.get("reason", ""),
                "metrics": issue.get("metrics", {}),
            }
            cand = self.engine.propose_patch_candidate(
                target=target,
                issue_summary=issue_summary,
                origin=origin,
            )
            if cand is not None:
                patches.append(cand)

        return patches

    # ---- 测试 Patch 并生成 Report ---- #
    def evaluate_patches(
        self,
        patches: List[PatchCandidate],
        *,
        test_suites: Optional[List[str]] = None,
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> List[PatchReport]:
        """
        遍历 PatchCandidate，调用 SelfProgrammingEngine.run_tests_for_patch。
        v1: 仍然是 stub 测试。
        """
        reports: List[PatchReport] = []
        for p in patches:
            report = self.engine.run_tests_for_patch(
                patch=p,
                test_suites=test_suites,
                extra_context=extra_context,
            )
            reports.append(report)
        return reports

    # ---- 顶层 run_once：单次演化轮次 ---- #
    def run_once(
        self,
        *,
        successful_traces: List[Dict[str, Any]],
        issues: List[Dict[str, Any]],
        test_suites: Optional[List[str]] = None,
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        一次完整的演化轮次（可由 SleepLoop / ResearchLayer 触发）：

        1. 从成功 Trace 抽取 SkillCandidate
        2. 从 Immune issue 生成 PatchCandidate
        3. 对 Patch 进行测试 & 生成 PatchReport
        4. 根据 PatchReport 选出少量“值得推进”的 Patch（当前只返回，真正上线仍需上层决策）
        """
        logger.info(
            "SkillEvolutionLoop.run_once: traces=%d issues=%d",
            len(successful_traces),
            len(issues),
        )

        skill_candidates = self.mine_skills_from_traces(successful_traces)
        patch_candidates = self.generate_patches_from_issues(issues)
        patch_reports = self.evaluate_patches(
            patch_candidates,
            test_suites=test_suites,
            extra_context=extra_context,
        )

        # 根据报告选择“可推进”的 patch（进入 Staging / Canary）
        promotable = self.engine.select_patches_to_promote(patch_reports, max_promote=extra_context.get("max_promote", 1) if extra_context else 1)

        return {
            "skill_candidates": [c.to_dict() for c in skill_candidates],
            "patch_candidates": [p.to_dict() for p in patch_candidates],
            "patch_reports": [r.to_dict() for r in patch_reports],
            "promotable_patches": [r.to_dict() for r in promotable],
        }
接下来只需要一个调度点（比如 SleepLoop）定期调用 SkillEvolutionLoop.run_once(...)，
就形成了：“执行 → Immune 标记问题 → Research / Sleep 调用演化 → 生成候选 → 等待 Gate / 宪法层审批”的闭环。
