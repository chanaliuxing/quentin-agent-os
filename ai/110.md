@dataclass
class LLMProbeResult:
    """
    LLM / 多模型会议 对某个 capability 的判断结果。
    """
    capability_id: str
    status_guess: str
    confidence: float
    evidence: str
    raw_answer: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class LLMBasedCapabilityProbe:
    """
    基于 LLM / MultiModelCouncil 的能力探测器。

    用法：
      probe = LLMBasedCapabilityProbe(repo_root=".", llm=..., council=...)
      result = probe.probe_capability(capability)
    """

    def __init__(
        self,
        *,
        repo_root: str | Path,
        llm: Optional[Any] = None,
        council: Optional[MultiModelCouncil] = None,
        max_bytes: int = 16_000,
    ) -> None:
        if llm is None and council is None:
            raise ValueError("Either llm or council must be provided for LLMBasedCapabilityProbe.")
        self.repo_root = Path(repo_root).resolve()
        self.llm = llm
        self.council = council
        self.max_bytes = max_bytes

    # --------------------------------------------------
    # 对外：批量探测
    # --------------------------------------------------
    def probe_all(
        self,
        capabilities: List[BlueprintCapability],
        *,
        only_cap_ids: Optional[List[str]] = None,
    ) -> List[LLMProbeResult]:
        results: List[LLMProbeResult] = []
        for cap in capabilities:
            if only_cap_ids is not None and cap.id not in only_cap_ids:
                continue
            try:
                pr = self.probe_capability(cap)
            except Exception:
                logger.exception("LLMBasedCapabilityProbe: error probing cap=%s", cap.id)
                continue
            if pr is not None:
                results.append(pr)
        return results

    # --------------------------------------------------
    # 对外：单条能力探测
    # --------------------------------------------------
    def probe_capability(self, cap: BlueprintCapability) -> Optional[LLMProbeResult]:
        file_paths, code_snippet = self._collect_code_context(cap.id)
        prompt = self._build_prompt(cap, file_paths, code_snippet)

        if self.council is not None:
            deliberation = self.council.deliberate(
                question=prompt,
                context={
                    "task": "blueprint_capability_probe",
                    "capability_id": cap.id,
                    "section": cap.section,
                },
            )
            raw = deliberation.final_answer
        else:
            raw = _call_llm(self.llm, prompt)

        if raw is None:
            return None

        try:
            data = json.loads(raw)
        except Exception:
            logger.warning(
                "LLMBasedCapabilityProbe: LLM answer not valid JSON for cap=%s, raw=%.200r",
                cap.id,
                raw,
            )
            return None

        # 期待的 JSON 字段：
        # {
        #   "status_guess": "done|partial|missing|unknown",
        #   "confidence": 0.0~1.0,
        #   "evidence": "..."
        # }
        status_guess = str(data.get("status_guess", "unknown")).lower()
        if status_guess not in ("done", "partial", "missing", "unknown"):
            status_guess = "unknown"

        try:
            confidence = float(data.get("confidence", 0.5))
        except Exception:
            confidence = 0.5
        # clamp
        confidence = max(0.0, min(1.0, confidence))

        evidence = str(data.get("evidence", "")).strip()
        if not evidence:
            evidence = "No detailed evidence provided by LLM."

        return LLMProbeResult(
            capability_id=cap.id,
            status_guess=status_guess,
            confidence=confidence,
            evidence=evidence,
            raw_answer=raw,
        )

    # --------------------------------------------------
    # 内部：收集代码上下文
    # --------------------------------------------------
    def _collect_code_context(self, capability_id: str) -> tuple[List[str], str]:
        hints = CAPABILITY_FILE_HINTS.get(capability_id, [])
        existing_files: List[str] = []
        chunks: List[str] = []

        remaining = self.max_bytes

        for rel in hints:
            path = self.repo_root / rel
            if not path.exists() or not path.is_file():
                continue
            try:
                b = path.read_bytes()
            except Exception:
                continue

            if remaining <= 0:
                break

            snippet_bytes = b[: remaining]
            remaining -= len(snippet_bytes)
            try:
                snippet = snippet_bytes.decode("utf-8", errors="replace")
            except Exception:
                snippet = ""
            existing_files.append(str(path.relative_to(self.repo_root)))
            chunks.append(f"# File: {path.relative_to(self.repo_root)}\n{snippet}\n")

        if not existing_files:
            return [], "## No code files found for this capability.\n"

        return existing_files, "\n\n".join(chunks)

    # --------------------------------------------------
    # 内部：Prompt 设计（中英混合）
    # --------------------------------------------------
    def _build_prompt(
        self,
        cap: BlueprintCapability,
        file_paths: List[str],
        code_snippet: str,
    ) -> str:
        cap_json = json.dumps(cap.to_dict(), ensure_ascii=False, indent=2)
        files_desc = ", ".join(file_paths) if file_paths else "（未找到相关文件）"

        return (
            "你是一个经验丰富的系统架构师 + 代码审查专家，任务是：\n"
            "根据『蓝皮书能力定义』以及『给定的代码片段』，判断这个能力在当前代码仓库中是否已经被实现。\n\n"
            "【蓝皮书能力定义 Blueprint Capability】\n"
            f"{cap_json}\n\n"
            "【可能相关的代码文件列表】\n"
            f"{files_desc}\n\n"
            "【相关代码片段 Code Snippet（可能被截断，仅供你判断）】\n"
            "---------------- CODE BEGIN ----------------\n"
            f"{code_snippet}\n"
            "---------------- CODE END ------------------\n\n"
            "请你给出极为简洁的、机器可读的判断结果，用 JSON 格式输出：\n"
            "{\n"
            '  \"status_guess\": \"done\" | \"partial\" | \"missing\" | \"unknown\",\n'
            '  \"confidence\": 0.0~1.0,\n'
            '  \"evidence\": \"用1-2句话说明你为什么这么判断，可以中英文混合\"\n'
            "}\n\n"
            "要求：\n"
            "  - 不要输出任何解释性文字，只输出一个 JSON 对象。\n"
            "  - 如果代码基本符合蓝皮书要求，选 \"done\" 并给出相对高的 confidence（>=0.7）。\n"
            "  - 如果只实现了部分关键点，选 \"partial\"。\n"
            "  - 如果基本找不到相关实现，选 \"missing\" 或 \"unknown\"（取决于你对判断的把握）。\n"
        )


def _call_llm(llm: Any, prompt: str) -> Optional[str]:
    """
    通用 LLM 调用适配器：
      - llm.complete(prompt=...)
      - llm.generate(prompt=...)
      - llm(prompt)
    """
    if llm is None:
        return None

    try:
        if hasattr(llm, "complete"):
            return llm.complete(prompt=prompt)
        if hasattr(llm, "generate"):
            return llm.generate(prompt=prompt)
        if callable(llm):
            return llm(prompt)
    except Exception:
        logger.exception("LLMBasedCapabilityProbe: error calling LLM")
        return None

    raise RuntimeError("LLM client has no complete/generate/callable interface")

2. 升级 scan_and_update_registry：聚合所有 probe
替换原来的 scan_and_update_registry 为下面这个版本（保留类名不变）：

    def scan_and_update_registry(self) -> CapabilityScanReport:
        """
        扫描所有能力项：

          - 静态规则（import / 文件检查）
          - 可选：LLM / 多模型会议 语义判断

        然后按能力 id 聚合所有 probe 结果，选出“最佳判断”，更新 CapabilitiesRegistry。

        聚合策略（简化版）：
          - status 优先级：done > partial > missing > unknown
          - 同一 status 下，confidence 高的优先
        """
        static_results = self._run_static_import_probes()
        llm_results: List[CapabilityProbeResult] = []

        if self._llm_probe is not None:
            # 先拿到当前 registry 中“还未 done”的能力 id，
            # 避免 LLM 在已经确认 done 的能力上过度消耗。
            not_done_ids: List[str] = []
            for cap in self.spec.capabilities:
                cs = self.registry.get(cap.id)
                if cs is None or cs.status != "done":
                    not_done_ids.append(cap.id)

            llm_raw_results = self._llm_probe.probe_all(
                self.spec.capabilities,
                only_cap_ids=not_done_ids,
            )

            # 转成 CapabilityProbeResult 统一类型
            for r in llm_raw_results:
                llm_results.append(
                    CapabilityProbeResult(
                        capability_id=r.capability_id,
                        status_guess=r.status_guess,
                        confidence=r.confidence,
                        evidence=f"[llm] {r.evidence}",
                        source="llm-probe",
                    )
                )

        # 合并两个来源的结果
        all_results: List[CapabilityProbeResult] = static_results + llm_results

        # 按能力 id 聚类
        by_cap: Dict[str, List[CapabilityProbeResult]] = {}
        for pr in all_results:
            by_cap.setdefault(pr.capability_id, []).append(pr)

        # 逐能力选出最佳判断并更新 registry
        for cap in self.spec.capabilities:
            cap_id = cap.id
            probes = by_cap.get(cap_id, [])
            if not probes:
                # 没有任何 probe 结果，则不强行覆盖现有状态
                continue

            best = self._select_best_probe(probes)

            current = self.registry.get(cap_id)
            current_status = current.status if current is not None else "unknown"

            new_status = current_status
            # 统一的“覆盖策略”：只要新判断比旧状态“更明确”，就覆盖
            if best.status_guess == "done":
                new_status = "done"
            elif best.status_guess == "partial":
                if current_status not in ("done", "partial"):
                    new_status = "partial"
            elif best.status_guess in ("missing", "unknown"):
                if current_status == "unknown":
                    new_status = best.status_guess

            if new_status != current_status:
                self.registry.set_status(
                    cap_id,
                    new_status,
                    evidence=best.evidence,
                )

        return CapabilityScanReport(
            spec_version=self.spec.version,
            repo_root=str(self.repo_root),
            probe_results=all_results,
        )
再在类里面增加一个辅助方法 _select_best_probe：

    def _select_best_probe(
        self,
        probes: List[CapabilityProbeResult],
    ) -> CapabilityProbeResult:
        """
        从多条 probe 结果中选出一个“最佳判断”：

          - 按 status 优先级排序：done > partial > missing > unknown
          - 同一 status 下，confidence 越高越优先
        """
        status_rank = {
            "done": 3,
            "partial": 2,
            "missing": 1,
            "unknown": 0,
        }

        def _key(p: CapabilityProbeResult) -> tuple[int, float]:
            return (status_rank.get(p.status_guess, 0), p.confidence)

        return max(probes, key=_key)
这样：

静态规则 + 多 AI 会议的判断会被“统一融合”，

如果 LLM 非常确定某能力是 done（status=done, confidence 高），就可以把之前的 unknown/missing 覆盖掉。

    def auto_update_capabilities_from_repo(
        self,
        *,
        repo_root: Optional[str | Path] = None,
    ) -> Dict[str, Any]:
        """
        使用 CapabilityScanner 对当前 repo 做一次扫描，
        自动更新 CapabilitiesRegistry（哪些蓝皮书能力已实现/部分实现）。

        这里会同时利用：
          - 静态规则（import / 文件存在性）
          - 如果可用：LLM / MultiModelCouncil 语义判断

          - scan_report: 本次扫描的详细结果（所有 probe）
          - registry_snapshot: 更新后的能力状态快照

        root = Path(repo_root or self.repo_root).resolve()

            repo_root=root,

            council=self.council,    # 优先使用多模型会议

        scan_report: CapabilityScanReport = scanner.scan_and_update_registry()

        return {
            "scan_report": scan_report.to_dict(),
            "registry_snapshot": self.capabilities_registry.export(),
        }
