# quentin/observability/service.py

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Optional
from collections import Counter
from datetime import datetime

from quentin.runtime.agent_runtime import AgentRuntime
from quentin.immune.schemas import ImmuneCaseStatus
# 下面这些 import 的类都已经在之前步骤里定义过或约定过
# 如果你的路径稍有不同，只需要对应调整即可。


@dataclass
class ObservabilityService:
    """
    系统统一观测与研究接口：
    - 模型路由表现（ModelRouter）
    - 记忆规模与层级（MemoryFabric）
    - 自编程 patch 情况（SelfProgrammingEngine）
    - 实验系统（ExperimentOrchestrator）
    - 免疫系统（ImmuneLoop）
    - 技能库（SkillLibrary）
    """

    runtime: AgentRuntime

    # ---------- Model / Routing 统计 ----------

    async def get_model_stats(self, limit: int = 2000) -> Dict[str, Any]:
        """
        从 ModelRouter 获取聚合后的模型使用情况。
        依赖：ModelRouter 提供 get_stats(limit=...) 方法。
        """
        router = getattr(self.runtime, "model_router", None)
        if router is None or not hasattr(router, "get_stats"):
            return {"models": {}, "note": "model_router or get_stats not available"}

        stats = await router.get_stats(limit=limit)
        # 约定 stats = {"models": {name: {...}}, "total_calls": ...}
        return stats

    # ---------- Memory / Fabric 统计 ----------

    async def get_memory_stats(self) -> Dict[str, Any]:
        """
        MemoryFabric 提供 stats() 方法，返回分层统计。
        如果暂时没有实现 stats()，就返回空。
        """
        memory = getattr(self.runtime, "memory", None)
        if memory is None:
            return {"tiers": {}, "note": "memory fabric not available"}

        if not hasattr(memory, "stats"):
            return {"tiers": {}, "note": "memory.stats() not implemented"}

        data = await memory.stats()
        # 建议 data 至少包括：
        # {
        #   "STS": {"count": ..., "hit_rate": ...},
        #   "MTS": {...},
        #   "LTS": {...}
        # }
        return data

    # ---------- Immune / 异常系统统计 ----------

    async def get_immune_stats(self, limit: int = 200) -> Dict[str, Any]:
        loop = getattr(self.runtime, "immune_loop", None)
        if loop is None or not hasattr(loop, "list_cases"):
            return {"cases": 0, "note": "immune_loop not available"}

        cases = await loop.list_cases(limit=limit)

        total = len(cases)
        if total == 0:
            return {
                "cases": 0,
                "by_status": {},
                "demo_cases": 0,
                "last_case": None,
            }

        by_status = Counter(c.status.value for c in cases)
        demo_cases = [c for c in cases if c.domain == "demo"]
        last = cases[-1]

        return {
            "cases": total,
            "by_status": dict(by_status),
            "demo_cases": len(demo_cases),
            "last_case": {
                "case_id": last.case_id,
                "status": last.status.value,
                "domain": last.domain,
                "title": last.title,
                "created_at": last.created_at.isoformat(),
            },
        }

    # ---------- 自编程 patch 统计 ----------

    async def get_patch_stats(self, limit: int = 200) -> Dict[str, Any]:
        """
        依赖：SelfProgrammingEngine 提供 list_patches(limit=...) 方法。
        如果你当前引擎还没实现 list_patches，可以先返回空字典。
        """
        engine = getattr(self.runtime, "self_programming", None)
        if engine is None or not hasattr(engine, "list_patches"):
            return {
                "patches": 0,
                "by_status": {},
                "note": "self_programming.list_patches() not available",
            }

        patches = await engine.list_patches(limit=limit)
        # 约定 patches 是 PatchRecord 列表（pydantic 模型）
        total = len(patches)
        if total == 0:
            return {"patches": 0, "by_status": {}, "applied": 0, "rolled_back": 0}

        from collections import Counter

        by_status = Counter(p.status.value for p in patches if hasattr(p, "status"))

        applied = sum(1 for p in patches if getattr(p, "status", None) == "applied")
        rolled_back = sum(
            1 for p in patches if getattr(p, "status", None) == "rolled_back"
        )

        return {
            "patches": total,
            "by_status": dict(by_status),
            "applied": applied,
            "rolled_back": rolled_back,
        }

    # ---------- 实验系统统计 ----------

    async def get_experiment_stats(self, limit: int = 200) -> Dict[str, Any]:
        """
        依赖：ExperimentOrchestrator 提供 list_experiments(limit=...) 方法，
        返回 ExperimentRecord 列表。
        """
        orch = getattr(self.runtime, "experiment_orchestrator", None)
        if orch is None or not hasattr(orch, "list_experiments"):
            return {
                "experiments": 0,
                "note": "experiment_orchestrator.list_experiments() not available",
            }

        exps = await orch.list_experiments(limit=limit)
        total = len(exps)
        if total == 0:
            return {"experiments": 0, "by_status": {}, "last_experiment": None}

        by_status = Counter(getattr(e, "status", "unknown") for e in exps)
        last = exps[-1]

        return {
            "experiments": total,
            "by_status": dict(by_status),
            "last_experiment": {
                "experiment_id": getattr(last, "experiment_id", None),
                "name": getattr(last, "name", None),
                "status": getattr(last, "status", None),
            },
        }

    # ---------- Skill / 技能库统计 ----------

    async def get_skill_stats(self, limit: int = 200) -> Dict[str, Any]:
        lib = getattr(self.runtime, "skill_library", None)
        if lib is None or not hasattr(lib, "list_skills"):
            return {
                "skills": 0,
                "note": "skill_library not available",
            }

        skills = await lib.list_skills(limit=limit)
        total = len(skills)
        if total == 0:
            return {"skills": 0, "by_kind": {}, "by_domain": {}}

        by_kind = Counter(s.kind.value for s in skills)
        by_domain = Counter((s.domain or "unknown") for s in skills)

        return {
            "skills": total,
            "by_kind": dict(by_kind),
            "by_domain": dict(by_domain),
        }

    # ---------- SleepLoop / Episodes 简要统计（可选） ----------

    async def get_sleep_stats(self, limit_episodes: int = 50) -> Dict[str, Any]:
        sleep_loop = getattr(self.runtime, "sleep_loop", None)
        provider = getattr(sleep_loop, "episodes_provider", None) if sleep_loop else None

        if provider is None or not hasattr(provider, "get_recent_episodes"):
            return {
                "episodes": 0,
                "note": "sleep_loop.episodes_provider not available",
            }

        episodes = await provider.get_recent_episodes(limit=limit_episodes)
        total = len(episodes)
        domains = Counter((ep.get("domain") or "unknown") for ep in episodes)

        return {
            "episodes": total,
            "by_domain": dict(domains),
        }

    # ---------- 总览 ----------

    async def get_overview(self) -> Dict[str, Any]:
        """
        汇总各子系统的统计信息。
        给 Dashboard / 前端 / 研究脚本使用。
        """
        # 并行优化可以以后用 asyncio.gather，这里先顺序调用，简单清晰
        model = await self.get_model_stats()
        memory = await self.get_memory_stats()
        immune = await self.get_immune_stats()
        patches = await self.get_patch_stats()
        exps = await self.get_experiment_stats()
        skills = await self.get_skill_stats()
        sleep = await self.get_sleep_stats()

        now = datetime.utcnow().isoformat()

        return {
            "timestamp": now,
            "model": model,
            "memory": memory,
            "immune": immune,
            "patches": patches,
            "experiments": exps,
            "skills": skills,
            "sleep": sleep,
        }
这里所有方法都严格只读，不改动任何状态，方便挂到任何地方（HTTP / CLI / Notebook）。

二、Observability API：FastAPI HTTP 服务
新建：quentin/observability/api.py

# quentin/observability/api.py

from __future__ import annotations

from typing import Optional

from fastapi import FastAPI, Depends

from quentin.runtime.settings import Settings
from quentin.runtime.agent_runtime import AgentRuntime
from quentin.observability.service import ObservabilityService


def create_app(settings: Optional[Settings] = None) -> FastAPI:
    """
    创建一个 FastAPI 应用，用于暴露 Observability HTTP 接口。

    用法（命令行）：
        uvicorn quentin.observability.api:create_app --factory --reload
    或在 qctl 里提供 serve-observability 命令（下面会给出）。
    """
    app = FastAPI(
        title="Project Quentin Observability API",
        version="0.1.0",
        description="Observability endpoints for the meta-agent system.",
    )

    settings = settings or Settings()
    runtime = AgentRuntime.create_from_settings(settings)
    obs_service = ObservabilityService(runtime=runtime)

    def get_obs() -> ObservabilityService:
        return obs_service

    # --------- 基本健康检查 ---------

    @app.get("/health")
    async def health():
        return {"status": "ok"}

    # --------- 各子系统指标 ---------

    @app.get("/overview")
    async def overview(service: ObservabilityService = Depends(get_obs)):
        return await service.get_overview()

    @app.get("/models")
    async def model_stats(service: ObservabilityService = Depends(get_obs)):
        return await service.get_model_stats()

    @app.get("/memory")
    async def memory_stats(service: ObservabilityService = Depends(get_obs)):
        return await service.get_memory_stats()

    @app.get("/immune")
    async def immune_stats(service: ObservabilityService = Depends(get_obs)):
        return await service.get_immune_stats()

    @app.get("/patches")
    async def patch_stats(service: ObservabilityService = Depends(get_obs)):
        return await service.get_patch_stats()

    @app.get("/experiments")
    async def experiment_stats(service: ObservabilityService = Depends(get_obs)):
        return await service.get_experiment_stats()

    @app.get("/skills")
    async def skill_stats(service: ObservabilityService = Depends(get_obs)):
        return await service.get_skill_stats()

    @app.get("/sleep")
    async def sleep_stats(service: ObservabilityService = Depends(get_obs)):
        return await service.get_sleep_stats()

    return app
这样任何人 clone 这个 repo 后，只要装上 fastapi + uvicorn，就可以跑一个本地监控 API，Grafana / 前端直接 HTTP 拉数据。

@cli.command("metrics-overview")
def metrics_overview_cmd():
    """
    打印当前系统的 Observability 总览（单次）。

    用于：
    - 快速检查当前模型 / 记忆 / patch / 实验 / 免疫 / 技能的整体情况
    - 做离线研究时 snapshot 一份 JSON

    async def _run():
        import json

        settings = Settings()
        runtime = AgentRuntime.create_from_settings(settings)

        from quentin.observability.service import ObservabilityService

        service = ObservabilityService(runtime=runtime)
        overview = await service.get_overview()

        print(json.dumps(overview, ensure_ascii=False, indent=2))

    asyncio.run(_run())

@cli.command("serve-observability")
@click.option("--host", default="127.0.0.1", show_default=True)
@click.option("--port", default=8000, show_default=True)
def serve_observability_cmd(host: str, port: int):
    """
    启动 Observability HTTP 服务（FastAPI + Uvicorn）。

    启动后可以访问：
    - GET /health
    - GET /overview
    - GET /models
    - GET /memory
    - GET /immune
    - GET /patches
    - GET /experiments
    - GET /skills
    - GET /sleep
    """

    import uvicorn
    from quentin.observability.api import create_app
    from quentin.runtime.settings import Settings

    settings = Settings()
    app = create_app(settings=settings)

    uvicorn.run(app, host=host, port=port)
