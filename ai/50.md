@dataclass
class PatchTestCaseResult:
    """
    单条测试用例结果（这里既可以表示单元测试，也可以表示一次整体命令的结果）。
    """
    name: str                       # 用例名或命令名，如 "pytest::all" / "unit.test_xxx"
    status: str                     # "passed" | "failed" | "error" | "skipped" | "not_run"
    duration_sec: Optional[float] = None
    error_message: Optional[str] = None
    extra: Dict[str, Any] = field(default_factory=dict)


@dataclass
class PatchTestReport:
    """
    针对某个 CodePatch 的测试报告。

    v1 只记录基本统计：
    - 成功/失败/错误/跳过数量
    - 总耗时
    - 静态分析结果（如 flake8/mypy/ruff 等，可后续扩展）
    """
    report_id: str
    patch_id: str
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    success: bool = False
    total_cases: int = 0
    passed: int = 0
    failed: int = 0
    errored: int = 0
    skipped: int = 0
    runtime_sec: Optional[float] = None

    static_analysis: Dict[str, Any] = field(default_factory=dict)
    cases: List[PatchTestCaseResult] = field(default_factory=list)

    def to_dict(self) -> Dict[str, Any]:
        return {
            "report_id": self.report_id,
            "patch_id": self.patch_id,
            "created_at": self.created_at,
            "success": self.success,
            "total_cases": self.total_cases,
            "passed": self.passed,
            "failed": self.failed,
            "errored": self.errored,
            "skipped": self.skipped,
            "runtime_sec": self.runtime_sec,
            "static_analysis": self.static_analysis,
            "cases": [asdict(c) for c in self.cases],
        }

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "PatchTestReport":
        cases = [
            PatchTestCaseResult(**c)
            for c in data.get("cases", [])
        ]
        return cls(
            report_id=data["report_id"],
            patch_id=data["patch_id"],
            created_at=data.get("created_at", ""),
            success=data.get("success", False),
            total_cases=data.get("total_cases", 0),
            passed=data.get("passed", 0),
            failed=data.get("failed", 0),
            errored=data.get("errored", 0),
            skipped=data.get("skipped", 0),
            runtime_sec=data.get("runtime_sec"),
            static_analysis=data.get("static_analysis", {}),
            cases=cases,
        )

新文件：quentin/self_programming/testing.py

这里做一个 v1 最小实现：

# quentin/self_programming/testing.py
from __future__ import annotations

import logging
import subprocess
import time
from datetime import datetime
from pathlib import Path
from typing import Optional, List, Dict, Any, Union

from quentin.self_programming.patches import (
    CodePatch,
    PatchTestCaseResult,
    PatchTestReport,
)

logger = logging.getLogger(__name__)


class BasePatchTestRunner:
    """
    Patch 测试执行器接口。
    后续你可以实现不同 runner：
    - 调 pytest
    - 调 tox / nox
    - 调自定义脚本
    - 甚至在 Docker/K8s 里跑
    """

    def run_for_patch(self, patch: CodePatch) -> PatchTestReport:
        raise NotImplementedError


class NoopPatchTestRunner(BasePatchTestRunner):
    """
    占位实现：标记“未执行任何测试”。
    用于本地还没配置测试时防止整个流程崩掉。
    """

    def run_for_patch(self, patch: CodePatch) -> PatchTestReport:
        report_id = f"{patch.patch_id}::tests::noop"
        case = PatchTestCaseResult(
            name="noop",
            status="not_run",
            duration_sec=None,
            error_message="No tests executed (NoopPatchTestRunner).",
            extra={},
        )
        report = PatchTestReport(
            report_id=report_id,
            patch_id=patch.patch_id,
            success=False,
            total_cases=1,
            passed=0,
            failed=0,
            errored=0,
            skipped=0,
            runtime_sec=None,
            static_analysis={},
            cases=[case],
        )
        return report


class ShellCommandPatchTestRunner(BasePatchTestRunner):
    """
    使用 shell 命令执行测试。
    默认命令：pytest -q

    注意：v1 并不会自动应用 CodePatch，只是针对当前代码库跑测试。
    以后可以在调用前后加上“临时应用补丁 + 回滚”的逻辑。
    """

    def __init__(
        self,
        repo_root: Union[str, Path] = ".",
        command: str = "pytest -q",
        env: Optional[Dict[str, str]] = None,
    ) -> None:
        self.repo_root = Path(repo_root)
        self.command = command
        self.env = env

    def run_for_patch(self, patch: CodePatch) -> PatchTestReport:
        report_id = f"{patch.patch_id}::tests::shell::{int(time.time())}"
        start_ts = time.perf_counter()

        case_name = f"shell:{self.command}"

        try:
            logger.info(
                "Running shell test command for patch %s: %s (cwd=%s)",
                patch.patch_id,
                self.command,
                self.repo_root,
            )
            proc = subprocess.run(
                self.command,
                shell=True,
                cwd=str(self.repo_root),
                env=self.env,
                capture_output=True,
                text=True,
            )
            end_ts = time.perf_counter()
            duration = end_ts - start_ts

            status = "passed" if proc.returncode == 0 else "failed"

            # 截断输出，避免 JSON 过大
            stdout = (proc.stdout or "")[:4000]
            stderr = (proc.stderr or "")[:4000]

            case = PatchTestCaseResult(
                name=case_name,
                status=status,
                duration_sec=duration,
                error_message=None if proc.returncode == 0 else (stderr or stdout),
                extra={
                    "returncode": proc.returncode,
                    "stdout_preview": stdout,
                    "stderr_preview": stderr,
                },
            )

            success = proc.returncode == 0
            report = PatchTestReport(
                report_id=report_id,
                patch_id=patch.patch_id,
                success=success,
                total_cases=1,
                passed=1 if success else 0,
                failed=0 if success else 1,
                errored=0,
                skipped=0,
                runtime_sec=duration,
                static_analysis={},
                cases=[case],
            )
            return report

        except Exception as e:
            end_ts = time.perf_counter()
            duration = end_ts - start_ts

            logger.exception("Error running shell test command for patch %s", patch.patch_id)
            case = PatchTestCaseResult(
                name=case_name,
                status="error",
                duration_sec=duration,
                error_message=str(e),
                extra={},
            )
            report = PatchTestReport(
                report_id=report_id,
                patch_id=patch.patch_id,
                success=False,
                total_cases=1,
                passed=0,
                failed=0,
                errored=1,
                skipped=0,
                runtime_sec=duration,
                static_analysis={},
                cases=[case],
            )
            return report

然后在 SelfProgrammingEngine 类里加方法：

class SelfProgrammingEngine:
    # ... 你之前的 __init__ / propose_patch_for_design / review_patch_with_council / list_patches 等保持不变

    # -------- Patch 测试主入口 --------

    def test_patch(
        self,
        patch_id: str,
        runner: Optional[BasePatchTestRunner] = None,
    ) -> CodePatch:
        """
        使用给定的 PatchTestRunner 对补丁进行测试，并把 PatchTestReport
        写回 CodePatch.tests。

        默认使用 NoopPatchTestRunner（也就是不实际跑测试），
        需要你在 CLI 或调用处显式传入 ShellCommandPatchTestRunner 才会真正跑 pytest 等。
        """
        patch = self.patch_store.get(patch_id)
        if not patch:
            raise ValueError(f"CodePatch not found: {patch_id}")

        runner = runner or NoopPatchTestRunner()
        report: PatchTestReport = runner.run_for_patch(patch)

        patch.tests = report
        patch.updated_at = datetime.utcnow().isoformat()

        # 这里暂时不改变 patch.status（仍然是 generated / reviewed 等），
        # 后面你可以根据测试结果 + 评审结果共同决定是否标记为 approved。
        self.patch_store.upsert(patch)

        logger.info(
            "Tests finished for patch %s: success=%s, total_cases=%s, failed=%s, errored=%s",
            patch.patch_id,
            report.success,
            report.total_cases,
            report.failed,
            report.errored,
        )
        return patch

定义 handler 函数 cmd_patch_test：

def cmd_patch_test(args: argparse.Namespace) -> None:
    engine = SelfProgrammingEngine(
        blueprint_path=args.blueprint,
    )

    mode = args.mode
    if mode == "shell":
        runner = ShellCommandPatchTestRunner(
            repo_root=args.repo_root,
            command=args.cmd,
        )
    else:
        runner = NoopPatchTestRunner()

    patch = engine.test_patch(
        patch_id=args.patch_id,
        runner=runner,
    )

    print(f"Patch tested: {patch.patch_id}")
    if patch.tests:
        r = patch.tests
        print(f"Success: {r.success}")
        print(f"Total cases: {r.total_cases}, passed={r.passed}, failed={r.failed}, errored={r.errored}, skipped={r.skipped}")
        if r.cases:
            print("---- Cases ----")
            for c in r.cases:
                print(f"- {c.name}: {c.status} (duration={c.duration_sec})")
                if c.error_message:
                    print(f"  error: {c.error_message[:200]}")
    else:
        print("No test report attached to patch.")
