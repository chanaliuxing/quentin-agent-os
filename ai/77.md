# ---------- 数据结构 ----------

@dataclass
class PatchTaskSpec:
    """
    一次自编程任务的规范描述（蓝皮书：自编程引擎输入）：

    - target_path     : 需要修改的 Python 文件（相对 repo 根目录）
    - reason          : 为什么要改？（来自 ImmuneLoop / SleepLoop / 人类）
    - domain          : core / desktop / legal / trading / ...
    - environment     : sandbox / staging / prod
    - risk_level      : low / medium / high / critical
    - context_snippet : 可选的上下文（trace 摘要、错误日志、性能指标等）
    """

    target_path: str
    reason: str
    domain: str = "core"
    environment: str = "sandbox"
    risk_level: str = "high"
    context_snippet: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class PatchTestResult:
    passed: bool
    command: str
    stdout: str
    stderr: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class PatchGateResult:
    approved: bool
    council_summary: str
    council_raw_answer: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class PatchResult:
    """
    自编程一次完整流程的结果（蓝皮书：Patch Report 的简化版）

    - success           : 最终是否成功应用 patch
    - applied           : 是否已经写入文件
    - guard             : SafetyKernel 的结果
    - test_result       : 测试结果（可为空）
    - gate_result       : 多 AI 审批结果（可为空）
    - new_code_preview  : 新代码前几行预览
    - error             : 失败时的错误信息
    """

    task_id: str
    target_path: str
    success: bool
    applied: bool
    guard: GuardResult
    test_result: Optional[PatchTestResult]
    gate_result: Optional[PatchGateResult]
    new_code_preview: str
    error: Optional[str] = None
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


# ---------- PatchGate：多 AI 审批 ----------

class PatchGateCouncil:
    """
    多 AI 审批委员会：

    - 输入：旧代码摘要 / 新代码摘要 / diff / 测试结果
    - 输出：是否批准 + 简要说明

    这里用 JSON 协议，方便后续自动分析。
    """

    def __init__(self, council: Optional[MultiAICouncil] = None, router: Optional[ModelRouter] = None) -> None:
        if council is None:
            router = router or ModelRouter()
            # 用 router 选模型，这里角色标上 patch_gate，方便之后做路由优化
            cfg = router.route(mode="critic", task_type="code_patch")
            members = [
                LLMConfig(provider=cfg.provider, model=cfg.model, role="patch_gate_reviewer_1"),
                LLMConfig(provider=cfg.provider, model=cfg.model, role="patch_gate_reviewer_2"),
            ]
            council = MultiAICouncil(members=members)
        self.council = council

    def review(
        self,
        *,
        target_path: str,
        reason: str,
        diff_text: str,
        test_result: Optional[PatchTestResult],
    ) -> PatchGateResult:
        context_parts = [
            f"TARGET PATH: {target_path}",
            "",
            f"REASON:\n{reason}",
            "",
            "UNIFIED DIFF (old -> new):",
            diff_text,
        ]
        if test_result is not None:
            context_parts.append("")
            context_parts.append("TEST RESULT:")
            context_parts.append(f"command: {test_result.command}")
            context_parts.append(f"passed : {test_result.passed}")
            context_parts.append("stdout :")
            context_parts.append(test_result.stdout[:2000])
            context_parts.append("stderr :")
            context_parts.append(test_result.stderr[:2000])

        context = "\n".join(context_parts)

        question = (
            "You are a PATCH REVIEW COUNCIL for a self-evolving AI system.\n"
            "Given the patch diff and test results, you must decide whether to approve it.\n\n"
            "You MUST return ONLY one JSON object with the following fields:\n"
            "{\n"
            "  \"approved\": true/false,\n"
            "  \"summary\": \"short human-readable explanation\"\n"
            "}\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="patch_gate",
        )
        ans = decision.final_answer or "{}"

        try:
            s = ans.strip()
            first = s.find("{")
            last = s.rfind("}")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)
            approved = bool(data.get("approved", False))
            summary = str(data.get("summary", "") or "")
        except Exception:
            logger.exception("PatchGateCouncil.review: failed to parse council JSON: %r", ans)
            approved = False
            summary = "Failed to parse council answer; default to not approved."

        return PatchGateResult(
            approved=approved,
            council_summary=summary,
            council_raw_answer=ans,
        )


# ---------- SelfProgrammingEngine 主体 ----------

class SelfProgrammingEngine:
    """
    SelfProgrammingEngine v1：

    一个“受控自改代码”流水线：

    1. 根据 PatchTaskSpec 构造 ActionRequest，调用 SafetyKernel.guard_action
       - 宪法检查 + 预算检查
    2. 如果被拒绝，直接结束
    3. 用 LLM 生成新代码（整文件重写策略）
    4. 生成 unified diff
    5. 运行基础测试（可选，可配置）
    6. 把 diff + 测试结果交给 PatchGateCouncil（多 AI 审批）
    7. 如果批准，写文件 + 记录结果到 ResearchManager + 写 ExecutionTrace
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        safety_kernel: Optional[SafetyKernel] = None,
        model_router: Optional[ModelRouter] = None,
        patch_gate: Optional[PatchGateCouncil] = None,
        research_manager: Optional[ResearchManager] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.safety_kernel = safety_kernel or SafetyKernel()
        self.model_router = model_router or ModelRouter()
        self.patch_gate = patch_gate or PatchGateCouncil(router=self.model_router)
        self.research_manager = research_manager or ResearchManager()

    # ---- 对外主入口 ----

    def propose_and_apply_patch(self, task: PatchTaskSpec) -> PatchResult:
        task_id = str(uuid.uuid4())
        target_file = (self.repo_root / task.target_path).resolve()

        # 1) SafetyKernel 审核
        estimated_tokens = self._estimate_tokens_for_task(task, target_file)
        action = ActionRequest.new(
            actor="selfprog_engine",
            action_type="code_patch",
            domain=task.domain,
            environment=task.environment,
            risk_level=task.risk_level,
            estimated_cost_tokens=estimated_tokens,
            estimated_cost_usd=0.0,  # 本地模型，先写 0
            description=task.reason,
            metadata={
                "target_path": task.target_path,
                "task_id": task_id,
            },
        )
        guard = self.safety_kernel.guard_action(action)
        if not guard.allow:
            return PatchResult(
                task_id=task_id,
                target_path=task.target_path,
                success=False,
                applied=False,
                guard=guard,
                test_result=None,
                gate_result=None,
                new_code_preview="",
                error=f"Denied by SafetyKernel: {guard.reason}",
            )

        # 2) 读取旧代码
        try:
            old_code = target_file.read_text(encoding="utf-8")
        except FileNotFoundError:
            return PatchResult(
                task_id=task_id,
                target_path=task.target_path,
                success=False,
                applied=False,
                guard=guard,
                test_result=None,
                gate_result=None,
                new_code_preview="",
                error=f"Target file not found: {target_file}",
            )

        # 3) LLM 生成新代码（全文件重写策略）
        new_code = self._generate_new_code(task=task, old_code=old_code)

        # 4) 生成 diff
        diff_text = self._make_unified_diff(
            old_code=old_code,
            new_code=new_code,
            filename=str(task.target_path),
        )

        # 5) 运行基础测试（v1 简单调用 pytest，失败不一定阻止 patch，但会影响 Gate 决策）
        test_result = self._run_basic_tests(task)

        # 6) 多 AI 审批
        gate_result = self.patch_gate.review(
            target_path=task.target_path,
            reason=task.reason,
            diff_text=diff_text,
            test_result=test_result,
        )

        if not gate_result.approved:
            result = PatchResult(
                task_id=task_id,
                target_path=task.target_path,
                success=False,
                applied=False,
                guard=guard,
                test_result=test_result,
                gate_result=gate_result,
                new_code_preview="\n".join(new_code.splitlines()[:20]),
                error=f"Patch not approved by PatchGateCouncil: {gate_result.council_summary}",
            )
            self._record_research(task, result, diff_text)
            self._record_trace(task, result, diff_text)
            return result

        # 7) 写入文件（备份 .bak）
        backup_file = target_file.with_suffix(target_file.suffix + ".bak")
        try:
            if not backup_file.exists():
                backup_file.write_text(old_code, encoding="utf-8")
            target_file.write_text(new_code, encoding="utf-8")
        except Exception as e:
            result = PatchResult(
                task_id=task_id,
                target_path=task.target_path,
                success=False,
                applied=False,
                guard=guard,
                test_result=test_result,
                gate_result=gate_result,
                new_code_preview="\n".join(new_code.splitlines()[:20]),
                error=f"Failed to write patched file: {e}",
            )
            self._record_research(task, result, diff_text)
            self._record_trace(task, result, diff_text)
            return result

        # 8) 成功结果 + 研究记录 + trace
        result = PatchResult(
            task_id=task_id,
            target_path=task.target_path,
            success=True,
            applied=True,
            guard=guard,
            test_result=test_result,
            gate_result=gate_result,
            new_code_preview="\n".join(new_code.splitlines()[:20]),
            error=None,
        )
        self._record_research(task, result, diff_text)
        self._record_trace(task, result, diff_text)
        return result

    # ---- 内部：估算 token / prompt ----

    def _estimate_tokens_for_task(self, task: PatchTaskSpec, target_file: Path) -> int:
        """
        非精确估算：用文件行数 + context 长度做个近似即可，主要用于预算控制。
        """
        try:
            txt = target_file.read_text(encoding="utf-8")
            n_lines = txt.count("\n") + 1
        except FileNotFoundError:
            n_lines = 100
        ctx_len = len(task.context_snippet or "")
        # 很粗糙的估算：1 行 ≈ 10 token，context 每 4 字符 ≈ 1 token
        return int(n_lines * 10 + ctx_len / 4)

    def _generate_new_code(self, *, task: PatchTaskSpec, old_code: str) -> str:
        cfg = self.model_router.route(mode="executor", task_type="code_patch")

        system = (
            "You are a SELF-PROGRAMMING ENGINE agent.\n"
            "Your job is to REWRITE the entire Python file to improve or fix it,\n"
            "based on the given reason and context.\n\n"
            "REQUIREMENTS:\n"
            "- Preserve public APIs unless obviously wrong.\n"
            "- Keep imports consistent, avoid adding unused ones.\n"
            "- Make code clear and well-structured.\n"
            "- Respond ONLY with the full new file content (pure code, no explanation).\n"
        )

        user = textwrap.dedent(f"""
        TARGET FILE PATH:
        {task.target_path}

        REASON FOR CHANGE:
        {task.reason}

        OPTIONAL CONTEXT (may be empty):
        {task.context_snippet or "(none)"}

        CURRENT FILE CONTENT:
        ```python
        {old_code}
        ```
        """)

        raw = call_llm(cfg, user_prompt=user, system_prompt=system)
        # 简单裁剪掉可能出现的 ```python ... ``` 包裹
        text = raw.strip()
        if "```" in text:
            # 找到第一对代码 fence
            parts = text.split("```")
            # 形如: ["", "python\ncode...", ""]
            if len(parts) >= 2:
                # 去掉可能的语言标签
                body = parts[1]
                if "\n" in body:
                    body = body.split("\n", 1)[1]
                text = body
        return text.strip()

    def _make_unified_diff(self, *, old_code: str, new_code: str, filename: str) -> str:
        old_lines = old_code.splitlines(keepends=True)
        new_lines = new_code.splitlines(keepends=True)
        diff = difflib.unified_diff(
            old_lines,
            new_lines,
            fromfile=f"{filename} (old)",
            tofile=f"{filename} (new)",
        )
        return "".join(diff)

    def _run_basic_tests(self, task: PatchTaskSpec) -> PatchTestResult:
        """
        v1：尝试运行 `pytest -q`，如果失败就把日志传给 PatchGateCouncil。
        将来可以按文件映射更精确地跑局部测试。
        """
        cmd = ["pytest", "-q"]
        try:
            proc = subprocess.run(
                cmd,
                cwd=self.repo_root,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                text=True,
                timeout=300,
            )
            passed = proc.returncode == 0
            return PatchTestResult(
                passed=passed,
                command=" ".join(cmd),
                stdout=proc.stdout,
                stderr=proc.stderr,
            )
        except FileNotFoundError:
            # 没装 pytest，当作未跑测试，但不阻止 patch
            return PatchTestResult(
                passed=True,
                command="pytest (not installed; tests skipped)",
                stdout="pytest not found; tests skipped.",
                stderr="",
            )
        except Exception as e:
            return PatchTestResult(
                passed=False,
                command=" ".join(cmd),
                stdout="",
                stderr=f"Exception while running tests: {e}",
            )

    # ---- 内部：写入 Research & Trace ----

    def _record_research(self, task: PatchTaskSpec, result: PatchResult, diff_text: str) -> None:
        rm = self.research_manager
        capability = "self_programming:code_patch"
        requirement_id = "patch_success_rate"

        exps = rm.list_experiments()
        exp = None
        for e in exps:
            if e.capability == capability and e.requirement_id == requirement_id:
                exp = e
                break
        if exp is None:
            exp = rm.create_experiment(
                name="SelfProgrammingEngine",
                purpose="self_programming",
                capability=capability,
                requirement_id=requirement_id,
                tags=["selfprog", "code_patch"],
            )

        metrics = {
            "success": 1.0 if result.success else 0.0,
            "applied": 1.0 if result.applied else 0.0,
        }
        if result.test_result is not None:
            metrics["tests_passed"] = 1.0 if result.test_result.passed else 0.0

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics=metrics,
            context={
                "task": task.to_dict(),
                "diff_preview": diff_text[:2000],
            },
        )

    def _record_trace(self, task: PatchTaskSpec, result: PatchResult, diff_text: str) -> None:
        from quentin.core.traces import ExecutionTraceStore  # 延迟 import 避免循环

        store = ExecutionTraceStore()
        trace = ExecutionTrace(
            trace_id=str(uuid.uuid4()),
            task_id=result.task_id,
            component_id="selfprog_engine.v1",
            blueprint_capability_id="self_programming_engine",
            target_module=str(task.target_path),
            target_symbol="SelfProgrammingEngine.propose_and_apply_patch",
            success=result.success,
            error_type=None if result.success else "patch_failed",
            error_message=result.error,
            start_time=result.created_at,
            end_time=datetime.utcnow().isoformat(),
            metrics={
                "applied": 1.0 if result.applied else 0.0,
                "tests_passed": 1.0
                if (result.test_result and result.test_result.passed)
                else 0.0,
            },
            tags={
                "domain": task.domain,
                "environment": task.environment,
                "risk_level": task.risk_level,
            },
        )
        store.append(trace)

def cmd_selfprog_patch(args: argparse.Namespace) -> None:
    engine = SelfProgrammingEngine()
    task = PatchTaskSpec(
        target_path=args.target,
        reason=args.reason,
        domain=args.domain,
        environment=args.environment,
        risk_level=args.risk_level,
    )
    result = engine.propose_and_apply_patch(task)

    print(f"Task ID    : {result.task_id}")
    print(f"Target     : {result.target_path}")
    print(f"Success    : {result.success}")
    print(f"Applied    : {result.applied}")
    print(f"Guard      : {result.guard.reason}")
    print(f"Policy ok? : {result.guard.policy_decision.allow}")
    print(f"Budget ok? : {result.guard.budget_decision.allow}")
    if result.test_result:
        print(f"Tests pass : {result.test_result.passed}")
    if result.gate_result:
        print(f"Gate appr? : {result.gate_result.approved}")
        print(f"Gate summary:\n{result.gate_result.council_summary}\n")

    if result.error:
        print("Error:")
        print(result.error)
    else:
        print("\nNew code preview (first 20 lines):\n")
        print(result.new_code_preview)
三、现在「自我迭代」这一块已经怎么运转？
到这一步，你的 repo 里已经实现了一个 可跑的、自带安全内核的自编程流程：

任何人 / 任何系统都可以触发自编程实验：

qctl selfprog-patch \
  --target quentin/core/cognitive_loop.py \
  --reason "Improve error handling and logging for failed planner JSON."
背后实际上跑了这一整条链路：

PatchTaskSpec 描述自编程任务；

ActionRequest → SafetyKernel.guard_action()

PolicyEngine 检查宪法：

这个 domain 是否允许自编程？

risk_level 在当前 environment 是否允许？

BudgetController 检查配额：

当天 & 本月 token / 费用是否超标？

通过后，SelfProgrammingEngine：

读取旧代码；

用本地 LLM（通过 ModelRouter）生成新代码；

生成 diff；

尝试跑 pytest（可以以后精细化）；

把 diff + 测试结果交给 PatchGateCouncil（多 AI 审批）；

只有通过 Gate 才真正写入文件，并备份旧版；

最后把整个过程写入：

ResearchManager（Experiment: SelfProgrammingEngine）

ExecutionTraceStore（用于 ImmuneLoop、SleepLoop 继续分析）。

自改代码也变成“可实验、可回滚、可观测”的一部分：

你可以用：

qctl exp-list 看自编程实验；

qctl exp-runs 看每次 patch 的 success / tests_passed；

qctl exp-compare 比较不同 variant（以后可以有不同自编程策略）；

qctl policy-show / qctl budget-show 检查安全与配额。

如果你再说「继续落地」，我会优先做两件事之一（都直接对照蓝皮书补 gap）：

在 ImmuneLoop / SleepLoop 里真正接入 SelfProgrammingEngine：

当某个组件长期 error rate 较高时，ImmuneLoop 自动拼出一个 PatchTaskSpec，调用 SelfProgrammingEngine；

SleepLoop 从成功 Trace 抽象出技能后，也可以触发对 mutable/skills/*.py 的自编程改写；

这些动作全部走 SafetyKernel & ResearchManager。

实现一个通用的 Goal & Definition-of-Done 模块（多 AI 开会判定“任务算不算完成”），让 repo 里的 Agent 能够在没有人工参与的情况下，对“成功/失败/何时停止”做更清晰的自我判断，同时把这个 DoD 也纳入实验与演化体系。
