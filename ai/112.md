一、新文件：Ollama LLM 客户端
文件：quentin/llm/ollama_client.py

这是一个很薄的封装，让上层只关心 complete(prompt) 和 __call__(prompt)。

# quentin/llm/ollama_client.py
from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Optional
import json
import logging
import requests

logger = logging.getLogger(__name__)


@dataclass
class OllamaModelConfig:
    name: str              # 模型名，如 "qwen3-vl:latest"
    base_url: str = "http://localhost:11434"
    # 你可以在这里扩展 temperature / top_p / num_ctx / num_predict 等参数
    temperature: float = 0.3
    top_p: float = 0.9
    num_ctx: int = 8192
    num_predict: int = 1024


class OllamaLLMClient:
    """
    一个简单的 Ollama HTTP 客户端封装，提供：
      - complete(prompt)
      - chat(messages)
      - __call__(prompt)
    """

    def __init__(self, config: OllamaModelConfig) -> None:
        self.config = config
        self._generate_url = f"{self.config.base_url}/api/generate"
        self._chat_url = f"{self.config.base_url}/api/chat"

    # -----------------------------
    # 核心：纯文本 prompt -> 文本输出
    # -----------------------------
    def complete(self, prompt: str, **extra_params: Any) -> str:
        """
        对应 Ollama /api/generate 接口。
        """
        payload: Dict[str, Any] = {
            "model": self.config.name,
            "prompt": prompt,
            "stream": False,
            "options": {
                "temperature": self.config.temperature,
                "top_p": self.config.top_p,
                "num_ctx": self.config.num_ctx,
                "num_predict": self.config.num_predict,
            },
        }
        # 允许上层按需覆盖参数
        if extra_params:
            payload.update(extra_params)

        try:
            resp = requests.post(self._generate_url, json=payload, timeout=600)
            resp.raise_for_status()
        except Exception as e:
            logger.exception("OllamaLLMClient.complete error: %s", e)
            raise

        data = resp.json()
        # /api/generate 的响应字段是 "response"
        text = data.get("response", "")
        return text

    # -----------------------------
    # Chat 模式（可选，用不到可以先不用）
    # -----------------------------
    def chat(self, messages: List[Dict[str, str]], **extra_params: Any) -> str:
        """
        对应 /api/chat 接口，messages 是 OpenAI 风格：
          [{"role": "system", "content": "..."}, {"role": "user", "content": "..."}]
        """
        payload: Dict[str, Any] = {
            "model": self.config.name,
            "messages": messages,
            "stream": False,
            "options": {
                "temperature": self.config.temperature,
                "top_p": self.config.top_p,
                "num_ctx": self.config.num_ctx,
                "num_predict": self.config.num_predict,
            },
        }
        if extra_params:
            payload.update(extra_params)

        try:
            resp = requests.post(self._chat_url, json=payload, timeout=600)
            resp.raise_for_status()
        except Exception as e:
            logger.exception("OllamaLLMClient.chat error: %s", e)
            raise

        data = resp.json()
        # /api/chat 返回 {"message": {"role": "...", "content": "..."}}
        msg = data.get("message") or {}
        return msg.get("content", "")

    # -----------------------------
    # 兼容之前我们写的“llm(prompt)”风格
    # -----------------------------
    def __call__(self, prompt: str, **extra_params: Any) -> str:
        return self.complete(prompt, **extra_params)

def create_runtime_for_blueprint_self_iteration(repo_root: str | Path = ".") -> MetaAgentRuntime:
    """
    最小可运行版本的 MetaAgentRuntime，用本地 Ollama 模型 + 极简 Memory/Tool/Critic。

    目标：让蓝皮书自迭代流水线可以在本地跑通，不绑定具体业务域。
    """
    repo_root = Path(repo_root).resolve()

    # 1) 主 LLM（给 runtime 直接用的 model_router）
    #    这里用 qwen3-vl:latest 作为“主脑”
    main_llm = OllamaLLMClient(
        OllamaModelConfig(
            name="qwen3-vl:latest",
            base_url="http://localhost:11434",
            temperature=0.3,
            top_p=0.9,
            num_ctx=8192,
            num_predict=1024,
        )
    )

    # 2) 多 AI 会议面板（MultiModelCouncil）
    council = create_default_ollama_council(base_url="http://localhost:11434")

    # 3) Memory Fabric —— 先用一个简单实现
    #    你可以在 quentin/memory/simple_fabric.py 里写一个基于 JSON/SQLite 的简单三层存储
    memory_fabric = SimpleMemoryFabric(
        repo_root=repo_root,
    )

    # 4) ToolManager —— 先空实现，后面再逐步接入真正工具
    tool_manager = ToolManager()

    # 5) Critic —— 做基础 self-reflection / 评分
    critic = SimpleCritic(
        llm=main_llm,
    )

    # 6) 先构造两个 runtime 变体（用于 A/B 测试）：
    #    - baseline: 不用 council，只用 main_llm
    #    - council: 规划/反省等重要环节用多模型会议
    baseline_runtime = MetaAgentRuntime(
        model_router=main_llm,
        memory_fabric=memory_fabric,
        tool_manager=tool_manager,
        critic=critic,
        council=None,
        repo_root=repo_root,
        research_layer=None,  # 稍后注入
    )

    council_runtime = MetaAgentRuntime(
        model_router=main_llm,
        memory_fabric=memory_fabric,
        tool_manager=tool_manager,
        critic=critic,
        council=council,
        repo_root=repo_root,
        research_layer=None,  # 稍后注入
    )

    variants_runtimes = {
        "baseline": baseline_runtime,
        "council": council_runtime,
    }

    # 7) ExperimentOrchestrator + ResearchLayer
    exp_orch = ExperimentOrchestrator(
        repo_root=repo_root,
        variants_runtimes=variants_runtimes,
    )
    research_layer = ResearchLayer(
        repo_root=repo_root,
        experiment_orchestrator=exp_orch,
    )

    # 8) 把 research_layer 注入到两个 runtime 中
    baseline_runtime.attach_research_layer(research_layer)
    council_runtime.attach_research_layer(research_layer)

    # 9) 作为“主中控 runtime”，我们选用 council 版本
    return council_runtime

# quentin/memory/simple_fabric.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import threading


@dataclass
class MemoryItem:
    id: str
    tier: str          # "STS" | "MTS" | "LTS"
    content: str
    meta: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class SimpleMemoryFabric:
    """
    一个极简版本的 Memory Fabric：
      - 只存 JSON 文件
      - 通过 tier 字段区分 STS / MTS / LTS
    """

    def __init__(self, repo_root: str | Path) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.path = self.repo_root / "data" / "memory" / "fabric.jsonl"
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._lock = threading.Lock()

    # 真正的实现会有复杂检索，这里只给最小 CRUD 接口

    def append(self, item: MemoryItem) -> None:
        with self._lock, self.path.open("a", encoding="utf-8") as f:
            f.write(json.dumps(item.to_dict(), ensure_ascii=False) + "\n")

    def list_all(self) -> List[MemoryItem]:
        if not self.path.exists():
            return []
        out: List[MemoryItem] = []
        with self.path.open("r", encoding="utf-8") as f:
            for line in f:
                try:
                    data = json.loads(line)
                    out.append(MemoryItem(**data))
                except Exception:
                    continue
        return out

    # 预留：向量检索/分层迁移等，以后蓝皮书自迭代可以自己来扩展

# quentin/critic/core.py
from __future__ import annotations

from typing import Any, Dict


class SimpleCritic:
    """
    极简 Critic，占位实现：
      - 通过 LLM 对一次执行的 summary 做简单打分 / 反思
      - 以后可以由蓝皮书自迭代去升级为更复杂的多指标 Critic
    """

    def __init__(self, llm: Any) -> None:
        self.llm = llm

    def evaluate(self, summary: str, context: Dict[str, Any] | None = None) -> Dict[str, Any]:
        prompt = (
            "你是一个批判性思考的评审者，请对下面这段执行总结做简短评价：\n\n"
            f"{summary}\n\n"
            "请用 JSON 输出：\n"
            "{\n"
            '  \"score\": 0~1 的浮点数,\n'
            '  \"comment\": \"一句话点评\"\n'
            "}\n"
            "不要输出其它文字。"
        )
        try:
            raw = self.llm(prompt)
        except Exception:
            return {"score": 0.5, "comment": "LLM critic error."}

        # 为了简单，不强制解析 JSON，留给后续迭代来优化
        return {"score": 0.5, "comment": raw[:200]}
