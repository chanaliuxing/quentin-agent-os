from __future__ import annotations

from typing import Dict, Any, List, Optional
from datetime import datetime
from enum import Enum
from pydantic import BaseModel, Field


class ImmuneCaseStatus(str, Enum):
    open = "open"
    investigating = "investigating"
    mitigated = "mitigated"
    ignored = "ignored"


class ImmuneCase(BaseModel):
    """
    Immune Loop 发现的“异常事件包”：
    - 一段时间内 error / latency 等异常
    - 对应的 trace_id 列表
    - 后续可以被 Research Layer / 自编程引擎使用
    """
    case_id: str

    window_start: datetime
    window_end: datetime

    trace_ids: List[str] = Field(default_factory=list)
    error_trace_ids: List[str] = Field(default_factory=list)

    # 聚合指标
    total_tasks: int = 0
    error_tasks: int = 0
    avg_latency_sec: float = 0.0

    # 暂时用字符串列表占位，后续可以换成结构化 suspects
    suspected_modules: List[str] = Field(default_factory=list)

    status: ImmuneCaseStatus = ImmuneCaseStatus.open
    created_at: datetime = Field(default_factory=datetime.utcnow)
    meta: Dict[str, Any] = Field(default_factory=dict)

# ============================================
# quentin/immune/loop.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass
from typing import Dict, Any, List, Optional
from datetime import datetime, timedelta
import uuid
import json

from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.kernel.safety import SafetyKernel
from quentin.immune.schemas import ImmuneCase, ImmuneCaseStatus
from quentin.cognitive.schemas import ExecutionSummary


@dataclass
class ImmuneLoop:
    """
    Immune Loop v1：

    - 从 ExecutionSummary 中计算最近一段时间的 error_rate / latency
    - 如果超过阈值，创建 ImmuneCase
    - 标记 SafetyKernel 进入 degraded 状态（占位）
    - 后续可以把 ImmuneCase 交给 Research / SelfProgramming 处理
    """
    kv: KVStore
    tracer: Tracer
    safety_kernel: SafetyKernel

    summary_prefix: str = "exec:summary:"
    summary_index_key: str = "exec:index:summary"
    immune_case_prefix: str = "immune:case:"
    immune_case_index_key: str = "immune:index:case"

    async def _load_summary(self, trace_id: str) -> Optional[ExecutionSummary]:
        data = await self.kv.get(f"{self.summary_prefix}{trace_id}")
        if not data:
            return None
        return ExecutionSummary(**data)

    async def _save_case(self, case: ImmuneCase) -> None:
        key = f"{self.immune_case_prefix}{case.case_id}"
        await self.kv.save(key, case.model_dump())
        await self.kv.append_list(self.immune_case_index_key, case.case_id)

    async def list_cases(self, limit: int = 50) -> List[ImmuneCase]:
        ids = await self.kv.get(self.immune_case_index_key) or []
        ids = list(dict.fromkeys(ids))[-limit:]
        res: List[ImmuneCase] = []
        for cid in ids:
            data = await self.kv.get(f"{self.immune_case_prefix}{cid}")
            if data:
                res.append(ImmuneCase(**data))
        return res

    async def get_case(self, case_id: str) -> Optional[ImmuneCase]:
        data = await self.kv.get(f"{self.immune_case_prefix}{case_id}")
        return ImmuneCase(**data) if data else None

    async def scan_recent(
        self,
        window_minutes: int = 60,
        error_rate_threshold: float = 0.3,
        min_tasks: int = 10,
    ) -> Optional[ImmuneCase]:
        """
        扫描最近 window_minutes 内的 ExecutionSummary：
        - total_tasks < min_tasks -> 不触发
        - error_rate >= 阈值 -> 生成 ImmuneCase + 调整 SafetyState（降级）
        """
        now = datetime.utcnow()
        window_start = now - timedelta(minutes=window_minutes)

        trace_ids = await self.kv.get(self.summary_index_key) or []
        trace_ids = list(dict.fromkeys(trace_ids))[-200:]  # 最近 200 条足够做监控

        summaries: List[ExecutionSummary] = []
        for tid in trace_ids:
            s = await self._load_summary(tid)
            if not s:
                continue
            if s.finished_at < window_start:
                continue
            summaries.append(s)

        total = len(summaries)
        if total < min_tasks:
            # 数据太少，不做任何动作
            self.tracer.span(
                "L2",
                "immune_scan_skip_insufficient_data",
                module="immune_loop",
                input={"total": total, "min_tasks": min_tasks},
            )
            return None

        error_summaries = [s for s in summaries if not s.success]
        error_count = len(error_summaries)
        error_rate = error_count / total if total else 0.0
        avg_latency = (
            sum(s.metrics.get("latency_sec", 0.0) for s in summaries) / total
        )

        if error_rate < error_rate_threshold:
            # 正常范围内
            self.tracer.span(
                "L2",
                "immune_scan_normal",
                module="immune_loop",
                input={
                    "total": total,
                    "error_rate": error_rate,
                    "threshold": error_rate_threshold,
                    "avg_latency_sec": avg_latency,
                },
            )
            return None

        # 触发 ImmuneCase
        case_id = f"immune_{uuid.uuid4().hex}"
        case = ImmuneCase(
            case_id=case_id,
            window_start=window_start,
            window_end=now,
            trace_ids=[s.trace_id for s in summaries],
            error_trace_ids=[s.trace_id for s in error_summaries],
            total_tasks=total,
            error_tasks=error_count,
            avg_latency_sec=avg_latency,
            suspected_modules=[],  # v1 先不细分模块
            status=ImmuneCaseStatus.open,
            meta={
                "error_rate": error_rate,
                "error_rate_threshold": error_rate_threshold,
                "window_minutes": window_minutes,
            },
        )
        await self._save_case(case)

        # 通知 SafetyKernel：进入降级模式（实现可以很简单）
        await self.safety_kernel.mark_degraded(
            reason="high_error_rate",
            details={
                "case_id": case_id,
                "error_rate": error_rate,
                "total": total,
                "error_tasks": error_count,
            },
        )

        self.tracer.span(
            "L2",
            "immune_case_created",
            module="immune_loop",
            input={
                "case_id": case_id,
                "error_rate": error_rate,
                "total": total,
                "error_tasks": error_count,
            },
        )

        return case

# ============================================
# quentin/sleep/loop.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass
from typing import Dict, Any, List, Optional
from datetime import datetime, timedelta
import uuid

from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.memory.fabric import MemoryFabric
from quentin.memory.schemas import MemoryItem, MemoryTier
from quentin.cognitive.schemas import ExecutionSummary
from quentin.runtime.model_router import ModelRouter


@dataclass
class SleepLoop:
    """
    Sleep / Consolidation Loop v1：

    每次运行做三件事：
    1) 选取最近一段时间的 ExecutionSummary
    2) 为每条 summary 生成一个简单“语义记忆” (semantic_summary)
    3) 调 MemoryFabric.run_consolidation 做晋升 / 遗忘
    """
    kv: KVStore
    tracer: Tracer
    memory: MemoryFabric
    model_router: ModelRouter

    summary_prefix: str = "exec:summary:"
    summary_index_key: str = "exec:index:summary"

    async def _load_summary(self, trace_id: str) -> Optional[ExecutionSummary]:
        data = await self.kv.get(f"{self.summary_prefix}{trace_id}")
        if not data:
            return None
        return ExecutionSummary(**data)

    async def _generate_semantic_summary_text(self, s: ExecutionSummary) -> str:
        """
        v1 占位实现：不用 LLM，简单拼接一句话。
        将来可以在这里接 ModelRouter，用 LLM 做更细致的语义压缩。
        """
        status = "succeeded" if s.success else "failed"
        return f"[{s.domain}] goal='{s.goal}' {status} in {s.metrics.get('latency_sec', 0.0):.2f}s."

    async def _write_semantic_memories(
        self,
        summaries: List[ExecutionSummary],
        max_items: int = 100,
    ) -> int:
        """
        为一批 ExecutionSummary 写语义记忆。
        返回写入数量。
        """
        count = 0
        for s in summaries[:max_items]:
            text = await self._generate_semantic_summary_text(s)
            mem = MemoryItem(
                memory_id=f"sem_{uuid.uuid4().hex}",
                kind="semantic_summary",
                content=text,
                metadata={
                    "source": "sleep_loop",
                    "trace_id": s.trace_id,
                    "task_id": s.task_id,
                    "domain": s.domain,
                    "goal": s.goal,
                },
                memory_tier=MemoryTier.STS,  # 先放 STS，后续由 consolidation 决定升 MTS/LTS
            )
            await self.memory.add_memory(mem)
            count += 1
        return count

    async def run_once(
        self,
        window_hours: int = 24,
        max_summaries: int = 200,
        max_semantic_items: int = 50,
        consolidation_limit_per_tier: int = 1000,
    ) -> Dict[str, Any]:
        """
        Sleep Loop 主入口。
        """
        now = datetime.utcnow()
        window_start = now - timedelta(hours=window_hours)

        trace_ids = await self.kv.get(self.summary_index_key) or []
        trace_ids = list(dict.fromkeys(trace_ids))[-max_summaries:]

        summaries: List[ExecutionSummary] = []
        for tid in trace_ids:
            s = await self._load_summary(tid)
            if not s:
                continue
            if s.finished_at < window_start:
                continue
            summaries.append(s)

        # 1) 生成语义记忆
        written = await self._write_semantic_memories(
            summaries,
            max_items=max_semantic_items,
        )

        # 2) 运行 MemoryFabric 合并 / 晋升 / 遗忘
        consolidation_result = await self.memory.run_consolidation(
            max_items_per_tier=consolidation_limit_per_tier
        )

        result = {
            "semantic_memories_written": written,
            "summaries_scanned": len(summaries),
            "consolidation": consolidation_result,
        }

        self.tracer.span(
            "L2",
            "sleep_loop_run",
            module="sleep_loop",
            output=result,
        )
        return result

class SafetyState(BaseModel):
    """
    全局安全状态。
    """
    self_programming_enabled: bool = True
    degraded: bool = False
    last_incident_at: Optional[datetime] = None
    meta: Dict[str, Any] = {}


@dataclass
class SafetyKernel:
    kv: KVStore
    tracer: Tracer

    state_key: str = "safety:state"

    async def get_state(self) -> SafetyState:
        data = await self.kv.get(self.state_key) or {}
        try:
            return SafetyState(**data)
        except Exception:
            return SafetyState()

    async def _save_state(self, state: SafetyState) -> None:
        await self.kv.save(self.state_key, state.model_dump())

    async def set_self_programming_enabled(self, enabled: bool) -> None:
        state = await self.get_state()
        state.self_programming_enabled = enabled
        await self._save_state(state)
        self.tracer.span(
            "L0",
            "safety_set_self_programming",
            module="safety_kernel",
            input={"enabled": enabled},
        )

    async def mark_degraded(self, reason: str, details: Dict[str, Any]) -> None:
        """
        被 ImmuneLoop 调用：记录一次异常事件，并标记系统进入 degraded 状态。
        （当前只是标记，不做更多动作；以后可以在这里关掉自编程 / 降低探索强度等）
        """
        state = await self.get_state()
        state.degraded = True
        state.last_incident_at = datetime.utcnow()
        state.meta.setdefault("incidents", [])
        state.meta["incidents"].append(
            {"reason": reason, "details": details, "ts": state.last_incident_at.isoformat()}
        )
        await self._save_state(state)

        self.tracer.span(
            "L0",
            "safety_mark_degraded",
            module="safety_kernel",
            input={"reason": reason, "details": details},
        )

    async def clear_degraded(self) -> None:
        """
        手动解除 degraded 状态（通过 CLI 或 Web 操作）。
        """
        state = await self.get_state()
        state.degraded = False
        await self._save_state(state)

        self.tracer.span(
            "L0",
            "safety_clear_degraded",
            module="safety_kernel",
            input={},
        )
