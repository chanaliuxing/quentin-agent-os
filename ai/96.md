from __future__ import annotations

from typing import List, Optional
from pathlib import Path
import json
import logging

from quentin.trace.spec import ExecutionTrace, StepTrace

logger = logging.getLogger(__name__)


class TraceStore:
    """
    v1: 简单 JSONL 存储，用于：
    - ImmuneLoop 检查最近一段时间的 trace
    - SleepLoop 做记忆巩固与技能挖掘
    """

    def __init__(self, *, repo_root: str | Path = ".") -> None:
        self.repo_root = Path(repo_root).resolve()
        self.base_dir = self.repo_root / "data" / "traces"
        self.base_dir.mkdir(parents=True, exist_ok=True)
        self.jsonl_path = self.base_dir / "traces.jsonl"

    def append(self, trace: ExecutionTrace) -> None:
        line = json.dumps(trace.to_dict(), ensure_ascii=False)
        try:
            with self.jsonl_path.open("a", encoding="utf-8") as f:
                f.write(line + "\n")
        except Exception:
            logger.exception("TraceStore: failed to append trace")

    def load_recent(self, limit: int = 200) -> List[ExecutionTrace]:
        """
        v1: 粗暴从尾部往前读 limit 行。
        后续可按时间索引优化。
        """
        traces: List[ExecutionTrace] = []
        if not self.jsonl_path.exists():
            return traces

        try:
            with self.jsonl_path.open("r", encoding="utf-8") as f:
                lines = f.readlines()[-limit:]
        except Exception:
            logger.exception("TraceStore: failed to read traces")
            return traces

        for line in lines:
            line = line.strip()
            if not line:
                continue
            try:
                obj = json.loads(line)
                steps = [StepTrace(**s) for s in obj.get("steps", [])]
                tr = ExecutionTrace(
                    trace_id=obj["trace_id"],
                    task_id=obj.get("task_id", ""),
                    domain=obj.get("domain", "general"),
                    status=obj.get("status", "unknown"),
                    steps=steps,
                    created_at=obj.get("created_at", ""),
                    execution_summary=obj.get("execution_summary", {}),
                    lessons_learned=obj.get("lessons_learned", {}),
                    top_level_metrics=obj.get("top_level_metrics", {}),
                    meta=obj.get("meta", {}),
                )
                traces.append(tr)
            except Exception:
                logger.exception("TraceStore: failed to parse trace line")

        return traces

    - 交给 SelfProgrammingEngine 生成 PatchCandidate

    created_by: str = "immune_loop"

class ImmuneLoop:
    """
    Immune Loop v1：

    职责：
    - 从 ExecutionTrace / 高层 metrics 中发现异常模式（v1: 简单规则）
    - 聚合成 ImmuneCase
    - 将 ImmuneCase 写入存储，并交给 Research / SelfProgramming 使用

    注意：
    - v1 不做复杂统计，只实现骨架和简单阈值逻辑。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        trace_store: Optional[TraceStore] = None,
        case_store: Optional[ImmuneCaseStore] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.trace_store = trace_store or TraceStore(repo_root=self.repo_root)
        self.case_store = case_store or ImmuneCaseStore(repo_root=self.repo_root)

        # 简单阈值配置（可被 ResearchLayer/自编程调整）
        self.failure_rate_threshold = 0.4      # 比如失败率 > 40%
        self.min_failures = 3                  # 至少 3 次失败才生成 case

    # ---- 入口1：观测单个任务的结果 ---- #
    def observe_trace(self, trace: ExecutionTrace) -> List[ImmuneSignal]:
        """
        对单个 ExecutionTrace 做初步检查，产生 ImmuneSignal 列表（可能为空）。
        v1 规则：
        - 如果任务 status="failed"，生成一个任务级信号
        - 如果 top_level_metrics 中某些值过大，也可做简单处理（预留）
        """
        signals: List[ImmuneSignal] = []

        if trace.status == "failed":
            sig = ImmuneSignal(
                signal_id=uuid.uuid4().hex,
                source="trace",
                component="task",
                key=trace.task_id,
                severity="medium",
                metrics=trace.top_level_metrics,
                context={"trace_id": trace.trace_id},
            )
            signals.append(sig)

        # TODO: 可以再从 steps 中检查某个 skill / model 的错误集中度

        return signals

    # ---- 入口2：周期性扫描最近 trace ---- #
    def scan_recent_traces(self, limit: int = 200) -> List[ImmuneCase]:
        """
        周期性调用：
        - 读取最近 N 条 trace
        - 统计某些 key（如 skill_id / model_id）的失败情况
        - 对超过阈值的模块生成 ImmuneCase
        """
        traces = self.trace_store.load_recent(limit=limit)
        logger.info("ImmuneLoop.scan_recent_traces: loaded %d traces", len(traces))

        # v1: 只按 task_id 聚合失败次数（实际应该按 component / skill / model 聚合）
        failure_counts: Dict[str, int] = {}
        context_by_task: Dict[str, Dict[str, Any]] = {}

        for tr in traces:
            if tr.status != "failed":
                continue
            failure_counts[tr.task_id] = failure_counts.get(tr.task_id, 0) + 1
            context_by_task.setdefault(tr.task_id, {})["last_trace_id"] = tr.trace_id

        cases: List[ImmuneCase] = []
        for task_id, fail_count in failure_counts.items():
            if fail_count < self.min_failures:
                continue
            severity = "medium"
            if fail_count >= 10:
                severity = "high"
            if fail_count >= 20:
                severity = "critical"

            sig = ImmuneSignal(
                signal_id=uuid.uuid4().hex,
                source="aggregate_trace",
                component="task",
                key=task_id,
                severity=severity,
                metrics={"failure_count": fail_count},
                context=context_by_task.get(task_id, {}),
            )
            case = ImmuneCase(
                case_id=uuid.uuid4().hex,
                component="task",
                target={
                    "target_type": "task",
                    "identifier": task_id,
                    "file_path": "",
                    "symbol_name": None,
                    "meta": {},
                },
                severity=severity,
                signals=[sig],
                summary=f"Task {task_id} failure count={fail_count}",
                suggested_actions=[
                    "Investigate planner/executor logic for this task pattern.",
                    "Consider creating patch candidate or new skill.",
                ],
            )
            self.case_store.append(case)
            cases.append(case)

        return cases

class MemoryFabric:
    """
    三层 Memory Fabric v1：

    - STS: task/session 级记忆（短期）
    - MTS: project/version 级记忆（中期）
    - LTS: 跨项目稳定知识（长期）

    v1: 用三个 JSONL 文件简单存储。
    """

    def __init__(self, *, repo_root: str | Path = ".") -> None:
        self.repo_root = Path(repo_root).resolve()
        base_dir = self.repo_root / "data" / "memory"
        base_dir.mkdir(parents=True, exist_ok=True)
        self.sts_path = base_dir / "sts.jsonl"
        self.mts_path = base_dir / "mts.jsonl"
        self.lts_path = base_dir / "lts.jsonl"

    # ---- 写入 ---- #
    def add_memory(self, item: MemoryItem) -> None:
        if item.tier == "STS":
            path = self.sts_path
        elif item.tier == "MTS":
            path = self.mts_path
        else:
            path = self.lts_path

        line = json.dumps(item.to_dict(), ensure_ascii=False)
        try:
            with path.open("a", encoding="utf-8") as f:
                f.write(line + "\n")
        except Exception:
            logger.exception("MemoryFabric: failed to append memory to %s", path)

    # ---- 简单查询（按 tier） ---- #
    def load_all(self, tier: str = "STS") -> List[MemoryItem]:
        if tier == "STS":
            path = self.sts_path
        elif tier == "MTS":
            path = self.mts_path
        else:
            path = self.lts_path

        items: List[MemoryItem] = []
        if not path.exists():
            return items
        try:
            with path.open("r", encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if not line:
                        continue
                    obj = json.loads(line)
                    items.append(MemoryItem(**obj))
        except Exception:
            logger.exception("MemoryFabric: failed to load from %s", path)
        return items

    # ---- 简单迁移逻辑（STS -> MTS / LTS） ---- #
    def promote(
        self,
        *,
        items: List[MemoryItem],
        target_tier: str,
    ) -> None:
        """
        v1：外部先筛选好 items，
        这里仅负责把它们以新 tier 写入对应文件。
        """
        for item in items:
            item.tier = target_tier
            self.add_memory(item)

# quentin/loops/sleep.py
from __future__ import annotations

from typing import Any, Dict, List, Optional
from pathlib import Path
from datetime import datetime
import logging
import uuid

from quentin.memory.fabric import MemoryFabric, MemoryItem
from quentin.trace.store import TraceStore
from quentin.trace.spec import ExecutionTrace
from quentin.evolution.skill_loop import SkillEvolutionLoop
from quentin.self_programming.engine import SelfProgrammingEngine

logger = logging.getLogger(__name__)


class SleepLoop:
    """
    Sleep / Consolidation Loop v1：

    职责：
    1. 从最近的 ExecutionTrace 中抽取：
       - 新的 episodic 记忆（STS）
       - 总结型的 semantic 记忆（MTS/LTS 候选）
    2. 按简单规则做 tier 迁移
    3. 触发 SkillEvolutionLoop，对成功 trace + Immune issues 做一次演化轮次
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        memory_fabric: Optional[MemoryFabric] = None,
        trace_store: Optional[TraceStore] = None,
        evolution_loop: Optional[SkillEvolutionLoop] = None,
        self_programming_engine: Optional[SelfProgrammingEngine] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.memory_fabric = memory_fabric or MemoryFabric(repo_root=self.repo_root)
        self.trace_store = trace_store or TraceStore(repo_root=self.repo_root)

        # 如果没传 evolution_loop，就根据 engine 构造一个默认的
        from quentin.evolution.skill_loop import SkillEvolutionLoop as _SkillLoop

        self.engine = self_programming_engine or SelfProgrammingEngine(repo_root=self.repo_root)
        self.evolution_loop = evolution_loop or _SkillLoop(self.engine)

        # 预算配置（蓝皮书要求有 budget 控制）
        self.max_traces_per_run = 100

    # ---- 从 trace 生成 STS 记忆 ---- #
    def _build_sts_memories(self, traces: List[ExecutionTrace]) -> List[MemoryItem]:
        items: List[MemoryItem] = []
        for tr in traces:
            mem = MemoryItem(
                memory_id=uuid.uuid4().hex,
                memory_type="episodic",
                content={
                    "trace_id": tr.trace_id,
                    "task_id": tr.task_id,
                    "domain": tr.domain,
                    "status": tr.status,
                    "execution_summary": tr.execution_summary,
                    "lessons_learned": tr.lessons_learned,
                    "top_level_metrics": tr.top_level_metrics,
                },
                tier="STS",
                domain=tr.domain,
                importance=0.7 if tr.status == "success" else 0.4,
                origin={"type": "trace"},
            )
            items.append(mem)
        return items

    # ---- 简单的 MTS/LTS 候选规则（stub） ---- #
    def _select_for_promotion(
        self,
        sts_items: List[MemoryItem],
    ) -> Dict[str, List[MemoryItem]]:
        """
        v1: 非常简单的规则：
        - status=success 且 importance>0.6 => 进入 MTS 候选
        - TODO：真正的 LTS 需要跨多任务复用的统计，这里先不做
        """
        mts_candidates: List[MemoryItem] = []

        for item in sts_items:
            content = item.content or {}
            if content.get("status") == "success" and item.importance > 0.6:
                mts_candidates.append(item)

        return {
            "MTS": mts_candidates,
            "LTS": [],  # v1 暂不自动升 LTS
        }

    # ---- Skill Evolution 集成 ---- #
    def _prepare_evolution_inputs(
        self,
        traces: List[ExecutionTrace],
    ) -> Dict[str, Any]:
        """
        为 SkillEvolutionLoop 准备输入：
        - successful_traces: 选出成功且可作为 skill_seed 的 trace 片段
        - issues: 这里先留空；真实场景中可以从 ImmuneCaseStore 读取
        """
        successful_traces: List[Dict[str, Any]] = []
        for tr in traces:
            if tr.status != "success":
                continue
            successful_traces.append(
                {
                    "trace_id": tr.trace_id,
                    "task_id": tr.task_id,
                    "domain": tr.domain,
                    "skill_seed": True,   # 先统一标记为可用，后续可用更精细逻辑
                }
            )

        issues: List[Dict[str, Any]] = []
        # v1: issues 由外部 ImmuneLoop 提供，这里不读取

        return {"successful_traces": successful_traces, "issues": issues}

    # ---- 对外主接口：run_once ---- #
    def run_once(
        self,
        *,
        traces: Optional[List[ExecutionTrace]] = None,
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        每次 Sleep 调用的主方法。

        通常由调度器定期触发（比如每天 N 点 / 每小时一次）。
        """
        ctx = extra_context or {}
        if traces is None:
            all_traces = self.trace_store.load_recent(limit=self.max_traces_per_run)
        else:
            all_traces = traces[: self.max_traces_per_run]

        logger.info("SleepLoop.run_once: traces=%d", len(all_traces))

        # 1) STS 写入
        sts_items = self._build_sts_memories(all_traces)
        for item in sts_items:
            self.memory_fabric.add_memory(item)

        # 2) 简单的 MTS/LTS 候选评估
        promotions = self._select_for_promotion(sts_items)
        if promotions["MTS"]:
            self.memory_fabric.promote(items=promotions["MTS"], target_tier="MTS")
        # LTS 暂不自动升

        # 3) Skill & Code Evolution 触发
        evo_inputs = self._prepare_evolution_inputs(all_traces)
        evo_result = self.evolution_loop.run_once(
            successful_traces=evo_inputs["successful_traces"],
            issues=evo_inputs["issues"],
            test_suites=ctx.get("test_suites"),
            extra_context=ctx.get("evolution_context"),
        )

        return {
            "sts_items_written": len(sts_items),
            "mts_promoted": len(promotions["MTS"]),
            "lts_promoted": len(promotions["LTS"]),
            "evolution_result": evo_result,
        }
