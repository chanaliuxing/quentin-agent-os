from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, List, Optional
from pathlib import Path
import logging
import time

from quentin.eval.harness import EvalHarness
from quentin.research.experiments import (
    ExperimentOrchestrator,
    ExperimentRecord,
    ExperimentRunResult,
    ExperimentConfig,
)
# 注意：ExperimentOrchestrator 里已经引用了 MetaAgentRuntime
# 这里只负责“调度研究”，不直接依赖 runtime 实现细节。

logger = logging.getLogger(__name__)


# ============================================================
# 配置与状态数据结构
# ============================================================

@dataclass
class ResearchBudgetConfig:
    """
    研究预算配置（蓝皮书里的 Budget & Cost Controller 的 v1 实现）：

    - max_experiments_per_cycle: 每个 research_cycle 允许跑多少个 experiment
    - max_runtime_sec_per_experiment: 单个 experiment 一轮运行的最大允许时间（软上限）
    - max_runs_per_experiment: 每个 experiment 在全局允许的最大 run 次数
    """
    max_experiments_per_cycle: int = 2
    max_runtime_sec_per_experiment: float = 600.0
    max_runs_per_experiment: int = 20

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ResearchCycleRequest:
    """
    一次研究周期的请求参数：

    - strategy: 使用哪种策略选择 experiment（目前支持 "simple"）
    - max_experiments: 本次 cycle 最多跑几个 experiment（覆盖默认预算）
    """
    strategy: str = "simple"
    max_experiments: Optional[int] = None

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ResearchCycleResult:
    """
    一次研究周期的执行结果：

    - started_at / finished_at: 起止时间
    - executed_experiments: 本轮实际执行的 experiment_id 列表
    - run_results: experiment_id -> ExperimentRunResult dict
    - variant_rankings: experiment_id -> compare_variants(...) 的结果
    """
    started_at: float
    finished_at: float
    executed_experiments: List[str]
    run_results: Dict[str, Dict[str, Any]]
    variant_rankings: Dict[str, Dict[str, Any]]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "started_at": self.started_at,
            "finished_at": self.finished_at,
            "executed_experiments": self.executed_experiments,
            "run_results": self.run_results,
            "variant_rankings": self.variant_rankings,
        }


# ============================================================
# Budget Controller：预算控制器
# ============================================================

class BudgetController:
    """
    控制研究预算的简单实现。

    目前的策略：
      - 限制每个 experiment 的 run 次数（max_runs_per_experiment）
      - 限制每个 cycle 跑多少个 experiment（由 ResearchBudgetConfig + Request 决定）

    以后可以扩展：
      - 跟踪 token 使用量 / 显式成本
      - 跟踪 CPU/GPU 时间
      - 根据“最近收益”调整优先级
    """

    def __init__(self, config: ResearchBudgetConfig) -> None:
        self.config = config

    def can_run_experiment(self, record: ExperimentRecord) -> bool:
        """
        根据历史 run 次数判断这个 experiment 是否还能继续跑。
        """
        num_runs = len(record.runs)
        if num_runs >= self.config.max_runs_per_experiment:
            logger.info(
                "Experiment %s reached max_runs_per_experiment=%d, skip",
                record.config.experiment_id,
                self.config.max_runs_per_experiment,
            )
            return False
        return True

    def limit_experiments_for_cycle(
        self,
        *,
        candidates: List[ExperimentRecord],
        request: ResearchCycleRequest,
    ) -> List[ExperimentRecord]:
        """
        按当前预算和请求限制本次 cycle 实际要跑的 experiment 数量。
        """
        max_exp = request.max_experiments or self.config.max_experiments_per_cycle
        if len(candidates) <= max_exp:
            return candidates
        return candidates[:max_exp]


# ============================================================
# Evolution Strategy Manager：选择“改哪里、怎么改”
# ============================================================

class EvolutionStrategyManager:
    """
    决定本次 research cycle 要优先优化哪些 experiment 的简单策略模块。

    v1 策略（对齐蓝皮书“研究友好”要求）：

      - strategy="simple":
          * 优先选择运行次数最少的 experiment（探索广度）
          * 其次按 meta.priority 排序（例如 1=最高）
    """

    def __init__(self, *, default_strategy: str = "simple") -> None:
        self.default_strategy = default_strategy

    def select_experiments_for_cycle(
        self,
        *,
        experiments: List[ExperimentRecord],
        strategy: Optional[str] = None,
    ) -> List[ExperimentRecord]:
        strat = strategy or self.default_strategy
        if strat == "simple":
            return self._select_simple(experiments)
        # 可以在这里增加其它策略（如 reward_based、error_hotspot 等）
        return self._select_simple(experiments)

    def _select_simple(
        self,
        experiments: List[ExperimentRecord],
    ) -> List[ExperimentRecord]:
        def _priority(rec: ExperimentRecord) -> int:
            meta = rec.config.meta or {}
            # priority 数字越小优先级越高，默认 100
            return int(meta.get("priority", 100))

        # 先按 run 次数（少的在前），再按 priority（小的在前）
        return sorted(
            experiments,
            key=lambda r: (len(r.runs), _priority(r)),
        )


# ============================================================
# Policy & Governance：宪法级规则 / 合规策略 v1
# ============================================================

class PolicyAndGovernance:
    """
    宪法级策略（v1 简化版）：

      - 现在只做非常宽松的约束（全允许），主要作用是预留接口。
      - 未来可以：
          * 把宪法写成 JSON / YAML，由 LLM 审阅
          * 使用 MultiModelCouncil 对“高风险实验”做额外审批
    """

    def __init__(
        self,
        *,
        allowed_domains: Optional[List[str]] = None,
        blocked_domains: Optional[List[str]] = None,
    ) -> None:
        self.allowed_domains = allowed_domains or []   # 空列表表示不限制
        self.blocked_domains = blocked_domains or []   # 显式禁止的 domain

    def approve_experiment(self, record: ExperimentRecord) -> bool:
        """
        判断某个 experiment 是否允许运行。

        默认逻辑：
          - 如果 meta.domain 在 blocked_domains → 拒绝
          - 如果 allowed_domains 非空且 domain 不在其中 → 拒绝
          - 否则允许
        """
        meta = record.config.meta or {}
        domain = meta.get("domain", "general")

        if domain in self.blocked_domains:
            logger.warning("Experiment %s domain=%s is blocked", record.config.experiment_id, domain)
            return False

        if self.allowed_domains and (domain not in self.allowed_domains):
            logger.info(
                "Experiment %s domain=%s not in allowed_domains, skip",
                record.config.experiment_id,
                domain,
            )
            return False

        return True

# ResearchLayer：研究与演化总控
# ============================================================

class ResearchLayer:
    """
    L3 Research & Evolution Layer v1 实现。

    目前它主要负责：

      - 持有一个 ExperimentOrchestrator（真正跑实验）
      - 持有一个 BudgetController（算力 / 次数预算）
      - 持有一个 EvolutionStrategyManager（选哪些 experiment 来跑）
      - 持有一个 PolicyAndGovernance（做“宪法级”允许/禁止判断）

    对外暴露的关键方法：

      - create_experiment(...)         # 代理给 ExperimentOrchestrator
      - run_research_cycle(request)    # 一次完整研究周期
      - list_experiments()             # 查看已有实验
      - get_experiment_record(id)      # 获取详细信息
    """

    def __init__(
        self,
        *,
        repo_root: str | Path,
        experiment_orchestrator: ExperimentOrchestrator,
        budget_config: Optional[ResearchBudgetConfig] = None,
        eval_harness: Optional[EvalHarness] = None,
        policy: Optional[PolicyAndGovernance] = None,
        evolution_strategy_manager: Optional[EvolutionStrategyManager] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.experiment_orchestrator = experiment_orchestrator
        self.eval_harness = eval_harness or experiment_orchestrator.eval_harness

        self.budget_config = budget_config or ResearchBudgetConfig()
        self.budget_controller = BudgetController(self.budget_config)
        self.policy = policy or PolicyAndGovernance()
        self.evolution_strategy_manager = (
            evolution_strategy_manager or EvolutionStrategyManager()
        )

    # --------------------------------------------------
    # 基础查询
    # --------------------------------------------------
    def list_experiments(self) -> List[str]:
        """
        返回当前已知的 experiment_id 列表。
        """
        return list(self.experiment_orchestrator._experiments.keys())

    def get_experiment_record(self, experiment_id: str) -> ExperimentRecord:
        """
        获取某个 experiment 的完整记录（配置 + 历史 runs）。
        """
        return self.experiment_orchestrator._get_experiment_or_raise(experiment_id)

    # --------------------------------------------------
    # 代理创建（方便外层统一从 ResearchLayer 入口用）
    # --------------------------------------------------
    def create_experiment(
        self,
        *,
        name: str,
        description: str,
        variants: Optional[List[str]] = None,
        tasks: Optional[List[Any]] = None,
        meta: Optional[Dict[str, Any]] = None,
        experiment_id: Optional[str] = None,
    ) -> ExperimentRecord:
        """
        直接转发给 ExperimentOrchestrator.create_experiment(...)。
        """
        return self.experiment_orchestrator.create_experiment(
            name=name,
            description=description,
            variants=variants,
            tasks=tasks,
            meta=meta,
            experiment_id=experiment_id,
        )

    # --------------------------------------------------
    # Research Cycle：一次完整研究周期
    # --------------------------------------------------
    def run_research_cycle(
        self,
        request: Optional[ResearchCycleRequest] = None,
    ) -> ResearchCycleResult:
        """
        Research Layer 对外的核心接口之一：

        - 选出一批 experiment（根据 EvolutionStrategyManager）
        - 过滤掉不符合 Policy 的 experiment
        - 再在 BudgetController 限制下，跑 run_experiment(...)
        - 每个 experiment 跑完后，自动调用 compare_variants(...)
        """
        req = request or ResearchCycleRequest()
        started_at = time.time()

        # 1) 收集所有 experiment record
        all_ids = self.list_experiments()
        all_records: List[ExperimentRecord] = [
            self.get_experiment_record(eid) for eid in all_ids
        ]

        if not all_records:
            logger.info("ResearchLayer: no experiments found, nothing to do.")
            return ResearchCycleResult(
                started_at=started_at,
                finished_at=time.time(),
                executed_experiments=[],
                run_results={},
                variant_rankings={},
            )

        # 2) 策略层筛选：优先哪些 experiment
        candidates = self.evolution_strategy_manager.select_experiments_for_cycle(
            experiments=all_records,
            strategy=req.strategy,
        )

        # 3) Policy 层过滤：不允许的直接丢弃
        allowed: List[ExperimentRecord] = []
        for rec in candidates:
            if self.policy.approve_experiment(rec) and self.budget_controller.can_run_experiment(rec):
                allowed.append(rec)

        # 4) Budget 限制：一个 cycle 最多跑多少 experiment
        to_run = self.budget_controller.limit_experiments_for_cycle(
            candidates=allowed,
            request=req,
        )

        executed_experiments: List[str] = []
        run_results: Dict[str, Dict[str, Any]] = {}
        variant_rankings: Dict[str, Dict[str, Any]] = {}

        # 5) 逐个 experiment 调用 ExperimentOrchestrator.run_experiment & compare_variants
        for rec in to_run:
            exp_id = rec.config.experiment_id
            logger.info("ResearchLayer: running experiment %s", exp_id)

            run_result: ExperimentRunResult = self.experiment_orchestrator.run_experiment(exp_id)
            executed_experiments.append(exp_id)
            run_results[exp_id] = run_result.to_dict()

            try:
                compare_res = self.experiment_orchestrator.compare_variants(exp_id)
            except Exception:
                logger.exception("ResearchLayer: compare_variants failed for %s", exp_id)
                compare_res = {"status": "error"}
            variant_rankings[exp_id] = compare_res

        finished_at = time.time()

        return ResearchCycleResult(
            started_at=started_at,
            finished_at=finished_at,
            executed_experiments=executed_experiments,
            run_results=run_results,
            variant_rankings=variant_rankings,
        )
