# quentin/core/traces.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from datetime import datetime
from typing import Any, Dict, List, Optional


@dataclass
class ExecutionTrace:
    """
    一次任务执行的最小结构化 Trace（供 ImmuneLoop / SleepLoop 使用）。

    注意：这不是完整日志，只是“免疫 / 研究”关心的那部分摘要。
    """

    trace_id: str
    task_id: str

    # 哪个模块/能力在工作（比如 "planner", "executor", "skill:open_file"）
    component_id: str

    # 蓝皮书中的能力 ID（可选），让 ImmuneLoop 能直接对上 blueprint capability
    blueprint_capability_id: Optional[str] = None

    # 代码位置（可选）——如果知道是哪个模块出了问题，可以填 module / symbol
    target_module: Optional[str] = None      # 如 "quentin.core.cognitive_loop"
    target_symbol: Optional[str] = None      # 如 "CognitiveLoop.run_task"

    success: bool = True
    error_type: Optional[str] = None         # e.g. "TimeoutError", "ValidationError"
    error_message: Optional[str] = None

    start_time: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    end_time: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    # 一些通用指标（可以按需扩展）
    metrics: Dict[str, float] = field(default_factory=dict)
    tags: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ImmuneCase:
    """
    ImmuneLoop 发现的一个“异常病例”。

    - severity          : 严重程度（low/medium/high）
    - action_type       : 建议动作类型（selfprog_patch / degrade / disable_feature / escalate / none）
    - related_traces    : 相关 trace_id 列表
    - proposed_module   : 建议优化的模块（可选）
    - proposed_symbol   : 建议优化的符号（可选）
    """

    case_id: str
    component_id: str
    severity: str
    action_type: str
    summary: str
    reason: str

    related_traces: List[str] = field(default_factory=list)
    blueprint_capability_id: Optional[str] = None
    proposed_module: Optional[str] = None
    proposed_symbol: Optional[str] = None

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

# quentin/core/immune_loop.py
from __future__ import annotations

import json
import logging
import uuid
from collections import defaultdict, deque
from dataclasses import asdict
from pathlib import Path
from typing import Deque, Dict, List, Optional, Tuple

from quentin.core.traces import ExecutionTrace, ImmuneCase
from quentin.llm.multi_council import MultiAICouncil, CouncilDecision
from quentin.llm.types import LLMConfig
from quentin.selfprog.engine import SelfProgrammingEngine
from quentin.selfprog.models import Patch
from quentin.research.experiments import ResearchManager, Experiment

logger = logging.getLogger(__name__)


class ImmuneLoop:
    """
    ImmuneLoop v1：

    - 接收 ExecutionTrace（任务执行结果）
    - 做简单异常检测（error rate、重复错误类型）
    - 出现异常时：
        1）用多 AI 会议模式生成 ImmuneCase（诊断 + 建议动作）
        2）如果建议是 selfprog_patch，自动调用 SelfProgrammingEngine.propose_patch(...)
    - 所有 ImmuneCase + 触发的 patch 记录进 ResearchManager，方便分析。

    注意：
    - 这里只做“发现+触发”，不做自动降级/熔断（那可以在后续版本扩展）。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        research_manager: Optional[ResearchManager] = None,
        selfprog_engine: Optional[SelfProgrammingEngine] = None,
        council: Optional[MultiAICouncil] = None,
        trace_store_path: str | Path = "data/traces.jsonl",
        window_size: int = 50,
        error_rate_threshold: float = 0.5,
        min_window: int = 5,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.research_manager = research_manager or ResearchManager()
        self.selfprog_engine = selfprog_engine or SelfProgrammingEngine(repo_root=self.repo_root)

        if council is None:
            members = [
                LLMConfig(
                    provider="ollama",
                    model="qwen3-vl:latest",
                    role="immune_diagnoser",
                ),
                LLMConfig(
                    provider="ollama",
                    model="deepseek-r1:1.5b",
                    role="root_cause_analyst",
                ),
            ]
            council = MultiAICouncil(members=members)
        self.council = council

        self.trace_store_path = self.repo_root / trace_store_path
        self.trace_store_path.parent.mkdir(parents=True, exist_ok=True)

        self.window_size = window_size
        self.error_rate_threshold = error_rate_threshold
        self.min_window = min_window

        # 每个 component_id 一条滑动窗口
        self._recent_traces: Dict[str, Deque[ExecutionTrace]] = defaultdict(
            lambda: deque(maxlen=self.window_size)
        )

    # ---------- 对外 API：处理一条 Trace ----------

    def process_trace(self, trace: ExecutionTrace) -> Tuple[Optional[ImmuneCase], Optional[Patch]]:
        """
        主入口：

        1. 持久化 trace（写 jsonl）
        2. 更新滑动窗口
        3. 检测是否异常
        4. 如异常 -> 多 AI 诊断 ImmuneCase
        5. 如 action_type=selfprog_patch -> 自动提自编程补丁

        返回：(immune_case | None, patch | None)
        """
        self._store_trace(trace)
        self._recent_traces[trace.component_id].append(trace)

        is_abnormal, reason = self._detect_anomaly(trace.component_id)
        if not is_abnormal:
            return None, None

        immune_case = self._diagnose_case(trace.component_id, reason)
        patch: Optional[Patch] = None

        if immune_case and immune_case.action_type == "selfprog_patch":
            patch = self._trigger_self_programming(immune_case)

        self._record_research(immune_case, patch)
        return immune_case, patch

    # ---------- 内部：存储 Trace ----------

    def _store_trace(self, trace: ExecutionTrace) -> None:
        try:
            with self.trace_store_path.open("a", encoding="utf-8") as f:
                f.write(json.dumps(trace.to_dict(), ensure_ascii=False) + "\n")
        except Exception:
            logger.exception("ImmuneLoop: failed to store trace to %s", self.trace_store_path)

    # ---------- 内部：异常检测 ----------

    def _detect_anomaly(self, component_id: str) -> Tuple[bool, str]:
        """
        v1 简单策略：

        - 窗口内至少 min_window 条 trace
        - 若 failure rate > error_rate_threshold -> 异常
        - 或者最近 N 条都是同一个 error_type -> 异常
        """
        traces = list(self._recent_traces[component_id])
        n = len(traces)
        if n < self.min_window:
            return False, "window too small"

        failures = [t for t in traces if not t.success]
        failure_rate = len(failures) / n

        # 条件 1：失败率过高
        if failure_rate > self.error_rate_threshold:
            return True, f"High failure rate: {len(failures)}/{n} ({failure_rate:.2f})"

        # 条件 2：最近 3 条都是同一错误类型
        recent = traces[-3:] if n >= 3 else traces
        if recent and all(not t.success for t in recent):
            err_types = {t.error_type for t in recent}
            if len(err_types) == 1 and list(err_types)[0] is not None:
                return True, f"Repeated same error_type in recent traces: {list(err_types)[0]}"

        return False, "no anomaly detected"

    # ---------- 内部：多 AI 诊断 ImmuneCase ----------

    def _diagnose_case(self, component_id: str, anomaly_reason: str) -> ImmuneCase:
        traces = list(self._recent_traces[component_id])
        failures = [t for t in traces if not t.success]

        # 构造诊断上下文
        ctx_lines: List[str] = [
            f"COMPONENT_ID: {component_id}",
            f"ANOMALY_REASON: {anomaly_reason}",
            "",
            "RECENT_TRACES_SUMMARY:",
        ]

        for t in traces[-10:]:  # 最近 10 条
            ctx_lines.append(
                f"- trace_id={t.trace_id}, success={t.success}, "
                f"error_type={t.error_type}, metrics={t.metrics}"
            )

        ctx_lines.append("")
        ctx_lines.append("FAILURE_DETAILS (up to 5):")
        for t in failures[-5:]:
            ctx_lines.append(
                f"- trace_id={t.trace_id}, error_type={t.error_type}, "
                f"error_message={t.error_message}, target_module={t.target_module}, "
                f"target_symbol={t.target_symbol}, blueprint_capability_id={t.blueprint_capability_id}"
            )

        context = "\n".join(ctx_lines)

        question = (
            "You are the IMMUNE COUNCIL of a self-evolving AI agent.\n"
            "You must diagnose the anomaly and propose a SAFE action.\n\n"
            "Possible action_type:\n"
            "- \"selfprog_patch\": generate a self-programming patch for a specific module/symbol.\n"
            "- \"degrade\": suggest temporarily turning off or simplifying this component.\n"
            "- \"disable_feature\": suggest disabling a specific feature.\n"
            "- \"escalate\": require human or external supervisor review.\n"
            "- \"none\": no action needed.\n\n"
            "You MUST be conservative. Only choose \"selfprog_patch\" when:\n"
            "- there is a clear module/symbol to target, AND\n"
            "- the failure pattern is stable and reproducible.\n\n"
            "Return ONLY ONE JSON object:\n"
            "{\n"
            "  \"severity\": \"low\" | \"medium\" | \"high\",\n"
            "  \"action_type\": \"selfprog_patch\" | \"degrade\" | \"disable_feature\" | \"escalate\" | \"none\",\n"
            "  \"summary\": \"short summary of the problem\",\n"
            "  \"reason\": \"explain why you choose this action\",\n"
            "  \"blueprint_capability_id\": \"id-or-null\",\n"
            "  \"proposed_module\": \"quentin.xxx\" or null,\n"
            "  \"proposed_symbol\": \"ClassOrFuncName\" or null\n"
            "}\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="immune_diagnosis",
        )
        ans = decision.final_answer or "{}"

        try:
            s = ans.strip()
            first = s.find("{")
            last = s.rfind("}")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)
        except Exception:
            logger.exception("ImmuneLoop._diagnose_case: failed to parse council JSON: %r", ans)
            data = {
                "severity": "medium",
                "action_type": "escalate",
                "summary": "Failed to parse council answer.",
                "reason": "Fallback due to JSON parse error.",
                "blueprint_capability_id": None,
                "proposed_module": None,
                "proposed_symbol": None,
            }

        severity = str(data.get("severity", "medium"))
        action_type = str(data.get("action_type", "escalate"))
        summary = str(data.get("summary", ""))
        reason = str(data.get("reason", ""))
        bp_id = data.get("blueprint_capability_id")
        proposed_module = data.get("proposed_module")
        proposed_symbol = data.get("proposed_symbol")

        case_id = str(uuid.uuid4())
        related_ids = [t.trace_id for t in failures[-10:]]

        immune_case = ImmuneCase(
            case_id=case_id,
            component_id=component_id,
            severity=severity,
            action_type=action_type,
            summary=summary,
            reason=reason,
            related_traces=related_ids,
            blueprint_capability_id=bp_id,
            proposed_module=proposed_module,
            proposed_symbol=proposed_symbol,
            metadata={
                "anomaly_reason": anomaly_reason,
            },
        )
        return immune_case

    # ---------- 内部：触发自编程 ----------

    def _trigger_self_programming(self, case: ImmuneCase) -> Optional[Patch]:
        """
        当 ImmuneCase.action_type == selfprog_patch 时被调用。
        """
        # 没有明确目标模块 -> 暂时只记录，不触发
        if not case.proposed_module:
            logger.warning(
                "ImmuneLoop._trigger_self_programming: no proposed_module for case %s; skip.",
                case.case_id,
            )
            return None

        target_module = case.proposed_module
        target_symbol = case.proposed_symbol

        # 简单把 module 转为文件路径
        from quentin.blueprint.auto_patcher import BlueprintGapAutoPatcher

        autopatcher = BlueprintGapAutoPatcher()  # 用它的 module_to_path 逻辑
        target_path = autopatcher._module_to_path(target_module)
        if not target_path:
            logger.warning(
                "ImmuneLoop._trigger_self_programming: cannot infer path from module %s; skip.",
                target_module,
            )
            return None

        # problem description 里包括 ImmuneCase 的 summary + reason
        problem_lines = [
            f"ImmuneCase ID: {case.case_id}",
            f"Component ID : {case.component_id}",
            f"Severity     : {case.severity}",
            f"Summary      : {case.summary}",
            f"Reason       : {case.reason}",
            "",
            "The immune loop has detected repeated failures and requests a self-programming patch",
            "to improve this module or symbol.",
        ]
        if case.blueprint_capability_id:
            problem_lines.append(f"Blueprint capability: {case.blueprint_capability_id}")

        problem = "\n".join(problem_lines)

        try:
            patch = self.selfprog_engine.propose_patch(
                target_path=target_path,
                problem_description=problem,
                change_type="refactor",
                target_symbol=target_symbol,
                extra_context=(
                    "This patch is triggered by ImmuneLoop due to repeated failures. "
                    "Focus on robustness, error handling, and safe fallback behaviors."
                ),
            )
            return patch
        except Exception:
            logger.exception(
                "ImmuneLoop._trigger_self_programming: propose_patch failed for case %s",
                case.case_id,
            )
            return None

    # ---------- 内部：记录 Research 实验 ----------

    def _record_research(
        self,
        case: Optional[ImmuneCase],
        patch: Optional[Patch],
    ) -> None:
        rm = self.research_manager

        capability = "immune:anomaly_handling"
        requirement_id = "immune_loop_effectiveness"

        exps = rm.list_experiments()
        exp: Optional[Experiment] = None
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "immune_loop"
            ):
                exp = e
                break

        if exp is None:
            exp = rm.create_experiment(
                name="ImmuneLoop",
                purpose="immune_loop",
                capability=capability,
                requirement_id=requirement_id,
                tags=["immune", "anomaly"],
            )

        metrics = {
            "has_case": int(case is not None),
            "has_patch": int(patch is not None),
        }
        ctx: Dict[str, any] = {}
        if case:
            ctx["case_id"] = case.case_id
            ctx["component_id"] = case.component_id
            ctx["severity"] = case.severity
            ctx["action_type"] = case.action_type
        if patch:
            ctx["patch_id"] = patch.patch_id
            ctx["patch_target_path"] = patch.target_path

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics=metrics,
            context=ctx,
        )
⚠️ 注意：这里有一个小“偷懒”的地方：_trigger_self_programming 里复用了 BlueprintGapAutoPatcher._module_to_path，这是为了避免重复写 module→path 的逻辑。如果你觉得不优雅，可以把这个方法提到一个公用 util 模块，比如 quentin/utils/modules.py。

def cmd_immune_process_trace(args: argparse.Namespace) -> None:
    import json as _json
    from pathlib import Path as _Path

    p = _Path(args.trace_file)
    if not p.exists():
        print(f"Trace file not found: {p}")
        return

    try:
        raw = _json.loads(p.read_text(encoding="utf-8") or "{}")
    except Exception as e:
        print(f"Failed to parse trace JSON: {e}")
        return

    try:
        trace = ExecutionTrace(
            trace_id=raw.get("trace_id"),
            task_id=raw.get("task_id"),
            component_id=raw.get("component_id"),
            blueprint_capability_id=raw.get("blueprint_capability_id"),
            target_module=raw.get("target_module"),
            target_symbol=raw.get("target_symbol"),
            success=bool(raw.get("success", True)),
            error_type=raw.get("error_type"),
            error_message=raw.get("error_message"),
            start_time=raw.get("start_time") or "",
            end_time=raw.get("end_time") or "",
            metrics=raw.get("metrics") or {},
            tags=raw.get("tags") or {},
        )
    except Exception as e:
        print(f"Failed to construct ExecutionTrace: {e}")
        return

    immune = ImmuneLoop()
    immune_case, patch = immune.process_trace(trace)

    if not immune_case:
        print("No anomaly detected by ImmuneLoop. No ImmuneCase created.")
        return

    print("ImmuneCase created:")
    print(f"- case_id      : {immune_case.case_id}")
    print(f"- component_id : {immune_case.component_id}")
    print(f"- severity     : {immune_case.severity}")
    print(f"- action_type  : {immune_case.action_type}")
    print(f"- summary      : {immune_case.summary}")
    print(f"- reason       : {immune_case.reason}")

    if patch:
        print("\nPatch proposed by ImmuneLoop:")
        print(f"- patch_id     : {patch.patch_id}")
        print(f"- target_path  : {patch.target_path}")
        print(f"- target_sym   : {patch.target_symbol}")
        print(f"- status       : {patch.status}")
        print(f"- risk_level   : {patch.risk_level}")
