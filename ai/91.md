from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional
from pathlib import Path
from datetime import datetime
import json
import uuid
import logging

logger = logging.getLogger(__name__)


# --------- 基本 Schema --------- #

@dataclass
class VariantConfig:
    """
    一个实验“变体”（Variant）的配置。

    - variant_id       : 唯一 ID（字符串）
    - name             : 人类可读名称
    - description      : 做什么改动，例如“planner 使用策略B”，“model_router 改成 bandit 策略”
    - target_component : 目标组件，例如 "planner" / "model_router" / "skill_evolution"
    - params           : 对该组件的参数改动（由对应 Runner 解释）
    - implementation_hint :
        可选，用于指明实现位置，比如 "quentin.planner.strategies:StrategyB"
        真正加载执行逻辑由 ExperimentRunner 处理。
    """
    variant_id: str
    name: str
    description: str
    target_component: str
    params: Dict[str, Any] = field(default_factory=dict)
    implementation_hint: Optional[str] = None

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ExperimentConfig:
    """
    实验配置：描述要对哪个组件做什么样的对比试验。

    - experiment_id    : 唯一 ID
    - name             : 标题，例如 "planner_strategies_v1"
    - purpose          : 实验目的（自然语言）
    - target_component : 实验作用的组件（与 VariantConfig.target_component 一致）
    - created_at       : ISO 时间
    - created_by       : 人/系统名（v1 可写 "system"）
    - status           : "pending" | "running" | "completed" | "failed"
    - metadata         : 任意附加信息（数据集信息 / 调度策略 / 域等）
    - variants         : 变体列表
    """
    experiment_id: str
    name: str
    purpose: str
    target_component: str
    created_at: str
    created_by: str
    status: str = "pending"
    metadata: Dict[str, Any] = field(default_factory=dict)
    variants: List[VariantConfig] = field(default_factory=list)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        # dataclasses 默认会把 dataclass 展开；这里 variants 已是 dict 列表了
        return d


@dataclass
class VariantResult:
    """
    单个 Variant 的评测结果。
    metrics 里可以放任意指标，如:
      - "success_rate"
      - "avg_latency"
      - "avg_cost"
      - "reward"
    """
    variant_id: str
    metrics: Dict[str, float]
    extra: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ExperimentResult:
    """
    整个 Experiment 的结果汇总。
    """
    experiment_id: str
    status: str  # "pending" | "running" | "completed" | "failed"
    started_at: Optional[str] = None
    finished_at: Optional[str] = None
    variant_results: List[VariantResult] = field(default_factory=list)
    best_variant_id: Optional[str] = None
    best_metric_name: Optional[str] = None
    notes: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


# --------- 存储层：ExperimentStore --------- #

class ExperimentStore:
    """
    v1: 用文件系统存 ExperimentConfig 和 ExperimentResult。
    将来可替换为 DB / 对象存储。
    """

    def __init__(self, *, repo_root: str | Path = ".") -> None:
        self.repo_root = Path(repo_root).resolve()
        self.base_dir = self.repo_root / "data" / "experiments"
        self.base_dir.mkdir(parents=True, exist_ok=True)

        self.config_dir = self.base_dir / "configs"
        self.config_dir.mkdir(parents=True, exist_ok=True)

        self.result_dir = self.base_dir / "results"
        self.result_dir.mkdir(parents=True, exist_ok=True)

    # ---- Config 操作 ---- #
    def create_experiment(
        self,
        *,
        name: str,
        purpose: str,
        target_component: str,
        created_by: str = "system",
        metadata: Optional[Dict[str, Any]] = None,
        variants: Optional[List[VariantConfig]] = None,
    ) -> ExperimentConfig:
        experiment_id = uuid.uuid4().hex
        cfg = ExperimentConfig(
            experiment_id=experiment_id,
            name=name,
            purpose=purpose,
            target_component=target_component,
            created_at=datetime.utcnow().isoformat(),
            created_by=created_by,
            status="pending",
            metadata=metadata or {},
            variants=variants or [],
        )
        self.save_config(cfg)
        return cfg

    def save_config(self, cfg: ExperimentConfig) -> None:
        path = self.config_dir / f"{cfg.experiment_id}.json"
        try:
            path.write_text(
                json.dumps(cfg.to_dict(), ensure_ascii=False, indent=2),
                encoding="utf-8",
            )
        except Exception:
            logger.exception("ExperimentStore: failed to save config %s", path)

    def load_config(self, experiment_id: str) -> Optional[ExperimentConfig]:
        path = self.config_dir / f"{experiment_id}.json"
        if not path.exists():
            return None
        try:
            obj = json.loads(path.read_text(encoding="utf-8"))
            variants = [
                VariantConfig(
                    variant_id=v["variant_id"],
                    name=v.get("name", ""),
                    description=v.get("description", ""),
                    target_component=v.get("target_component", ""),
                    params=v.get("params", {}),
                    implementation_hint=v.get("implementation_hint"),
                )
                for v in obj.get("variants", [])
            ]
            return ExperimentConfig(
                experiment_id=obj["experiment_id"],
                name=obj["name"],
                purpose=obj.get("purpose", ""),
                target_component=obj.get("target_component", ""),
                created_at=obj.get("created_at", ""),
                created_by=obj.get("created_by", ""),
                status=obj.get("status", "pending"),
                metadata=obj.get("metadata", {}),
                variants=variants,
            )
        except Exception:
            logger.exception("ExperimentStore: failed to load config %s", path)
            return None

    # ---- Result 操作 ---- #
    def save_result(self, result: ExperimentResult) -> None:
        path = self.result_dir / f"{result.experiment_id}.json"
        try:
            path.write_text(
                json.dumps(result.to_dict(), ensure_ascii=False, indent=2),
                encoding="utf-8",
            )
        except Exception:
            logger.exception("ExperimentStore: failed to save result %s", path)

    def load_result(self, experiment_id: str) -> Optional[ExperimentResult]:
        path = self.result_dir / f"{experiment_id}.json"
        if not path.exists():
            return None
        try:
            obj = json.loads(path.read_text(encoding="utf-8"))
            variant_results = [
                VariantResult(
                    variant_id=v["variant_id"],
                    metrics=v.get("metrics", {}),
                    extra=v.get("extra", {}),
                )
                for v in obj.get("variant_results", [])
            ]
            return ExperimentResult(
                experiment_id=obj["experiment_id"],
                status=obj.get("status", "pending"),
                started_at=obj.get("started_at"),
                finished_at=obj.get("finished_at"),
                variant_results=variant_results,
                best_variant_id=obj.get("best_variant_id"),
                best_metric_name=obj.get("best_metric_name"),
                notes=obj.get("notes", ""),
            )
        except Exception:
            logger.exception("ExperimentStore: failed to load result %s", path)
            return None

# quentin/research/orchestrator.py
from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Optional, Protocol, runtime_checkable
from datetime import datetime
import logging

from quentin.research.experiments import (
    ExperimentStore,
    ExperimentConfig,
    VariantConfig,
    ExperimentResult,
    VariantResult,
)

logger = logging.getLogger(__name__)


@runtime_checkable
class ExperimentRunner(Protocol):
    """
    某一类组件的实验执行器接口。

    比如：
    - PlannerRunner：对不同 planner 策略的 Variant 做评测
    - ModelRouterRunner：对不同路由策略做评测
    - SkillEvolutionRunner：对不同技能演化策略做评测

    run_variant 必须返回 VariantResult。
    """

    def run_variant(
        self,
        experiment: ExperimentConfig,
        variant: VariantConfig,
        context: Optional[Dict[str, Any]] = None,
    ) -> VariantResult:
        ...


@dataclass
class OrchestratorConfig:
    """
    Orchestrator 的配置占位。
    目前没啥字段，以后可以放：
    - 默认评测 metric 名称
    - 同时运行的最大实验数量等
    """
    default_score_metric: str = "score"


class ExperimentOrchestrator:
    """
    v1：轻量实验编排器。

    特点：
    - 不绑定具体领域：只知道有“组件组件(component)”和“变体(variant)”；
    - 具体怎么评测由 ExperimentRunner 决定；
    - 结果统一写入 ExperimentStore。
    """

    def __init__(
        self,
        *,
        repo_root: str | None = None,
        store: Optional[ExperimentStore] = None,
        config: Optional[OrchestratorConfig] = None,
    ) -> None:
        from pathlib import Path

        self.repo_root = Path(repo_root or ".").resolve()
        self.store = store or ExperimentStore(repo_root=self.repo_root)
        self.config = config or OrchestratorConfig()

        # 每个 target_component 对应一个 Runner
        self.runners: Dict[str, ExperimentRunner] = {}

    # -------- Runner 注册 -------- #
    def register_runner(self, component: str, runner: ExperimentRunner) -> None:
        """
        注册某个组件的 Runner。

        component 如：
        - "planner"
        - "model_router"
        - "skill_evolution"
        """
        self.runners[component] = runner

    # -------- Experiment 管理 -------- #
    def create_experiment(
        self,
        *,
        name: str,
        purpose: str,
        target_component: str,
        created_by: str = "system",
        metadata: Optional[Dict[str, Any]] = None,
        variants: Optional[List[VariantConfig]] = None,
    ) -> ExperimentConfig:
        cfg = self.store.create_experiment(
            name=name,
            purpose=purpose,
            target_component=target_component,
            created_by=created_by,
            metadata=metadata,
            variants=variants,
        )
        logger.info(
            "ExperimentOrchestrator: created experiment %s for component=%s",
            cfg.experiment_id,
            target_component,
        )
        return cfg

    def run_experiment(
        self,
        experiment_id: str,
        *,
        context: Optional[Dict[str, Any]] = None,
        score_metric: Optional[str] = None,
    ) -> ExperimentResult:
        """
        顺序执行指定实验的所有 variants，并产生 ExperimentResult。

        - context 由具体 Runner 自行解释（可包含数据集路径、离线样本集等）
        """
        cfg = self.store.load_config(experiment_id)
        if cfg is None:
            raise ValueError(f"ExperimentOrchestrator: experiment {experiment_id} not found")

        runner = self.runners.get(cfg.target_component)
        if runner is None:
            raise RuntimeError(
                f"ExperimentOrchestrator: no runner registered for component={cfg.target_component}"
            )

        result = ExperimentResult(
            experiment_id=experiment_id,
            status="running",
            started_at=datetime.utcnow().isoformat(),
            finished_at=None,
            variant_results=[],
            best_variant_id=None,
            best_metric_name=None,
            notes="",
        )
        self.store.save_result(result)  # 立刻写一遍“running”状态

        metric_name = score_metric or self.config.default_score_metric
        best_score = None
        best_variant_id = None

        for v in cfg.variants:
            try:
                vr = runner.run_variant(cfg, v, context=context)
            except Exception as e:
                logger.exception(
                    "ExperimentOrchestrator: runner failed for experiment %s variant %s",
                    experiment_id,
                    v.variant_id,
                )
                # 失败也要写入一个结果占位
                vr = VariantResult(
                    variant_id=v.variant_id,
                    metrics={},
                    extra={"error": str(e)},
                )

            result.variant_results.append(vr)

            # 比较 score
            score = vr.metrics.get(metric_name)
            if score is not None:
                if best_score is None or score > best_score:
                    best_score = score
                    best_variant_id = v.variant_id

            # 每个 variant 跑完都可以增量保存一次（方便中断恢复）
            self.store.save_result(result)

        result.status = "completed"
        result.finished_at = datetime.utcnow().isoformat()
        result.best_variant_id = best_variant_id
        result.best_metric_name = metric_name

        if best_variant_id is None:
            result.notes = (
                "No variant produced a valid score metric; "
                "best_variant_id remains None. Check metrics or runner implementation."
            )

        self.store.save_result(result)
        logger.info(
            "ExperimentOrchestrator: experiment %s completed. best_variant=%s metric=%s score=%s",
            experiment_id,
            best_variant_id,
            metric_name,
            best_score,
        )
        return result

    def compare_variants(
        self,
        experiment_id: str,
        metric_name: Optional[str] = None,
    ) -> Optional[Dict[str, Any]]:
        """
        简单比较实验中各变体在指定 metric 上的表现。
        返回结构：
        {
          "experiment_id": ...,
          "metric": ...,
          "best_variant_id": ...,
          "scores": { variant_id: score_or_None, ... }
        }
        """
        result = self.store.load_result(experiment_id)
        if result is None:
            return None

        m = metric_name or result.best_metric_name or self.config.default_score_metric
        scores: Dict[str, Optional[float]] = {}
        best_variant_id = None
        best_score = None

        for vr in result.variant_results:
            score = vr.metrics.get(m)
            scores[vr.variant_id] = score
            if score is not None:
                if best_score is None or score > best_score:
                    best_score = score
                    best_variant_id = vr.variant_id

        return {
            "experiment_id": experiment_id,
            "metric": m,
            "best_variant_id": best_variant_id,
            "scores": scores,
        }

# quentin/research/interface.py
from __future__ import annotations

from typing import Any, Dict, Optional
import logging

from quentin.immune.store import ImmuneCase
from quentin.research.experiments import ExperimentStore, VariantConfig
from quentin.research.orchestrator import ExperimentOrchestrator

logger = logging.getLogger(__name__)


class ResearchInterface:
    """
    v1：一个轻量的 Research Layer 接口壳子 + ExperimentOrchestrator 适配层。

    主要职责：
    - 接收 ImmuneCase，做统一入口（当前打日志 + 可选自动生成实验）
    - 对上层暴露 create_experiment / run_experiment / compare_variants 等方法
    """

    def __init__(
        self,
        *,
        repo_root: str | None = None,
        orchestrator: Optional[ExperimentOrchestrator] = None,
        store: Optional[ExperimentStore] = None,
    ) -> None:
        from pathlib import Path

        self.repo_root = Path(repo_root or ".").resolve()
        self.store = store or ExperimentStore(repo_root=self.repo_root)
        self.orchestrator = orchestrator or ExperimentOrchestrator(
            repo_root=self.repo_root,
            store=self.store,
        )

    # ---- ImmuneCase 接口 ---- #
    def submit_immune_case(self, case: ImmuneCase) -> None:
        """
        ImmuneLoop 检测到异常后调用：
        - 当前：打日志；
        - 将来：可以自动生成实验，或推送到队列，由人/AI 决定是否开实验。
        """
        logger.info(
            "ResearchInterface: received immune case %s (type=%s severity=%s)",
            case.case_id,
            case.anomaly.get("anomaly_type"),
            case.anomaly.get("severity"),
        )
        # v1 不自动建实验，只是记录。
        # 若你想试验“全自动自演化”，可以在这里调用 create_experiment_from_case。

    # ---- 实验 API ---- #
    def create_experiment_from_case(self, case: ImmuneCase) -> str:
        """
        根据 ImmuneCase 创建一个简单的实验占位：
        比如：
        - 目标组件：planner / model_router / skill_evolution 等
        - 暂时创建两个占位 Variant（baseline / candidate），后续可以由 AI 来填充 params。
        """
        anomaly_type = case.anomaly.get("anomaly_type", "unknown")
        target_component = self._infer_target_component_from_case(case)

        name = f"auto_from_case_{case.case_id[:8]}"
        purpose = f"Auto experiment derived from immune case {case.case_id} ({anomaly_type})"

        # v1: 先生成两个占位 Variant，具体 params 由你后续填
        variants = [
            VariantConfig(
                variant_id="baseline",
                name="baseline",
                description="Current / previous stable configuration",
                target_component=target_component,
                params={"mode": "baseline"},
            ),
            VariantConfig(
                variant_id="candidate",
                name="candidate",
                description="New configuration to try (to be filled by AI/human).",
                target_component=target_component,
                params={"mode": "candidate"},
            ),
        ]

        cfg = self.orchestrator.create_experiment(
            name=name,
            purpose=purpose,
            target_component=target_component,
            created_by="immune_auto",
            metadata={
                "source_case_id": case.case_id,
                "anomaly_type": anomaly_type,
            },
            variants=variants,
        )
        logger.info(
            "ResearchInterface: created experiment %s from immune case %s",
            cfg.experiment_id,
            case.case_id,
        )
        return cfg.experiment_id

    def run_experiment(self, experiment_id: str, context: Optional[Dict[str, Any]] = None) -> Dict[str, Any]:
        """
        运行某个实验，并返回结果的 dict。
        """
        result = self.orchestrator.run_experiment(experiment_id, context=context)
        return result.to_dict()

    def compare_variants(
        self,
        experiment_id: str,
        metric_name: Optional[str] = None,
    ) -> Optional[Dict[str, Any]]:
        """
        比较实验中的各变体表现。
        """
        return self.orchestrator.compare_variants(experiment_id, metric_name=metric_name)

    # ---- 内部辅助 ---- #
    def _infer_target_component_from_case(self, case: ImmuneCase) -> str:
        """
        从 ImmuneCase 推断目标组件。
        v1: 极简版——只看 anomaly_type 和 goal_summary.domain。
        将来可以接更智能的推断（甚至让 AI 决定）。
        """
        domain = (case.goal_summary.get("domain") or "").lower()
        anomaly_type = case.anomaly.get("anomaly_type", "")

        # 示例规则：Trading 域优先给 model_router / risk_manager 做实验
        if "trading" in domain:
            return "model_router"
        # Legal 域优先给 planner / reasoning 做实验
        if "legal" in domain:
            return "planner"
        # 其他默认 planner
        if "high_failure_rate" in anomaly_type:
            return "planner"
        return "planner"
