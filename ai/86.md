# quentin/core/executor.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional
from datetime import datetime
import logging
import importlib
import json

from pathlib import Path

from quentin.skills.manager import SkillManager, SkillSpec
from quentin.models.router import ModelRouter
from quentin.models.llm import call_llm

logger = logging.getLogger(__name__)


@dataclass
class StepExecutionRecord:
    """用于写入 Trace / 日志的执行结果结构。"""

    step_id: str
    description: str
    skill_id: Optional[str]
    executor_mode: str  # "skill" | "llm" | "noop"
    success: bool
    result: Any = None
    error: Optional[str] = None
    started_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    finished_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class Executor:
    """
    Executor v1.1：支持 Skill 调用 + LLM fallback 的通用执行器。

    设计目标：
    - 先尝试执行 skill（如果 step 指定了 skill_id 且已实现）；
    - skill 调用失败或不存在时，回退到 LLM 执行；
    - 将每个步骤的执行记录包装成 StepExecutionRecord，便于 Trace & SleepLoop 使用。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        model_router: Optional[ModelRouter] = None,
        skill_manager: Optional[SkillManager] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.model_router = model_router or ModelRouter()
        self.skill_manager = skill_manager or SkillManager(repo_root=self.repo_root)

    # ---------- 对外主入口 ----------

    def execute_plan(
        self,
        *,
        goal: Any,
        plan: Any,
        context: Dict[str, Any],
    ) -> Dict[str, Any]:
        """
        执行一个 PlannerResult 风格的 plan。

        预期 plan 结构（和前面 _run_planner 的输出保持一致）：
        - plan.steps: List[{"step_id", "description", "skill_id", "notes"}]
        """
        records: List[StepExecutionRecord] = []
        overall_success = True

        steps = getattr(plan, "steps", None) or plan.get("steps", [])
        for step in steps:
            rec = self._execute_step(step=step, goal=goal, context=context)
            records.append(rec)
            if not rec.success:
                overall_success = False

        return {
            "success": overall_success,
            "steps": [r.to_dict() for r in records],
        }

    # ---------- 执行单步 ----------

    def _execute_step(
        self,
        *,
        step: Dict[str, Any],
        goal: Any,
        context: Dict[str, Any],
    ) -> StepExecutionRecord:
        desc = step.get("description") or ""
        step_id = step.get("step_id") or "step"
        skill_id = step.get("skill_id")

        rec = StepExecutionRecord(
            step_id=step_id,
            description=desc,
            skill_id=skill_id,
            executor_mode="noop",
            success=False,
        )
        rec.started_at = datetime.utcnow().isoformat()

        # 1) 有 skill_id 的优先尝试 Skill 调用
        if skill_id:
            logger.info("Executor: step %s suggests skill_id=%s", step_id, skill_id)
            ok = self._try_execute_skill(step=step, goal=goal, context=context, record=rec)
            if ok:
                rec.executor_mode = "skill"
                rec.finished_at = datetime.utcnow().isoformat()
                return rec
            else:
                # skill 调用失败，fallback 到 LLM
                logger.warning("Executor: skill %s failed, fallback to LLM for step %s", skill_id, step_id)

        # 2) LLM fallback
        self._fallback_llm_execute(step=step, goal=goal, context=context, record=rec)
        rec.finished_at = datetime.utcnow().isoformat()
        return rec

    # ---------- Skill 调用 ----------

    def _try_execute_skill(
        self,
        *,
        step: Dict[str, Any],
        goal: Any,
        context: Dict[str, Any],
        record: StepExecutionRecord,
    ) -> bool:
        """
        如果 skill 存在且已实现，则尝试动态导入并调用：

        约定的函数签名（后面 SkillImplementationEngine 也会遵守）：
        def <func_name>(step: dict, goal: Optional[dict] = None, context: Optional[dict] = None) -> dict:
            return {
                "status": "ok" | "error",
                "result": ...,
                "meta": {...}
            }
        """
        skill_id = record.skill_id
        if not skill_id:
            return False

        try:
            sk: Optional[SkillSpec] = self.skill_manager.registry.get(skill_id)
        except Exception:
            logger.exception("Executor._try_execute_skill: failed to read skill %s", skill_id)
            record.error = "skill_registry_error"
            return False

        if sk is None:
            record.error = "skill_not_found"
            return False
        if not sk.is_active:
            record.error = f"skill_not_active(status={sk.status})"
            return False
        if not sk.implementation_module or not sk.implementation_symbol:
            record.error = "skill_not_implemented"
            return False

        try:
            module = importlib.import_module(sk.implementation_module)
            func = getattr(module, sk.implementation_symbol)
        except Exception as e:
            logger.exception("Executor._try_execute_skill: import/attr error for %s", skill_id)
            record.error = f"import_error: {e}"
            return False

        # 将 goal 简单转成 dict，避免 dataclass/复杂对象在 skill 里不好处理
        goal_dict: Dict[str, Any] = self._goal_to_dict(goal)

        try:
            result = func(step, goal_dict, context or {})
        except Exception as e:
            logger.exception("Executor._try_execute_skill: skill %s raised exception", skill_id)
            record.error = f"runtime_error: {e}"
            return False

        # 约定返回结构
        if isinstance(result, dict):
            status = result.get("status", "ok")
            record.result = result
            if status == "ok":
                record.success = True
                return True
            else:
                record.error = f"skill_status={status}"
                return False
        else:
            # 如果执行结果不是 dict，就简单认为成功但提醒一下
            record.result = result
            record.success = True
            return True

    # ---------- LLM fallback ----------

    def _fallback_llm_execute(
        self,
        *,
        step: Dict[str, Any],
        goal: Any,
        context: Dict[str, Any],
        record: StepExecutionRecord,
    ) -> None:
        """
        使用 LLM 作为通用执行路径。

        它不会真正操作外部世界，只是给出“下一步应该做什么/思考结果”，
        适合在自我反思/思维链阶段或无工具可调用时使用。
        """
        record.executor_mode = "llm"

        # route executor 模式
        domain = getattr(goal, "domain", None) or "core"
        cfg = self.model_router.route(mode="executor", task_type=domain)

        payload = {
            "step": step,
            "goal": self._goal_to_dict(goal),
            "context": context or {},
        }
        user_prompt = json.dumps(payload, ensure_ascii=False, indent=2)

        system_prompt = (
            "You are the EXECUTION AGENT in a self-improving system.\n"
            "You receive a single planned step and the current goal/context.\n"
            "You MUST return a JSON object only, describing:\n"
            "{\n"
            "  \"status\": \"ok\" | \"error\",\n"
            "  \"result\": \"short natural language description of what you did or concluded\",\n"
            "  \"meta\": {\n"
            "    \"warnings\": [\"...\"],\n"
            "    \"notes\": \"optional extra info\"\n"
            "  }\n"
            "}\n"
            "Do not output anything outside of this JSON.\n"
        )

        try:
            raw = call_llm(cfg, user_prompt=user_prompt, system_prompt=system_prompt).strip()
        except Exception as e:
            logger.exception("Executor._fallback_llm_execute: LLM call failed")
            record.success = False
            record.error = f"llm_error: {e}"
            return

        try:
            first = raw.find("{")
            last = raw.rfind("}")
            if first != -1 and last != -1 and last > first:
                raw_json = raw[first : last + 1]
            else:
                raw_json = raw
            obj = json.loads(raw_json)
        except Exception:
            logger.exception("Executor._fallback_llm_execute: failed to parse JSON")
            record.success = False
            record.error = "llm_parse_error"
            record.result = raw
            return

        record.result = obj
        status = obj.get("status", "ok")
        if status == "ok":
            record.success = True
        else:
            record.success = False
            record.error = f"llm_status={status}"

    # ---------- 辅助：Goal 序列化 ----------

    def _goal_to_dict(self, goal: Any) -> Dict[str, Any]:
        # 这里假定 Goal 是 dataclass 或有常见字段；尽量宽松
        if goal is None:
            return {}
        if isinstance(goal, dict):
            return goal
        out: Dict[str, Any] = {}
        for attr in ("id", "title", "description", "domain", "metadata"):
            if hasattr(goal, attr):
                out[attr] = getattr(goal, attr)
        return out

四、现在这一步完成后，系统具备了什么新能力？
从蓝皮书视角，我们又往前推了一大步：

「技能和自编程不只是写在 JSON 和代码文件里，而是真的被行动层调用。」

当前闭环已经是可运行闭环：

CognitiveLoop 处理任务 → Planner 生成带 skill_id 的 plan；

SleepLoop 从成功 Trace 中抽技能 → SkillManager 管理 → SkillImplementationEngine 生成代码 → auto_skills.py；

Executor 在执行 plan 时：

如果 step 指定了 skill_id，并且 Skill 已 active + implemented：
→ 动态 import 对应函数 → 调用（返回结构写进 Trace）

如果 skill 调用失败或不存在：
→ fallback 到 LLM 执行，结果仍然结构化写进记录。

这意味着：

新生成的技能一旦被实现，就能被后续任务直接复用；

自编程引擎不是“写完代码就算了”，而是产生真实行为；

SleepLoop / ResearchLayer 可以从 StepExecutionRecord 中看到“哪些技能被成功调用、收益如何”，为下一轮优化提供数据。
