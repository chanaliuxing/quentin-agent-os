一、Eval Harness：从 CodePatch 提炼指标
新文件：quentin/evaluation/harness.py

这个模块只做一件事：

# quentin/evaluation/harness.py
from __future__ import annotations

import logging
from dataclasses import dataclass, asdict
from typing import Dict, Any, Optional

from quentin.self_programming.patches import CodePatch

logger = logging.getLogger(__name__)


@dataclass
class PatchOutcomeMetrics:
    """
    针对单个 CodePatch 的评估结果。

    v1 是一个简单但统一的结构：
    - success        : 是否“总体成功”（测试过 + 通过 + 没有明显高风险）
    - score          : 0~1 的综合分（方便做排序/对比）
    - tests_success  : 测试是否通过
    - tests_run      : 是否有跑过测试
    - review_present : 是否有多 AI 评审
    - high_risk_flag : 是否从评审文本中 heuristically 检测出“高风险”提示
    - stats          : 一些基本统计（cases 数、pass/fail 等）
    - excerpts       : 截断的测试输出 / 评审摘要，方便 dashboard 展示
    """
    metrics_id: str
    patch_id: str

    success: bool
    score: float

    tests_success: bool
    tests_run: bool
    review_present: bool
    high_risk_flag: bool

    stats: Dict[str, Any]
    excerpts: Dict[str, str]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class EvalHarness:
    """
    v1 版评估框架：
    - 只负责对 CodePatch 做“离线评分”（不调用 LLM），方便 Research Layer 聚合分析。
    """

    def summarize_patch(self, patch: CodePatch) -> PatchOutcomeMetrics:
        patch_id = patch.patch_id
        metrics_id = f"{patch_id}::metrics"

        # ---- 测试相关 ----
        tests = patch.tests
        if tests is None:
            tests_run = False
            tests_success = False
            test_stats = {
                "total_cases": 0,
                "passed": 0,
                "failed": 0,
                "errored": 0,
                "skipped": 0,
                "runtime_sec": None,
            }
            test_excerpt = ""
        else:
            tests_run = True
            tests_success = bool(tests.success)
            test_stats = {
                "total_cases": tests.total_cases,
                "passed": tests.passed,
                "failed": tests.failed,
                "errored": tests.errored,
                "skipped": tests.skipped,
                "runtime_sec": tests.runtime_sec,
            }
            # 简单截取第一条用例的错误信息/输出预览
            if tests.cases:
                c0 = tests.cases[0]
                err = (c0.error_message or "")[:400]
                test_excerpt = err
            else:
                test_excerpt = ""

        # ---- 评审相关 ----
        review = patch.review
        review_present = review is not None
        high_risk_flag = False
        review_excerpt = ""

        if review is not None:
            summary = (review.summary or "")[:800]
            risks = (review.risks or "")[:800]
            review_excerpt = summary + "\n\n" + risks

            text = (summary + " " + risks).lower()
            # 非常简单的高风险 heuristics，可以之后用 LLM 再精细化
            risk_keywords = [
                "high risk",
                "严重风险",
                "highly unsafe",
                "重大缺陷",
                "严重 bug",
            ]
            for kw in risk_keywords:
                if kw in text:
                    high_risk_flag = True
                    break

        # ---- 综合评分逻辑 v1 ----
        # 这是一个简单可解释的 heuristics，后面可以用 ML / bandit 替换：
        #   - 测试通过 +0.6
        #   - 有评审       +0.2
        #   - 无测试但有评审视为 0.3
        #   - 检测到高风险 -0.3
        score = 0.0
        if tests_run and tests_success:
            score += 0.6
        elif tests_run and not tests_success:
            score += 0.1  # 至少说明跑过测试，有信号
        elif (not tests_run) and review_present:
            score += 0.3

        if review_present:
            score += 0.2

        if high_risk_flag:
            score -= 0.3

        if score < 0:
            score = 0.0
        if score > 1:
            score = 1.0

        success = tests_success and (not high_risk_flag)

        stats = {
            "tests": test_stats,
            "review_present": review_present,
            "high_risk_flag": high_risk_flag,
        }
        excerpts = {
            "tests": test_excerpt,
            "review": review_excerpt,
        }

        return PatchOutcomeMetrics(
            metrics_id=metrics_id,
            patch_id=patch_id,
            success=success,
            score=score,
            tests_success=tests_success,
            tests_run=tests_run,
            review_present=review_present,
            high_risk_flag=high_risk_flag,
            stats=stats,
            excerpts=excerpts,
        )

from __future__ import annotations

import json
import logging
from dataclasses import dataclass, asdict, field
from datetime import datetime
from pathlib import Path
from statistics import mean
from typing import Dict, Any, List, Optional

from quentin.self_programming.patches import (
    CodePatch,
    FileBackedPatchStore,
)
from quentin.evaluation.harness import EvalHarness, PatchOutcomeMetrics

logger = logging.getLogger(__name__)


@dataclass
class ExperimentVariant:
    """
    一个实验变体，对应一个 CodePatch。

    这里主要关心：
    - variant_id: 通常就用 patch_id
    - patch_id  : 用于回溯
    - score     : EvalHarness 计算出来的综合分
    - success   : 是否成功
    - metrics   : EvalHarness 的完整 metrics 字典（方便后续扩展）
    """
    variant_id: str
    patch_id: str
    created_at: str
    score: float
    success: bool
    metrics: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "ExperimentVariant":
        return cls(
            variant_id=data["variant_id"],
            patch_id=data["patch_id"],
            created_at=data.get("created_at", ""),
            score=float(data.get("score", 0.0)),
            success=bool(data.get("success", False)),
            metrics=data.get("metrics", {}),
        )


@dataclass
class Experiment:
    """
    一个实验集合，对应蓝皮书里的某个能力 + requirement。
    kind 目前固定为 "self_programming_patch"，未来可以扩展成 routing/agent 等其他类型。

    - experiment_id: 唯一 id，例如 "self_programming_patch::memory_fabric::mf.search_semantic"
    - capability   : e.g. "memory_fabric"
    - requirement_id: e.g. "mf.search_semantic" 或 None
    - variants     : 多个 patch 作为变体
    """
    experiment_id: str
    kind: str
    capability: str
    requirement_id: Optional[str]

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    variants: List[ExperimentVariant] = field(default_factory=list)

    def to_dict(self) -> Dict[str, Any]:
        return {
            "experiment_id": self.experiment_id,
            "kind": self.kind,
            "capability": self.capability,
            "requirement_id": self.requirement_id,
            "created_at": self.created_at,
            "updated_at": self.updated_at,
            "variants": [v.to_dict() for v in self.variants],
        }

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "Experiment":
        variants = [
            ExperimentVariant.from_dict(v)
            for v in data.get("variants", [])
        ]
        return cls(
            experiment_id=data["experiment_id"],
            kind=data.get("kind", "self_programming_patch"),
            capability=data["capability"],
            requirement_id=data.get("requirement_id"),
            created_at=data.get("created_at", ""),
            updated_at=data.get("updated_at", ""),
            variants=variants,
        )

    def aggregate_score(self) -> Dict[str, Any]:
        """
        聚合这组变体的整体信息，用于 ranking：
        - avg_score, best_score, num_variants, success_rate
        """
        if not self.variants:
            return {
                "avg_score": 0.0,
                "best_score": 0.0,
                "num_variants": 0,
                "success_rate": 0.0,
            }
        scores = [v.score for v in self.variants]
        successes = [1 for v in self.variants if v.success]

        avg_score = float(mean(scores)) if scores else 0.0
        best_score = float(max(scores)) if scores else 0.0
        num_variants = len(self.variants)
        success_rate = len(successes) / num_variants if num_variants > 0 else 0.0

        return {
            "avg_score": avg_score,
            "best_score": best_score,
            "num_variants": num_variants,
            "success_rate": success_rate,
        }


class ExperimentStore:
    """
    实验存储：一个 JSON 文件 {experiment_id: experiment_dict}
    """

    def __init__(self, path: str | Path = "data/experiments.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._cache: Dict[str, Dict[str, Any]] = {}
        self._loaded = False

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                with self.path.open("r", encoding="utf-8") as f:
                    self._cache = json.load(f)
            except Exception as e:
                logger.exception("Failed to load experiments from %s", self.path)
                self._cache = {}
        else:
            self._cache = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self.path.open("w", encoding="utf-8") as f:
            json.dump(self._cache, f, ensure_ascii=False, indent=2)

    def get_all(self) -> Dict[str, Experiment]:
        self._load()
        return {
            exp_id: Experiment.from_dict(d)
            for exp_id, d in self._cache.items()
        }

    def upsert(self, experiment: Experiment) -> None:
        self._load()
        self._cache[experiment.experiment_id] = experiment.to_dict()
        self._save()
ResearchManager：同步 & 分析
# 同一个文件 quentin/research/experiments.py 继续追加

class ResearchManager:
    """
    Research & Evolution Layer v1：

    现在只做一件事：
    - 从所有 CodePatch + EvalHarness -> 构建/更新 Experiment 集合
    - 提供一些排名查询接口（top capabilities / top patches）

    后续可以在这里加：
    - Budget & Cost Controller
    - Evolution Strategy Manager
    """

    def __init__(
        self,
        patch_store: Optional[FileBackedPatchStore] = None,
        experiment_store: Optional[ExperimentStore] = None,
        eval_harness: Optional[EvalHarness] = None,
    ) -> None:
        self.patch_store = patch_store or FileBackedPatchStore()
        self.experiment_store = experiment_store or ExperimentStore()
        self.eval_harness = eval_harness or EvalHarness()

    # ---------- 同步：从 patches -> experiments ----------

    def sync_patch_experiments(self) -> int:
        """
        遍历所有 CodePatch：
        - 通过 EvalHarness 计算 PatchOutcomeMetrics
        - 将其归入对应 Experiment（按 capability + requirement_id 分组）

        返回处理过的 patch 数量。
        """
        raw = self.patch_store.get_all()
        patches = [CodePatch.from_dict(d) for d in raw.values()]

        experiments = self.experiment_store.get_all()

        count = 0
        for patch in patches:
            metrics = self.eval_harness.summarize_patch(patch)
            exp_id = self._experiment_id_for_patch(patch)
            experiment = experiments.get(exp_id)

            if experiment is None:
                experiment = Experiment(
                    experiment_id=exp_id,
                    kind="self_programming_patch",
                    capability=patch.capability,
                    requirement_id=patch.requirement_id,
                )

            # 如果该 patch 已经作为 variant 存在，则更新指标；否则追加
            existing = None
            for v in experiment.variants:
                if v.patch_id == patch.patch_id:
                    existing = v
                    break

            if existing is None:
                variant = ExperimentVariant(
                    variant_id=patch.patch_id,
                    patch_id=patch.patch_id,
                    created_at=datetime.utcnow().isoformat(),
                    score=metrics.score,
                    success=metrics.success,
                    metrics=metrics.to_dict(),
                )
                experiment.variants.append(variant)
            else:
                existing.score = metrics.score
                existing.success = metrics.success
                existing.metrics = metrics.to_dict()

            experiment.updated_at = datetime.utcnow().isoformat()
            experiments[exp_id] = experiment
            count += 1

        # 写回存储
        for exp in experiments.values():
            self.experiment_store.upsert(exp)

        logger.info("Synced experiments from %d patches.", count)
        return count

    def _experiment_id_for_patch(self, patch: CodePatch) -> str:
        # requirement_id 可能为 None
        req = patch.requirement_id or "ALL"
        return f"self_programming_patch::{patch.capability}::{req}"

    # ---------- 查询：top capabilities ----------

    def top_capabilities(self, limit: int = 10) -> List[Dict[str, Any]]:
        """
        返回按 avg_score 排序的 capability 列表：
        [
          {
            "experiment_id": "...",
            "capability": "memory_fabric",
            "requirement_id": "mf.search_semantic",
            "avg_score": 0.83,
            "best_score": 0.95,
            "success_rate": 0.75,
            "num_variants": 4,
            "updated_at": "..."
          },
          ...
        ]
        """
        exps = self.experiment_store.get_all()
        rows: List[Dict[str, Any]] = []

        for exp in exps.values():
            agg = exp.aggregate_score()
            row = {
                "experiment_id": exp.experiment_id,
                "capability": exp.capability,
                "requirement_id": exp.requirement_id,
                "avg_score": agg["avg_score"],
                "best_score": agg["best_score"],
                "success_rate": agg["success_rate"],
                "num_variants": agg["num_variants"],
                "updated_at": exp.updated_at,
            }
            rows.append(row)

        rows.sort(key=lambda r: r["avg_score"], reverse=True)
        return rows[:limit]

    # ---------- 查询：top patches ----------

    def top_patches(self, limit: int = 10) -> List[Dict[str, Any]]:
        """
        直接从所有 ExperimentVariant 里选出得分最高的若干 patch。
        返回：
        [
          {
            "patch_id": "...",
            "capability": "...",
            "requirement_id": "...",
            "score": 0.93,
            "success": true,
            "experiment_id": "...",
          },
          ...
        ]
        """
        exps = self.experiment_store.get_all()
        rows: List[Dict[str, Any]] = []

        for exp in exps.values():
            for v in exp.variants:
                rows.append(
                    {
                        "patch_id": v.patch_id,
                        "capability": exp.capability,
                        "requirement_id": exp.requirement_id,
                        "score": v.score,
                        "success": v.success,
                        "experiment_id": exp.experiment_id,
                    }
                )

        rows.sort(key=lambda r: r["score"], reverse=True)
        return rows[:limit]

def cmd_research_sync(args: argparse.Namespace) -> None:
    mgr = ResearchManager()
    count = mgr.sync_patch_experiments()
    print(f"Synced experiments from {count} patches.")


def cmd_research_top_capabilities(args: argparse.Namespace) -> None:
    mgr = ResearchManager()
    rows = mgr.top_capabilities(limit=args.limit)
    if not rows:
        print("No experiments found.")
        return
    print("Top capabilities by avg_score:")
    for r in rows:
        cap = r["capability"]
        req = r["requirement_id"] or "ALL"
        print(
            f"- {cap} / {req} :: avg_score={r['avg_score']:.3f}, "
            f"best={r['best_score']:.3f}, success_rate={r['success_rate']:.2%}, "
            f"variants={r['num_variants']}, updated_at={r['updated_at']}"
        )


def cmd_research_top_patches(args: argparse.Namespace) -> None:
    mgr = ResearchManager()
    rows = mgr.top_patches(limit=args.limit)
    if not rows:
        print("No experiment variants found.")
        return
    print("Top patches by score:")
    for r in rows:
        cap = r["capability"]
        req = r["requirement_id"] or "ALL"
        status = "OK" if r["success"] else "FAIL"
        print(
            f"- [{status}] patch={r['patch_id']} :: cap={cap}/{req} :: "
            f"score={r['score']:.3f} :: exp={r['experiment_id']}"
        )
