2.5 get_stats()：按模型聚合统计
    async def get_stats(self, limit: int = 1000) -> Dict[str, Any]:
        """
        聚合最近若干次调用的统计信息：
        - 每个 model 的调用次数 / 成功率 / 平均延迟 / 平均 cost
        """
        ids = await self.kv.get(self.usage_index_key) or []
        ids = list(dict.fromkeys(ids))[-limit:]

        per_model: Dict[str, Dict[str, Any]] = {}
        for uid in ids:
            data = await self.kv.get(f"{self.usage_prefix}{uid}")
            if not data:
                continue
            rec = ModelUsageRecord(**data)
            m = per_model.setdefault(
                rec.model_name,
                {
                    "model_name": rec.model_name,
                    "calls": 0,
                    "success": 0,
                    "total_latency": 0.0,
                    "total_cost": 0.0,
                    "cost_samples": 0,
                },
            )
            m["calls"] += 1
            if rec.success:
                m["success"] += 1
            m["total_latency"] += rec.latency_sec
            if rec.cost is not None:
                m["total_cost"] += rec.cost
                m["cost_samples"] += 1

        result_models: List[Dict[str, Any]] = []
        for m in per_model.values():
            calls = m["calls"]
            success = m["success"]
            cost_samples = m["cost_samples"]
            result_models.append(
                {
                    "model_name": m["model_name"],
                    "calls": calls,
                    "success_rate": success / calls if calls else 0.0,
                    "avg_latency_sec": m["total_latency"] / calls if calls else 0.0,
                    "avg_cost": m["total_cost"] / cost_samples if cost_samples else None,
                }
            )

        return {
            "models": result_models,
            "total_calls": len(ids),
        }
三、多智能体角色 & 协调器
蓝皮书要求固定 4 种基座角色：

Speed-Agent：追求快

Safety-Agent：追求稳 / 白名单

Cost-Agent：追求便宜

Domain-Expert-Agent：按 domain 做偏好

我们做的是一个通用协调器：

给一组候选模型名 + task 元信息，
每个角色投票，最后选出一个「推荐模型」。

from __future__ import annotations

from enum import Enum
from typing import Dict, Any, List
from pydantic import BaseModel, Field


class AgentRole(str, Enum):
    SPEED = "speed"
    SAFETY = "safety"
    COST = "cost"
    DOMAIN_EXPERT = "domain_expert"


class RoleVote(BaseModel):
    """
    某个角色对某个模型的打分 / 投票。
    """
    role: AgentRole
    model_name: str
    score: float = 0.0
    reason: str = ""


class RoutingDecision(BaseModel):
    """
    MultiAgentCoordinator 的总决策结果。
    """
    chosen_model: str
    votes: List[RoleVote] = Field(default_factory=list)
    meta: Dict[str, Any] = Field(default_factory=dict)
3.2 协调器实现：quentin/multi_agent/coordinator.py
# ============================================
# quentin/multi_agent/coordinator.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass
from typing import List, Dict, Any, Optional

from quentin.infra.logging import Tracer
from quentin.runtime.model_router import ModelRouter
from quentin.multi_agent.schemas import AgentRole, RoleVote, RoutingDecision
from quentin.runtime.settings import Settings  # 你之前定义的 Settings


@dataclass
class MultiAgentCoordinator:
    """
    Multi-Agent Coordinator v1：

    输入：
      - candidate_models: ["gpt-4", "deepseek-r1", ...]
      - task_meta: {domain, goal, role, ...}

    输出：
      - RoutingDecision: chosen_model + 各角色投票详情

    当前实现：
      - 利用 ModelRouter 的 get_stats() 做简单“数据驱动”
      - Speed: 优先平均延迟低的
      - Cost: 优先成本低的
      - Safety: 强制在白名单 + penalty 黑名单
      - Domain-Expert: 按 settings 中的 domain→model 偏好
    """
    tracer: Tracer
    model_router: ModelRouter
    settings: Settings

    async def _get_recent_stats(self) -> Dict[str, Any]:
        return await self.model_router.get_stats(limit=1000)

    def _get_model_stat(
        self,
        stats: Dict[str, Any],
        model_name: str,
    ) -> Dict[str, Any]:
        for m in stats.get("models", []):
            if m["model_name"] == model_name:
                return m
        return {
            "model_name": model_name,
            "calls": 0,
            "success_rate": 0.0,
            "avg_latency_sec": 0.0,
            "avg_cost": None,
        }

    # ---------- 各角色策略 ----------

    def _speed_vote(
        self,
        model_stats: Dict[str, Dict[str, Any]],
        candidates: List[str],
    ) -> List[RoleVote]:
        votes: List[RoleVote] = []
        # 越快分数越高（简单用 1 / latency）
        for m in candidates:
            s = model_stats[m]
            lat = s["avg_latency_sec"] or 0.0
            score = 0.0
            if lat > 0:
                score = 1.0 / lat
            votes.append(
                RoleVote(
                    role=AgentRole.SPEED,
                    model_name=m,
                    score=score,
                    reason=f"avg_latency_sec={lat}",
                )
            )
        return votes

    def _cost_vote(
        self,
        model_stats: Dict[str, Dict[str, Any]],
        candidates: List[str],
    ) -> List[RoleVote]:
        votes: List[RoleVote] = []
        # 越便宜分数越高（用 1 / cost，没 cost 的视作 0 分）
        for m in candidates:
            s = model_stats[m]
            cost = s.get("avg_cost")
            if cost is None or cost <= 0:
                score = 0.0
            else:
                score = 1.0 / cost
            votes.append(
                RoleVote(
                    role=AgentRole.COST,
                    model_name=m,
                    score=score,
                    reason=f"avg_cost={cost}",
                )
            )
        return votes

    def _safety_vote(
        self,
        candidates: List[str],
    ) -> List[RoleVote]:
        # 从 settings 读取白名单 / 黑名单（如果没配就都设为 neutral）
        whitelist = set(getattr(self.settings, "safety_model_whitelist", []) or [])
        blacklist = set(getattr(self.settings, "safety_model_blacklist", []) or [])

        votes: List[RoleVote] = []
        for m in candidates:
            if blacklist and m in blacklist:
                score = -1.0
                reason = "blacklisted"
            elif whitelist and m in whitelist:
                score = 1.0
                reason = "whitelisted"
            else:
                score = 0.0
                reason = "neutral"
            votes.append(
                RoleVote(
                    role=AgentRole.SAFETY,
                    model_name=m,
                    score=score,
                    reason=reason,
                )
            )
        return votes

    def _domain_expert_vote(
        self,
        candidates: List[str],
        domain: Optional[str],
    ) -> List[RoleVote]:
        """
        从 settings.domain_model_preferences 中查出对当前 domain 的偏好。
        格式例子：
          domain_model_preferences = {
            "legal": ["gpt-4", "gpt-4-mini"],
            "trading": ["deepseek-r1"],
          }
        """
        prefs: Dict[str, List[str]] = getattr(self.settings, "domain_model_preferences", {}) or {}
        preferred = prefs.get(domain or "", [])

        votes: List[RoleVote] = []
        for m in candidates:
            if preferred and m in preferred:
                score = 1.0
                reason = "preferred_for_domain"
            else:
                score = 0.0
                reason = "neutral"
            votes.append(
                RoleVote(
                    role=AgentRole.DOMAIN_EXPERT,
                    model_name=m,
                    score=score,
                    reason=reason,
                )
            )
        return votes

    # ---------- 总协调 ----------

    async def choose_model(
        self,
        candidates: List[str],
        task_meta: Optional[Dict[str, Any]] = None,
    ) -> RoutingDecision:
        """
        Multi-Agent 总入口：

        - 输入候选模型 + 任务元信息
        - 输出最终选择 + 角色投票详情
        """
        task_meta = task_meta or {}
        domain = task_meta.get("domain")

        stats = await self._get_recent_stats()
        # 整理成 model_name -> stat
        model_stats: Dict[str, Dict[str, Any]] = {}
        for m in candidates:
            model_stats[m] = self._get_model_stat(stats, m)

        # 各角色投票
        votes: List[RoleVote] = []
        votes.extend(self._speed_vote(model_stats, candidates))
        votes.extend(self._cost_vote(model_stats, candidates))
        votes.extend(self._safety_vote(candidates))
        votes.extend(self._domain_expert_vote(candidates, domain))

        # 聚合：简单求和（后续可用 weight / bandit 替换）
        agg: Dict[str, float] = {m: 0.0 for m in candidates}
        for v in votes:
            agg[v.model_name] += v.score

        # 选择总分最高的
        chosen_model = max(agg.items(), key=lambda kv: kv[1])[0]

        decision = RoutingDecision(
            chosen_model=chosen_model,
            votes=votes,
            meta={
                "scores": agg,
                "task_meta": task_meta,
            },
        )

        self.tracer.span(
            "L2",
            "multi_agent_choose_model",
            module="multi_agent",
            input={"candidates": candidates, "domain": domain},
            output={"chosen_model": chosen_model},
        )

        return decision
现在你有了一个完全通用 MultiAgent 协调器：

不绑定任何具体 LLM provider，只依赖 ModelRouter.get_stats() + Settings。

以后你只要在调用 LLM 前问一句 decision = multi_agent.choose_model([...], task_meta)，就能既可控又可实验。

@cli.command("routing-choose")
@click.option("--models", required=True, help="Comma-separated candidate models, e.g. 'gpt-4,deepseek-r1'")
@click.option("--domain", default=None, help="Optional domain name (legal/trading/desktop/...)")
def routing_choose_cmd(models: str, domain: Optional[str]):
    """
    用 MultiAgentCoordinator 对一组候选模型做一次路由决策。
    """
    async def _run():
        import json
        settings = Settings()
        runtime = AgentRuntime.create_from_settings(settings)

        candidates = [m.strip() for m in models.split(",") if m.strip()]
        decision = await runtime.multi_agent.choose_model(
            candidates=candidates,
            task_meta={"domain": domain},
        )

        print(json.dumps(decision.model_dump(), ensure_ascii=False, indent=2))

    asyncio.run(_run())
