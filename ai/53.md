from __future__ import annotations

from dataclasses import dataclass, field, asdict
from datetime import datetime
from typing import Dict, Any, List, Literal, Optional

MemoryTier = Literal["STS", "MTS", "LTS"]
MemoryKind = Literal["episodic", "semantic", "skill", "meta"]

@dataclass
class MemoryUpdatePolicy:
    """
    对应蓝皮书里的 update_policy：
    - promotion_rule   : STS/MTS 升级条件的文字描述（后续可以变成可执行规则）
    - decay_rule       : 何时衰减/删除
    - consolidation_batch: nightly / weekly / manual
    v1 先存字符串，方便人/AI读；真正的逻辑在 Sleep Loop 里实现。
    """
    promotion_rule: str = ""
    decay_rule: str = ""
    consolidation_batch: str = "nightly"

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "MemoryUpdatePolicy":
        return cls(
            promotion_rule=data.get("promotion_rule", ""),
            decay_rule=data.get("decay_rule", ""),
            consolidation_batch=data.get("consolidation_batch", "nightly"),
        )


@dataclass
class MemoryEntry:
    """
    Memory Fabric v1 的核心数据结构。

    对齐蓝皮书：
    - memory_tier : STS/MTS/LTS
    - kind        : episodic/semantic/skill/meta
    - content     : 统一放在一个 dict 里，至少有 'text' 字段；以后可以再加 'json'/'trace_ref'
    - importance  : 0~1，重要程度
    - usage_count : 被命中/使用次数（用于 Sleep Loop）
    - last_accessed_at: 上次被访问时间
    - update_policy  : promotion/decay 规则描述
    """
    memory_id: str
    memory_tier: MemoryTier
    kind: MemoryKind

    content: Dict[str, Any] = field(default_factory=dict)

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    last_accessed_at: Optional[str] = None

    importance: float = 0.5
    usage_count: int = 0

    source: str = "unknown"                # e.g. "task_execution", "sleep_consolidation"
    tags: List[str] = field(default_factory=list)
    relations: Dict[str, float] = field(default_factory=dict)  # memory_id -> weight

    update_policy: MemoryUpdatePolicy = field(default_factory=MemoryUpdatePolicy)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        # dataclasses 已经递归展开 update_policy
        return d

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "MemoryEntry":
        upd = data.get("update_policy", {})
        policy = MemoryUpdatePolicy.from_dict(upd)
        return cls(
            memory_id=data["memory_id"],
            memory_tier=data.get("memory_tier", "STS"),
            kind=data.get("kind", "episodic"),
            content=data.get("content", {}),
            created_at=data.get("created_at", ""),
            updated_at=data.get("updated_at", ""),
            last_accessed_at=data.get("last_accessed_at"),
            importance=float(data.get("importance", 0.5)),
            usage_count=int(data.get("usage_count", 0)),
            source=data.get("source", "unknown"),
            tags=list(data.get("tags", [])),
            relations=dict(data.get("relations", {})),
            update_policy=policy,
        )

# quentin/memory/store.py
from __future__ import annotations

import json
import logging
from pathlib import Path
from typing import Dict, Any, List, Optional

from quentin.memory.schemas import MemoryEntry, MemoryTier

logger = logging.getLogger(__name__)


class FileBackedMemoryStore:
    """
    最简单版的 Memory 存储：一个 JSON 文件 {memory_id: MemoryEntryDict}
    将来可以替换成 SQLite / 向量库，但对上层 API 不变。
    """

    def __init__(self, path: str | Path = "data/memory.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._cache: Dict[str, Dict[str, Any]] = {}
        self._loaded = False

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                with self.path.open("r", encoding="utf-8") as f:
                    self._cache = json.load(f)
            except Exception as e:
                logger.exception("Failed to load memory from %s", self.path)
                self._cache = {}
        else:
            self._cache = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self.path.open("w", encoding="utf-8") as f:
            json.dump(self._cache, f, ensure_ascii=False, indent=2)

    # ---------- CRUD ----------

    def get(self, memory_id: str) -> Optional[MemoryEntry]:
        self._load()
        d = self._cache.get(memory_id)
        return MemoryEntry.from_dict(d) if d else None

    def upsert(self, entry: MemoryEntry) -> None:
        self._load()
        self._cache[entry.memory_id] = entry.to_dict()
        self._save()

    def delete(self, memory_id: str) -> None:
        self._load()
        if memory_id in self._cache:
            del self._cache[memory_id]
            self._save()

    def all_entries(self) -> List[MemoryEntry]:
        self._load()
        return [MemoryEntry.from_dict(d) for d in self._cache.values()]

    def list_by_tier(self, tier: MemoryTier) -> List[MemoryEntry]:
        return [e for e in self.all_entries() if e.memory_tier == tier]

    def count_by_tier(self) -> Dict[str, int]:
        counts: Dict[str, int] = {"STS": 0, "MTS": 0, "LTS": 0}
        for e in self.all_entries():
            if e.memory_tier in counts:
                counts[e.memory_tier] += 1
        return counts

# quentin/memory/fabric.py
from __future__ import annotations

import logging
from datetime import datetime
from typing import Dict, Any, List, Optional

from quentin.memory.schemas import MemoryEntry, MemoryUpdatePolicy, MemoryTier, MemoryKind
from quentin.memory.store import FileBackedMemoryStore

logger = logging.getLogger(__name__)


class MemoryFabric:
    """
    蓝皮书里的 Memory Fabric v1：

    - 提供统一接口给系统其它部分使用（Planner / Critic / SleepLoop 等）
    - 底层用 FileBackedMemoryStore，未来可以换成更复杂的实现
    """

    def __init__(
        self,
        store: Optional[FileBackedMemoryStore] = None,
    ) -> None:
        self.store = store or FileBackedMemoryStore()

    # ---------- 基础 CRUD ----------

    def add_memory(
        self,
        tier: MemoryTier,
        kind: MemoryKind,
        text: str,
        *,
        importance: float = 0.5,
        source: str = "unknown",
        tags: Optional[List[str]] = None,
        update_policy: Optional[MemoryUpdatePolicy] = None,
        extra_content: Optional[Dict[str, Any]] = None,
    ) -> MemoryEntry:
        """
        添加一个新的记忆条目。

        v1 只支持 content['text']，extra_content 用于扩展。
        """
        import uuid

        memory_id = str(uuid.uuid4())
        now = datetime.utcnow().isoformat()

        content: Dict[str, Any] = {"text": text}
        if extra_content:
            content.update(extra_content)

        entry = MemoryEntry(
            memory_id=memory_id,
            memory_tier=tier,
            kind=kind,
            content=content,
            created_at=now,
            updated_at=now,
            last_accessed_at=now,
            importance=importance,
            usage_count=0,
            source=source,
            tags=tags or [],
            relations={},
            update_policy=update_policy or MemoryUpdatePolicy(),
        )

        self.store.upsert(entry)
        logger.info("Memory added: id=%s tier=%s kind=%s", memory_id, tier, kind)
        return entry

    def get(self, memory_id: str) -> Optional[MemoryEntry]:
        return self.store.get(memory_id)

    def update(self, entry: MemoryEntry) -> None:
        entry.updated_at = datetime.utcnow().isoformat()
        self.store.upsert(entry)

    def delete(self, memory_id: str) -> None:
        self.store.delete(memory_id)

    # ---------- 访问记录 ----------

    def record_access(self, memory_id: str) -> None:
        """
        在每次命中记忆时调用：增加 usage_count，更新 last_accessed_at。
        """
        entry = self.store.get(memory_id)
        if not entry:
            return
        entry.usage_count += 1
        entry.last_accessed_at = datetime.utcnow().isoformat()
        self.store.upsert(entry)

    # ---------- 简单文本搜索 (非向量) ----------

    def search_text(
        self,
        query: str,
        tiers: Optional[List[MemoryTier]] = None,
        limit: int = 20,
    ) -> List[MemoryEntry]:
        """
        v1 用最简单的 substring 匹配。
        未来可以替换成向量检索，接口不变。
        """
        query_lower = query.lower()
        results: List[MemoryEntry] = []
        for e in self.store.all_entries():
            if tiers and e.memory_tier not in tiers:
                continue
            text = str(e.content.get("text", "")).lower()
            if query_lower in text:
                results.append(e)
        results.sort(key=lambda e: (-e.importance, -(e.usage_count or 0)))
        return results[:limit]

    # ---------- 统计 ----------

    def stats(self) -> Dict[str, Any]:
        """
        返回 Memory Fabric 的基础统计信息，方便 Sleep Loop 和 Dashboard。
        """
        entries = self.store.all_entries()
        by_tier = {"STS": 0, "MTS": 0, "LTS": 0}
        by_kind: Dict[str, int] = {}
        for e in entries:
            if e.memory_tier in by_tier:
                by_tier[e.memory_tier] += 1
            by_kind[e.kind] = by_kind.get(e.kind, 0) + 1

        total = len(entries)
        return {
            "total": total,
            "by_tier": by_tier,
            "by_kind": by_kind,
        }
三、Sleep / Consolidation Loop：记忆升级 + 安全遗忘
新文件：quentin/memory/consolidation.py

实现一个 SleepLoop v1，按照蓝皮书：

STS 高频命中 → 升到 MTS

MTS 在多个任务中复用且稳定 → 升到 LTS（现在用 usage_count/importance 近似）

STS 低频 / 低重要度 / 很久未用 → 删除（安全遗忘）

@dataclass
class SleepConsolidationConfig:
    """
    Sleep Loop 的阈值配置：

    - sts_to_mts_min_usage      : STS -> MTS 最小 usage_count
    - sts_to_mts_min_importance : STS -> MTS 最小 importance
    - sts_decay_days            : STS 存活天数（超过且使用很少则删除）

    - mts_to_lts_min_usage      : MTS -> LTS 最小 usage_count
    - mts_to_lts_min_importance : MTS -> LTS 最小 importance
    - mts_min_age_days          : MTS -> LTS 最小存活天数
    """
    sts_to_mts_min_usage: int = 3
    sts_to_mts_min_importance: float = 0.6
    sts_decay_days: int = 7

    mts_to_lts_min_usage: int = 10
    mts_to_lts_min_importance: float = 0.7
    mts_min_age_days: int = 7

    def to_dict(self) -> Dict[str, Any]:

3）SleepLoop 实现
class SleepLoop:
    """
    Sleep / Consolidation Loop v1：

    - 每次运行：
      1. 扫描 STS，决定哪些升级到 MTS，哪些删除
      2. 扫描 MTS，决定哪些升级到 LTS

    v1 的逻辑完全是 heuristics，可以之后通过 Self-Programming 再迭代。
    """

    def __init__(
        self,
        fabric: Optional[MemoryFabric] = None,
        config: Optional[SleepConsolidationConfig] = None,
    ) -> None:
        self.fabric = fabric or MemoryFabric()
        self.config = config or SleepConsolidationConfig()

    # ---------- 主入口 ----------

    def run_once(self) -> SleepLoopResult:
        result = SleepLoopResult()
        now = datetime.utcnow()

        # 1) 处理 STS
        sts_entries = self.fabric.store.list_by_tier("STS")  # type: ignore[arg-type]
        for e in sts_entries:
            result.processed_sts += 1
            if self._should_promote_sts_to_mts(e):
                self._promote(e, "MTS")
                result.promoted_sts_to_mts += 1
            elif self._should_decay_sts(e, now):
                self.fabric.delete(e.memory_id)
                result.deleted_sts += 1

        # 2) 处理 MTS
        mts_entries = self.fabric.store.list_by_tier("MTS")  # type: ignore[arg-type]
        for e in mts_entries:
            result.processed_mts += 1
            if self._should_promote_mts_to_lts(e, now):
                self._promote(e, "LTS")
                result.promoted_mts_to_lts += 1

        logger.info(
            "SleepLoop finished: %s",
            result.to_dict(),
        )
        return result

    # ---------- 内部规则 ----------

    def _should_promote_sts_to_mts(self, e: MemoryEntry) -> bool:
        cfg = self.config
        if e.usage_count >= cfg.sts_to_mts_min_usage and e.importance >= cfg.sts_to_mts_min_importance:
            return True
        return False

    def _should_decay_sts(self, e: MemoryEntry, now: datetime) -> bool:
        cfg = self.config
        # 如果根本没被访问过，则视 last_accessed_at 为 created_at
        last_ts_str = e.last_accessed_at or e.created_at
        try:
            last_ts = datetime.fromisoformat(last_ts_str)
        except Exception:
            return False

        age_days = (now - last_ts).days
        if age_days >= cfg.sts_decay_days and e.usage_count <= 1 and e.importance < 0.5:
            return True
        return False

    def _should_promote_mts_to_lts(self, e: MemoryEntry, now: datetime) -> bool:
        cfg = self.config
        try:
            created_ts = datetime.fromisoformat(e.created_at)
        except Exception:
            return False

        age_days = (now - created_ts).days
        if (
            e.usage_count >= cfg.mts_to_lts_min_usage
            and e.importance >= cfg.mts_to_lts_min_importance
            and age_days >= cfg.mts_min_age_days
        ):
            return True
        return False

    def _promote(self, e: MemoryEntry, new_tier: MemoryTier) -> None:
        e.memory_tier = new_tier
        e.updated_at = datetime.utcnow().isoformat()
        self.fabric.update(e)
