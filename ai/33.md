Example skill used by the demo workflow.

It *deliberately* contains a bug so that:
- Demo runs can observe wrong results
- ImmuneLoop can open ImmuneCase
- Later, SelfProgrammingEngine can propose a patch to fix it

2. 最小版 CognitiveLoop：执行 demo 任务
文件：quentin/cognitive/loop.py

# quentin/cognitive/loop.py

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, Optional

from quentin.infra.logging import Tracer


@dataclass
class Task:
    task_id: str
    goal: str
    payload: Dict[str, Any]
    domain: str = "demo"  # 默认 demo 域


@dataclass
class TaskResult:
    task_id: str
    success: bool
    output: Any = None
    error: Optional[str] = None


@dataclass
class CognitiveLoop:
    """
    Minimal CognitiveLoop v1 for demo:
    - Only supports domain == 'demo'
    - Calls mutable.skills.example_skill.add_numbers(a, b)
    """
    tracer: Tracer

    async def run_task(self, task: Task) -> TaskResult:
        self.tracer.span(
            "L1",
            "task_start",
            module="cognitive",
            input={"task_id": task.task_id, "goal": task.goal, "domain": task.domain},
        )

        try:
            if task.domain == "demo":
                a = task.payload["a"]
                b = task.payload["b"]
                # Import inside to avoid hard dependency at import time
                from mutable.skills import example_skill

                out = example_skill.add_numbers(a, b)
            else:
                raise NotImplementedError(f"Unknown domain: {task.domain}")

            success = True
            error = None
        except Exception as e:
            out = None
            success = False
            error = repr(e)

        self.tracer.span(
            "L1",
            "task_end",
            module="cognitive",
            input={"task_id": task.task_id},
            output={"success": success},
        )

        return TaskResult(task_id=task.task_id, success=success, output=out, error=error)
这里只是一个“最小能跑”的 CognitiveLoop：

不用 LLM、不用 tools，只是直接调用一个 skill。

但架子是对的：有 Task / TaskResult / Trace。

3.2 ImmuneLoop 实现
文件：quentin/immune/loop.py

# quentin/immune/loop.py

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime
from typing import List, Optional

import uuid

from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.immune.schemas import ImmuneCase, ImmuneCaseStatus


@dataclass
class ImmuneLoop:
    """
    Minimal ImmuneLoop v1:

    - Stores ImmuneCase in KV
    - For demo, any incorrect result will open a simple ImmuneCase
    """
    tracer: Tracer
    kv: KVStore

    case_prefix: str = "immune:case:"
    case_index_key: str = "immune:index:cases"

    async def _save_case(self, case: ImmuneCase) -> None:
        case.updated_at = datetime.utcnow()
        key = f"{self.case_prefix}{case.case_id}"
        await self.kv.save(key, case.model_dump())
        await self.kv.append_list(self.case_index_key, case.case_id)

    async def list_cases(self, limit: int = 50) -> List[ImmuneCase]:
        ids = await self.kv.get(self.case_index_key) or []
        ids = list(dict.fromkeys(ids))[-limit:]
        res: List[ImmuneCase] = []
        for cid in ids:
            data = await self.kv.get(f"{self.case_prefix}{cid}")
            if data:
                res.append(ImmuneCase(**data))
        return res

    async def get_case(self, case_id: str) -> Optional[ImmuneCase]:
        data = await self.kv.get(f"{self.case_prefix}{case_id}")
        if not data:
            return None
        return ImmuneCase(**data)

    async def open_case_if_needed(
        self,
        title: str,
        description: str,
        domain: Optional[str] = None,
        tags: Optional[List[str]] = None,
        metrics: Optional[dict] = None,
        meta: Optional[dict] = None,
    ) -> ImmuneCase:
        """
        v1 简化：每次调用都新建一个 ImmuneCase。
        未来可以根据 title/domain 去合并或去重。
        """
        case_id = "case_" + uuid.uuid4().hex
        case = ImmuneCase(
            case_id=case_id,
            title=title,
            description=description,
            domain=domain,
            tags=tags or [],
            metrics=metrics or {},
            meta=meta or {},
        )
        await self._save_case(case)

        self.tracer.span(
            "L2",
            "immune_case_open",
            module="immune",
            input={"case_id": case_id, "title": title, "domain": domain},
        )

        return case

    async def record_demo_task_result(
        self,
        task_id: str,
        success: bool,
        expected_ok: bool,
        error: Optional[str] = None,
    ) -> None:
        """
        Demo 专用：只要结果不符合预期，就开一个 ImmuneCase。
        """
        if success and expected_ok:
            return

        title = "demo_add_numbers_bug"
        description = "Demo skill 'add_numbers' produced incorrect results."
        metrics = {"fail": 1.0}
        meta = {"task_id": task_id, "error": error}

        await self.open_case_if_needed(
            title=title,
            description=description,
            domain="demo",
            metrics=metrics,
            meta=meta,
        )
这版 ImmuneLoop 很“傻”：每次错误都新建一个 Case。
但对于 v3 蓝皮书来说够用了——SelfImprovementManager 可以拿这些 Case 当“热点”。

5. Demo 工作流：让 bug 跑起来
文件：quentin/demo/toy_workflow.py

# quentin/demo/toy_workflow.py

from __future__ import annotations

import random
import uuid
from typing import Optional

from quentin.runtime.agent_runtime import AgentRuntime
from quentin.cognitive.loop import Task


async def run_demo(runtime: AgentRuntime, episodes: int = 5) -> None:
    """
    最小 Demo：
    - 随机生成 a, b
    - 调用 CognitiveLoop（内部用 example_skill.add_numbers）
    - 对比 expected = a + b
    - 错误时 -> ImmuneLoop.record_demo_task_result
    """
    tracer = runtime.tracer

    for i in range(episodes):
        a = random.randint(0, 10)
        b = random.randint(0, 10)
        expected = a + b

        task_id = f"demo_task_{uuid.uuid4().hex[:8]}"
        task = Task(
            task_id=task_id,
            goal=f"add {a} + {b}",
            payload={"a": a, "b": b},
            domain="demo",
        )

        result = await runtime.cognitive_loop.run_task(task)
        ok = result.success and (result.output == expected)

        tracer.span(
            "L1",
            "demo_eval",
            module="demo",
            input={
                "task_id": task.task_id,
                "a": a,
                "b": b,
                "output": result.output,
                "expected": expected,
            },
            output={"ok": ok},
        )

        status = "OK" if ok else "BUG"
        print(f"[EP {i}] {a} + {b} -> {result.output} (expected {expected}) :: {status}")

        if not ok:
            await runtime.immune_loop.record_demo_task_result(
                task_id=task_id,
                success=result.success,
                expected_ok=True,
                error=result.error,
            )
因为 example_skill.add_numbers 是错的，所以绝大多数情况下都会打印 BUG，并产生 ImmuneCase。

import asyncio
import click

from quentin.runtime.settings import Settings
from quentin.runtime.agent_runtime import AgentRuntime


# ... 之前已经有的 @click.group() cli 定义 ...


@cli.command("demo-toy")
@click.option("--episodes", default=5, show_default=True, help="Number of demo episodes to run.")
def demo_toy_cmd(episodes: int):
    """
    Run the minimal toy demo:

    - Uses buggy example_skill.add_numbers
    - Observes incorrect results
    - Records ImmuneCase via ImmuneLoop
    """
    async def _run():
        settings = Settings()
        runtime = AgentRuntime.create_from_settings(settings)

        from quentin.demo.toy_workflow import run_demo
        await run_demo(runtime, episodes=episodes)

    asyncio.run(_run())

7. 下一步怎么“自我迭代”？
现在这个最小 Demo 已经实现了：

有一个明确、可验证错误的 skill（example_skill.add_numbers）。

有 CognitiveLoop 执行任务，写 Trace。

有 ImmuneLoop 在看到错误时创建 ImmuneCase。

有 CLI 命令可以反复触发这个过程。

在你现有的 SelfProgrammingEngine + SelfImprovementManager 基础上：

你可以手动调用 CLI：

qctl patch-propose-text 把 a - b 改成 a + b

qctl patch-run-tests

qctl patch-apply

再跑 qctl demo-toy，观察 BUG 是否消失

从 ImmuneCase 自动触发 patch 生成（用 LLM 读源码+Case，自动写 patch）
