# quentin/router/metrics.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional
from pathlib import Path
from datetime import datetime
import json
import uuid
import logging

logger = logging.getLogger(__name__)


@dataclass
class ModelRouteRequestRecord:
    """
    一次路由请求的结构化表示，用于记录和离线训练。

    注意：这里只是“记录”，真正的调用参数由 ModelRouter 决定。
    """
    task_type: str
    importance: str = "normal"     # "low" / "normal" / "high"
    domain: str = "general"
    constraints: Dict[str, Any] = field(default_factory=dict)
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ModelRouteDecisionRecord:
    """
    ModelRouter 的决策结果（可持久化）。
    """
    model_id: str
    provider: str
    reason: str
    params: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ModelRouteOutcomeRecord:
    """
    路由决策的结果，用于学习路由策略。
    """
    success: Optional[bool] = None
    reward: Optional[float] = None
    cost_tokens: Optional[float] = None
    latency_ms: Optional[float] = None
    quality_score: Optional[float] = None  # 人工/AI 评价

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ModelRouteLogRecord:
    """
    完整的一次路由日志，用于 JSONL 持久化和训练数据。
    """
    log_id: str
    created_at: str
    request: Dict[str, Any]
    decision: Dict[str, Any]
    outcome: Dict[str, Any]
    runtime_flags: Dict[str, Any] = field(default_factory=dict)
    experiment_tags: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class ModelRouterLogStore:
    """
    v1：简单的 JSONL 日志，用于后续 Sleep Loop / Research 的离线训练。

    - append(log_record)
    - 可扩展：按需做简单统计聚合。
    """

    def __init__(self, *, repo_root: str | Path = ".") -> None:
        self.repo_root = Path(repo_root).resolve()
        self.base_dir = self.repo_root / "data" / "model_router"
        self.base_dir.mkdir(parents=True, exist_ok=True)
        self.jsonl_path = self.base_dir / "routes.jsonl"

    def append(
        self,
        *,
        request: ModelRouteRequestRecord,
        decision: ModelRouteDecisionRecord,
        outcome: Optional[ModelRouteOutcomeRecord] = None,
        runtime_flags: Optional[Dict[str, Any]] = None,
        experiment_tags: Optional[Dict[str, Any]] = None,
    ) -> ModelRouteLogRecord:
        log = ModelRouteLogRecord(
            log_id=uuid.uuid4().hex,
            created_at=datetime.utcnow().isoformat(),
            request=request.to_dict(),
            decision=decision.to_dict(),
            outcome=(outcome or ModelRouteOutcomeRecord()).to_dict(),
            runtime_flags=runtime_flags or {},
            experiment_tags=experiment_tags or {},
        )
        line = json.dumps(log.to_dict(), ensure_ascii=False)
        try:
            with self.jsonl_path.open("a", encoding="utf-8") as f:
                f.write(line + "\n")
        except Exception:
            logger.exception("ModelRouterLogStore: failed to append route log to %s", self.jsonl_path)
        return log
现在每一次路由，都可以写成一个 JSONL 记录，后面 Sleep Loop / Research Layer 就有数据可以挖掘。

# quentin/core/model_router.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, Optional
from pathlib import Path
import logging

from quentin.router.metrics import (
    ModelRouteRequestRecord,
    ModelRouteDecisionRecord,
    ModelRouteOutcomeRecord,
    ModelRouterLogStore,
)

logger = logging.getLogger(__name__)


@dataclass
class ModelRouteRequest:
    """
    CognitiveLoop / 其他组件请求路由时使用的结构。
    """
    task_type: str            # 如 "planning" / "execution" / "reflection" / "done_check"
    importance: str = "normal"
    domain: str = "general"   # "desktop" / "legal" / "trading" / "general" 等
    constraints: Dict[str, Any] = field(default_factory=dict)
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_record(self) -> ModelRouteRequestRecord:
        return ModelRouteRequestRecord(
            task_type=self.task_type,
            importance=self.importance,
            domain=self.domain,
            constraints=self.constraints,
            metadata=self.metadata,
        )


@dataclass
class ModelRouteDecision:
    model_id: str
    provider: str
    reason: str
    params: Dict[str, Any] = field(default_factory=dict)

    def to_record(self) -> ModelRouteDecisionRecord:
        return ModelRouteDecisionRecord(
            model_id=self.model_id,
            provider=self.provider,
            reason=self.reason,
            params=self.params,
        )

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class ModelRouterConfig:
    """
    ModelRouter 的配置容器。

    v1：主要是静态映射 + 默认模型 ID。
    将来可以被 self-programming 引擎自动修改。
    """

    def __init__(self, raw: Optional[Dict[str, Any]] = None) -> None:
        cfg = raw or {}
        self.default_model: str = cfg.get("default_model", "gpt_default")
        self.safe_model: str = cfg.get("safe_model", "gpt_safe")
        self.cheap_model: str = cfg.get("cheap_model", "gpt_cheap")

        # 按 task_type 映射
        self.by_task_type: Dict[str, str] = cfg.get("by_task_type", {})
        # 按 domain 映射
        self.by_domain: Dict[str, str] = cfg.get("by_domain", {})

        # 预留更多策略参数
        self.extra: Dict[str, Any] = cfg.get("extra", {})

    def to_dict(self) -> Dict[str, Any]:
        return {
            "default_model": self.default_model,
            "safe_model": self.safe_model,
            "cheap_model": self.cheap_model,
            "by_task_type": self.by_task_type,
            "by_domain": self.by_domain,
            "extra": self.extra,
        }


class ModelRouter:
    """
    通用 ModelRouter：

    - 支持 degrade_mode / runtime_flags
    - 支持通过 config 和 Variant params 调整策略
    - 每次决策都会写入 ModelRouterLogStore
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        base_config: Optional[Dict[str, Any]] = None,
        log_store: Optional[ModelRouterLogStore] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.config = ModelRouterConfig(base_config)
        self.log_store = log_store or ModelRouterLogStore(repo_root=self.repo_root)

    # --- 对外主接口 --- #
    def route(
        self,
        request: ModelRouteRequest,
        *,
        runtime_flags: Optional[Dict[str, Any]] = None,
        variant_params: Optional[Dict[str, Any]] = None,
    ) -> ModelRouteDecision:
        """
        主路由函数：
        - request           : 当前任务的元信息
        - runtime_flags     : 来自 CognitiveLoop 的运行状态（如 degrade_mode / self_programming_enabled）
        - variant_params    : 可选，用于实验时覆盖策略（来自 Experiment Variant）
        """
        flags = runtime_flags or {}
        params = variant_params or {}

        degrade_mode = bool(flags.get("degrade_mode", False))
        cost_sensitive = bool(params.get("cost_sensitive", False))
        safety_sensitive = bool(params.get("safety_sensitive", False))

        # 1) 选择候选模型 ID（简单规则版）
        model_id = self._select_model_id(request, degrade_mode, cost_sensitive, safety_sensitive)
        provider = self._infer_provider_from_model_id(model_id)

        reason_parts = []
        if degrade_mode:
            reason_parts.append("degrade_mode")
        if cost_sensitive:
            reason_parts.append("cost_sensitive")
        if safety_sensitive:
            reason_parts.append("safety_sensitive")
        if not reason_parts:
            reason_parts.append("default_routing")

        decision = ModelRouteDecision(
            model_id=model_id,
            provider=provider,
            reason=",".join(reason_parts),
            params={
                "degrade_mode": degrade_mode,
                "cost_sensitive": cost_sensitive,
                "safety_sensitive": safety_sensitive,
            },
        )

        # v1: outcome 先留空，后续可以由调用者补上（成功率 / cost 等）
        outcome = ModelRouteOutcomeRecord()

        # 写日志（便于后续离线分析）
        try:
            req_record = request.to_record()
            dec_record = decision.to_record()
            self.log_store.append(
                request=req_record,
                decision=dec_record,
                outcome=outcome,
                runtime_flags=flags,
                experiment_tags=params.get("experiment_tags") or {},
            )
        except Exception:
            logger.exception("ModelRouter.route: failed to log route record")

        return decision

    # --- 内部策略（v1：简单规则）--- #
    def _select_model_id(
        self,
        request: ModelRouteRequest,
        degrade_mode: bool,
        cost_sensitive: bool,
        safety_sensitive: bool,
    ) -> str:
        # 优先 degrade_mode：只用 safe_model
        if degrade_mode:
            return self.config.safe_model

        # 安全优先（例如 legal 域）
        if safety_sensitive or request.domain.lower() in ("legal", "compliance"):
            return self.config.safe_model

        # 成本优先
        if cost_sensitive or request.constraints.get("budget") == "low":
            return self._select_cheap_model(request)

        # 按 task_type 精细控制（如 planning/logging 等）
        if request.task_type in self.config.by_task_type:
            return self.config.by_task_type[request.task_type]

        # 按 domain 控制
        if request.domain in self.config.by_domain:
            return self.config.by_domain[request.domain]

        # 默认
        return self.config.default_model

    def _select_cheap_model(self, request: ModelRouteRequest) -> str:
        # v1：简单返回 cheap_model，后面可以根据 task_type / domain 做更细粒度策略
        return self.config.cheap_model

    def _infer_provider_from_model_id(self, model_id: str) -> str:
        """
        根据 model_id 推断 provider（只是示例逻辑）。
        例如：
        - "gpt-4o"         -> "openai"
        - "deepseek-r1"    -> "deepseek"
        - "qwen2.5:7b"     -> "ollama"
        """
        if "gpt" in model_id:
            return "openai"
        if "deepseek" in model_id:
            return "deepseek"
        if "qwen" in model_id or ":" in model_id:
            return "ollama"
        return "unknown"
后续自编程引擎可以通过修改 ModelRouterConfig.by_task_type/by_domain/extra 来“学会”更好的路由策略；
ModelRouterLogStore 里的 JSONL 文件，就是训练数据。

# quentin/research/runners/model_router_offline_runner.py
from __future__ import annotations

from typing import Any, Dict, Optional, Callable
import logging

from quentin.research.orchestrator import ExperimentRunner
from quentin.research.experiments import (
    ExperimentConfig,
    VariantConfig,
    VariantResult,
)
from quentin.eval.spec import EvalDatasetSpec, EvalTaskSpec
from quentin.eval.harness import EvalHarness
from quentin.core.model_router import ModelRouter, ModelRouteRequest

logger = logging.getLogger(__name__)

# eval_router_fn: (task, variant, context, router) -> {success, score, notes, ...}
EvalRouterFn = Callable[
    [EvalTaskSpec, VariantConfig, Dict[str, Any], ModelRouter],
    Dict[str, Any]
]


class ModelRouterOfflineRunner(ExperimentRunner):
    """
    针对 ModelRouter 的离线评测 Runner。

    用法：
    - context 里提供：
        - "router_base_config": 可选，初始化 ModelRouterConfig 用
        - "router_eval_dataset": EvalDatasetSpec（component="model_router"）
        - "eval_router_fn": 可选，自定义评测函数
    - 若未提供 eval_router_fn，则使用 stub 版本：
        - 如果 expected_output.expected_model_id 存在且与 router 选择一致 => score=1
        - 否则 score=0
    """

    def __init__(
        self,
        *,
        eval_harness: Optional[EvalHarness] = None,
        eval_router_fn: Optional[EvalRouterFn] = None,
        repo_root: str | None = None,
    ) -> None:
        self.eval_harness = eval_harness or EvalHarness()
        self.eval_router_fn = eval_router_fn or self._stub_eval_router_fn
        self.repo_root = repo_root

    def run_variant(
        self,
        experiment: ExperimentConfig,
        variant: VariantConfig,
        context: Optional[Dict[str, Any]] = None,
    ) -> VariantResult:
        ctx = dict(context or {})

        dataset: Optional[EvalDatasetSpec] = (
            ctx.get("router_eval_dataset")
            or ctx.get("eval_dataset")
        )
        if dataset is None:
            logger.warning(
                "ModelRouterOfflineRunner: no EvalDatasetSpec provided in context "
                "(router_eval_dataset / eval_dataset missing). Using empty dataset."
            )
            dataset = EvalDatasetSpec(
                dataset_id="empty_router",
                name="empty_router",
                description="Empty dataset for ModelRouter.",
                component="model_router",
                tasks=[],
            )

        if dataset.component != "model_router":
            logger.warning(
                "ModelRouterOfflineRunner: dataset.component=%s but runner is for model_router.",
                dataset.component,
            )

        base_cfg = ctx.get("router_base_config") or {}
        router = ModelRouter(
            repo_root=self.repo_root or ".",
            base_config=base_cfg,
        )

        def _eval_fn(task: EvalTaskSpec, variant_cfg: VariantConfig, local_ctx: Dict[str, Any]) -> Dict[str, Any]:
            return self.eval_router_fn(task, variant_cfg, local_ctx, router)

        return self.eval_harness.run_dataset(
            experiment=experiment,
            variant=variant,
            dataset=dataset,
            eval_fn=_eval_fn,
            context=ctx,
        )

    # ---------- 默认 stub 评测逻辑 ---------- #

    def _stub_eval_router_fn(
        self,
        task: EvalTaskSpec,
        variant: VariantConfig,
        context: Dict[str, Any],
        router: ModelRouter,
    ) -> Dict[str, Any]:
        """
        stub 版评测逻辑：

        - 从 task.input_payload 中读取：
            - task_type
            - importance
            - domain
            - constraints
        - 从 expected_output 中读取 expected_model_id，如果 router 选的一致 => success=True, score=1。
        """
        ip = task.input_payload or {}
        req = ModelRouteRequest(
            task_type=ip.get("task_type", "generic"),
            importance=ip.get("importance", "normal"),
            domain=ip.get("domain", "general"),
            constraints=ip.get("constraints", {}),
            metadata=ip.get("metadata", {}),
        )

        # variant.params 作为 variant_params 传给 router（例如 cost_sensitive / safety_sensitive）
        decision = router.route(
            req,
            runtime_flags=context.get("runtime_flags") or {},
            variant_params=variant.params,
        )

        expected = task.expected_output or {}
        expected_model_id = expected.get("expected_model_id")

        success = (expected_model_id is not None and decision.model_id == expected_model_id)
        score = 1.0 if success else 0.0

        return {
            "success": success,
            "score": score,
            "notes": "stub router eval; replace with real scoring logic if needed.",
            "chosen_model_id": decision.model_id,
            "expected_model_id": expected_model_id,
            "variant_id": variant.variant_id,
            "task_id": task.task_id,
        }
后续你要做更高级的评测（例如结合质量评分、成本、延迟等），只需要把 eval_router_fn 换成自己的函数即可。
