from __future__ import annotations

from dataclasses import dataclass, asdict
from datetime import datetime, timedelta
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging
import uuid

logger = logging.getLogger(__name__)

MemoryTier = str  # "STS" | "MTS" | "LTS"
MemoryKind = str  # "episodic" | "semantic" | "procedural"


@dataclass
class MemoryItem:
    """
    Memory Fabric 的核心单元。

    - tier: STS/MTS/LTS
    - kind: episodic / semantic / procedural
    - content: 具体内容（可以是字符串或结构化 dict）
    - importance: 0.0~1.0 粗略重要度
    - usage_count: 被命中/引用次数
    - created_at / last_used_at: ISO 时间戳
    - update_policy: 用于升级/衰减规则的配置（蓝皮书里的 promotion_rule/decay_rule）
    """
    memory_id: str
    tier: MemoryTier
    kind: MemoryKind
    content: Any
    importance: float
    usage_count: int
    created_at: str
    last_used_at: Optional[str]
    tags: List[str]
    update_policy: Dict[str, Any]
    meta: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class MemoryFabricStats:
    tier_counts: Dict[str, int]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class MemoryFabric:
    """
    最小可运行版 Memory Fabric：

    - 用 JSONL 文件实现 STS/MTS/LTS 存储（data/memory/STS.jsonl 等）
    - 暂不实现向量检索，只提供简单 query/filter
    - 提供 consolidate_from_traces() 给 SleepLoop 用，按蓝皮书生成 episodic 记忆
    - 提供 run_maintenance_cycle() 做 promotion / decay（研究友好）

    以后你可以：
      - 把底层换成 SQLite / 向量库
      - 用 embeddings + 相似度做 query
      - 升级 promotion/decay 规则
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        storage_dir: Optional[str | Path] = None,
        default_update_policy: Optional[Dict[str, Any]] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.storage_dir = (
            Path(storage_dir).resolve()
            if storage_dir is not None
            else self.repo_root / "data" / "memory"
        )
        self.storage_dir.mkdir(parents=True, exist_ok=True)

        self.tier_files: Dict[str, Path] = {
            "STS": self.storage_dir / "STS.jsonl",
            "MTS": self.storage_dir / "MTS.jsonl",
            "LTS": self.storage_dir / "LTS.jsonl",
        }

        self.default_update_policy = default_update_policy or {
            "promotion_rule": "auto_basic_v1",
            "decay_rule": "time_based_v1",
            "consolidation_batch": "nightly",
        }

    # --------------------------------------------------
    # 基础 API
    # --------------------------------------------------
    def store(self, item: MemoryItem) -> MemoryItem:
        """
        追加写入 JSONL 文件。
        """
        tier = item.tier
        if tier not in self.tier_files:
            raise ValueError(f"Unknown memory tier: {tier}")

        path = self.tier_files[tier]
        path.parent.mkdir(parents=True, exist_ok=True)

        line = json.dumps(item.to_dict(), ensure_ascii=False)
        with path.open("a", encoding="utf-8") as f:
            f.write(line + "\n")

        return item

    def query(
        self,
        *,
        tier: Optional[MemoryTier] = None,
        kind: Optional[MemoryKind] = None,
        tag: Optional[str] = None,
        limit: int = 50,
    ) -> List[MemoryItem]:
        """
        最简单的过滤查询（没有向量相似度）：

        - 可以按 tier/kind/tag 过滤
        - 从对应 JSONL 文件中顺序读取，取前 limit 条
        """
        tiers = [tier] if tier else list(self.tier_files.keys())
        results: List[MemoryItem] = []

        for t in tiers:
            path = self.tier_files.get(t)
            if not path or not path.exists():
                continue

            with path.open("r", encoding="utf-8") as f:
                for line in f:
                    try:
                        data = json.loads(line)
                    except Exception:
                        continue
                    if kind and data.get("kind") != kind:
                        continue
                    if tag and tag not in (data.get("tags") or []):
                        continue

                    results.append(_memory_from_dict(data))
                    if len(results) >= limit:
                        return results

        return results

    def get_stats(self) -> MemoryFabricStats:
        """
        简单统计各 tier 的条数（按 JSONL 行数估算）。
        """
        counts: Dict[str, int] = {}
        for tier, path in self.tier_files.items():
            if not path.exists():
                counts[tier] = 0
            else:
                try:
                    with path.open("r", encoding="utf-8") as f:
                        counts[tier] = sum(1 for _ in f)
                except Exception:
                    counts[tier] = -1  # 标记读取失败

        return MemoryFabricStats(tier_counts=counts)

    # --------------------------------------------------
    # SleepLoop 调用入口：从 Trace 巩固记忆
    # --------------------------------------------------
    def consolidate_from_traces(
        self,
        traces: List[Dict[str, Any]],
        context: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        SleepLoop 调用的“记忆巩固”入口。

        v1 策略（简单但结构齐全）：
          - 对每个 trace 生成一个 STS 级 episodic 记忆条目：
              content = {trace_id, task_id, goal, domain, summary, lessons}
          - importance 粗略按 trace.status:
              success -> 0.7, failed -> 0.5, error -> 0.3
          - 更新 stats 并返回

        以后可以：
          - 在这里添加 promotion_rule：符合条件的 STS -> MTS/LTS
          - 把多次出现的模式压缩为 semantic/procedural 记忆
        """
        ctx = context or {}
        now = _utc_iso()
        new_items: List[MemoryItem] = []

        for t in traces:
            status = t.get("status") or "unknown"
            importance = _estimate_importance(status)

            trace_id = t.get("trace_id") or t.get("id") or str(uuid.uuid4())
            task_id = t.get("task_id", "unknown_task")
            goal = t.get("goal", "")
            domain = t.get("domain", "general")

            summary = t.get("execution_summary") or {}
            lessons = t.get("lessons_learned") or None

            content = {
                "trace_id": trace_id,
                "task_id": task_id,
                "goal": goal,
                "domain": domain,
                "status": status,
                "summary": summary,
                "lessons_learned": lessons,
            }

            tags = [
                "trace",
                f"status:{status}",
                f"domain:{domain}",
            ]

            item = MemoryItem(
                memory_id=f"mem::{trace_id}",
                tier="STS",
                kind="episodic",
                content=content,
                importance=importance,
                usage_count=0,
                created_at=now,
                last_used_at=None,
                tags=tags,
                update_policy=self.default_update_policy.copy(),
                meta={
                    "source": "sleep_loop",
                    "context": ctx,
                },
            )
            self.store(item)
            new_items.append(item)

        stats = self.get_stats()

        return {
            "status": "ok",
            "num_new_items": len(new_items),
            "tier_counts": stats.to_dict()["tier_counts"],
        }

    # --------------------------------------------------
    # 维护循环：promotion / decay
    # --------------------------------------------------
    def run_maintenance_cycle(
        self,
        *,
        now: Optional[datetime] = None,
        config: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        记忆维护循环：升阶 + 衰减 / 删除。

        v1 实现（简单可研究）：

          1. 从 STS/MTS/LTS 全量读入内存（只读 JSONL，一轮重写）
          2. 应用规则：
             - STS -> MTS:
                 importance >= 0.7 或 usage_count >= 3
             - MTS -> LTS:
                 importance >= 0.8 且 usage_count >= 5
             - STS decay:
                 created_at 距今 > 3 天 且 usage_count == 0 → 删除
             - MTS decay:
                 created_at 距今 > 14 天 且 usage_count <= 1 且 importance < 0.5 → 删除
             - LTS 默认不删除（以后可以扩展）

          3. 重写三层 JSONL 文件，返回统计结果。

        config 可以覆盖默认阈值，比如：
        {
          "sts_max_age_days": 3,
          "mts_max_age_days": 14,
          "sts_promotion_importance": 0.7,
          "mts_promotion_importance": 0.8,
          ...
        }
        """
        now = now or datetime.utcnow()
        cfg = {
            "sts_max_age_days": 3,
            "mts_max_age_days": 14,
            "sts_promotion_importance": 0.7,
            "mts_promotion_importance": 0.8,
            "sts_promotion_usage": 3,
            "mts_promotion_usage": 5,
        }
        if config:
            cfg.update(config)

        # 1) 读入所有 tier
        sts_items = self._load_tier_items("STS")
        mts_items = self._load_tier_items("MTS")
        lts_items = self._load_tier_items("LTS")

        promoted_sts_to_mts: List[str] = []
        promoted_mts_to_lts: List[str] = []
        decayed_ids: List[str] = []

        # 2) STS: promotion + decay
        new_sts: List[MemoryItem] = []
        for item in sts_items:
            created_at = _parse_iso(item.created_at)
            age_days = (now - created_at).days if created_at else 0

            # decay 条件
            if (
                age_days > cfg["sts_max_age_days"]
                and item.usage_count == 0
            ):
                decayed_ids.append(item.memory_id)
                continue

            # promotion 条件：STS -> MTS
            if (
                item.importance >= cfg["sts_promotion_importance"]
                or item.usage_count >= cfg["sts_promotion_usage"]
            ):
                item.tier = "MTS"
                promoted_sts_to_mts.append(item.memory_id)
                mts_items.append(item)
            else:
                new_sts.append(item)

        sts_items = new_sts

        # 3) MTS: promotion + decay
        new_mts: List[MemoryItem] = []
        for item in mts_items:
            created_at = _parse_iso(item.created_at)
            age_days = (now - created_at).days if created_at else 0

            # decay 条件
            if (
                age_days > cfg["mts_max_age_days"]
                and item.usage_count <= 1
                and item.importance < 0.5
            ):
                decayed_ids.append(item.memory_id)
                continue

            # promotion 条件：MTS -> LTS
            if (
                item.importance >= cfg["mts_promotion_importance"]
                and item.usage_count >= cfg["mts_promotion_usage"]
            ):
                item.tier = "LTS"
                promoted_mts_to_lts.append(item.memory_id)
                lts_items.append(item)
            else:
                new_mts.append(item)

        mts_items = new_mts

        # 4) LTS：暂不做 decay，保留全部
        # （以后可以在这里实现长期知识的“替换/合并”，而不是简单删除）

        # 5) 重写 JSONL 文件
        self._rewrite_tier("STS", sts_items)
        self._rewrite_tier("MTS", mts_items)
        self._rewrite_tier("LTS", lts_items)

        stats = self.get_stats()

        return {
            "status": "ok",
            "promoted_sts_to_mts": promoted_sts_to_mts,
            "promoted_mts_to_lts": promoted_mts_to_lts,
            "decayed_ids": decayed_ids,
            "tier_counts": stats.to_dict()["tier_counts"],
        }

    # --------------------------------------------------
    # 内部工具：加载/重写 tier
    # --------------------------------------------------
    def _load_tier_items(self, tier: MemoryTier) -> List[MemoryItem]:
        path = self.tier_files.get(tier)
        if not path or not path.exists():
            return []

        items: List[MemoryItem] = []
        with path.open("r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                try:
                    data = json.loads(line)
                    items.append(_memory_from_dict(data))
                except Exception:
                    continue
        return items

    def _rewrite_tier(self, tier: MemoryTier, items: List[MemoryItem]) -> None:
        path = self.tier_files.get(tier)
        if not path:
            return
        path.parent.mkdir(parents=True, exist_ok=True)
        with path.open("w", encoding="utf-8") as f:
            for item in items:
                line = json.dumps(item.to_dict(), ensure_ascii=False)
                f.write(line + "\n")


def _utc_iso() -> str:
    return datetime.utcnow().isoformat()


def _parse_iso(ts: Optional[str]) -> Optional[datetime]:
    if not ts:
        return None
    try:
        return datetime.fromisoformat(ts)
    except Exception:
        return None


def _estimate_importance(status: str) -> float:
    if status == "success":
        return 0.7
    if status == "failed":
        return 0.5
    if status == "error":
        return 0.3
    return 0.4


def _memory_from_dict(data: Dict[str, Any]) -> MemoryItem:
    return MemoryItem(
        memory_id=data.get("memory_id", f"mem::{uuid.uuid4().hex}"),
        tier=data.get("tier", "STS"),
        kind=data.get("kind", "episodic"),
        content=data.get("content"),
        importance=float(data.get("importance", 0.5)),
        usage_count=int(data.get("usage_count", 0)),
        created_at=data.get("created_at", _utc_iso()),
        last_used_at=data.get("last_used_at"),
        tags=list(data.get("tags") or []),
        update_policy=dict(data.get("update_policy") or {}),
        meta=dict(data.get("meta") or {}),
    )

# quentin/loops/sleep.py
from __future__ import annotations

from dataclasses import asdict, is_dataclass
from typing import Any, Dict, List, Optional
import logging

from quentin.evolution.skill_loop import SkillEvolutionLoop

logger = logging.getLogger(__name__)


def _trace_to_dict(t: Any) -> Dict[str, Any]:
    """
    兼容：
    - ExecutionTrace dataclass
    - dict
    - 其它对象（作为 raw 包一层）
    """
    if isinstance(t, dict):
        return t
    if is_dataclass(t):
        return asdict(t)
    return {"raw": t}


class SleepLoop:
    """
    睡眠循环（Sleep / Consolidation Loop）v1 实现。

    职责：
      1. 聚合最近一段时间的 ExecutionTrace
      2. 统计成功 / 失败等概况
      3. 为 MemoryFabric 准备“巩固候选”（先以摘要形式给出）
      4. 触发 MemoryFabric 维护循环（promotion / decay）
      5. 调用 SkillEvolutionLoop（成功 Trace + issues）做技能/代码进化

    这里刻意保持“研究友好”：
      - 不强绑定任何具体 MemoryFabric API，只是约定性地调用 consolidate_from_traces() / run_maintenance_cycle()
      - 所有输入输出都是结构化 dict，方便以后接不同算法
    """

    def __init__(
        self,
        *,
        memory_fabric: Optional[Any] = None,
        skill_evolution_loop: Optional[SkillEvolutionLoop] = None,
        max_traces: int = 500,
    ) -> None:
        self.memory_fabric = memory_fabric
        self.skill_evolution_loop = skill_evolution_loop
        self.max_traces = max_traces

    def run_once(
        self,
        *,
        recent_traces: List[Any],
        issues_from_immune: Optional[List[Dict[str, Any]]] = None,
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        由 MetaAgentRuntime 调用：

        - recent_traces: 最近一段时间的 ExecutionTrace 列表（新在前或新在后均可）
        - issues_from_immune: 来自 ImmuneLoop 的问题列表（可为空）
        - extra_context: 额外信息（如运行批次 ID 等）

        返回：
        {
          "trace_stats": {...},
          "memory_consolidation": {...},
          "memory_maintenance": {...},
          "skill_evolution": {...},
        }
        """
        ctx = extra_context or {}
        issues = issues_from_immune or []

        # 1) 预处理 trace（截断到 max_traces）
        if len(recent_traces) > self.max_traces:
            # 假设 recent_traces 是按时间排序的：我们保留最新的 max_traces
            recent_traces = recent_traces[-self.max_traces :]

        trace_dicts = [_trace_to_dict(t) for t in recent_traces]

        # 2) 统计成功 / 失败 / error 概况
        stats = self._summarize_traces(trace_dicts)

        # 3) MemoryFabric 巩固
        mem_consolidation: Dict[str, Any] = {}
        mem_maintenance: Dict[str, Any] = {}
        if self.memory_fabric is not None and hasattr(self.memory_fabric, "consolidate_from_traces"):
            try:
                mem_consolidation = self.memory_fabric.consolidate_from_traces(
                    trace_dicts,
                    context={"source": "sleep_loop", **ctx},
                )
            except Exception:
                logger.exception("SleepLoop: memory_fabric.consolidate_from_traces failed")
                mem_consolidation = {"status": "error"}

            # 4) MemoryFabric 维护循环（promotion / decay）
            if hasattr(self.memory_fabric, "run_maintenance_cycle"):
                try:
                    mem_maintenance = self.memory_fabric.run_maintenance_cycle(
                        config={"source": "sleep_loop"},
                    )
                except Exception:
                    logger.exception("SleepLoop: memory_fabric.run_maintenance_cycle failed")
                    mem_maintenance = {"status": "error"}

        # 5) Skill & Code Evolution（如果有）
        se_result: Dict[str, Any] = {}
        if self.skill_evolution_loop is not None:
            successful = [t for t in trace_dicts if t.get("status") == "success"]
            try:
                se_result = self.skill_evolution_loop.run_once(
                    successful_traces=successful,
                    issues=issues,
                    test_suites=["unit", "static_analysis"],
                    extra_context={"source": "sleep_loop", **ctx},
                )
            except Exception:
                logger.exception("SleepLoop: skill_evolution_loop.run_once failed")
                se_result = {"status": "error"}

        return {
            "trace_stats": stats,
            "memory_consolidation": mem_consolidation,
            "memory_maintenance": mem_maintenance,
            "skill_evolution": se_result,
        }

    # --------------------------------------------------
    # 内部：Trace 概况统计
    # --------------------------------------------------
    def _summarize_traces(self, traces: List[Dict[str, Any]]) -> Dict[str, Any]:
        total = len(traces)
        success = sum(1 for t in traces if t.get("status") == "success")
        failed = sum(1 for t in traces if t.get("status") == "failed")
        error = sum(1 for t in traces if t.get("status") == "error")
        unknown = total - success - failed - error

        domains: Dict[str, int] = {}
        for t in traces:
            d = t.get("domain", "unknown")
            domains[d] = domains.get(d, 0) + 1

        return {
            "total_traces": total,
            "success": success,
            "failed": failed,
            "error": error,
            "unknown": unknown,
            "by_domain": domains,
        }

@dataclass
class CouncilDeliberationResult:

    context: Dict[str, Any]
    final_answer: str

class MultiModelCouncil:
    """
    多模型“开会”协调器。

    - members: [(name, llm_client)]
    - aggregator: 一个负责汇总的 llm（通常是最强/最稳的模型）

    提供：
      - deliberate(question, context) -> CouncilDeliberationResult
    """

    def __init__(
        self,
        members: List[Tuple[str, Any]],
        aggregator: Any,
        member_roles: Optional[Dict[str, str]] = None,
    ) -> None:
        self.members = members
        self.aggregator = aggregator
        self.member_roles = member_roles or {}

    # --------------------------------------------------
    # 对外主接口：多模型协商
    # --------------------------------------------------
    def deliberate(
        self,
        question: str,
        context: Optional[Dict[str, Any]] = None,
    ) -> CouncilDeliberationResult:
        """
        让多个模型对同一个问题独立给出意见，然后由 aggregator 汇总。

        适用场景：
          - 重要计划（planner）
          - 多次失败的任务重试前
          - 代码生成/自编程时的风险评估
        """
        ctx = context or {}
        member_responses: List[CouncilMemberResponse] = []

        # 1) 每个成员单独思考
        for name, llm in self.members:
            role = self.member_roles.get(name, "general-specialist")
            model_name = getattr(llm, "model", getattr(llm, "name", "unknown"))

            prompt = self._build_member_prompt(
                member_name=name,
                member_role=role,
                question=question,
                context=ctx,
            )
            try:
                answer_text = _call_llm(llm, prompt)
            except Exception:
                logger.exception("MultiModelCouncil: member %s failed", name)
                answer_text = "[ERROR] member failed to respond."

            member_responses.append(
                CouncilMemberResponse(
                    name=name,
                    model_name=model_name,
                    role=role,
                    answer=answer_text,
                )
            )

        # 2) aggregator 汇总 / 做最终决策建议
        summary_prompt = self._build_aggregator_prompt(
            question=question,
            context=ctx,
            member_responses=member_responses,
        )
        try:
            summary_text = _call_llm(self.aggregator, summary_prompt)
        except Exception:
            logger.exception("MultiModelCouncil: aggregator failed, fallback to simple merge")
            summary_text = self._fallback_summary(member_responses)

        final_answer, reasoning = self._parse_summary(summary_text)

        return CouncilDeliberationResult(
            question=question,
            context=ctx,
            final_answer=final_answer,
            reasoning=reasoning,
            members=member_responses,
            raw_summary_text=summary_text,
        )

    # --------------------------------------------------
    # Prompt 构造
    # --------------------------------------------------
    def _build_member_prompt(
        self,
        *,
        member_name: str,
        member_role: str,
        question: str,
        context: Dict[str, Any],
    ) -> str:
        ctx_json = json.dumps(context, ensure_ascii=False, indent=2)
        return (
            f"你是一个多模型委员会中的成员：{member_name}，角色：{member_role}。\n"
            f"现在有一个问题需要你独立思考并给出专业意见：\n\n"
            f"【问题】\n{question}\n\n"
            f"【上下文（可能为英文或结构化 JSON）】\n{ctx_json}\n\n"
            f"请你：\n"
            f"1. 简要说明你如何理解这个问题。\n"
            f"2. 给出你的建议方案（可以是步骤、代码策略或风险分析）。\n"
            f"3. 明确指出你认为的主要风险和不确定性。\n"
            f"请用尽量清晰、结构化的中文或英文回答，不要问我问题。"
        )

    def _build_aggregator_prompt(
        self,
        *,
        question: str,
        context: Dict[str, Any],
        member_responses: List[CouncilMemberResponse],
    ) -> str:
        ctx_json = json.dumps(context, ensure_ascii=False, indent=2)

        body_parts = []
        for m in member_responses:
            body_parts.append(
                f"[成员 {m.name} / 模型 {m.model_name} / 角色 {m.role} 的意见]\n{m.answer}\n"
            )
        body = "\n\n".join(body_parts)

        return (
            "你现在扮演多模型委员会的主席（Aggregator）。\n\n"
            "下面是多个不同模型对同一问题给出的独立回答，你需要：\n"
            "1. 找出它们的共识部分。\n"
            "2. 指出主要分歧点。\n"
            "3. 在充分考虑风险的前提下，给出一个综合性的、谨慎的最终建议。\n"
            "4. 你的输出需要是一个 JSON 对象，字段包括：\n"
            '   - \"final_answer\": 综合建议（字符串，尽量简洁）。\n'
            '   - \"reasoning\": 你综合和权衡的过程说明。\n\n"
            "【问题】\n"
            f"{question}\n\n"
            "【上下文】\n"
            f"{ctx_json}\n\n"
            "【成员回答】\n"
            f"{body}\n\n"
            "请只输出一个 JSON 对象，不要有其它文字。"
        )

    # --------------------------------------------------
    # 汇总结果解析 & fallback
    # --------------------------------------------------
    def _parse_summary(self, summary_text: str) -> (str, str):
        """
        解析 aggregator 输出的 JSON，如果失败则做一个宽松降级。
        """
        summary_text = summary_text.strip()
        try:
            data = json.loads(summary_text)
            final_answer = data.get("final_answer") or ""
            reasoning = data.get("reasoning") or ""
            return final_answer, reasoning
        except Exception:
            # fallback：当成普通文本，final_answer=全文，reasoning=简单说明
            logger.warning("MultiModelCouncil: summary not valid JSON, use fallback")
            return summary_text, "Aggregator summary was not valid JSON; using raw text as final_answer."

    def _fallback_summary(self, members: List[CouncilMemberResponse]) -> str:
        parts = []
        for m in members:
            parts.append(f"[{m.name}] {m.answer}")
        return "\n\n".join(parts)


def _call_llm(llm: Any, prompt: str) -> str:
    """
    统一 LLM 调用接口，兼容多种实现：

    优先：
      - llm.complete(prompt=...)
      - llm.generate(prompt=...)
      - llm(prompt)
    """
    if hasattr(llm, "complete"):
        return llm.complete(prompt)
    if hasattr(llm, "generate"):
        return llm.generate(prompt)
    if callable(llm):
        return llm(prompt)
    raise RuntimeError("LLM client has no complete/generate/callable interface")

from __future__ import annotations

from dataclasses import dataclass, asdict, is_dataclass
from typing import Any, Dict, List, Optional
import json
import logging
import time
import uuid

from quentin.orchestrator.council import MultiModelCouncil

logger = logging.getLogger(__name__)


@dataclass
class OrchestratorStep:
    step_id: str
    step_type: str            # "plan" | "execute" | "council" | "reflect"
    llm_name: Optional[str]
    description: str
    prompt: Optional[str]
    raw_output: Any
    parsed: Optional[Dict[str, Any]]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class OrchestratorResult:
    trace_id: str
    status: str
    steps: List[OrchestratorStep]
    execution_summary: Dict[str, Any]
    lessons_learned: Optional[Any]
    top_level_metrics: Dict[str, Any]
    final_output: Any

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        # steps 也转 dict
        d["steps"] = [s.to_dict() for s in self.steps]
        return d


class SimpleOrchestrator:
    """
    通用版 Orchestrator：

    - 不依赖具体 domain（Desktop/Legal/Trading 等都可以用）
    - 只需要一个 LLM（主模型） + 可选的 MultiModelCouncil
    - 负责在一次任务内完成：
        1. 规划（plan）
        2. 执行（generate answer / plan-of-steps）
        3. 多模型开会（council）
        4. 自我反思（reflection）

    输出结构直接喂给 MetaAgentRuntime._build_trace_from_orchestrator。
    """

    def __init__(
        self,
        *,
        llm: Any,
        council: Optional[MultiModelCouncil] = None,
        name: str = "simple-orchestrator",
    ) -> None:
        self.llm = llm
        self.council = council
        self.name = name

    # --------------------------------------------------
    # 对外主接口：执行一次任务
    # --------------------------------------------------
    def run_task(
        self,
        goal: str,
        context: Dict[str, Any],
        runtime: Optional[Any] = None,
    ) -> OrchestratorResult:
        """
        MetaAgentRuntime 会调用这个方法。

        - goal: 用户任务目标（自然语言）
        - context: 附加上下文（domain, user_profile, task_id 等）
        - runtime: 运行时（可以访问 memory_fabric / tools 等）
        """
        trace_id = f"orch-trace::{uuid.uuid4().hex}"
        steps: List[OrchestratorStep] = []
        llm_calls = 0

        start_t = time.perf_counter()
        domain = context.get("domain", "general")

        # 1) 规划阶段（plan）
        plan_prompt = self._build_plan_prompt(goal, context)
        raw_plan = _call_llm(self.llm, plan_prompt)
        llm_calls += 1
        plan = self._parse_plan(raw_plan)

        steps.append(
            OrchestratorStep(
                step_id=f"step::{uuid.uuid4().hex}",
                step_type="plan",
                llm_name=_llm_name(self.llm),
                description="Initial plan generated by main LLM.",
                prompt=plan_prompt,
                raw_output=raw_plan,
                parsed=plan,
            )
        )

        # 2) 执行阶段（execute）
        exec_prompt = self._build_execute_prompt(goal, context, plan)
        raw_exec = _call_llm(self.llm, exec_prompt)
        llm_calls += 1

        steps.append(
            OrchestratorStep(
                step_id=f"step::{uuid.uuid4().hex}",
                step_type="execute",
                llm_name=_llm_name(self.llm),
                description="Execution by main LLM according to the plan.",
                prompt=exec_prompt,
                raw_output=raw_exec,
                parsed=None,
            )
        )

        provisional_output = raw_exec

        # 3) 多模型委员会（可选）
        council_result_dict: Optional[Dict[str, Any]] = None
        if self.council is not None:
            question_for_council = self._build_council_question(
                goal=goal,
                context=context,
                plan=plan,
                provisional_output=provisional_output,
            )
            council_result = self.council.deliberate(
                question=question_for_council,
                context={"domain": domain, "goal": goal, **context},
            )
            council_result_dict = council_result.to_dict()
            steps.append(
                OrchestratorStep(
                    step_id=f"step::{uuid.uuid4().hex}",
                    step_type="council",
                    llm_name="multi-model-council",
                    description="Multi-LLM council deliberation.",
                    prompt=question_for_council,
                    raw_output=council_result_dict,
                    parsed=None,
                )
            )
            final_output = council_result.final_answer or provisional_output
        else:
            final_output = provisional_output

        # 4) 自我反思阶段（reflection）
        reflection_prompt = self._build_reflection_prompt(
            goal=goal,
            context=context,
            plan=plan,
            final_output=final_output,
        )
        raw_reflection = _call_llm(self.llm, reflection_prompt)
        llm_calls += 1

        reflection = {
            "raw_text": raw_reflection,
        }

        steps.append(
            OrchestratorStep(
                step_id=f"step::{uuid.uuid4().hex}",
                step_type="reflect",
                llm_name=_llm_name(self.llm),
                description="Self-reflection about the task and potential improvements.",
                prompt=reflection_prompt,
                raw_output=raw_reflection,
                parsed=reflection,
            )
        )

        duration = time.perf_counter() - start_t

        execution_summary = {
            "final_output": final_output,
            "goal": goal,
            "domain": domain,
            "plan": plan,
            "council_used": self.council is not None,
            "council_result": council_result_dict,
        }

        top_level_metrics = {
            "latency_sec": duration,
            "llm_calls": llm_calls,
        }

        result = OrchestratorResult(
            trace_id=trace_id,
            status="success",
            steps=steps,
            execution_summary=execution_summary,
            lessons_learned=reflection,
            top_level_metrics=top_level_metrics,
            final_output=final_output,
        )
        return result
