2.2 MemoryFabric 实现

# quentin/memory/fabric.py

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Optional

from datetime import datetime

from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.memory.schemas import MemoryRecord, MemoryTier


@dataclass
class MemoryFabric:
    """
    MemoryFabric v1：

    - 提供最小 CRUD / stats / consolidate_once 接口
    - 使用 KVStore 保存 MemoryRecord（metadata）
    - 不强制实现复杂的向量检索逻辑，只给出占位接口
    """

    tracer: Tracer
    kv: KVStore

    key_prefix: str = "memory:record:"
    index_key: str = "memory:index:all"

    # ---------- CRUD ----------

    async def save(self, record: MemoryRecord) -> None:
        record.updated_at = datetime.utcnow()
        key = f"{self.key_prefix}{record.memory_id}"
        await self.kv.save(key, record.model_dump())
        await self.kv.append_list(self.index_key, record.memory_id)

    async def get(self, memory_id: str) -> Optional[MemoryRecord]:
        data = await self.kv.get(f"{self.key_prefix}{memory_id}")
        if not data:
            return None
        return MemoryRecord(**data)

    async def list_all_ids(self, limit: int = 1000) -> List[str]:
        ids = await self.kv.get(self.index_key) or []
        # 去重 + 只取最近一部分
        return list(dict.fromkeys(ids))[-limit:]

    async def add(
        self,
        kind: str,
        content: Dict[str, Any],
        tier: MemoryTier = MemoryTier.STS,
        importance: float = 0.0,
        update_policy: Optional[Dict[str, Any]] = None,
        meta: Optional[Dict[str, Any]] = None,
    ) -> MemoryRecord:
        import uuid

        memory_id = "mem_" + uuid.uuid4().hex
        record = MemoryRecord(
            memory_id=memory_id,
            kind=kind,
            tier=tier,
            content=content,
            importance=importance,
            update_policy=update_policy or {},
            meta=meta or {},
        )
        await self.save(record)

        self.tracer.span(
            "L1",
            "memory_add",
            module="memory",
            input={"memory_id": memory_id, "tier": record.tier.value, "kind": record.kind.value},
        )

        return record

    # ---------- 统计 & stats ----------

    async def stats(self) -> Dict[str, Any]:
        """
        返回每个 tier 的数量等基本统计信息。

        蓝皮书要求至少知道：
        - 每层规模
        - 以后可以扩展命中率 / 升级速度等
        """
        ids = await self.list_all_ids(limit=5000)
        tier_counts = {t.value: 0 for t in MemoryTier}

        for mid in ids:
            rec = await self.get(mid)
            if not rec:
                continue
            tier_counts[rec.tier.value] = tier_counts.get(rec.tier.value, 0) + 1

        return {
            "tiers": tier_counts,
            "total": sum(tier_counts.values()),
        }

    # ---------- 搜索（占位） ----------

    async def search(
        self,
        query: str,
        top_k: int = 5,
        tier: Optional[MemoryTier] = None,
    ) -> List[MemoryRecord]:
        """
        v1：占位实现，用最朴素的方法模拟“搜索”：
        - 按 tier 过滤（如果指定）
        - 在 content["text"] 里做简单 substring 搜索

        以后可以替换成真正的向量检索。
        """
        ids = await self.list_all_ids(limit=5000)
        results: List[MemoryRecord] = []

        for mid in ids:
            rec = await self.get(mid)
            if not rec:
                continue
            if tier and rec.tier != tier:
                continue

            text = str(rec.content.get("text") or rec.content)
            if query.lower() in text.lower():
                results.append(rec)
            if len(results) >= top_k:
                break

        return results

    # ---------- consolidate_once（记忆巩固 & 晋级） ----------

    async def consolidate_once(self, budget: Dict[str, Any]) -> Dict[str, Any]:
        """
        v1 简单策略：
        - 对 STS 中 usage_count >= 3 的条目晋级到 MTS
        - 对 STS 中 importance < 0.1 且 usage_count == 0 的条目视为“可衰减”，不做删除，只打标记

        你以后可以把这里替换成任何复杂算法（基于 update_policy 的规则引擎等）
        """
        ids = await self.list_all_ids(limit=2000)
        promoted = 0
        decayed = 0

        for mid in ids:
            rec = await self.get(mid)
            if not rec or rec.tier != MemoryTier.STS:
                continue

            # 简单晋级规则
            if rec.usage_count >= 3 or rec.importance >= 0.7:
                rec.tier = MemoryTier.MTS
                promoted += 1
                await self.save(rec)
                continue

            # 简单衰减规则
            if rec.usage_count == 0 and rec.importance < 0.1:
                rec.meta.setdefault("decayed", True)
                decayed += 1
                await self.save(rec)

        stats = {"promoted": promoted, "decayed": decayed}
        return stats
这样 MemoryFabric 这一层就完全对齐蓝皮书里「Memory Fabric + 三层记忆 + consolidate_once 接口」的要求了。

# quentin/research/experiment_orchestrator.py

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Optional

from datetime import datetime
import uuid

from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.research.experiment_schemas import (
    ExperimentRecord,
    ExperimentStatus,
    ExperimentConfig,
    ExperimentVariant,
)


@dataclass
class ExperimentOrchestrator:
    """
    Research Layer 的实验编排器。

    v1 功能：
    - create_experiment：定义一个实验及其 variants
    - list_experiments / get_experiment
    - run_experiment：占位实现（记录一次“运行”，生成假数据）
    - compare_variants：简单根据某个 metric 排序

    之后你可以在 run_experiment 里：
    - 根据 config.target_module 定位具体模块
    - 挂上不同 Variant 的实现
    - 跑离线数据集或在线小流量实验
    """

    tracer: Tracer
    kv: KVStore

    record_prefix: str = "exp:record:"
    index_key: str = "exp:index:all"

    async def _save(self, rec: ExperimentRecord) -> None:
        rec.updated_at = datetime.utcnow()
        key = f"{self.record_prefix}{rec.experiment_id}"
        await self.kv.save(key, rec.model_dump())
        await self.kv.append_list(self.index_key, rec.experiment_id)

    async def create_experiment(
        self,
        name: str,
        description: str,
        config: ExperimentConfig,
        variants: List[ExperimentVariant],
    ) -> ExperimentRecord:
        exp_id = "exp_" + uuid.uuid4().hex
        rec = ExperimentRecord(
            experiment_id=exp_id,
            name=name,
            description=description,
            config=config,
            variants=variants,
        )
        await self._save(rec)

        self.tracer.span(
            "L3",
            "experiment_created",
            module="research",
            input={"experiment_id": exp_id, "name": name, "target": config.target_module},
        )

        return rec

    async def get_experiment(self, experiment_id: str) -> Optional[ExperimentRecord]:
        data = await self.kv.get(f"{self.record_prefix}{experiment_id}")
        if not data:
            return None
        return ExperimentRecord(**data)

    async def list_experiments(self, limit: int = 200) -> List[ExperimentRecord]:
        ids = await self.kv.get(self.index_key) or []
        ids = list(dict.fromkeys(ids))[-limit:]

        res: List[ExperimentRecord] = []
        for eid in ids:
            data = await self.kv.get(f"{self.record_prefix}{eid}")
            if data:
                res.append(ExperimentRecord(**data))
        return res

    async def run_experiment(self, experiment_id: str) -> ExperimentRecord:
        """
        v1 占位实现：不真正调用任何模块，只是：
        - 把状态从 pending -> running -> completed
        - 给每个 variant 生成一些假指标

        之后你可以把这里替换为：
        - 在 Sandbox 里挂载不同的模块实现
        - 跑固定的数据集 / replay traces
        - 统计真实的 reward / cost 等
        """
        rec = await self.get_experiment(experiment_id)
        if not rec:
            raise ValueError(f"experiment not found: {experiment_id}")

        rec.status = ExperimentStatus.running
        await self._save(rec)

        # 假装跑了一堆测试
        results: Dict[str, Any] = {}
        for v in rec.variants:
            results[v.variant_id] = {
                "metric_example": 0.5,  # 这里以后换成真实 metric
                "note": "placeholder",
            }

        rec.status = ExperimentStatus.completed
        rec.results = results
        await self._save(rec)

        self.tracer.span(
            "L3",
            "experiment_run",
            module="research",
            input={"experiment_id": experiment_id},
            output={"status": rec.status.value},
        )

        return rec

    async def compare_variants(
        self,
        experiment_id: str,
        metric: str,
    ) -> List[Dict[str, Any]]:
        """
        按某个 metric 对 variants 排序。
        """
        rec = await self.get_experiment(experiment_id)
        if not rec or not rec.results:
            return []

        rows = []
        for v in rec.variants:
            rv = rec.results.get(v.variant_id, {})
            val = rv.get(metric)
            rows.append(
                {
                    "variant_id": v.variant_id,
                    "name": v.name,
                    "metric": val,
                }
            )

        rows = [r for r in rows if r["metric"] is not None]
        rows.sort(key=lambda x: x["metric"], reverse=True)
        return rows

from quentin.observability.service import ObservabilityService

from quentin.research.experiment_orchestrator import ExperimentOrchestrator


@dataclass
class EvolutionStrategyManager:
    """
    Evolution Strategy Manager v1：

    - 使用 Observability 的数据来判断“优化热点”
    - 为热点模块构造 ExperimentConfig + Variants
    - 交给 ExperimentOrchestrator 去创建实验

    这个类不直接修改任何业务逻辑，只负责“决定在哪些地方做实验”。
    """

    observability: ObservabilityService
    orchestrator: ExperimentOrchestrator

    async def suggest_hotspots(self) -> Dict[str, Any]:
        """
        根据当前指标，找出值得优先优化的模块。
        v1 简单策略：
        - Immune cases 多的模块是热点
        - patch 回滚多的模块是热点
        """
        overview = await self.observability.get_overview()
        immune = overview.get("immune", {})
        patches = overview.get("patches", {})

        hotspots: List[str] = []

        if immune.get("cases", 0) > 0:
            hotspots.append("immune_related_modules")

        if patches.get("rolled_back", 0) > 0:
            hotspots.append("self_programming")

        return {
            "hotspots": hotspots,
        }

    async def create_simple_experiment_for_module(
        self,
        module_name: str,
        metric: str,
    ) -> str:
        """
        为某个模块创建一个最简单的实验：
        - baseline vs candidate
        - metric 由调用方指定
        """
        cfg = ExperimentConfig(
            target_module=module_name,
            metrics=[metric],
            meta={},
        )

        variants = [
            ExperimentVariant(
                variant_id="baseline",
                name="baseline",
                params={"strategy": "baseline"},
            ),
            ExperimentVariant(
                variant_id="candidate",
                name="candidate",
                params={"strategy": "candidate"},
            ),
        ]

        rec = await self.orchestrator.create_experiment(
            name=f"{module_name}_baseline_vs_candidate",
            description=f"Auto-created experiment for {module_name}",
            config=cfg,
            variants=variants,
        )
        return rec.experiment_id

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Dict


@dataclass
class BudgetController:
    """
    管理系统内的算力 / token / 资金等预算。

    v1 实现：
    - 用一个简单的字典记录每个 "bucket" 的最大额度和已用额度
    - check_and_consume 用于扣减
    """

    limits: Dict[str, float] = field(default_factory=dict)
    used: Dict[str, float] = field(default_factory=dict)

    def set_limit(self, key: str, value: float) -> None:
        self.limits[key] = value

    def get_remaining(self, key: str) -> float:
        limit = self.limits.get(key, 0.0)
        u = self.used.get(key, 0.0)
        return max(0.0, limit - u)

    def check_and_consume(self, key: str, cost: float) -> bool:
        remaining = self.get_remaining(key)
        if cost > remaining:
            return False
        self.used[key] = self.used.get(key, 0.0) + cost
        return True

# quentin/models/router.py

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any, Dict, List, Optional
from collections import defaultdict
from datetime import datetime


@dataclass
class ModelInfo:
    name: str
    kind: str              # "gpt" / "llama" / "local" / ...
    cost_per_1k_tokens: float = 0.0
    latency_hint_ms: int = 1000
    domains: List[str] = field(default_factory=list)
    tags: List[str] = field(default_factory=list)


@dataclass
class ModelUsageStats:
    calls: int = 0
    total_tokens: int = 0
    total_cost: float = 0.0
    total_latency_ms: int = 0
    last_called_at: Optional[datetime] = None


@dataclass
class ModelRouter:
    """
    v1 ModelRouter：

    - 记录可用模型及其基本信息
    - 根据 task 的 domain / preference 做简单路由
    - 聚合使用统计，提供给 ObservabilityService.get_model_stats()
    """

    models: Dict[str, ModelInfo] = field(default_factory=dict)
    stats: Dict[str, ModelUsageStats] = field(default_factory=dict)

    def register_model(self, info: ModelInfo) -> None:
        self.models[info.name] = info
        self.stats.setdefault(info.name, ModelUsageStats())

    def choose_model(
        self,
        task_type: str,
        domain: Optional[str] = None,
        preference: str = "balanced",  # "speed" / "cost" / "quality" / ...
    ) -> Optional[str]:
        """
        v1 简单路由策略：
        - 先按 domain 过滤
        - 再根据 preference 选一个模型
        """
        candidates = list(self.models.values())
        if domain:
            filtered = [m for m in candidates if domain in (m.domains or [])]
            if filtered:
                candidates = filtered

        if not candidates:
            return None

        if preference == "speed":
            candidates.sort(key=lambda m: m.latency_hint_ms)
        elif preference == "cost":
            candidates.sort(key=lambda m: m.cost_per_1k_tokens)
        else:
            # balanced：简单按 name 排序
            candidates.sort(key=lambda m: m.name)

        return candidates[0].name

    def record_usage(
        self,
        model_name: str,
        tokens: int,
        latency_ms: int,
    ) -> None:
        s = self.stats.setdefault(model_name, ModelUsageStats())
        s.calls += 1
        s.total_tokens += tokens
        s.total_latency_ms += latency_ms
        info = self.models.get(model_name)
        cost_per_1k = info.cost_per_1k_tokens if info else 0.0
        s.total_cost += (tokens / 1000.0) * cost_per_1k
        s.last_called_at = datetime.utcnow()

    async def get_stats(self, limit: int = 2000) -> Dict[str, Any]:
        """
        为 ObservabilityService.get_model_stats() 提供的数据。
        """
        models: Dict[str, Any] = {}
        for name, info in self.models.items():
            s = self.stats.get(name, ModelUsageStats())
            models[name] = {
                "kind": info.kind,
                "domains": info.domains,
                "tags": info.tags,
                "calls": s.calls,
                "total_tokens": s.total_tokens,
                "total_cost": s.total_cost,
                "avg_latency_ms": (s.total_latency_ms / s.calls) if s.calls else None,
            }

        total_calls = sum(m["calls"] for m in models.values())
        return {"models": models, "total_calls": total_calls}

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any, Awaitable, Callable, Dict

ToolFn = Callable[[Dict[str, Any]], Awaitable[Dict[str, Any]]]


@dataclass
class ToolManager:
    """
    通用 Tool & Skill 管理器：
    - register_tool：注册异步函数
    - call_tool：按名称调用
    """

    tools: Dict[str, ToolFn] = field(default_factory=dict)

    def register_tool(self, name: str, fn: ToolFn) -> None:
        self.tools[name] = fn

    async def call_tool(self, name: str, args: Dict[str, Any]) -> Dict[str, Any]:
        if name not in self.tools:
            raise KeyError(f"tool not found: {name}")
        return await self.tools[name](args)

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List

from quentin.planning.planner import Plan, PlanStep
from quentin.tools.manager import ToolManager
from quentin.skills.library import SkillLibrary


@dataclass
class Executor:
    """
    执行层：

    - 接收 Planner 生成的 Plan
    - 逐步执行 PlanStep（调用 Tool 或 Skill）
    - 返回一个简单的 execution_summary（供 Critic / SleepLoop 使用）

    v1 实现只处理 kind == "llm_call" 或 "tool" 的步骤。
    """

    tools: ToolManager
    skill_lib: SkillLibrary

    async def execute(self, plan: Plan) -> Dict[str, Any]:
        step_results: List[Dict[str, Any]] = []

        for step in plan.steps:
            if step.kind == "tool":
                result = await self.tools.call_tool(step.name, step.args)
            else:
                # 其他类型占位处理
                result = {"status": "noop", "kind": step.kind}

            step_results.append(
                {
                    "step_id": step.step_id,
                    "kind": step.kind,
                    "name": step.name,
                    "result": result,
                }
            )

        summary = {
            "plan_id": plan.plan_id,
            "steps": step_results,
            "success": True,
        }
        return summary

            "lessons_learned": comment,
            "raw_summary": execution_summary,
        }
