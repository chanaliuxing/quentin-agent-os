# quentin/self_improvement/auto_bridge.py
from __future__ import annotations

import logging
import uuid
from typing import Optional, List

from quentin.core.task import TaskSpec, TaskResult
from quentin.core.critic import CriticFeedback
from quentin.self_improvement.manager import SelfImprovementManager
from quentin.self_improvement.schemas import ImprovementTask

logger = logging.getLogger(__name__)


class AutoImprovementBridge:
    """
    把 CognitiveLoop 中的“失败任务 / 低置信任务”，自动转成 ImprovementTask。

    设计原则：
      - 不对“每一个小错误”都建任务，否则会淹没自己；
      - 针对“明显失败 / 置信度很低 / 有 error_message”的任务，生成一条 TODO；
      - capability 命名为：cogloop:<domain>
      - requirement_id 固定先用：task_quality
    """

    def __init__(
        self,
        *,
        si_manager: Optional[SelfImprovementManager] = None,
        min_confidence: float = 0.6,
    ) -> None:
        self.si_manager = si_manager or SelfImprovementManager()
        self.min_confidence = min_confidence

    # ---------- 主入口 ----------

    def maybe_create_from_cog_result(
        self,
        task: TaskSpec,
        result: TaskResult,
        feedback: CriticFeedback,
    ) -> Optional[ImprovementTask]:
        """
        根据 TaskResult + CriticFeedback 判断是否需要创建 ImprovementTask。
        """
        need = self._should_create(task, result, feedback)
        if not need:
            return None

        # 防止重复：同一个 original_task_id 只建一次
        existing = self._find_existing_for_task(task.task_id)
        if existing:
            logger.info(
                "AutoImprovementBridge: task_id=%s already has ImprovementTask=%s",
                task.task_id,
                existing.task_id,
            )
            return existing

        capability = f"cogloop:{task.domain}"
        requirement_id = "task_quality"

        desc = (
            f"Auto-created from CognitiveLoop failure.\n"
            f"- Original task_id: {task.task_id}\n"
            f"- Goal: {task.goal}\n\n"
            f"- Error: {result.error_message}\n\n"
            f"- Critic lessons (truncated):\n"
            f"{(feedback.lessons_learned or '')[:1200]}"
        )

        it = ImprovementTask(
            task_id=str(uuid.uuid4()),
            capability=capability,
            requirement_id=requirement_id,
            description=desc,
            source="cognitive_loop.auto",
            overall_status="todo",
            metadata={
                "original_task_id": task.task_id,
                "domain": task.domain,
                "goal": task.goal,
            },
        )

        self.si_manager.add_task(it)
        logger.info(
            "AutoImprovementBridge: created ImprovementTask=%s for original_task_id=%s",
            it.task_id,
            task.task_id,
        )
        return it

    # ---------- 内部逻辑 ----------

    def _should_create(
        self,
        task: TaskSpec,
        result: TaskResult,
        feedback: CriticFeedback,
    ) -> bool:
        if result.error_message:
            return True
        if not result.success:
            return True
        if feedback is None:
            return False
        if not feedback.ok:
            return True
        if feedback.confidence < self.min_confidence:
            return True
        return False

    def _find_existing_for_task(self, original_task_id: str) -> Optional[ImprovementTask]:
        tasks: List[ImprovementTask] = self.si_manager.list_tasks()
        for t in tasks:
            meta = getattr(t, "metadata", {}) or {}
            if meta.get("original_task_id") == original_task_id:
                return t
        return None
注意：这里假设 ImprovementTask 有字段 task_id/capability/requirement_id/description/source/overall_status/metadata，和我们前面设计的一致。如果你之后稍微改 schema，这里改一改字段名即可。

# quentin/research/cog_bridge.py
from __future__ import annotations

import logging
from datetime import datetime
from typing import Optional, Dict, Any

from quentin.core.task import TaskSpec, TaskResult
from quentin.core.critic import CriticFeedback
from quentin.research.experiments import ResearchManager, Experiment

logger = logging.getLogger(__name__)


class CognitiveResearchBridge:
    """
    把 CognitiveLoop 的 TaskResult 作为在线实验数据写入 ResearchManager。

    策略：
      - capability  : "cogloop:<domain>"
      - requirement : "task_success"
      - purpose     : "cogloop_online"

      - variant_id  : "default"（后续可以扩展为不同 agent_version / 策略）
      - metrics:
          success           : bool
          critic_confidence : float
          has_error         : bool
          duration_seconds  : float
    """

    def __init__(self, rm: Optional[ResearchManager] = None) -> None:
        self.rm = rm or ResearchManager()

    def record_task_outcome(
        self,
        task: TaskSpec,
        result: TaskResult,
        feedback: Optional[CriticFeedback],
    ) -> None:
        capability = f"cogloop:{task.domain}"
        requirement_id = "task_success"

        exp = self._get_or_create_experiment(
            capability=capability,
            requirement_id=requirement_id,
        )

        duration = self._compute_duration_seconds(
            result.started_at,
            result.finished_at,
        )

        metrics: Dict[str, Any] = {
            "success": bool(result.success),
            "critic_confidence": float(feedback.confidence) if feedback else None,
            "has_error": bool(result.error_message),
            "duration_seconds": duration,
        }

        context = {
            "task_id": task.task_id,
            "goal": task.goal[:200],
            "domain": task.domain,
        }

        self.rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="default",
            metrics=metrics,
            context=context,
        )

    # ---------- 内部 ----------

    def _get_or_create_experiment(
        self,
        *,
        capability: str,
        requirement_id: str,
    ) -> Experiment:
        exps = self.rm.list_experiments()
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "cogloop_online"
            ):
                return e

        return self.rm.create_experiment(
            name=f"CognitiveLoop-{capability}-{requirement_id}",
            purpose="cogloop_online",
            capability=capability,
            requirement_id=requirement_id,
            tags=["cognitive_loop", "online"],
        )

    def _compute_duration_seconds(
        self,
        start: str,
        end: str,
    ) -> Optional[float]:
        try:
            s = datetime.fromisoformat(start)
            e = datetime.fromisoformat(end)
            return (e - s).total_seconds()
        except Exception:
            return None

# quentin/kernel/safety.py
from __future__ import annotations

import json
import logging
from dataclasses import dataclass, field, asdict
from datetime import datetime
from pathlib import Path
from typing import List, Dict, Any, Optional

from quentin.self_programming.patches import CodePatch

logger = logging.getLogger(__name__)


def _norm_path(p: str) -> str:
    return p.replace("\\", "/").lstrip("./")


@dataclass
class SafetyConfig:
    """
    安全内核配置（可通过 data/safety_kernel.json 修改）：

    - protected_prefixes : 完全禁止自动修改的路径前缀（kernel / 宪法 / 蓝皮书等）
    - high_risk_prefixes : 强风险路径（如 trading / legal 等业务关键模块）
    - auto_apply_max_risk: 自动应用 patch 能接受的最大风险级别（"low" / "medium"）
    - enabled            : 是否启用安全内核（不建议关）
    """
    protected_prefixes: List[str] = field(
        default_factory=lambda: [
            "quentin/kernel",
            "quentin/blueprint",
        ]
    )
    high_risk_prefixes: List[str] = field(
        default_factory=lambda: [
            "quentin/trading",
            "quentin/legal",
        ]
    )
    auto_apply_max_risk: str = "medium"  # "low" / "medium"
    enabled: bool = True

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "SafetyConfig":
        if d is None:
            return cls()
        return cls(
            protected_prefixes=list(d.get("protected_prefixes", []) or []),
            high_risk_prefixes=list(d.get("high_risk_prefixes", []) or []),
            auto_apply_max_risk=d.get("auto_apply_max_risk", "medium"),
            enabled=bool(d.get("enabled", True)),
        )


@dataclass
class PatchRiskAssessment:
    """
    对一个 CodePatch 的风险评估结果。

    risk_level: "low" / "medium" / "high" / "blocked"
    blocked   : 是否禁止自动应用
    reasons   : 文字说明
    """
    patch_id: str
    risk_level: str
    blocked: bool
    reasons: List[str] = field(default_factory=list)
    capability: Optional[str] = None
    requirement_id: Optional[str] = None
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class SafetyKernel:
    """
    v1 安全内核：

      - 不负责“如何写 patch”，只负责“哪些 patch 可以、哪些不可以自动上线”；
      - 不负责“应用 patch”的细节（那是 future 的 PatchApplier），只提供 can_auto_apply 决策；
      - 至少保证：
          * quentin/kernel / quentin/blueprint 等目录永远不允许自动改；
          * trading / legal 等敏感目录，被标记为 high risk；
    """

    def __init__(
        self,
        *,
        config_path: str | Path = "data/safety_kernel.json",
        decisions_path: str | Path = "data/safety_kernel_decisions.json",
    ) -> None:
        self.config_path = Path(config_path)
        self.decisions_path = Path(decisions_path)
        self.config_path.parent.mkdir(parents=True, exist_ok=True)
        self.decisions_path.parent.mkdir(parents=True, exist_ok=True)

        self.config: SafetyConfig = self._load_config()
        self._decisions: Dict[str, Dict[str, Any]] = self._load_decisions()

    # ---------- 配置 & 决策持久化 ----------

    def _load_config(self) -> SafetyConfig:
        if self.config_path.exists():
            try:
                raw = json.loads(self.config_path.read_text(encoding="utf-8") or "{}")
                return SafetyConfig.from_dict(raw)
            except Exception:
                logger.exception("SafetyKernel: failed to load config, using default.")
        return SafetyConfig()

    def _save_config(self) -> None:
        self.config_path.write_text(
            json.dumps(self.config.to_dict(), ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    def _load_decisions(self) -> Dict[str, Dict[str, Any]]:
        if self.decisions_path.exists():
            try:
                raw = json.loads(
                    self.decisions_path.read_text(encoding="utf-8") or "{}"
                )
                return raw.get("decisions", {}) or {}
            except Exception:
                logger.exception("SafetyKernel: failed to load decisions.")
        return {}

    def _save_decisions(self) -> None:
        raw = {"decisions": self._decisions}
        self.decisions_path.write_text(
            json.dumps(raw, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    # ---------- 风险评估 & 决策 ----------

    def assess_patch(self, patch: CodePatch) -> PatchRiskAssessment:
        """
        纯规则版风险评估（不调用 LLM）。未来可以在外层挂多 AI 评审再修正 risk_level。
        """
        cap = getattr(patch, "capability", None)
        req = getattr(patch, "requirement_id", None)

        # 尝试从 patch 中找目标路径（约定字段）
        target_path = getattr(patch, "target_path", None)
        if target_path is None:
            meta = getattr(patch, "metadata", {}) or {}
            target_path = meta.get("target_path")

        reasons: List[str] = []
        risk = "low"
        blocked = False

        if not self.config.enabled:
            reasons.append("SafetyKernel disabled in config.")
            risk = "medium"

        # 路径检查
        if target_path:
            norm = _norm_path(str(target_path))

            for prefix in self.config.protected_prefixes:
                if norm.startswith(prefix.rstrip("/")):
                    risk = self._max_risk(risk, "blocked")
                    blocked = True
                    reasons.append(
                        f"target_path={norm} is under protected_prefix={prefix}"
                    )

            for prefix in self.config.high_risk_prefixes:
                if norm.startswith(prefix.rstrip("/")):
                    risk = self._max_risk(risk, "high")
                    reasons.append(
                        f"target_path={norm} is under high_risk_prefix={prefix}"
                    )
        else:
            reasons.append("No target_path in patch metadata; defaulting to medium risk.")
            risk = self._max_risk(risk, "medium")

        # capability 级别规则
        cap_str = (cap or "").lower()
        if "trading" in cap_str or "legal" in cap_str:
            risk = self._max_risk(risk, "high")
            reasons.append(f"capability={cap} indicates high-risk domain.")

        # blocked 优先级最高
        if risk == "blocked":
            blocked = True

        assessment = PatchRiskAssessment(
            patch_id=patch.patch_id,
            risk_level=risk,
            blocked=blocked,
            reasons=reasons,
            capability=cap,
            requirement_id=req,
        )
        return assessment

    def can_auto_apply(self, assessment: PatchRiskAssessment) -> bool:
        """
        根据 config + assessment 决定是否允许“自动应用 patch”。
        当前我们的自编程引擎还没自动 apply patch，但这个接口是预留给未来的。
        """
        if not self.config.enabled:
            return False
        if assessment.blocked:
            return False

        level = assessment.risk_level
        max_level = self.config.auto_apply_max_risk

        order = {"low": 0, "medium": 1, "high": 2, "blocked": 3}
        if order.get(level, 2) > order.get(max_level, 1):
            return False
        return True

    def record_decision(
        self,
        patch: CodePatch,
        assessment: PatchRiskAssessment,
        *,
        approved_by: str,
    ) -> None:
        """
        记录一次决策（包括“拒绝自动应用”的情况），方便审计与可追溯。
        """
        d = {
            "assessment": assessment.to_dict(),
            "approved_by": approved_by,
            "recorded_at": datetime.utcnow().isoformat(),
        }
        self._decisions[patch.patch_id] = d
        self._save_decisions()

    @staticmethod
    def _max_risk(a: str, b: str) -> str:
        order = {"low": 0, "medium": 1, "high": 2, "blocked": 3}
        va = order.get(a, 0)
        vb = order.get(b, 0)
        return a if va >= vb else b
