# quentin/traces/core.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from datetime import datetime
from enum import Enum
from typing import Any, Dict, List, Optional
import uuid


class TraceEventType(str, Enum):
    TASK_RECEIVED = "task_received"
    PLANNING = "planning"
    EXECUTION = "execution"
    TOOL_CALL = "tool_call"
    MODEL_CALL = "model_call"
    CRITIC = "critic"
    IMMUNE = "immune"
    SLEEP = "sleep"
    SKILL_EVOLUTION = "skill_evolution"
    SELF_PROGRAMMING = "self_programming"
    SUMMARY = "summary"
    ERROR = "error"


@dataclass
class TraceEvent:
    event_id: str
    trace_id: str
    event_type: str
    timestamp: str
    payload: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ExecutionSummary:
    """
    一次任务完成后的结构化总结。
    """
    trace_id: str
    task_id: str
    status: str  # "success" | "failed" | "partial"
    result_preview: str
    meta: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class LessonsLearned:
    """
    Critic / Sleep Loop 生成的“经验教训”，可以写入 Memory Fabric。
    """
    trace_id: str
    task_id: str
    content: str
    tags: List[str] = field(default_factory=list)
    meta: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class TraceRecorder:
    """
    Trace 记录接口：所有 loop 都只依赖这个接口，不依赖具体存储。
    """

    def record_event(self, event: TraceEvent) -> None:  # pragma: no cover - interface
        raise NotImplementedError

    def record_summary(self, summary: ExecutionSummary) -> None:  # pragma: no cover
        raise NotImplementedError

    def record_lessons(self, lessons: LessonsLearned) -> None:  # pragma: no cover
        raise NotImplementedError


class JsonlTraceRecorder(TraceRecorder):
    """
    非常简单的 JSONL Trace 记录器：
      - data/traces/events.jsonl
      - data/traces/summaries.jsonl
      - data/traces/lessons.jsonl

    后面可以由自迭代去升级成 ClickHouse / DuckDB / 向量库等。
    """

    def __init__(self, repo_root: str | None = None) -> None:
        from pathlib import Path
        import json
        self._json = json
        from pathlib import Path as _P
        root = _P(repo_root or ".").resolve()
        traces_dir = root / "data" / "traces"
        traces_dir.mkdir(parents=True, exist_ok=True)

        self._events_path = traces_dir / "events.jsonl"
        self._summaries_path = traces_dir / "summaries.jsonl"
        self._lessons_path = traces_dir / "lessons.jsonl"

    def _append_line(self, path, obj: Dict[str, Any]) -> None:
        with path.open("a", encoding="utf-8") as f:
            f.write(self._json.dumps(obj, ensure_ascii=False) + "\n")

    def record_event(self, event: TraceEvent) -> None:
        self._append_line(self._events_path, event.to_dict())

    def record_summary(self, summary: ExecutionSummary) -> None:
        self._append_line(self._summaries_path, summary.to_dict())

    def record_lessons(self, lessons: LessonsLearned) -> None:
        self._append_line(self._lessons_path, lessons.to_dict())


def new_trace_event(
    trace_id: str,
    event_type: TraceEventType,
    payload: Optional[Dict[str, Any]] = None,
) -> TraceEvent:
    return TraceEvent(
        event_id=str(uuid.uuid4()),
        trace_id=trace_id,
        event_type=event_type.value,
        timestamp=datetime.utcnow().isoformat(),
        payload=payload or {},
    )

# quentin/runtime/cognitive_loop.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, Optional
from datetime import datetime
import uuid

from quentin.traces.core import (
    TraceEventType,
    TraceRecorder,
    ExecutionSummary,
    LessonsLearned,
    new_trace_event,
)
from quentin.memory.simple_fabric import SimpleMemoryFabric  # 或者你自己的 MemoryFabric
from quentin.critic.core import SimpleCritic


@dataclass
class TaskRequest:
    """
    通用任务请求结构，不绑定任何具体 domain。
    """
    task_id: str
    description: str
    payload: Dict[str, Any]
    context: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class TaskResult:
    task_id: str
    trace_id: str
    status: str
    output: Any
    summary: ExecutionSummary
    lessons: Optional[LessonsLearned]


class CognitiveLoop:
    """
    认知循环（Cognitive Loop）骨架：

      - 统一入口：run_task(...)
      - 强制：有 Trace、有 Summary、有 Critic、自反思可以写入 Memory Fabric。
    """

    def __init__(
        self,
        *,
        trace_recorder: TraceRecorder,
        memory_fabric: SimpleMemoryFabric,
        critic: SimpleCritic,
        runtime: Any,  # MetaAgentRuntime 或其他：负责真正的“怎么执行任务”
    ) -> None:
        self.trace_recorder = trace_recorder
        self.memory_fabric = memory_fabric
        self.critic = critic
        self.runtime = runtime

    # --------------------------------------------------
    # 对外：统一入口
    # --------------------------------------------------
    def run_task(
        self,
        description: str,
        payload: Dict[str, Any],
        context: Optional[Dict[str, Any]] = None,
    ) -> TaskResult:
        task_id = str(uuid.uuid4())
        trace_id = f"trace-{task_id}"

        ctx = context or {}

        # 1) 记录 TASK_RECEIVED
        self.trace_recorder.record_event(
            new_trace_event(
                trace_id,
                TraceEventType.TASK_RECEIVED,
                {"task_id": task_id, "description": description, "context": ctx},
            )
        )

        # 2) 真正执行任务（调用 runtime）
        #    这里我们假设 runtime 有一个 execute_task(...) 方法，
        #    由你根据实际情况去实现（可以是 planner + executor 的封装）。
        try:
            self.trace_recorder.record_event(
                new_trace_event(
                    trace_id,
                    TraceEventType.PLANNING,
                    {"task_id": task_id, "note": "planning & execution started"},
                )
            )
            output = self.runtime.execute_task(
                task_id=task_id,
                description=description,
                payload=payload,
                context=ctx,
                trace_id=trace_id,
                trace_recorder=self.trace_recorder,
            )
            status = "success"
        except Exception as e:
            output = {"error": repr(e)}
            status = "failed"
            self.trace_recorder.record_event(
                new_trace_event(
                    trace_id,
                    TraceEventType.ERROR,
                    {"task_id": task_id, "error": repr(e)},
                )
            )

        # 3) 生成 ExecutionSummary
        result_preview = str(output)[:400]
        summary = ExecutionSummary(
            trace_id=trace_id,
            task_id=task_id,
            status=status,
            result_preview=result_preview,
            meta={
                "description": description,
                "timestamp": datetime.utcnow().isoformat(),
            },
        )
        self.trace_recorder.record_summary(summary)

        # 4) 用 Critic 做简单自反思（LessonsLearned）
        critic_out = self.critic.evaluate(summary=result_preview, context={"trace_id": trace_id})
        comment = str(critic_out.get("comment", ""))[:1000]
        lessons = LessonsLearned(
            trace_id=trace_id,
            task_id=task_id,
            content=comment,
            tags=["cognitive_loop", f"status:{status}"],
            meta={"critic_score": critic_out.get("score", None)},
        )
        self.trace_recorder.record_lessons(lessons)

        # 5) 可以把 Lessons 写入 Memory Fabric（STS 层）
        from quentin.memory.simple_fabric import MemoryItem
        mem_item = MemoryItem(
            id=f"lesson-{trace_id}",
            tier="STS",
            content=lessons.content,
            meta={
                "trace_id": trace_id,
                "task_id": task_id,
                "status": status,
                "source": "cognitive_loop",
            },
        )
        self.memory_fabric.append(mem_item)

        return TaskResult(
            task_id=task_id,
            trace_id=trace_id,
            status=status,
            output=output,
            summary=summary,
            lessons=lessons,
        )

3. 免疫循环骨架：quentin/runtime/immune_loop.py
新文件：quentin/runtime/immune_loop.py
功能：统计 error rate / latency / cost，超过阈值就生成“降级 / 回滚 / 减少自改”的动作。

# quentin/runtime/immune_loop.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, Optional, List
from collections import deque
from datetime import datetime

from quentin.traces.core import TraceRecorder, new_trace_event, TraceEventType


@dataclass
class ImmuneAction:
    """
    当检测到异常时，免疫循环返回的“建议动作”。

    action_type:
      - "none"             : 不做任何操作
      - "degrade"          : 降级（关闭部分功能/自改）
      - "rollback"         : 回滚到上一稳定版本
      - "reduce_exploration": 降低探索强度
      - "pause_self_programming": 暂停自编程功能
    """
    action_type: str
    reason: str
    metadata: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class AnomalyDetectorConfig:
    window_size: int = 100
    error_rate_threshold: float = 0.3
    max_latency_ms: Optional[float] = None


class ImmuneLoop:
    """
    Immune Loop 骨架：

      - 维护一个滑动窗口（最近 N 次 task 的结果）
      - 计算 error rate / latency 等指标
      - 当超过阈值时，给出 ImmuneAction（降级/回滚/暂停自编程）
      - 同时写 Trace，方便后续 Research Layer 分析免疫事件
    """

    def __init__(
        self,
        *,
        config: AnomalyDetectorConfig,
        trace_recorder: TraceRecorder,
    ) -> None:
        self.config = config
        self.trace_recorder = trace_recorder
        self._window: deque[Dict[str, Any]] = deque(maxlen=config.window_size)

    def record_task_outcome(
        self,
        *,
        trace_id: str,
        task_id: str,
        status: str,
        latency_ms: Optional[float] = None,
        cost_tokens: Optional[float] = None,
    ) -> None:
        self._window.append(
            {
                "trace_id": trace_id,
                "task_id": task_id,
                "status": status,
                "latency_ms": latency_ms,
                "cost_tokens": cost_tokens,
                "timestamp": datetime.utcnow().isoformat(),
            }
        )

    def analyze_and_maybe_act(self) -> ImmuneAction:
        if not self._window:
            return ImmuneAction(action_type="none", reason="no_data", metadata={})

        total = len(self._window)
        errors = sum(1 for x in self._window if x["status"] == "failed")
        error_rate = errors / total

        max_lat = None
        if self.config.max_latency_ms is not None:
            latencies = [x["latency_ms"] for x in self._window if x["latency_ms"] is not None]
            if latencies:
                max_lat = max(latencies)

        # 简单规则：可以以后改成 ML
        if error_rate >= self.config.error_rate_threshold:
            action = ImmuneAction(
                action_type="degrade",
                reason=f"high_error_rate={error_rate:.3f}",
                metadata={"window_size": total, "errors": errors},
            )
        elif max_lat is not None and max_lat > self.config.max_latency_ms:  # type: ignore
            action = ImmuneAction(
                action_type="reduce_exploration",
                reason=f"high_latency={max_lat}ms",
                metadata={"window_size": total},
            )
        else:
            action = ImmuneAction(
                action_type="none",
                reason="within_threshold",
                metadata={"error_rate": error_rate},
            )

        # 写 Trace
        self.trace_recorder.record_event(
            new_trace_event(
                trace_id="immune-loop",
                event_type=TraceEventType.IMMUNE,
                payload=action.to_dict(),
            )
        )
        return action

# quentin/runtime/sleep_loop.py
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Any, Dict, List

import json

from quentin.traces.core import TraceRecorder, TraceEventType, new_trace_event
from quentin.memory.simple_fabric import SimpleMemoryFabric, MemoryItem


@dataclass
class SleepLoopConfig:
    max_traces_per_run: int = 500
    promotion_threshold: int = 3  # Lessons 被引用次数达到多少次就升级 MTS/LTS（以后再真实现）
    run_mode: str = "nightly"     # "nightly" | "manual" 等


class SleepLoop:
    """
    Sleep / Consolidation Loop 骨架：

      - 每次运行：
        * 从 STS/MTS 读取一定数量的记忆 / lessons
        * 做简单的摘要/合并（目前 stub）
        * 决定是否将部分记忆升级为 MTS / LTS

      目前实现是非常简单的占位逻辑：
        - 把所有 STS 记忆做一次“打包摘要”，写一条新的 MTS 记忆
        - 真正的语义聚类/筛选，可以以后由 Self-Programming / 研究层进化。
    """

    def __init__(
        self,
        *,
        config: SleepLoopConfig,
        trace_recorder: TraceRecorder,
        memory_fabric: SimpleMemoryFabric,
        repo_root: str | Path = ".",
    ) -> None:
        self.config = config
        self.trace_recorder = trace_recorder
        self.memory_fabric = memory_fabric
        self.repo_root = Path(repo_root).resolve()

    def run_once(self) -> Dict[str, Any]:
        # 1) 读取所有记忆
        all_items = self.memory_fabric.list_all()
        sts_items = [m for m in all_items if m.tier == "STS"]

        # 采样一部分
        sts_items = sts_items[: self.config.max_traces_per_run]

        if not sts_items:
            result = {
                "sts_processed": 0,
                "created_mts_items": 0,
            }
            self._record_sleep_trace(result)
            return result

        # 2) 做一个非常简单的“合并摘要”
        combined_text = "\n\n".join(m.content for m in sts_items)
        summary_content = f"[SleepLoop summary] {len(sts_items)} STS memories merged."

        # TODO：可以使用 LLM 来真正做摘要，
        # 当前版本先只记录 meta。

        mts_item = MemoryItem(
            id=f"sleep-mts-{len(all_items)}",
            tier="MTS",
            content=summary_content,
            meta={
                "source": "sleep_loop",
                "num_sts_merged": len(sts_items),
            },
        )
        self.memory_fabric.append(mts_item)

        result = {
            "sts_processed": len(sts_items),
            "created_mts_items": 1,
        }
        self._record_sleep_trace(result)
        return result

    def _record_sleep_trace(self, payload: Dict[str, Any]) -> None:
        self.trace_recorder.record_event(
            new_trace_event(
                trace_id="sleep-loop",
                event_type=TraceEventType.SLEEP,
                payload=payload,
            )
        )

class SkillEvolutionLoop:
    """
    Skill & Code Evolution Loop 骨架：

      - 从成功 Trace 中挖出模式（pattern mining）
      - 由 LLM 生成候选技能代码 + 测试
      - 使用 ExperimentOrchestrator 在 Sandbox 中验证候选技能
      - 决定：接受 / 拒绝 / 并存
    """

    def __init__(
        self,
        *,
        trace_recorder: TraceRecorder,
        experiment_orchestrator: ExperimentOrchestrator,
        llm: Any,  # 可以是 OllamaLLMClient 或 MultiModelCouncil
    ) -> None:
        self.trace_recorder = trace_recorder
        self.experiment_orchestrator = experiment_orchestrator
        self.llm = llm

    def run_once(self) -> SkillEvolutionStats:
        """
        最小可运行版本：
          - 目前不真正挖 Trace，只是做一个占位流程：
            1. 构造一个虚拟 pattern
            2. 让 LLM 生成一个技能候选代码 stub
            3. 用 ExperimentOrchestrator 注册一个“技能评估实验”（stub）
        """
        patterns: List[SkillPattern] = []
        candidates: List[SkillCandidate] = []

        # 1)（占位）构造一个虚拟 Pattern，后续由你/系统改成真正的 pattern-mining
        pattern = SkillPattern(
            pattern_id="pattern-demo-001",
            description="A generic pattern for analyzing blueprint gaps and suggesting improvements.",
            source_trace_ids=[],
        )
        patterns.append(pattern)

        # 2) 让 LLM 根据 pattern 生成一个候选技能代码 stub
        prompt = (
            "你是一个资深 Python 工程师，请为下面这个“技能模式”生成一个技能函数骨架：\n\n"
            f"{pattern.description}\n\n"
            "要求：\n"
            "  - 使用 Python 函数\n"
            "  - 不要依赖具体业务域\n"
            "  - 函数签名示例：def skill_fn(input_data: dict) -> dict:\n"
            "  - 只返回代码本身，不要解释\n"
        )
        try:
            code_stub = self.llm(prompt)
        except Exception as e:
            code_stub = f"# LLM error: {repr(e)}"

        tests_stub = "# TODO: add tests for this skill."

        candidate = SkillCandidate(
            candidate_id="candidate-demo-001",
            pattern=pattern,
            code_stub=code_stub,
            tests_stub=tests_stub,
        )
        candidates.append(candidate)

        # 3) 使用 ExperimentOrchestrator 注册一个“技能评估实验”（占位）
        exp_id = self.experiment_orchestrator.register_skill_experiment(
            skill_candidate=candidate.to_dict()
        )

        # 记录 Trace
        self.trace_recorder.record_event(
            new_trace_event(
                trace_id="skill-evolution",
                event_type=TraceEventType.SKILL_EVOLUTION,
                payload={
                    "patterns": [p.to_dict() for p in patterns],
                    "candidates": [c.to_dict() for c in candidates],
                    "registered_experiment_id": exp_id,
                },
            )
        )

        # 目前先不真正执行 experiment，也不决定 accept/reject
        stats = SkillEvolutionStats(
            patterns_mined=len(patterns),
            candidates_generated=len(candidates),
            experiments_created=1,
            candidates_accepted=0,
            candidates_rejected=0,
        )
        return stats
