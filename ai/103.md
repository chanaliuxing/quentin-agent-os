# quentin/memory/fabric.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging
import uuid

logger = logging.getLogger(__name__)

MemoryTier = str  # "STS" | "MTS" | "LTS"
MemoryKind = str  # "episodic" | "semantic" | "procedural"


@dataclass
class MemoryItem:
    """
    Memory Fabric 的核心单元。

    - tier: STS/MTS/LTS
    - kind: episodic / semantic / procedural
    - content: 具体内容（可以是字符串或结构化 dict）
    - importance: 0.0~1.0 粗略重要度
    - usage_count: 被命中/引用次数
    - created_at / last_used_at: ISO 时间戳
    - update_policy: 用于升级/衰减规则的配置（蓝皮书里的 promotion_rule/decay_rule）
    """
    memory_id: str
    tier: MemoryTier
    kind: MemoryKind
    content: Any
    importance: float
    usage_count: int
    created_at: str
    last_used_at: Optional[str]
    tags: List[str]
    update_policy: Dict[str, Any]
    meta: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class MemoryFabricStats:
    tier_counts: Dict[str, int]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class MemoryFabric:
    """
    最小可运行版 Memory Fabric：

    - 用 JSONL 文件实现 STS/MTS/LTS 存储（data/memory/STS.jsonl 等）
    - 暂不实现向量检索，只提供简单 query/filter
    - 提供 consolidate_from_traces() 给 SleepLoop 用，按蓝皮书生成 episodic 记忆

    以后你可以：
      - 把底层换成 SQLite / 向量库
      - 用 embeddings + 相似度做 query
      - 升级 promotion/decay 规则
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        storage_dir: Optional[str | Path] = None,
        default_update_policy: Optional[Dict[str, Any]] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.storage_dir = (
            Path(storage_dir).resolve()
            if storage_dir is not None
            else self.repo_root / "data" / "memory"
        )
        self.storage_dir.mkdir(parents=True, exist_ok=True)

        self.tier_files: Dict[str, Path] = {
            "STS": self.storage_dir / "STS.jsonl",
            "MTS": self.storage_dir / "MTS.jsonl",
            "LTS": self.storage_dir / "LTS.jsonl",
        }

        self.default_update_policy = default_update_policy or {
            "promotion_rule": "auto_basic_v1",
            "decay_rule": "time_based_v1",
            "consolidation_batch": "nightly",
        }

    # --------------------------------------------------
    # 基础 API
    # --------------------------------------------------
    def store(self, item: MemoryItem) -> MemoryItem:
        """
        追加写入 JSONL 文件。
        """
        tier = item.tier
        if tier not in self.tier_files:
            raise ValueError(f"Unknown memory tier: {tier}")

        path = self.tier_files[tier]
        path.parent.mkdir(parents=True, exist_ok=True)

        line = json.dumps(item.to_dict(), ensure_ascii=False)
        with path.open("a", encoding="utf-8") as f:
            f.write(line + "\n")

        return item

    def query(
        self,
        *,
        tier: Optional[MemoryTier] = None,
        kind: Optional[MemoryKind] = None,
        tag: Optional[str] = None,
        limit: int = 50,
    ) -> List[MemoryItem]:
        """
        最简单的过滤查询（没有向量相似度）：

        - 可以按 tier/kind/tag 过滤
        - 从对应 JSONL 文件中顺序读取，取前 limit 条
        """
        tiers = [tier] if tier else list(self.tier_files.keys())
        results: List[MemoryItem] = []

        for t in tiers:
            path = self.tier_files.get(t)
            if not path or not path.exists():
                continue

            with path.open("r", encoding="utf-8") as f:
                for line in f:
                    try:
                        data = json.loads(line)
                    except Exception:
                        continue
                    if kind and data.get("kind") != kind:
                        continue
                    if tag and tag not in (data.get("tags") or []):
                        continue

                    results.append(_memory_from_dict(data))
                    if len(results) >= limit:
                        return results

        return results

    def get_stats(self) -> MemoryFabricStats:
        """
        简单统计各 tier 的条数（按 JSONL 行数估算）。
        """
        counts: Dict[str, int] = {}
        for tier, path in self.tier_files.items():
            if not path.exists():
                counts[tier] = 0
            else:
                try:
                    with path.open("r", encoding="utf-8") as f:
                        counts[tier] = sum(1 for _ in f)
                except Exception:
                    counts[tier] = -1  # 标记读取失败

        return MemoryFabricStats(tier_counts=counts)

    # --------------------------------------------------
    # SleepLoop 调用入口：从 Trace 巩固记忆
    # --------------------------------------------------
    def consolidate_from_traces(
        self,
        traces: List[Dict[str, Any]],
        context: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        SleepLoop 调用的“记忆巩固”入口。

        v1 策略（简单但结构齐全）：
          - 对每个 trace 生成一个 STS 级 episodic 记忆条目：
              content = {trace_id, task_id, goal, domain, summary, lessons}
          - importance 粗略按 trace.status:
              success -> 0.7, failed -> 0.5, error -> 0.3
          - 更新 stats 并返回

        以后可以：
          - 在这里添加 promotion_rule：符合条件的 STS -> MTS/LTS
          - 把多次出现的模式压缩为 semantic/procedural 记忆
        """
        ctx = context or {}
        now = _utc_iso()
        new_items: List[MemoryItem] = []

        for t in traces:
            status = t.get("status") or "unknown"
            importance = _estimate_importance(status)

            trace_id = t.get("trace_id") or t.get("id") or str(uuid.uuid4())
            task_id = t.get("task_id", "unknown_task")
            goal = t.get("goal", "")
            domain = t.get("domain", "general")

            summary = t.get("execution_summary") or {}
            lessons = t.get("lessons_learned") or None

            content = {
                "trace_id": trace_id,
                "task_id": task_id,
                "goal": goal,
                "domain": domain,
                "status": status,
                "summary": summary,
                "lessons_learned": lessons,
            }

            tags = [
                "trace",
                f"status:{status}",
                f"domain:{domain}",
            ]

            item = MemoryItem(
                memory_id=f"mem::{trace_id}",
                tier="STS",
                kind="episodic",
                content=content,
                importance=importance,
                usage_count=0,
                created_at=now,
                last_used_at=None,
                tags=tags,
                update_policy=self.default_update_policy.copy(),
                meta={
                    "source": "sleep_loop",
                    "context": ctx,
                },
            )
            self.store(item)
            new_items.append(item)

        stats = self.get_stats()

        return {
            "status": "ok",
            "num_new_items": len(new_items),
            "tier_counts": stats.to_dict()["tier_counts"],
        }


def _utc_iso() -> str:
    return datetime.utcnow().isoformat()


def _estimate_importance(status: str) -> float:
    if status == "success":
        return 0.7
    if status == "failed":
        return 0.5
    if status == "error":
        return 0.3
    return 0.4


def _memory_from_dict(data: Dict[str, Any]) -> MemoryItem:
    return MemoryItem(
        memory_id=data.get("memory_id", f"mem::{uuid.uuid4().hex}"),
        tier=data.get("tier", "STS"),
        kind=data.get("kind", "episodic"),
        content=data.get("content"),
        importance=float(data.get("importance", 0.5)),
        usage_count=int(data.get("usage_count", 0)),
        created_at=data.get("created_at", _utc_iso()),
        last_used_at=data.get("last_used_at"),
        tags=list(data.get("tags") or []),
        update_policy=dict(data.get("update_policy") or {}),
        meta=dict(data.get("meta") or {}),
    )
这版 MemoryFabric 是“文件版、最小可跑”的实现：不依赖外部 DB，不做向量检索，但接口已经和蓝皮书对齐，以后可以在不改调用方的前提下换成真正的向量库/嵌入版本。

# quentin/runtime/system.py
from __future__ import annotations

from dataclasses import dataclass, asdict, is_dataclass
from pathlib import Path
from typing import Any, Dict, List, Optional
import logging
import time
import uuid

from quentin.memory.fabric import MemoryFabric
from quentin.self_programming.engine import SelfProgrammingEngine
from quentin.evolution.skill_loop import SkillEvolutionLoop
from quentin.loops.sleep import SleepLoop
from quentin.loops.immune import ImmuneLoop, ImmuneCase

logger = logging.getLogger(__name__)


@dataclass
class ExecutionTrace:
    """
    标准化的执行轨迹（Cognitive Loop 输出）。

    - steps: 每一步的详情（规划/调用工具/代码执行/反思等）
    - execution_summary: 任务结束后的结构化总结
    - lessons_learned: 可选的“经验教训”，给 SleepLoop/MemoryFabric 用
    - top_level_metrics: 成本/耗时等
    """
    trace_id: str
    task_id: str
    goal: str
    domain: str
    status: str
    steps: List[Dict[str, Any]]
    execution_summary: Dict[str, Any]
    lessons_learned: Optional[Any]
    top_level_metrics: Dict[str, Any]
    raw_orchestrator_output: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class TaskResult:
    """
    MetaAgentRuntime.run_task 的返回值。
    """
    task_id: str
    output: Any
    trace: ExecutionTrace

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class MetaAgentRuntime:
    """
    v3 的最小运行时中枢（L1/L2/L3 的粘合层）。

    功能：
      - 对外暴露 run_task / run_sleep_cycle / run_immune_cycle / (run_research_cycle)
      - 维护统一的 ExecutionTrace 列表
      - 接入 MemoryFabric / SkillEvolutionLoop / SelfProgrammingEngine / ImmuneLoop
    """

    def __init__(
        self,
        *,
        repo_root: str | Path,
        orchestrator: Any,
        council: Optional[Any] = None,
        memory_fabric: Optional[MemoryFabric] = None,
        self_programming_engine: Optional[SelfProgrammingEngine] = None,
        skill_evolution_loop: Optional[SkillEvolutionLoop] = None,
        sleep_loop: Optional[SleepLoop] = None,
        immune_loop: Optional[ImmuneLoop] = None,
        blueprint_improver: Optional[Any] = None,
        max_traces_kept: int = 1000,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.orchestrator = orchestrator
        self.council = council

        # L0: Memory & Self-Programming
        self.memory_fabric = memory_fabric or MemoryFabric(repo_root=self.repo_root)
        self.self_programming_engine = (
            self_programming_engine or SelfProgrammingEngine(repo_root=self.repo_root)
        )
        self.skill_evolution_loop = (
            skill_evolution_loop
            or SkillEvolutionLoop(self.self_programming_engine, repo_root=self.repo_root)
        )

        # Loops
        self.sleep_loop = sleep_loop or SleepLoop(
            memory_fabric=self.memory_fabric,
            skill_evolution_loop=self.skill_evolution_loop,
        )
        self.immune_loop = immune_loop or ImmuneLoop()

        # Research Layer（蓝皮书改进器等）
        self.blueprint_improver = blueprint_improver

        # Trace 存储（简单列表版）
        self._traces: List[ExecutionTrace] = []
        self._max_traces_kept = max_traces_kept

        # 最近一次免疫/睡眠结果
        self._last_immune_cases: List[ImmuneCase] = []
        self._last_immune_issues: List[Dict[str, Any]] = []
        self._last_sleep_result: Optional[Dict[str, Any]] = None

    # --------------------------------------------------
    # 认知循环：执行一个任务
    # --------------------------------------------------
    def run_task(
        self,
        goal: str,
        context: Optional[Dict[str, Any]] = None,
    ) -> TaskResult:
        """
        对外入口：执行一次“任务”，由 orchestrator 负责具体规划/调用工具。

        - goal: 用户的自然语言目标
        - context: 附加上下文（domain、user_profile 等）

        返回 TaskResult（包含 ExecutionTrace）。
        """
        ctx = context or {}
        task_id = f"task::{uuid.uuid4().hex}"
        domain = ctx.get("domain", "general")

        start_t = time.perf_counter()
        try:
            orch_result = self._call_orchestrator(goal, ctx, task_id=task_id)
            duration = time.perf_counter() - start_t
            trace = self._build_trace_from_orchestrator(
                task_id=task_id,
                goal=goal,
                domain=domain,
                orch_result=orch_result,
                duration_sec=duration,
            )
            output = _extract_final_output(orch_result)
        except Exception as e:
            logger.exception("MetaAgentRuntime.run_task: orchestrator error")
            duration = time.perf_counter() - start_t
            trace = self._build_error_trace(
                task_id=task_id,
                goal=goal,
                domain=domain,
                error=e,
                duration_sec=duration,
            )
            output = None

        self._append_trace(trace)
        return TaskResult(task_id=task_id, output=output, trace=trace)

    # --------------------------------------------------
    # Sleep Loop：记忆巩固 + 技能/代码进化
    # --------------------------------------------------
    def run_sleep_cycle(self) -> Dict[str, Any]:
        """
        一次 Sleep / Consolidation 循环。

        把最近的 trace 丢给 SleepLoop：
          - 生成 MemoryFabric 记忆
          - 调用 SkillEvolutionLoop（携带 ImmuneLoop issues）
        """
        recent_traces = list(self._traces)  # 当前简单实现：全部
        result = self.sleep_loop.run_once(
            recent_traces=recent_traces,
            issues_from_immune=self._last_immune_issues,
            extra_context={"source": "runtime.sleep_cycle"},
        )
        self._last_sleep_result = result
        return result

    # --------------------------------------------------
    # Immune Loop：异常检测与 ImmuneCase
    # --------------------------------------------------
    def run_immune_cycle(self) -> Dict[str, Any]:
        """
        一次 Immune 循环：

        - 在最近 trace 中查找异常
        - 生成 ImmuneCase 列表
        - 转成 issue 列表供 SkillEvolution/Research 使用
        """
        recent_traces = list(self._traces)
        raw = self.immune_loop.run_once(
            recent_traces=recent_traces,
            extra_context={"source": "runtime.immune_cycle"},
        )

        cases_dicts = raw.get("cases") or []
        cases: List[ImmuneCase] = []
        for cd in cases_dicts:
            # ImmuneCase dataclass 结构：case_id, trace_id, reason, severity, trace_snapshot, meta
            try:
                case = ImmuneCase(
                    case_id=cd.get("case_id"),
                    trace_id=cd.get("trace_id"),
                    reason=cd.get("reason"),
                    severity=cd.get("severity"),
                    trace_snapshot=cd.get("trace_snapshot") or {},
                    meta=cd.get("meta") or {},
                )
                cases.append(case)
            except Exception:
                logger.exception("MetaAgentRuntime.run_immune_cycle: invalid ImmuneCase dict: %r", cd)

        issues = self.immune_loop.cases_to_issues(cases)

        self._last_immune_cases = cases
        self._last_immune_issues = issues

        return {
            "count": raw.get("count", len(cases)),
            "cases": [c.to_dict() for c in cases],
            "issues": issues,
        }

    # --------------------------------------------------
    # Research / Blueprint 循环（可选）
    # --------------------------------------------------
    def run_research_cycle(self, *, max_gaps: int = 20) -> Dict[str, Any]:
        """
        研究与蓝皮书自改循环的统一入口。

        如果在构造时注入了 blueprint_improver，就会调用 improver.run_once()；
        否则只返回一个占位结果（不影响主流程）。
        """
        if self.blueprint_improver is None:
            return {
                "status": "no_blueprint_improver",
                "message": "No blueprint improver attached to runtime.",
            }

        try:
            result = self.blueprint_improver.run_once(
                max_gaps=max_gaps,
                extra_context={"source": "runtime.research_cycle"},
            )
            return {"status": "ok", "blueprint_result": result}
        except Exception:
            logger.exception("MetaAgentRuntime.run_research_cycle: blueprint_improver failed")
            return {"status": "error"}

    # --------------------------------------------------
    # Trace 相关辅助方法
    # --------------------------------------------------
    def get_recent_traces(self, limit: int = 200) -> List[ExecutionTrace]:
        """
        返回最近 limit 条 ExecutionTrace。
        """
        if limit >= len(self._traces):
            return list(self._traces)
        return self._traces[-limit:]

    def _append_trace(self, trace: ExecutionTrace) -> None:
        self._traces.append(trace)
        if len(self._traces) > self._max_traces_kept:
            # 简单丢掉最老的
            overflow = len(self._traces) - self._max_traces_kept
            if overflow > 0:
                self._traces = self._traces[overflow:]

    # --------------------------------------------------
    # orchestrator 调用 & trace 构造
    # --------------------------------------------------
    def _call_orchestrator(
        self,
        goal: str,
        context: Dict[str, Any],
        *,
        task_id: str,
    ) -> Any:
        """
        尝试多种调用方式，以兼容不同实现的 orchestrator：

        优先顺序：
          1. orchestrator.run_task(goal, context=..., task_id=..., runtime=self)
          2. orchestrator.plan_and_execute(goal=..., context=..., task_id=..., runtime=self)
          3. orchestrator(goal, context)  // 如果是可调用对象
        """
        ctx = {**context, "task_id": task_id}

        # 1) run_task
        if hasattr(self.orchestrator, "run_task"):
            return self.orchestrator.run_task(goal, context=ctx, runtime=self)

        # 2) plan_and_execute
        if hasattr(self.orchestrator, "plan_and_execute"):
            return self.orchestrator.plan_and_execute(goal=goal, context=ctx, runtime=self)

        # 3) 可调用对象
        if callable(self.orchestrator):
            return self.orchestrator(goal, ctx)

        raise RuntimeError("Orchestrator has no run_task/plan_and_execute/callable interface")

    def _build_trace_from_orchestrator(
        self,
        *,
        task_id: str,
        goal: str,
        domain: str,
        orch_result: Any,
        duration_sec: float,
    ) -> ExecutionTrace:
        """
        将 orchestrator 的原始输出转成标准 ExecutionTrace。
        """
        if is_dataclass(orch_result):
            data = asdict(orch_result)
        elif isinstance(orch_result, dict):
            data = orch_result
        else:
            # 兜底
            data = {"raw": orch_result}

        trace_id = data.get("trace_id") or f"trace::{uuid.uuid4().hex}"
        status = data.get("status") or "success"
        steps = data.get("steps") or []
        exec_summary = data.get("execution_summary") or {"final_output": _extract_final_output(data)}
        lessons = data.get("lessons_learned") or None

        metrics = data.get("top_level_metrics") or {}
        # 补充耗时信息
        metrics.setdefault("latency_sec", duration_sec)

        return ExecutionTrace(
            trace_id=trace_id,
            task_id=task_id,
            goal=goal,
            domain=domain,
            status=status,
            steps=steps,
            execution_summary=exec_summary,
            lessons_learned=lessons,
            top_level_metrics=metrics,
            raw_orchestrator_output=data,
        )

    def _build_error_trace(
        self,
        *,
        task_id: str,
        goal: str,
        domain: str,
        error: Exception,
        duration_sec: float,
    ) -> ExecutionTrace:
        trace_id = f"trace::{uuid.uuid4().hex}"
        exec_summary = {
            "final_output": None,
            "error": str(error),
        }
        metrics = {
            "latency_sec": duration_sec,
        }
        return ExecutionTrace(
            trace_id=trace_id,
            task_id=task_id,
            goal=goal,
            domain=domain,
            status="error",
            steps=[],
            execution_summary=exec_summary,
            lessons_learned=None,
            top_level_metrics=metrics,
            raw_orchestrator_output={"error": str(error)},
        )


def _extract_final_output(orch_result: Any) -> Any:
    """
    尝试从 orchestrator 输出中提取“最终结果”：
      - 优先 final_output / output / result
      - 兜底返回原始对象
    """
    if is_dataclass(orch_result):
        d = asdict(orch_result)
    elif isinstance(orch_result, dict):
        d = orch_result
    else:
        return orch_result

    for key in ("final_output", "output", "result"):
        if key in d:
            return d[key]
    return d
