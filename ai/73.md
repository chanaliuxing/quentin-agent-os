# quentin/memory/fabric.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging
import uuid

logger = logging.getLogger(__name__)


@dataclass
class MemoryItem:
    """
    v1 通用记忆条目结构：

    - tier   : STS/MTS/LTS（短/中/长期）
    - kind   : "semantic" | "procedural" | "episode" | ...
    - content: 具体内容（自然语言或简短结构）
    """

    memory_id: str
    tier: str              # "STS" | "MTS" | "LTS"
    kind: str              # "semantic" | "procedural" | "episode"
    content: str

    tags: Dict[str, Any] = field(default_factory=dict)
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class FileBackedMemoryFabric:
    """
    v1 Memory Fabric：

    - 所有记忆写入 data/memory.jsonl
    - 暂时不做复杂检索，只提供最基础的 append + 简单 filter
    - SleepLoop 主要用它来写入：
        - semantic lessons（语义总结）
        - procedural hints（程序化提示/技能索引）
    """

    def __init__(self, *, repo_root: str | Path = ".", store_path: str | Path = "data/memory.jsonl") -> None:
        self.repo_root = Path(repo_root).resolve()
        self.store_path = self.repo_root / store_path
        self.store_path.parent.mkdir(parents=True, exist_ok=True)

    # ---------- 写入 ----------

    def add_item(
        self,
        *,
        tier: str,
        kind: str,
        content: str,
        tags: Optional[Dict[str, Any]] = None,
    ) -> MemoryItem:
        item = MemoryItem(
            memory_id=str(uuid.uuid4()),
            tier=tier,
            kind=kind,
            content=content,
            tags=tags or {},
        )
        self._append_raw(item)
        return item

    def add_semantic_lesson(
        self,
        *,
        tier: str,
        content: str,
        domain: Optional[str] = None,
        source: Optional[str] = None,
    ) -> MemoryItem:
        tags = {}
        if domain:
            tags["domain"] = domain
        if source:
            tags["source"] = source
        tags["type"] = "lesson"
        return self.add_item(tier=tier, kind="semantic", content=content, tags=tags)

    def add_procedural_hint(
        self,
        *,
        tier: str,
        content: str,
        skill_id: Optional[str] = None,
        component_id: Optional[str] = None,
    ) -> MemoryItem:
        tags: Dict[str, Any] = {"type": "procedural_hint"}
        if skill_id:
            tags["skill_id"] = skill_id
        if component_id:
            tags["component_id"] = component_id
        return self.add_item(tier=tier, kind="procedural", content=content, tags=tags)

    # ---------- 读取（简单版） ----------

    def load_all(self) -> List[MemoryItem]:
        items: List[MemoryItem] = []
        if not self.store_path.exists():
            return items
        try:
            with self.store_path.open("r", encoding="utf-8") as f:
                for line in f:
                    if not line.strip():
                        continue
                    d = json.loads(line)
                    items.append(
                        MemoryItem(
                            memory_id=d["memory_id"],
                            tier=d["tier"],
                            kind=d["kind"],
                            content=d["content"],
                            tags=d.get("tags", {}) or {},
                            created_at=d.get("created_at", ""),
                            updated_at=d.get("updated_at", ""),
                        )
                    )
        except Exception:
            logger.exception("FileBackedMemoryFabric.load_all: failed to read %s", self.store_path)
        return items

    # ---------- 内部 ----------

    def _append_raw(self, item: MemoryItem) -> None:
        try:
            with self.store_path.open("a", encoding="utf-8") as f:
                f.write(json.dumps(item.to_dict(), ensure_ascii=False) + "\n")
        except Exception:
            logger.exception("FileBackedMemoryFabric._append_raw: failed to append to %s", self.store_path)

from dataclasses import dataclass, asdict, field
from datetime import datetime
from typing import Any, Dict, Optional


@dataclass
class SkillPerformanceStats:
    success_rate: float = 0.0
    usage_count: int = 0
    last_score: float = 0.0
    last_used_at: Optional[str] = None


@dataclass
class Skill:
    """
    v1 Skill 定义：

    - skill_id      : 全局唯一 ID
    - name          : 人类可读名字
    - domain        : 领域标签（"core" | "desktop" | "legal" | "trading" | "cross"）
    - module        : python module 路径
    - symbol        : 类名或函数名
    - description   : 这个技能做什么
    - origin        : "human" | "evolved"
    - safety_level  : "low" | "medium" | "high"
    - supersedes    : 被它取代的旧 skill_id（如有）
    """

    skill_id: str
    name: str
    domain: str
    module: str
    symbol: str
    description: str

    origin: str = "human"
    safety_level: str = "medium"
    supersedes: Optional[str] = None

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    performance_stats: SkillPerformanceStats = field(default_factory=SkillPerformanceStats)
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        d["performance_stats"] = asdict(self.performance_stats)
        return d

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "Skill":
        perf = d.get("performance_stats") or {}
        stats = SkillPerformanceStats(
            success_rate=float(perf.get("success_rate", 0.0)),
            usage_count=int(perf.get("usage_count", 0)),
            last_score=float(perf.get("last_score", 0.0)),
            last_used_at=perf.get("last_used_at"),
        )
        return cls(
            skill_id=d["skill_id"],
            name=d["name"],
            domain=d.get("domain", "core"),
            module=d["module"],
            symbol=d["symbol"],
            description=d.get("description", ""),
            origin=d.get("origin", "human"),
            safety_level=d.get("safety_level", "medium"),
            supersedes=d.get("supersedes"),
            created_at=d.get("created_at", ""),
            updated_at=d.get("updated_at", ""),
            performance_stats=stats,
            metadata=d.get("metadata", {}) or {},
        )

# quentin/skills/registry.py
from __future__ import annotations

from pathlib import Path
from typing import Dict, List, Optional
import json
import logging

from quentin.skills.models import Skill, SkillPerformanceStats

logger = logging.getLogger(__name__)


class FileBackedSkillRegistry:
    """
    v1 SkillRegistry：

    - 所有技能存 data/skills.json
    - 提供 list/get/upsert/mark_supersedes 最基本操作
    """

    def __init__(self, *, repo_root: str | Path = ".", store_path: str | Path = "data/skills.json") -> None:
        self.repo_root = Path(repo_root).resolve()
        self.store_path = self.repo_root / store_path
        self.store_path.parent.mkdir(parents=True, exist_ok=True)
        self._cache: Dict[str, Skill] = {}
        self._loaded = False

    # ---------- 公共 API ----------

    def list_skills(self) -> List[Skill]:
        self._ensure_loaded()
        return list(self._cache.values())

    def get_skill(self, skill_id: str) -> Optional[Skill]:
        self._ensure_loaded()
        return self._cache.get(skill_id)

    def upsert_skill(self, skill: Skill) -> None:
        self._ensure_loaded()
        self._cache[skill.skill_id] = skill
        self._flush()

    def mark_supersedes(self, new_skill_id: str, old_skill_id: str) -> None:
        self._ensure_loaded()
        new_skill = self._cache.get(new_skill_id)
        old_skill = self._cache.get(old_skill_id)
        if not new_skill or not old_skill:
            logger.warning(
                "FileBackedSkillRegistry.mark_supersedes: missing new_skill or old_skill (%s, %s)",
                new_skill_id,
                old_skill_id,
            )
            return
        new_skill.supersedes = old_skill.skill_id
        self._cache[new_skill_id] = new_skill
        self._flush()

    def update_performance(
        self,
        skill_id: str,
        *,
        success: bool,
        score: float,
    ) -> None:
        self._ensure_loaded()
        skill = self._cache.get(skill_id)
        if not skill:
            return
        s = skill.performance_stats
        s.usage_count += 1
        s.last_score = score
        s.last_used_at = skill.updated_at
        # 简单滚动 success_rate
        alpha = 0.2
        success_val = 1.0 if success else 0.0
        s.success_rate = (1 - alpha) * s.success_rate + alpha * success_val
        skill.performance_stats = s
        self._cache[skill_id] = skill
        self._flush()

    # ---------- 内部 ----------

    def _ensure_loaded(self) -> None:
        if self._loaded:
            return
        self._load()
        self._loaded = True

    def _load(self) -> None:
        if not self.store_path.exists():
            self._cache = {}
            return
        try:
            raw = json.loads(self.store_path.read_text(encoding="utf-8") or "{}")
            self._cache = {sid: Skill.from_dict(sdata) for sid, sdata in raw.items()}
        except Exception:
            logger.exception("FileBackedSkillRegistry._load: failed to read %s", self.store_path)
            self._cache = {}

    def _flush(self) -> None:
        try:
            raw = {sid: s.to_dict() for sid, s in self._cache.items()}
            self.store_path.write_text(json.dumps(raw, ensure_ascii=False, indent=2), encoding="utf-8")
        except Exception:
            logger.exception("FileBackedSkillRegistry._flush: failed to write %s", self.store_path)

# quentin/skills/evolution.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging
import uuid

from quentin.core.traces import ExecutionTrace
from quentin.llm.multi_council import MultiAICouncil, CouncilDecision
from quentin.llm.types import LLMConfig
from quentin.skills.models import Skill
from quentin.skills.registry import FileBackedSkillRegistry
from quentin.selfprog.engine import SelfProgrammingEngine
from quentin.selfprog.models import Patch
from quentin.memory.fabric import FileBackedMemoryFabric
from quentin.research.experiments import ResearchManager, Experiment

logger = logging.getLogger(__name__)


@dataclass
class SkillEvolutionResult:
    skill: Optional[Skill]
    patch: Optional[Patch]
    reason: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class SkillEvolutionEngine:
    """
    v1 Skill Evolution：

    - 输入：某个 component_id 相关的高成功率 ExecutionTrace 列表
    - 通过 MultiAICouncil 总结出一个“候选技能描述 + 接口建议”
    - 调 SelfProgrammingEngine 生成 skill 代码文件（建议放 mutable/skills/ 下）
    - 在 SkillRegistry 中注册新 Skill
    - 可选：把这次技能抽象过程写入 MemoryFabric（semantic/procedural）

    注意：
    - v1 不做聚类，只是对传入的一批“已经筛好”的 trace 做抽象
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        skill_registry: Optional[FileBackedSkillRegistry] = None,
        memory_fabric: Optional[FileBackedMemoryFabric] = None,
        selfprog_engine: Optional[SelfProgrammingEngine] = None,
        research_manager: Optional[ResearchManager] = None,
        council: Optional[MultiAICouncil] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.skill_registry = skill_registry or FileBackedSkillRegistry(repo_root=self.repo_root)
        self.memory_fabric = memory_fabric or FileBackedMemoryFabric(repo_root=self.repo_root)
        self.selfprog_engine = selfprog_engine or SelfProgrammingEngine(repo_root=self.repo_root)
        self.research_manager = research_manager or ResearchManager()

        if council is None:
            members = [
                LLMConfig(
                    provider="ollama",
                    model="qwen3-vl:latest",
                    role="skill_designer",
                ),
                LLMConfig(
                    provider="ollama",
                    model="deepseek-r1:1.5b",
                    role="robustness_checker",
                ),
            ]
            council = MultiAICouncil(members=members)
        self.council = council

    # ---------- 主入口 ----------

    def propose_skill_from_traces(
        self,
        *,
        component_id: str,
        traces: List[ExecutionTrace],
        domain: str = "core",
    ) -> SkillEvolutionResult:
        """
        输入：某一 component 对应的一批“高成功率 trace”。

        返回：Skill + Patch（如果成功）或 None + None。
        """
        if not traces:
            return SkillEvolutionResult(skill=None, patch=None, reason="no traces provided")

        # 1) 多 AI 会议：生成技能抽象 spec
        spec = self._deliberate_skill_spec(component_id, traces)
        if not spec or not spec.get("module") or not spec.get("symbol"):
            return SkillEvolutionResult(skill=None, patch=None, reason="council did not produce valid spec")

        module = spec["module"]
        symbol = spec["symbol"]
        skill_name = spec.get("name") or f"{component_id}_skill"
        description = spec.get("description") or "Auto-evolved skill from high-success traces."

        # 建议把 auto skill 放到 mutable/skills 下
        target_path = self._module_to_path(module)
        if not target_path:
            return SkillEvolutionResult(
                skill=None,
                patch=None,
                reason=f"Cannot infer target_path from module {module}",
            )

        # 2) 构造给 SelfProgrammingEngine 的问题描述
        problem = self._build_problem_description(component_id, traces, spec, target_path)

        try:
            patch: Patch = self.selfprog_engine.propose_patch(
                target_path=target_path,
                problem_description=problem,
                change_type="new_module",
                target_symbol=symbol,
                extra_context=(
                    "This code should implement a reusable SKILL function/class. "
                    "Focus on clarity, robustness, and good docstrings. "
                    "Include at least one minimal test or usage example."
                ),
            )
        except Exception:
            logger.exception("SkillEvolutionEngine.propose_skill_from_traces: propose_patch failed.")
            return SkillEvolutionResult(skill=None, patch=None, reason="selfprog propose_patch failed")

        # 3) 注册 Skill
        skill_id = str(uuid.uuid4())
        skill = Skill(
            skill_id=skill_id,
            name=skill_name,
            domain=domain,
            module=module,
            symbol=symbol,
            description=description,
            origin="evolved",
            safety_level="low",  # 默认 low，后续可以调
        )
        self.skill_registry.upsert_skill(skill)

        # 4) 把这次技能抽象过程记进 MemoryFabric（semantic + procedural）
        self._store_memory(component_id, skill, traces, spec)

        # 5) 记录 Research 实验
        self._record_research(skill, patch, component_id)

        return SkillEvolutionResult(skill=skill, patch=patch, reason="ok")

    # ---------- 内部：多 AI 会议生成 skill spec ----------

    def _deliberate_skill_spec(
        self,
        component_id: str,
        traces: List[ExecutionTrace],
    ) -> Dict[str, Any]:
        ctx_lines = [
            f"COMPONENT_ID: {component_id}",
            "",
            "SAMPLE SUCCESS TRACES (up to 5):",
        ]
        sample = traces[-5:] if len(traces) > 5 else traces
        for t in sample:
            ctx_lines.append(
                f"- trace_id={t.trace_id}, metrics={t.metrics}, tags={t.tags}"
            )

        context = "\n".join(ctx_lines)

        question = (
            "You are a SKILL DESIGN COUNCIL for a self-evolving agent.\n"
            "Given these successful executions for the same component_id, you must propose\n"
            "ONE reusable skill abstraction.\n\n"
            "The skill should be a single Python function or class method with a clear name,\n"
            "inputs and outputs, and a short description.\n\n"
            "Return ONLY one JSON object:\n"
            "{\n"
            "  \"name\": \"Human readable skill name\",\n"
            "  \"description\": \"What this skill does and when to use it\",\n"
            "  \"module\": \"mutable.skills.skill_xxx\",   // Python module path\n"
            "  \"symbol\": \"SkillClassOrFunctionName\"\n"
            "}\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="skill_design",
        )
        ans = decision.final_answer or "{}"

        try:
            s = ans.strip()
            first = s.find("{")
            last = s.rfind("}")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)
            return {
                "name": str(data.get("name", "")),
                "description": str(data.get("description", "")),
                "module": str(data.get("module", "")),
                "symbol": str(data.get("symbol", "")),
            }
        except Exception:
            logger.exception("SkillEvolutionEngine._deliberate_skill_spec: parse error: %r", ans)
            return {}

    # ---------- 内部：problem description ----------

    def _build_problem_description(
        self,
        component_id: str,
        traces: List[ExecutionTrace],
        spec: Dict[str, Any],
        target_path: str,
    ) -> str:
        lines = [
            f"Component ID : {component_id}",
            "",
            "You are asked to implement a reusable SKILL inferred from high-success traces.",
            "Skill Spec (from council):",
            json.dumps(spec, ensure_ascii=False, indent=2),
            "",
            "TARGET FILE:",
            target_path,
            "",
            "SAMPLE TRACES (success cases, up to 5):",
        ]
        sample = traces[-5:] if len(traces) > 5 else traces
        for t in sample:
            lines.append(
                f"- trace_id={t.trace_id}, metrics={t.metrics}, tags={t.tags}"
            )
        return "\n".join(lines)

    # ---------- 内部：module -> path ----------

    def _module_to_path(self, module_name: str) -> Optional[str]:
        if not module_name:
            return None
        parts = module_name.split(".")
        path = Path(*parts[:-1]) / f"{parts[-1]}.py"
        return str(path).replace("\\", "/")

    # ---------- 内部：存 Memory ----------

    def _store_memory(
        self,
        component_id: str,
        skill: Skill,
        traces: List[ExecutionTrace],
        spec: Dict[str, Any],
    ) -> None:
        # semantic lesson
        lesson = (
            f"Skill '{skill.name}' was evolved from component '{component_id}'. "
            f"It is implemented at {skill.module}:{skill.symbol}. "
            "It captures a reusable pattern of successful executions."
        )
        self.memory_fabric.add_semantic_lesson(
            tier="MTS",
            content=lesson,
            domain="meta",
            source="skill_evolution",
        )

        # procedural hint
        hint = (
            f"When handling tasks for component '{component_id}', "
            f"consider using skill '{skill.name}' ({skill.module}:{skill.symbol}) "
            "for similar sub-problems."
        )
        self.memory_fabric.add_procedural_hint(
            tier="MTS",
            content=hint,
            skill_id=skill.skill_id,
            component_id=component_id,
        )

    # ---------- 内部：记录 Research ----------

    def _record_research(
        self,
        skill: Skill,
        patch: Patch,
        component_id: str,
    ) -> None:
        rm = self.research_manager
        capability = "skill:evolution"
        requirement_id = "skill_evolution_yield"

        exps = rm.list_experiments()
        exp: Optional[Experiment] = None
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "skill_evolution"
            ):
                exp = e
                break

        if exp is None:
            exp = rm.create_experiment(
                name="SkillEvolution",
                purpose="skill_evolution",
                capability=capability,
                requirement_id=requirement_id,
                tags=["skill", "evolution"],
            )

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics={"skill_generated": 1},
            context={
                "skill_id": skill.skill_id,
                "skill_module": skill.module,
                "skill_symbol": skill.symbol,
                "component_id": component_id,
                "patch_id": patch.patch_id,
                "patch_target_path": patch.target_path,
            },
        )

# quentin/core/sleep_loop.py
from __future__ import annotations

from collections import defaultdict
from datetime import datetime, timedelta
from pathlib import Path
from typing import Dict, List, Optional
import json
import logging

from quentin.core.traces import ExecutionTrace
from quentin.memory.fabric import FileBackedMemoryFabric
from quentin.skills.evolution import SkillEvolutionEngine, SkillEvolutionResult
from quentin.research.experiments import ResearchManager, Experiment

logger = logging.getLogger(__name__)


class SleepLoop:
    """
    Sleep / Consolidation Loop v1：

    职责：
      1. 从 data/traces.jsonl 中加载最近一段时间的执行 Trace
      2. 按 component_id 统计成功/失败与成功率
      3. 对高成功率且样本数足够的 component：
           → 调用 SkillEvolutionEngine.propose_skill_from_traces
      4. 同时写一些 Nightly Semantic Lessons 到 MemoryFabric

    注意：
      - 不做在线调度，只是一个“批处理”，可以通过 qctl sleep-run 调用。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        trace_store_path: str | Path = "data/traces.jsonl",
        memory_fabric: Optional[FileBackedMemoryFabric] = None,
        skill_evolution: Optional[SkillEvolutionEngine] = None,
        research_manager: Optional[ResearchManager] = None,
        lookback_hours: int = 24,
        min_traces_per_component: int = 10,
        min_success_rate: float = 0.8,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.trace_store_path = self.repo_root / trace_store_path
        self.memory_fabric = memory_fabric or FileBackedMemoryFabric(repo_root=self.repo_root)
        self.skill_evolution = skill_evolution or SkillEvolutionEngine(repo_root=self.repo_root)
        self.research_manager = research_manager or ResearchManager()

        self.lookback_hours = lookback_hours
        self.min_traces_per_component = min_traces_per_component
        self.min_success_rate = min_success_rate

    # ---------- 主入口 ----------

    def run_once(self) -> Dict[str, SkillEvolutionResult]:
        """
        运行一次 SleepLoop：

        返回：每个 component_id 对应的 SkillEvolutionResult（如果有）
        """
        traces = self._load_recent_traces()
        by_component: Dict[str, List[ExecutionTrace]] = defaultdict(list)
        for t in traces:
            by_component[t.component_id].append(t)

        results: Dict[str, SkillEvolutionResult] = {}

        for component_id, comp_traces in by_component.items():
            stats = self._compute_stats(comp_traces)
            if (
                stats["count"] >= self.min_traces_per_component
                and stats["success_rate"] >= self.min_success_rate
            ):
                # 触发技能演化
                res = self.skill_evolution.propose_skill_from_traces(
                    component_id=component_id,
                    traces=comp_traces,
                    domain="core",
                )
                results[component_id] = res

                # 写一条 semantic lesson 总结这次 consolidation
                self._store_consolidation_lesson(component_id, stats, res)
            else:
                # 也可以写一些简单统计，但 v1 可以略过
                continue

        self._record_research(results)
        return results

    # ---------- 内部：加载最近 Trace ----------

    def _load_recent_traces(self) -> List[ExecutionTrace]:
        traces: List[ExecutionTrace] = []
        if not self.trace_store_path.exists():
            return traces

        cutoff = datetime.utcnow() - timedelta(hours=self.lookback_hours)

        try:
            with self.trace_store_path.open("r", encoding="utf-8") as f:
                for line in f:
                    if not line.strip():
                        continue
                    d = json.loads(line)
                    start_time_str = d.get("start_time") or ""
                    try:
                        start_time = datetime.fromisoformat(start_time_str)
                    except Exception:
                        # 如果没写时间，就全部算进来
                        start_time = datetime.utcnow()
                    if start_time < cutoff:
                        continue

                    traces.append(
                        ExecutionTrace(
                            trace_id=d["trace_id"],
                            task_id=d.get("task_id", ""),
                            component_id=d["component_id"],
                            blueprint_capability_id=d.get("blueprint_capability_id"),
                            target_module=d.get("target_module"),
                            target_symbol=d.get("target_symbol"),
                            success=bool(d.get("success", True)),
                            error_type=d.get("error_type"),
                            error_message=d.get("error_message"),
                            start_time=start_time_str,
                            end_time=d.get("end_time") or "",
                            metrics=d.get("metrics") or {},
                            tags=d.get("tags") or {},
                        )
                    )
        except Exception:
            logger.exception("SleepLoop._load_recent_traces: failed to read %s", self.trace_store_path)

        return traces

    # ---------- 内部：统计 ----------

    def _compute_stats(self, traces: List[ExecutionTrace]) -> Dict[str, float]:
        n = len(traces)
        succ = sum(1 for t in traces if t.success)
        fail = n - succ
        success_rate = succ / n if n > 0 else 0.0
        return {"count": n, "success": succ, "fail": fail, "success_rate": success_rate}

    # ---------- 内部：写 semantic lesson ----------

    def _store_consolidation_lesson(
        self,
        component_id: str,
        stats: Dict[str, float],
        evo_result: SkillEvolutionResult,
    ) -> None:
        if not evo_result.skill:
            return
        s = evo_result.skill
        lesson = (
            f"During sleep consolidation, component '{component_id}' had "
            f"{stats['count']} traces with success_rate={stats['success_rate']:.2f}. "
            f"A new skill '{s.name}' ({s.module}:{s.symbol}) was evolved to capture "
            "this behavior."
        )
        self.memory_fabric.add_semantic_lesson(
            tier="MTS",
            content=lesson,
            domain="meta",
            source="sleep_loop",
        )

    # ---------- 内部：Research 记录 ----------

    def _record_research(self, results: Dict[str, SkillEvolutionResult]) -> None:
        rm = self.research_manager
        capability = "sleep:consolidation"
        requirement_id = "sleep_loop_effectiveness"

        exps = rm.list_experiments()
        exp: Optional[Experiment] = None
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "sleep_loop"
            ):
                exp = e
                break

        if exp is None:
            exp = rm.create_experiment(
                name="SleepLoop",
                purpose="sleep_loop",
                capability=capability,
                requirement_id=requirement_id,
                tags=["sleep", "consolidation"],
            )

        n_components = len(results)
        n_skills = sum(1 for r in results.values() if r.skill is not None)

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics={
                "components_processed": n_components,
                "skills_evolved": n_skills,
            },
            context={},
        )
