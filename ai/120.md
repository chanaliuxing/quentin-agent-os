# scripts/run_blueprint_cycle.py
"""
运行一次完整的“蓝皮书自我改进循环”。

功能：
  1. 初始化 MetaAgentRuntime（带 council、多 loop、自编程引擎等）
  2. 调用 ResearchLayer.run_blueprint_improvement_cycle(...)
  3. 把这次 cycle 的结果写入 data/blueprint/runs/ 下的 JSON 报告

用法（在 repo 根目录）：

  # 只跑一次，默认最多处理 5 个 gap
  python -m scripts.run_blueprint_cycle

  # 指定最多处理的 gap 数量
  python -m scripts.run_blueprint_cycle --max-gaps 10

  # 指定 repo 根目录（不写就是当前目录）
  python -m scripts.run_blueprint_cycle --repo-root /path/to/repo
"""

from __future__ import annotations

import argparse
import json
from datetime import datetime
from pathlib import Path
from typing import Any, Dict

from quentin.runtime.system import MetaAgentRuntime
from scripts.blueprint_self_iterate import create_runtime_for_blueprint_self_iteration


def _now_ts() -> str:
    # 用 UTC 时间生成文件名友好的时间戳
    return datetime.utcnow().strftime("%Y%m%dT%H%M%SZ")


def _print_summary(result: Dict[str, Any]) -> None:
    status = result.get("status", "unknown")
    gaps = result.get("gaps", [])
    modules = result.get("modules_to_improve", [])
    sp = result.get("self_programming_summary", {}) or {}

    print("=== Blueprint Improvement Cycle ===")
    print("status             :", status)
    print("num_gaps           :", len(gaps))
    print("num_modules_to_fix :", len(modules))
    print("self_programming   :")
    print("  status           :", sp.get("status", "n/a"))
    print("  num_targets      :", sp.get("num_targets", 0))
    print("  num_proposals    :", sp.get("num_proposals", 0))
    print("  experiment_ids   :", sp.get("experiment_ids", []))


def main() -> None:
    parser = argparse.ArgumentParser(description="Run one blueprint improvement cycle.")
    parser.add_argument(
        "--repo-root",
        type=str,
        default=".",
        help="Repository root (default: current directory).",
    )
    parser.add_argument(
        "--max-gaps",
        type=int,
        default=5,
        help="Max number of blueprint gaps to handle in this cycle.",
    )
    args = parser.parse_args()

    repo_root = Path(args.repo_root).resolve()

    # 1) 初始化 runtime（带 council + loops + research_layer + self_programming_engine）
    runtime: MetaAgentRuntime = create_runtime_for_blueprint_self_iteration(repo_root=repo_root)

    if runtime.research_layer is None:
        raise RuntimeError("runtime.research_layer is None. 请检查 create_runtime_for_blueprint_self_iteration 实现。")

    research_layer = runtime.research_layer

    # 2) 构造 research_request（这里先用占位，可以后接入真实 metrics）
    research_request = {
        "metrics": {
            # TODO: 未来可以接入 ImmuneLoop/EvalHarness 汇总的指标
            "note": "initial blueprint improvement cycle",
        }
    }

    # 3) 运行蓝皮书自迭代循环
    result = research_layer.run_blueprint_improvement_cycle(
        runtime=runtime,
        max_gaps=args.max_gaps,
        research_request=research_request,
    )

    # 4) 写入 JSON 报告到 data/blueprint/runs/
    runs_dir = repo_root / "data" / "blueprint" / "runs"
    runs_dir.mkdir(parents=True, exist_ok=True)

    ts = _now_ts()
    out_path = runs_dir / f"blueprint_cycle_{ts}.json"
    tmp = out_path.with_suffix(".tmp")

    with tmp.open("w", encoding="utf-8") as f:
        json.dump(result, f, ensure_ascii=False, indent=2)
    tmp.replace(out_path)

    # 5) 控制台输出简要 summary
    _print_summary(result)
    print("report_path        :", out_path)


if __name__ == "__main__":
    main()
2. 配套的小说明（你可以直接用）
准备蓝皮书 spec：

在 blueprint/meta_agent_v3_spec.json 里先放一版简化 spec，例如：

{
  "version": "3.0",
  "requirements": [
    {
      "id": "loop.cognitive.v1",
      "category": "loop",
      "description": "认知循环必须记录 Trace / Summary / Lessons，并将有价值信息写入 STS。",
      "priority": "high",
      "module_hint": "quentin/runtime/cognitive_loop.py",
      "symbol_hint": "CognitiveLoop.run_task"
    }
  ]
}
之后可以让系统自己迭代扩充这个 spec。

第一次跑：

cd /path/to/your/meta-agent-repo

python -m scripts.run_blueprint_cycle
如果 data/blueprint/capabilities.json 还不存在，那就相当于“蓝皮书所有 requirement 都是 gap”，会被扫描出来。

输出和文件：

控制台会看到本轮：

num_gaps

num_modules_to_fix

num_proposals（自编程生成的 patch 数量）

experiment_ids（注册的 patch 实验 ID 列表）

报告会写到：

data/blueprint/runs/blueprint_cycle_YYYYMMDDTHHMMSSZ.json

里边包含：

这轮处理的 gaps 全部详情

modules_to_improve（每个 gap 对应的 PatchTarget 建议）

self_programming_summary（多少 target → 多少 proposals + 实验 id）
