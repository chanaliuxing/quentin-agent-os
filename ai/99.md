一、接入本地 Ollama 模型：OllamaLLM
新文件：quentin/llm/ollama_client.py

这个类实现了前面 SimpleOrchestrator 里定义的 LanguageModel 接口。
默认用 /api/chat，你可以改成 /api/generate。

# quentin/llm/ollama_client.py
from __future__ import annotations

from typing import Any, Dict, List, Optional
import logging
import requests

from quentin.orchestrator.core import LanguageModel

logger = logging.getLogger(__name__)


class OllamaLLM(LanguageModel):
    """
    使用本地 Ollama 模型的 LanguageModel 实现。

    默认走 /api/chat 接口：
      POST {base_url}/api/chat
      {
        "model": "...",
        "messages": [...],
        "stream": false,
        ...
      }

    你可以在 default_params 里配置 temperature / top_p 等。
    """

    def __init__(
        self,
        *,
        model: str = "qwen3-vl:latest",
        base_url: str = "http://localhost:11434",
        timeout: float = 120.0,
        default_params: Optional[Dict[str, Any]] = None,
    ) -> None:
        self.model = model
        self.base_url = base_url.rstrip("/")
        self.timeout = timeout
        self.default_params = default_params or {}

    def generate(self, messages: List[Dict[str, str]], **kwargs: Any) -> str:
        """
        messages: [{"role": "system"/"user"/"assistant", "content": "..."}]
        """
        payload: Dict[str, Any] = {
            "model": self.model,
            "messages": messages,
            "stream": False,
        }
        # 合并默认参数和调用时的参数
        payload.update(self.default_params)
        payload.update(kwargs)

        url = f"{self.base_url}/api/chat"
        try:
            resp = requests.post(url, json=payload, timeout=self.timeout)
            resp.raise_for_status()
        except Exception as e:
            logger.exception("OllamaLLM: request failed to %s", url)
            raise RuntimeError(f"OllamaLLM request failed: {e}") from e

        try:
            data = resp.json()
        except Exception as e:
            logger.exception("OllamaLLM: invalid JSON response")
            raise RuntimeError(f"OllamaLLM invalid JSON: {e}") from e

        # /api/chat 返回 {"message": {"role": "...", "content": "..."}, ...}
        # 某些版本也可能有 "response" 字段，做一个兼容
        content = None
        if isinstance(data, dict):
            msg = data.get("message")
            if isinstance(msg, dict):
                content = msg.get("content")
            if content is None:
                content = data.get("response")

        if not isinstance(content, str):
            logger.error("OllamaLLM: no 'message.content' or 'response' in response: %r", data)
            raise RuntimeError("OllamaLLM: missing content in response")

        return content

二、多 AI 开会（Council）骨架
这是一个简单的 Multi-LLM Council，可以挂你所有本地模型：deepseek-r1 / qwen2.5 / qwen2.5vl / my-deepseek-uncensored。

新文件：quentin/orchestrator/council.py

# quentin/orchestrator/council.py
from __future__ import annotations

from typing import Any, Dict, List, Tuple, Optional
import logging

from quentin.orchestrator.core import LanguageModel

logger = logging.getLogger(__name__)


class MultiModelCouncil:
    """
    多模型“开会”骨架。

    思路：
    1. 每个模型先独立给出一个 answer；
    2. 用一个“仲裁模型”（可以是其中之一）做聚合：
       - 把 question + context + 所有候选答案一起丢给 aggregator；
       - 让它给出最终 answer + 对每个候选的简短点评。

    返回：
      final_answer: str
      debug_info: { "candidates": [...], "aggregator_notes": "..." }
    """

    def __init__(
        self,
        *,
        members: List[Tuple[str, LanguageModel]],
        aggregator: Optional[LanguageModel] = None,
    ) -> None:
        if not members:
            raise ValueError("MultiModelCouncil needs at least one member")

        self.members = members
        # 默认用第一个模型作为 aggregator
        self.aggregator = aggregator or members[0][1]

    def deliberate(
        self,
        *,
        topic: str,
        question: str,
        context: Dict[str, Any],
    ) -> Tuple[str, Dict[str, Any]]:
        """
        topic: "task_planning" / "task_reflection" / "coding_fix" / ...
        question: 本轮讨论的核心问题（比如：如何规划任务？如何修复这个 bug？）
        context: 附加信息（当前代码片段、错误日志、历史尝试等）
        """
        # 1) 每个模型独立作答
        candidates: List[Dict[str, Any]] = []
        for name, llm in self.members:
            try:
                answer = llm.generate(
                    [
                        {
                            "role": "system",
                            "content": (
                                "You are one expert member of a multi-agent council. "
                                f"Topic: {topic}. Respond concisely with your best answer. "
                                "Do not mention that you are part of a council."
                            ),
                        },
                        {
                            "role": "user",
                            "content": f"Question: {question}\nContext: {context}",
                        },
                    ]
                )
            except Exception:
                logger.exception("MultiModelCouncil: member %s failed to answer", name)
                answer = f"[{name} failed to answer due to an error.]"

            candidates.append(
                {
                    "model_name": name,
                    "answer": answer,
                }
            )

        # 2) 仲裁模型聚合
        try:
            agg_input = (
                f"Topic: {topic}\n\n"
                f"Question: {question}\n\n"
                f"Context: {context}\n\n"
                "Here are candidate answers from different experts:\n\n"
            )
            for idx, cand in enumerate(candidates, start=1):
                agg_input += (
                    f"--- Candidate {idx} (model={cand['model_name']}) ---\n"
                    f"{cand['answer']}\n\n"
                )

            agg_input += (
                "Please analyze the above candidates and produce:\n"
                "1) A final best answer.\n"
                "2) A brief critique of each candidate (what is good / missing / risky).\n\n"
                "Return your response in two parts separated by a line '--- CRITIQUE ---'."
            )

            agg_text = self.aggregator.generate(
                [
                    {
                        "role": "system",
                        "content": (
                            "You are an aggregator and judge of multiple expert agents. "
                            "You must synthesize the best answer and critique each candidate briefly."
                        ),
                    },
                    {
                        "role": "user",
                        "content": agg_input,
                    },
                ]
            )
        except Exception:
            logger.exception("MultiModelCouncil: aggregator failed, fallback to first candidate")
            # fallback：直接用第一个候选
            return candidates[0]["answer"], {
                "candidates": candidates,
                "aggregator_notes": "Aggregator failed; fallback to first candidate.",
            }

        # 3) 解析 aggregator 输出
        final_answer, critique = self._split_agg_output(agg_text)

        debug_info = {
            "candidates": candidates,
            "aggregator_notes": critique,
        }
        return final_answer, debug_info

    def _split_agg_output(self, text: str) -> Tuple[str, str]:
        """
        简单按分隔线 '--- CRITIQUE ---' 拆成：
        - final_answer
        - critique
        """
        marker = "--- CRITIQUE ---"
        if marker in text:
            parts = text.split(marker, 1)
            final = parts[0].strip()
            critique = parts[1].strip()
            return final, critique
        return text.strip(), ""

三、BlueprintGap → 自编程改进循环：BlueprintImprover
接下来，把蓝皮书 gap 自动接入 自编程引擎（SkillEvolutionLoop + SelfProgrammingEngine）。

新文件：quentin/blueprint/improver.py

# quentin/blueprint/improver.py
from __future__ import annotations

from typing import Any, Dict, List, Optional
from pathlib import Path
import logging

from quentin.blueprint.gap_tracker import BlueprintGapTracker, BlueprintGapReport, BlueprintGap
from quentin.evolution.skill_loop import SkillEvolutionLoop
from quentin.self_programming.engine import SelfProgrammingEngine

logger = logging.getLogger(__name__)


class BlueprintImprover:
    """
    根据 BlueprintGapTracker 的报告：
    - 把 gap 转成 issues 列表
    - 让 SkillEvolutionLoop / SelfProgrammingEngine 尝试生成 patch
    - 跑测试 + 选出 promotable patches

    v1：流程打通，patch 生成本身还是 stub（需要 LLM + 具体 repo context）。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        tracker: Optional[BlueprintGapTracker] = None,
        engine: Optional[SelfProgrammingEngine] = None,
        skill_evolution: Optional[SkillEvolutionLoop] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.tracker = tracker or BlueprintGapTracker(repo_root=self.repo_root)
        self.engine = engine or SelfProgrammingEngine(repo_root=self.repo_root)
        self.skill_evolution = skill_evolution or SkillEvolutionLoop(self.engine)

    # ---- 对外主入口 ---- #
    def run_once(
        self,
        *,
        max_gaps: int = 20,
        test_suites: Optional[List[str]] = None,
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        一次 Blueprint 改进循环：

        1) scan 蓝皮书与 repo 的 gap
        2) 选前 max_gaps 个 gap
        3) 转成 issues
        4) 生成 patch candidates
        5) 跑测试
        6) 选出可推广 patch

        返回结构化结果，方便写入 Trace / Memory / Dashboard。
        """
        report = self.tracker.scan()
        gaps = report.gaps[:max_gaps]

        logger.info("BlueprintImprover.run_once: gaps=%d (total=%d)", len(gaps), len(report.gaps))

        issues = self._gaps_to_issues(gaps)

        # 2) 生成 PatchCandidate
        patch_candidates = self.skill_evolution.generate_patches_from_issues(issues)

        # 3) 评估 Patch
        suites = test_suites or ["unit", "static_analysis"]
        patch_reports = self.skill_evolution.evaluate_patches(
            patch_candidates,
            test_suites=suites,
            extra_context=extra_context or {"source": "blueprint_improver"},
        )

        # 4) 选出可推广 patch
        promotable = self.engine.select_patches_to_promote(
            patch_reports,
            max_promote=1,
        )

        return {
            "blueprint_report": report.to_dict(),
            "issues": issues,
            "patch_candidates": [p.to_dict() for p in patch_candidates],
            "patch_reports": [r.to_dict() for r in patch_reports],
            "promotable_patches": [r.to_dict() for r in promotable],
        }

    # ---- 内部：把 gap 转 issue ---- #
    def _gaps_to_issues(self, gaps: List[BlueprintGap]) -> List[Dict[str, Any]]:
        """
        将 BlueprintGap 映射为自编程引擎使用的 issue 结构。
        """
        issues: List[Dict[str, Any]] = []
        for gap in gaps:
            severity = "medium"
            if gap.issue_type == "missing_file":
                severity = "high"
            elif gap.issue_type == "stub_implementation":
                severity = "medium"

            target = {
                "target_type": "python_module",  # v1 先假设按模块级别修
                "identifier": gap.path,
                "file_path": gap.path,
                "symbol_name": None,
                "meta": {
                    "component_id": gap.component_id,
                    "issue_type": gap.issue_type,
                },
            }

            issue = {
                "issue_id": f"blueprint_gap::{gap.component_id}",
                "source": "blueprint_gap",
                "target": target,
                "reason": gap.details,
                "metrics": {
                    "severity": severity,
                },
            }
            issues.append(issue)

        return issues
