# quentin/loops/immune_loop.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional
from datetime import datetime, timedelta
from pathlib import Path
import json
import logging
import uuid
import statistics

from quentin.core.traces import ExecutionTraceStore, ExecutionTrace
from quentin.research.experiments import ResearchManager
from quentin.selfprog.engine import SelfProgrammingEngine, PatchTaskSpec

logger = logging.getLogger(__name__)


# ---------- 数据结构 ----------

@dataclass
class ImmuneConfig:
    """
    ImmuneLoop 的基础配置（v1）：

    - window_minutes      : 分析时间窗口长度
    - min_samples         : 每个 component 至少多少条 Trace 才做判断
    - error_rate_threshold: 若失败率超过该阈值，视为异常
    - latency_ms_threshold: 若平均 latency_ms 超过该阈值，视为异常（可选）
    - enable_self_fix     : 是否允许自动触发 SelfProgrammingEngine
    - max_self_fix_per_run: 每次运行最多触发多少个自修补
    - component_to_module : 映射 component_id -> 代码路径（用于自修补）
    """

    window_minutes: int = 60
    min_samples: int = 10
    error_rate_threshold: float = 0.3
    latency_ms_threshold: Optional[float] = None
    enable_self_fix: bool = False
    max_self_fix_per_run: int = 1
    component_to_module: Dict[str, str] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ImmuneCase:
    """
    一次异常检测记录：

    - case_id        : 唯一 ID
    - component_id   : 哪个组件/loop 出现问题
    - blueprint_cap  : 对应蓝皮书 capability_id（若有）
    - time_window    : 分析窗口 [start, end]
    - num_samples    : 统计中参与的 Trace 数量
    - error_rate     : 失败率
    - avg_latency_ms : 平均延迟（若有）
    - severity       : 严重程度（low/medium/high）
    - suggested_action: 建议动作（降级/自修补/仅观察）
    - created_at     : 记录时间
    """

    case_id: str
    component_id: str
    blueprint_cap: Optional[str]
    time_window_start: str
    time_window_end: str
    num_samples: int
    error_rate: float
    avg_latency_ms: Optional[float]
    severity: str
    suggested_action: str
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


# ---------- ImmuneLoop ----------

class ImmuneLoop:
    """
    ImmuneLoop v1：

    - 从 ExecutionTraceStore 中读取最近一段时间的 Trace
    - 按 component_id 分组，统计：
        - 样本数
        - error_rate (1 - success_ratio)
        - avg_latency_ms (如果 metrics 中有 latency_ms)
    - 若某个 component 超过阈值，则生成 ImmuneCase：
        - severity: 根据 error_rate 和 latency 进行简单判断
        - suggested_action: e.g. 'observe', 'degrade', 'self_fix'
    - 将 ImmuneCase:
        - 写入 data/immune_cases.jsonl
        - 记录到 ResearchManager（capability: immune_loop）
        - 记录 ExecutionTrace（blueprint_capability_id="immune_loop"）
    - 如果启用 enable_self_fix:
        - 根据 component_to_module 映射，触发 SelfProgrammingEngine 做一次自修补实验
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        config: Optional[ImmuneConfig] = None,
        traces_store: Optional[ExecutionTraceStore] = None,
        research_manager: Optional[ResearchManager] = None,
        selfprog_engine: Optional[SelfProgrammingEngine] = None,
        immune_cases_path: str | Path = "data/immune_cases.jsonl",
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.config = config or ImmuneConfig()
        self.traces_store = traces_store or ExecutionTraceStore()
        self.research_manager = research_manager or ResearchManager()
        self.selfprog_engine = selfprog_engine or SelfProgrammingEngine(repo_root=self.repo_root)
        self.immune_cases_path = self.repo_root / immune_cases_path
        self.immune_cases_path.parent.mkdir(parents=True, exist_ok=True)

    # ---- 对外入口：执行一次 Immune 检查 ----

    def run_once(self) -> List[ImmuneCase]:
        now = datetime.utcnow()
        window_start = now - timedelta(minutes=self.config.window_minutes)

        # 1) 收集窗口内 Trace
        traces = list(self._iter_traces_since(window_start))

        # 2) 按 component 聚合
        stats = self._aggregate_by_component(traces)

        # 3) 检测异常并生成 ImmuneCase
        cases: List[ImmuneCase] = []
        for comp_id, comp_stats in stats.items():
            if comp_stats["num"] < self.config.min_samples:
                continue
            case = self._detect_anomaly_for_component(
                component_id=comp_id,
                stats=comp_stats,
                window_start=window_start,
                window_end=now,
            )
            if case is not None:
                cases.append(case)
                self._save_immune_case(case)
                self._record_research(case)
                self._record_trace(case)

        # 4) 触发自修补（有限个）
        if self.config.enable_self_fix and self.config.max_self_fix_per_run > 0:
            self._run_self_fix_for_cases(cases)

        return cases

    # ---- 内部：Trace 读取 & 聚合 ----

    def _iter_traces_since(self, since: datetime):
        """
        简单实现：读取最近 N 条 Trace，然后按时间过滤。
        ExecutionTraceStore 具体实现可以用 SQLite / JSONL / 其它。
        """
        # v1 先用 limit=5000 做一个粗略的时间过滤
        for tr in self.traces_store.iter_traces(limit=5000):
            try:
                t = datetime.fromisoformat(tr.end_time)
            except Exception:
                continue
            if t >= since:
                yield tr

    def _aggregate_by_component(self, traces: List[ExecutionTrace]) -> Dict[str, Dict[str, Any]]:
        stats: Dict[str, Dict[str, Any]] = {}
        for tr in traces:
            comp_id = tr.component_id or "unknown"
            s = stats.setdefault(
                comp_id,
                {
                    "num": 0,
                    "num_success": 0,
                    "latencies": [],
                    "blueprint_caps": set(),
                },
            )
            s["num"] += 1
            if tr.success:
                s["num_success"] += 1
            # latency_ms 可能存在于 metrics 里
            if tr.metrics and "latency_ms" in tr.metrics:
                try:
                    s["latencies"].append(float(tr.metrics["latency_ms"]))
                except Exception:
                    pass
            if tr.blueprint_capability_id:
                s["blueprint_caps"].add(tr.blueprint_capability_id)
        return stats

    # ---- 内部：检测异常 & 生成 ImmuneCase ----

    def _detect_anomaly_for_component(
        self,
        *,
        component_id: str,
        stats: Dict[str, Any],
        window_start: datetime,
        window_end: datetime,
    ) -> Optional[ImmuneCase]:
        num = stats["num"]
        num_success = stats["num_success"]
        error_rate = 1.0 - (num_success / max(1, num))
        latencies = stats["latencies"]
        avg_latency = statistics.mean(latencies) if latencies else None
        caps = stats["blueprint_caps"]
        blueprint_cap = next(iter(caps)) if caps else None

        if error_rate < self.config.error_rate_threshold and (
            self.config.latency_ms_threshold is None
            or avg_latency is None
            or avg_latency < self.config.latency_ms_threshold
        ):
            # 没有触发异常
            return None

        severity = self._classify_severity(error_rate=error_rate, avg_latency=avg_latency)
        suggested_action = self._suggest_action(severity=severity, component_id=component_id)

        case = ImmuneCase(
            case_id=str(uuid.uuid4()),
            component_id=component_id,
            blueprint_cap=blueprint_cap,
            time_window_start=window_start.isoformat(),
            time_window_end=window_end.isoformat(),
            num_samples=num,
            error_rate=error_rate,
            avg_latency_ms=avg_latency,
            severity=severity,
            suggested_action=suggested_action,
        )
        return case

    def _classify_severity(
        self,
        *,
        error_rate: float,
        avg_latency: Optional[float],
    ) -> str:
        """
        非常粗糙的严重度划分，可以后续用 ML / 学习策略替换。
        """
        if error_rate >= 0.8:
            return "high"
        if error_rate >= 0.5:
            return "medium"
        if self.config.latency_ms_threshold is not None and avg_latency is not None:
            if avg_latency >= self.config.latency_ms_threshold * 1.5:
                return "medium"
        return "low"

    def _suggest_action(self, *, severity: str, component_id: str) -> str:
        """
        v1 策略：
        - low    : "observe"
        - medium : 若有模块映射，则 "self_fix", 否则 "degrade"
        - high   : 若有模块映射，则 "self_fix", 否则 "degrade"
        """
        has_module = component_id in self.config.component_to_module
        if severity == "low":
            return "observe"
        if severity in ("medium", "high"):
            if has_module:
                return "self_fix"
            return "degrade"
        return "observe"

    # ---- 内部：ImmuneCase 持久化 ----

    def _save_immune_case(self, case: ImmuneCase) -> None:
        try:
            with self.immune_cases_path.open("a", encoding="utf-8") as f:
                f.write(json.dumps(case.to_dict(), ensure_ascii=False) + "\n")
        except Exception:
            logger.exception("ImmuneLoop._save_immune_case: failed to append case")

    # ---- 内部：Research & Trace ----

    def _record_research(self, case: ImmuneCase) -> None:
        rm = self.research_manager
        capability = "immune_loop"
        requirement_id = "anomaly_detection_quality"

        try:
            exps = rm.list_experiments()
        except Exception:
            logger.exception("ImmuneLoop._record_research: failed to list experiments")
            return

        exp = None
        for e in exps:
            if e.capability == capability and e.requirement_id == requirement_id:
                exp = e
                break
        if exp is None:
            exp = rm.create_experiment(
                name="ImmuneLoop",
                purpose="immune_anomaly_detection",
                capability=capability,
                requirement_id=requirement_id,
                tags=["immune", "anomaly"],
            )

        metrics = {
            "error_rate": case.error_rate,
            "severity_high_flag": 1.0 if case.severity == "high" else 0.0,
        }

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="immune_v1",
            metrics=metrics,
            context={
                "immune_case": case.to_dict(),
            },
        )

    def _record_trace(self, case: ImmuneCase) -> None:
        trace = ExecutionTrace(
            trace_id=str(uuid.uuid4()),
            task_id=case.case_id,
            component_id="immune_loop.v1",
            blueprint_capability_id="immune_loop",  # 对应 blueprint/spec_v3.json
            target_module="quentin/loops/immune_loop.py",
            target_symbol="ImmuneLoop.run_once",
            success=True,
            error_type=None,
            error_message=None,
            start_time=case.time_window_start,
            end_time=case.time_window_end,
            metrics={
                "error_rate": case.error_rate,
            },
            tags={
                "component_id": case.component_id,
                "severity": case.severity,
                "suggested_action": case.suggested_action,
            },
        )
        self.traces_store.append(trace)

    # ---- 内部：自修补触发 ----

    def _run_self_fix_for_cases(self, cases: List[ImmuneCase]) -> None:
        """
        按严重程度排序，从高到低，最多触发 max_self_fix_per_run 个自修补。
        """
        if not cases:
            return

        candidates = [c for c in cases if c.suggested_action == "self_fix"]
        if not candidates:
            return

        # 简单按 error_rate 排序
        candidates.sort(key=lambda c: c.error_rate, reverse=True)
        limit = min(self.config.max_self_fix_per_run, len(candidates))

        for i in range(limit):
            case = candidates[i]
            module_path = self.config.component_to_module.get(case.component_id)
            if not module_path:
                continue

            reason = (
                f"ImmuneLoop detected high error_rate={case.error_rate:.2f} "
                f"for component {case.component_id} in the last "
                f"{self.config.window_minutes} minutes. "
                f"Attempting a self-fix patch."
            )

            task = PatchTaskSpec(
                target_path=module_path,
                reason=reason,
                domain="core",
                environment="sandbox",
                risk_level="high",
                context_snippet=json.dumps(
                    {
                        "immune_case": case.to_dict(),
                    },
                    ensure_ascii=False,
                    indent=2,
                ),
            )

            try:
                result = self.selfprog_engine.propose_and_apply_patch(task)
                logger.info(
                    "ImmuneLoop self-fix result for %s: success=%s applied=%s",
                    case.component_id,
                    result.success,
                    result.applied,
                )
            except Exception:
                logger.exception("ImmuneLoop._run_self_fix_for_cases: self fix failed for %s", case.component_id)
二、睡眠循环 v1：quentin/loops/sleep_loop.py
新文件：quentin/loops/sleep_loop.py

这个版本做三件事情：

从短期记忆 / Trace 里抽样，调用 LLM 生成 “知识总结” → 写入 MemoryFabric 的 MTS/LTS；

用 LLM 从最近 Trace 中抽取“候选技能”（纯文本），先存在 data/skills.jsonl（未来交给 SkillManager 处理）；

调用 BlueprintGapTracker.scan()，找出蓝皮书完成度最低的一些 item，为其中一部分触发自编程 Patch（SelfProgrammingEngine）。

# quentin/loops/sleep_loop.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional
from datetime import datetime, timedelta
from pathlib import Path
import json
import logging
import uuid

from quentin.core.traces import ExecutionTraceStore, ExecutionTrace
from quentin.memory.fabric import MemoryFabric
from quentin.research.experiments import ResearchManager
from quentin.llm.router import ModelRouter
from quentin.llm.client import call_llm
from quentin.blueprint.tracker import BlueprintGapTracker
from quentin.selfprog.engine import SelfProgrammingEngine, PatchTaskSpec

logger = logging.getLogger(__name__)


# ---------- 数据结构 ----------

@dataclass
class SleepConfig:
    """
    SleepLoop 配置 v1：

    - trace_window_hours       : 抽样的 Trace 时间窗口
    - max_traces_for_summary   : 知识总结时最多使用的 Trace 数
    - max_traces_for_skills    : 技能抽取时最多使用的 Trace 数
    - enable_blueprint_scan    : 是否启用蓝皮书 gap 扫描
    - blueprint_min_score      : 低于该分数的 item 视为“缺口”
    - blueprint_patch_limit    : 每次 SleepLoop 触发的自编程 patch 数量上限
    """

    trace_window_hours: int = 24
    max_traces_for_summary: int = 100
    max_traces_for_skills: int = 50
    enable_blueprint_scan: bool = True
    blueprint_min_score: float = 0.6
    blueprint_patch_limit: int = 1

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class SkillCandidate:
    """
    SleepLoop 抽取出的技能候选（暂存为纯文本）：

    - skill_id      : 唯一 ID
    - title         : 技能标题
    - description   : 技能描述（自然语言）
    - domain        : 适用领域
    - blueprint_cap : 若能对齐到某条蓝皮书条目，则记录
    - source_traces : 用于抽取本技能的 trace_id 列表
    """

    skill_id: str
    title: str
    description: str
    domain: str
    blueprint_cap: Optional[str]
    source_traces: List[str]
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


# ---------- SleepLoop ----------

class SleepLoop:
    """
    SleepLoop v1：

    - 知识巩固：
        - 抽样最近一段时间的 Trace
        - 用 LLM 生成高层次知识总结
        - 写入 MemoryFabric 的 MTS / LTS

    - 技能抽取：
        - 用 LLM 从 Trace 中抽取“可重复的技能步骤”
        - 暂存到 data/skills.jsonl（以后交给 SkillManager 进一步结构化）

    - 蓝皮书 gap 修补：
        - BlueprintGapTracker.scan() 找分数最低的条目
        - 为其中部分 item 创建 PatchTaskSpec，调用 SelfProgrammingEngine 做自编程实验
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        config: Optional[SleepConfig] = None,
        traces_store: Optional[ExecutionTraceStore] = None,
        memory_fabric: Optional[MemoryFabric] = None,
        research_manager: Optional[ResearchManager] = None,
        model_router: Optional[ModelRouter] = None,
        blueprint_tracker: Optional[BlueprintGapTracker] = None,
        selfprog_engine: Optional[SelfProgrammingEngine] = None,
        skills_path: str | Path = "data/skills.jsonl",
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.config = config or SleepConfig()
        self.traces_store = traces_store or ExecutionTraceStore()
        self.memory_fabric = memory_fabric
        self.research_manager = research_manager or ResearchManager()
        self.model_router = model_router or ModelRouter()
        self.blueprint_tracker = blueprint_tracker or BlueprintGapTracker(repo_root=self.repo_root)
        self.selfprog_engine = selfprog_engine or SelfProgrammingEngine(repo_root=self.repo_root)
        self.skills_path = self.repo_root / skills_path
        self.skills_path.parent.mkdir(parents=True, exist_ok=True)

    # ---- 对外入口 ----

    def run_once(self) -> Dict[str, Any]:
        """
        返回一个简要报告 dict，包含本次 SleepLoop 做了什么。
        """
        start_ts = datetime.utcnow().isoformat()

        traces_for_summary = list(self._sample_traces_for_summary())
        traces_for_skills = list(self._sample_traces_for_skills())

        # 1) 知识总结 + 写入 MTS/LTS
        knowledge_summary = self._build_knowledge_summary(traces_for_summary)
        self._store_knowledge(knowledge_summary)

        # 2) 技能抽取 + 存入技能候选库
        skill_candidates = self._extract_skills(traces_for_skills)
        for sk in skill_candidates:
            self._save_skill_candidate(sk)

        # 3) 蓝皮书 gap 扫描 + 自编程 patch
        patch_results = []
        if self.config.enable_blueprint_scan:
            patch_results = self._run_blueprint_gap_patches()

        # 4) 写 Research & Trace
        self._record_research(
            knowledge_summary=knowledge_summary,
            skill_candidates=skill_candidates,
            patch_results=patch_results,
        )
        self._record_trace(
            start_ts=start_ts,
            knowledge_summary=knowledge_summary,
            skill_candidates=skill_candidates,
            patch_results=patch_results,
        )

        return {
            "knowledge_summary_len": len(knowledge_summary or ""),
            "num_skill_candidates": len(skill_candidates),
            "num_patches_attempted": len(patch_results),
        }

    # ---- 内部：Trace 抽样 ----

    def _sample_traces_for_summary(self):
        since = datetime.utcnow() - timedelta(hours=self.config.trace_window_hours)
        traces = []
        for tr in self.traces_store.iter_traces(limit=5000):
            try:
                t = datetime.fromisoformat(tr.end_time)
            except Exception:
                continue
            if t >= since:
                traces.append(tr)
        # 简单截断
        return traces[: self.config.max_traces_for_summary]

    def _sample_traces_for_skills(self):
        since = datetime.utcnow() - timedelta(hours=self.config.trace_window_hours)
        traces = []
        for tr in self.traces_store.iter_traces(limit=5000):
            try:
                t = datetime.fromisoformat(tr.end_time)
            except Exception:
                continue
            if t >= since:
                traces.append(tr)
        return traces[: self.config.max_traces_for_skills]

    # ---- 内部：知识总结 ----

    def _build_knowledge_summary(self, traces: List[ExecutionTrace]) -> str:
        if not traces:
            return ""

        cfg = self.model_router.route(mode="sleep", task_type="knowledge_summary")

        # 构造简化的 trace 摘要
        snippets = []
        for tr in traces:
            snippets.append(
                {
                    "component_id": tr.component_id,
                    "capability": tr.blueprint_capability_id,
                    "success": tr.success,
                    "metrics": tr.metrics,
                    "tags": tr.tags,
                }
            )

        user = json.dumps(snippets, ensure_ascii=False, indent=2)
        system = (
            "You are a SLEEP CONSOLIDATION agent.\n"
            "You receive a JSON list of execution trace summaries from the last period.\n"
            "Your job is to produce a concise KNOWLEDGE SUMMARY:\n"
            "- key patterns that seem useful for future decisions\n"
            "- lessons learned from failures and successes\n"
            "- general heuristics that should be promoted to mid/long-term memory\n"
            "Return only the summary as plain text, 300-800 words.\n"
        )

        summary = call_llm(cfg, user_prompt=user, system_prompt=system)
        return summary.strip()

    def _store_knowledge(self, summary: str) -> None:
        if not summary:
            return
        if self.memory_fabric is None:
            return

        record = {
            "type": "semantic_summary",
            "tier": "MTS",
            "content": summary,
            "created_at": datetime.utcnow().isoformat(),
        }

        try:
            if hasattr(self.memory_fabric, "store_mid_term"):
                self.memory_fabric.store_mid_term(record)
            elif hasattr(self.memory_fabric, "add_episode"):
                self.memory_fabric.add_episode("MTS", record)
            else:
                logger.debug("SleepLoop._store_knowledge: MemoryFabric has no MTS method, skipping.")
        except Exception:
            logger.exception("SleepLoop._store_knowledge: failed to store knowledge summary")

    # ---- 内部：技能抽取 ----

    def _extract_skills(self, traces: List[ExecutionTrace]) -> List[SkillCandidate]:
        if not traces:
            return []

        cfg = self.model_router.route(mode="sleep", task_type="skill_extraction")

        # 挑一些成功的 trace（done_flag=1 或 success=True）
        snippets = []
        src_ids = []
        for tr in traces:
            # 假设 done_flag 存在于 metrics，否则就用 success
            done_flag = 1.0 if tr.metrics and tr.metrics.get("done_flag") else (1.0 if tr.success else 0.0)
            if done_flag <= 0.0:
                continue
            snippets.append(
                {
                    "trace_id": tr.trace_id,
                    "component_id": tr.component_id,
                    "capability": tr.blueprint_capability_id,
                    "metrics": tr.metrics,
                }
            )
            src_ids.append(tr.trace_id)

        if not snippets:
            return []

        user = json.dumps(snippets, ensure_ascii=False, indent=2)
        system = (
            "You are a SKILL DISCOVERY agent.\n"
            "You receive a list of successful execution trace summaries.\n"
            "Your job is to propose 1-5 reusable SKILLS.\n"
            "Return a JSON list, each item:\n"
            "{\n"
            "  \"title\": \"short skill name\",\n"
            "  \"description\": \"how this skill works and when to use it\",\n"
            "  \"domain\": \"core/desktop/legal/trading/...\",\n"
            "  \"blueprint_cap\": \"optional capability id or null\"\n"
            "}\n"
        )

        raw = call_llm(cfg, user_prompt=user, system_prompt=system)
        try:
            text = raw.strip()
            first = text.find("[")
            last = text.rfind("]")
            if first != -1 and last != -1 and last > first:
                text = text[first : last + 1]
            data = json.loads(text)
        except Exception:
            logger.exception("SleepLoop._extract_skills: failed to parse skills JSON: %r", raw)
            return []

        skills: List[SkillCandidate] = []
        for item in data:
            try:
                sk = SkillCandidate(
                    skill_id=str(uuid.uuid4()),
                    title=str(item.get("title") or "Untitled Skill"),
                    description=str(item.get("description") or ""),
                    domain=str(item.get("domain") or "core"),
                    blueprint_cap=item.get("blueprint_cap"),
                    source_traces=src_ids,
                )
                skills.append(sk)
            except Exception:
                logger.exception("SleepLoop._extract_skills: invalid skill item: %r", item)
        return skills

    def _run_blueprint_gap_patches(self):
        report = self.blueprint_tracker.scan()
        # 找出 score 最低的一些 item
        items = sorted(report.items, key=lambda s: s.score)
        results = []

        count = 0
        for st in items:
            if st.score >= self.config.blueprint_min_score:
                continue
            if count >= self.config.blueprint_patch_limit:
                break

            modules = st.spec.hints.get("modules") or []
            if not modules:
                continue
            # 选第一个模块文件
            target_rel = modules[0]
            reason = (
                f"SleepLoop blueprint gap fix for capability {st.spec.id}. "
                f"Current score={st.score:.2f} < threshold={self.config.blueprint_min_score:.2f}.\n"
                f"Attempting to improve implementation of {target_rel}."
            )

            task = PatchTaskSpec(
                target_path=target_rel,
                reason=reason,
                domain="core",
                environment="sandbox",
                risk_level="medium",
                context_snippet=json.dumps(
                    {
                        "blueprint_item": st.to_dict(),
                        "blueprint_report": {
                            "version": report.version,
                            "avg_score": report.avg_score,
                        },
                    },
                    ensure_ascii=False,
                    indent=2,
                ),
            )

            try:
                res = self.selfprog_engine.propose_and_apply_patch(task)
                results.append(
                    {
                        "task_id": res.task_id,
                        "target_path": res.target_path,
                        "success": res.success,
                        "applied": res.applied,
                    }
                )
                count += 1
            except Exception:
                logger.exception("SleepLoop._run_blueprint_gap_patches: patch failed for %s", target_rel)
        return results

    def _record_research(
        self,
        *,
        knowledge_summary: str,
        skill_candidates: List[SkillCandidate],
        patch_results: List[Dict[str, Any]],
    ) -> None:
        rm = self.research_manager
        capability = "sleep_loop"
        requirement_id = "consolidation_quality"

        try:
            exps = rm.list_experiments()
        except Exception:
            logger.exception("SleepLoop._record_research: failed to list experiments")
            return

        exp = None
        for e in exps:
            if e.capability == capability and e.requirement_id == requirement_id:
                exp = e
                break
        if exp is None:
            exp = rm.create_experiment(
                name="SleepLoop",
                purpose="sleep_consolidation",
                capability=capability,
                requirement_id=requirement_id,
                tags=["sleep", "memory", "skills"],
            )

        metrics = {
            "knowledge_summary_len": len(knowledge_summary or ""),
            "num_skill_candidates": len(skill_candidates),
            "num_patches_attempted": len(patch_results),
        }

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="sleep_v1",
            metrics=metrics,
            context={
                "skill_candidates": [sk.to_dict() for sk in skill_candidates],
                "patch_results": patch_results,
            },
        )
