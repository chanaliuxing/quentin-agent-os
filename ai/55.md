二、BlueprintSpec & Loader：quentin/blueprint/spec.py
这个模块负责：

定义蓝皮书的 dataclass 结构（Capability / Requirement）。

从 YAML 文件加载成 BlueprintSpec。

如果 YAML 缺失，则给一个空 spec（不阻塞运行）。

# quentin/blueprint/spec.py
from __future__ import annotations

import logging
from dataclasses import dataclass, field, asdict
from pathlib import Path
from typing import Dict, Any, List, Optional

import yaml  # 需要安装 PyYAML

logger = logging.getLogger(__name__)


@dataclass
class RequirementSpec:
    id: str
    title: str
    description: str
    status: str  # "missing" / "placeholder" / "partial" / "ready"
    priority: float = 0.5
    impl: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "RequirementSpec":
        return cls(
            id=data["id"],
            title=data.get("title", data["id"]),
            description=data.get("description", ""),
            status=data.get("status", "missing"),
            priority=float(data.get("priority", 0.5)),
            impl=data.get("impl", {}) or {},
        )


@dataclass
class CapabilitySpec:
    id: str
    name: str
    layer: str
    description: str
    owner: str = "core"
    tags: List[str] = field(default_factory=list)
    requirements: List[RequirementSpec] = field(default_factory=list)

    def to_dict(self) -> Dict[str, Any]:
        return {
            "id": self.id,
            "name": self.name,
            "layer": self.layer,
            "description": self.description,
            "owner": self.owner,
            "tags": list(self.tags),
            "requirements": [r.to_dict() for r in self.requirements],
        }

    @classmethod
    def from_dict(cls, cap_id: str, data: Dict[str, Any]) -> "CapabilitySpec":
        reqs = [
            RequirementSpec.from_dict(r)
            for r in data.get("requirements", [])
        ]
        return cls(
            id=cap_id,
            name=data.get("name", cap_id),
            layer=data.get("layer", "L1"),
            description=data.get("description", ""),
            owner=data.get("owner", "core"),
            tags=list(data.get("tags", [])),
            requirements=reqs,
        )


@dataclass
class BlueprintSpec:
    name: str
    version: str
    description: str = ""
    capabilities: Dict[str, CapabilitySpec] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return {
            "name": self.name,
            "version": self.version,
            "description": self.description,
            "capabilities": {
                cid: cap.to_dict() for cid, cap in self.capabilities.items()
            },
        }

    def all_requirements(self) -> List[RequirementSpec]:
        reqs: List[RequirementSpec] = []
        for cap in self.capabilities.values():
            reqs.extend(cap.requirements)
        return reqs


class BlueprintSpecLoader:
    """
    负责从 YAML 文件加载 BlueprintSpec。
    """

    def __init__(self, path: str | Path = "blueprints/meta_agent_v3.yaml") -> None:
        self.path = Path(path)

    def load(self) -> BlueprintSpec:
        if not self.path.exists():
            logger.warning(
                "Blueprint spec file %s not found. Returning empty spec.",
                self.path,
            )
            return BlueprintSpec(
                name="EMPTY",
                version="0.0.0",
                description="Empty blueprint spec (file not found).",
                capabilities={},
            )

        with self.path.open("r", encoding="utf-8") as f:
            raw = yaml.safe_load(f) or {}

        caps_raw = raw.get("capabilities", {}) or {}
        caps: Dict[str, CapabilitySpec] = {}
        for cap_id, cap_data in caps_raw.items():
            caps[cap_id] = CapabilitySpec.from_dict(cap_id, cap_data)

        spec = BlueprintSpec(
            name=raw.get("name", "Meta-Agent"),
            version=str(raw.get("version", "0.0.0")),
            description=raw.get("description", ""),
            capabilities=caps,
        )
        return spec
三、BlueprintGapTracker.scan()：quentin/blueprint/gap_tracker.py
这个模块做的事：

加载蓝皮书 spec。

对每个 capability.requirement 生成一条 BlueprintGap：

带上 spec 里的 status / priority / impl 信息。

用简单反射检查 impl.module / impl.symbol 是否真的存在。

提供 scan() 返回所有 gaps。

提供 summarize() 做统计（方便 CLI / Dashboard 用）。

1）BlueprintGap 数据结构
# quentin/blueprint/gap_tracker.py
from __future__ import annotations

import importlib
import logging
from dataclasses import dataclass, asdict
from typing import Dict, Any, List, Optional, Tuple

from quentin.blueprint.spec import BlueprintSpecLoader, BlueprintSpec, CapabilitySpec, RequirementSpec

logger = logging.getLogger(__name__)


@dataclass
class BlueprintGap:
    """
    对应蓝皮书中一条 capability + requirement 的“实现状态快照”。

    - capability_id   : e.g. "memory_fabric"
    - requirement_id  : e.g. "mf.storage"
    - title           : requirement 标题
    - description     : requirement 描述
    - spec_status     : 蓝皮书里声明的状态（missing / placeholder / partial / ready）
    - priority        : 蓝皮书里的优先级（0~1）
    - impl            : {module, symbol, ...}
    - impl_module_ok  : 模块是否可以 import
    - impl_symbol_ok  : symbol 是否存在
    - auto_hint       : 根据 impl 检查给出的自动状态建议（不会覆盖 spec_status）
    """

    capability_id: str
    requirement_id: str
    title: str
    description: str

    spec_status: str
    priority: float

    impl: Dict[str, Any]
    impl_module_ok: bool
    impl_symbol_ok: bool

    auto_hint: str  # "no_impl_info" / "module_missing" / "symbol_missing" / "impl_present"

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)
2）BlueprintGapTracker 实现
class BlueprintGapTracker:
    """
    根据蓝皮书 spec + 代码实况，生成 BlueprintGap 列表。

    v1 只做“静态检查”：
    - spec_status 完全尊重 YAML（人/AI 可以手动调整）
    - impl_module_ok / impl_symbol_ok 用于辅助判断
    - auto_hint 给出自动建议，不直接改 spec_status
    """

    def __init__(self, spec_loader: Optional[BlueprintSpecLoader] = None) -> None:
        self.spec_loader = spec_loader or BlueprintSpecLoader()

    # ---------- 主入口 ----------

    def scan(self) -> List[BlueprintGap]:
        spec = self.spec_loader.load()
        gaps: List[BlueprintGap] = []

        for cap_id, cap in spec.capabilities.items():
            for req in cap.requirements:
                impl_module_ok, impl_symbol_ok, auto_hint = self._check_impl(req.impl)
                gap = BlueprintGap(
                    capability_id=cap_id,
                    requirement_id=req.id,
                    title=req.title,
                    description=req.description,
                    spec_status=req.status,
                    priority=req.priority,
                    impl=req.impl,
                    impl_module_ok=impl_module_ok,
                    impl_symbol_ok=impl_symbol_ok,
                    auto_hint=auto_hint,
                )
                gaps.append(gap)

        return gaps

    # ---------- 辅助：统计汇总 ----------

    def summarize(self, gaps: List[BlueprintGap]) -> Dict[str, Any]:
        by_status: Dict[str, int] = {}
        by_capability: Dict[str, int] = {}
        missing_impl: List[Tuple[str, str]] = []

        for g in gaps:
            st = g.spec_status
            by_status[st] = by_status.get(st, 0) + 1
            by_capability[g.capability_id] = by_capability.get(g.capability_id, 0) + 1

            if not g.impl_module_ok or not g.impl_symbol_ok:
                missing_impl.append((g.capability_id, g.requirement_id))

        return {
            "total_requirements": len(gaps),
            "by_status": by_status,
            "by_capability": by_capability,
            "missing_impl": missing_impl,
        }

    # ---------- 内部：检查 impl ----------

    def _check_impl(self, impl: Dict[str, Any]) -> Tuple[bool, bool, str]:
        """
        根据 impl 信息简单检查 module/symbol 是否存在。
        返回: (impl_module_ok, impl_symbol_ok, auto_hint)
        """
        if not impl:
            return False, False, "no_impl_info"

        module_name = impl.get("module")
        symbol_name = impl.get("symbol")

        if not module_name:
            return False, False, "no_impl_info"

        try:
            mod = importlib.import_module(module_name)
        except Exception:
            return False, False, "module_missing"

        if not symbol_name:
            return True, False, "symbol_missing"

        if hasattr(mod, symbol_name):
            return True, True, "impl_present"
        else:
            return True, False, "symbol_missing"

def cmd_blueprint_scan(args: argparse.Namespace) -> None:
    tracker = BlueprintGapTracker()
    gaps = tracker.scan()
    summary = tracker.summarize(gaps)

    print("Blueprint gap summary:")
    print(f"- total requirements: {summary['total_requirements']}")
    print("- by status:")
    for st, cnt in summary["by_status"].items():
        print(f"  * {st}: {cnt}")
    print("- by capability:")
    for cap, cnt in summary["by_capability"].items():
        print(f"  * {cap}: {cnt}")
    if summary["missing_impl"]:
        print("- requirements with missing impl (module or symbol):")
        for cap, req in summary["missing_impl"]:
            print(f"  * {cap} / {req}")


def cmd_blueprint_gaps(args: argparse.Namespace) -> None:
    tracker = BlueprintGapTracker()
    gaps = tracker.scan()

    status_filter = args.status.lower() if args.status else None

    rows = []
    for g in gaps:
        if status_filter and g.spec_status.lower() != status_filter:
            continue
        rows.append(g)

    if not rows:
        print("No matching blueprint requirements found.")
        return

    print("Blueprint requirements:")
    for g in rows:
        impl_info = g.impl or {}
        mod = impl_info.get("module", "-")
        sym = impl_info.get("symbol", "-")
        mod_status = "OK" if g.impl_module_ok else "MISSING"
        sym_status = "OK" if g.impl_symbol_ok else "MISSING"
        print(
            f"- {g.capability_id} / {g.requirement_id} "
            f"[status={g.spec_status}, priority={g.priority}]"
        )
        print(f"  title: {g.title}")
        print(f"  impl : {mod}.{sym}  (module={mod_status}, symbol={sym_status}, auto_hint={g.auto_hint})")
