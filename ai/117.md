1. Patch Schema：quentin/self_programming/schema.py
用 dataclass 把 patch 的核心概念固定下来，之后无论是 LLM 生成，还是你手工写，都用这套协议。

# quentin/self_programming/schema.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from datetime import datetime
from typing import Any, Dict, List, Optional


@dataclass
class PatchTarget:
    """
    Patch 的作用对象：
      - 只允许修改 /mutable/ 下被标记为 auto_updatable 的模块
      - path 一律用 repo_root 的相对路径
    """
    kind: str                     # "module" | "skill" | "config" 等
    path: str                     # 例: "mutable/skills/desktop_clicker.py"
    symbol: Optional[str] = None  # 可选: 具体函数/类名
    constraints: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class PatchTestPlan:
    """
    Patch 对应的测试计划：
      - 用于描述需要跑哪些测试（单测/回归/静态分析等）
      - 真实执行逻辑可以后面再补
    """
    unit_tests: List[str] = field(default_factory=list)       # 例如 pytest -k "test_xxx"
    regression_scenarios: List[str] = field(default_factory=list)
    static_checks: List[str] = field(default_factory=list)    # 例如 "mypy", "ruff"
    fuzzing: List[str] = field(default_factory=list)          # 预留

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class PatchProposal:
    """
    自编程引擎生成的“补丁提案”，不会直接改代码，只作为候选。
    """

    proposal_id: str
    created_at: str
    created_by: str              # "self_programming_engine" | "human" 等
    target: PatchTarget
    rationale: str               # 为什么要这么改（来自 LLM/ResearchLayer）
    # patch_content 可以有多种形式：
    #   - unified_diff: 纯 diff
    #   - full_file: 整个文件替换
    #   - snippet: 局部函数/类定义
    patch_type: str              # "unified_diff" | "full_file" | "snippet"
    patch_content: str
    test_plan: PatchTestPlan
    meta: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        # 手动展开嵌套 dataclass，方便 JSON 序列化
        d["target"] = self.target.to_dict()
        d["test_plan"] = self.test_plan.to_dict()
        return d


@dataclass
class PatchExecutionResult:
    """
    在 Sandbox / CI 中跑 patch 相关测试的结果（占位）
    """

    proposal_id: str
    tests_run: List[str]
    tests_passed: List[str]
    tests_failed: List[str]
    static_issues: List[str]
    # 指标变化，例如成功率/延迟/成本的变化（由 Eval Harness 写入）
    metrics_delta: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class PatchReport:
    """
    综合报告：Patch 从建议 -> 测试 -> 是否推荐接受
    """

    proposal: PatchProposal
    execution_result: Optional[PatchExecutionResult]
    recommended_action: str     # "accept" | "reject" | "needs_review"
    reasoning: str              # 对这个结论的解释

    def to_dict(self) -> Dict[str, Any]:
        d = {
            "proposal": self.proposal.to_dict(),
            "execution_result": self.execution_result.to_dict() if self.execution_result else None,
            "recommended_action": self.recommended_action,
            "reasoning": self.reasoning,
        }
        return d


def now_iso() -> str:
    return datetime.utcnow().isoformat()

# quentin/self_programming/engine.py
from __future__ import annotations

from dataclasses import asdict
from pathlib import Path
from typing import Any, Dict, List

import json

from quentin.self_programming.schema import (
    PatchTarget,
    PatchTestPlan,
    PatchProposal,
    PatchExecutionResult,
    PatchReport,
    now_iso,
)
from quentin.traces.core import TraceRecorder, TraceEventType, new_trace_event
from quentin.research.experiments import ExperimentOrchestrator


class SelfProgrammingEngine:
    """
    自编程引擎（Self-Programming Engine）骨架：

      - 输入：一组“需要改进的模块/技能”（PatchTarget 列表）+ 上下文（指标/Trace 摘要等）
      - 输出：
          * PatchProposal 列表（写入 data/patches/proposals/*.json）
          * 对应的 patch 实验（register_patch_experiment）
      - 目前不直接 apply patch，只负责：
          * 生成 patch 候选
          * 建立测试计划
          * 交给 ExperimentOrchestrator & ResearchLayer 后续处理
    """

    def __init__(
        self,
        *,
        repo_root: str | Path,
        llm: Any,  # OllamaLLMClient 或 MultiModelCouncil
        experiment_orchestrator: ExperimentOrchestrator,
        trace_recorder: TraceRecorder,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.llm = llm
        self.experiment_orchestrator = experiment_orchestrator
        self.trace_recorder = trace_recorder

        self._patches_dir = self.repo_root / "data" / "patches"
        self._patches_dir.mkdir(parents=True, exist_ok=True)

    # --------------------------------------------------
    # 对外：run_cycle
    # --------------------------------------------------
    def run_cycle(
        self,
        modules_to_improve: List[PatchTarget],
        context: Dict[str, Any],
    ) -> Dict[str, Any]:
        """
        完整的一次自编程周期（简化版）：

          1. 为每个 target 生成 PatchProposal
          2. 将每个 proposal 写入 data/patches/proposals
          3. 为每个 proposal 注册一个 patch 实验
          4. 写入 TraceEvent，返回一个 summary
        """
        proposals: List[PatchProposal] = []
        experiments: List[str] = []

        for target in modules_to_improve:
            proposal = self._generate_patch_proposal_for_target(target, context)
            self._persist_patch_proposal(proposal)
            proposals.append(proposal)

            exp_id = self.experiment_orchestrator.register_patch_experiment(
                patch_proposal=proposal.to_dict()
            )
            experiments.append(exp_id)

        payload = {
            "modules_to_improve": [t.to_dict() for t in modules_to_improve],
            "proposals": [p.to_dict() for p in proposals],
            "experiments": experiments,
        }

        self.trace_recorder.record_event(
            new_trace_event(
                trace_id="self-programming",
                event_type=TraceEventType.SELF_PROGRAMMING,
                payload=payload,
            )
        )

        return {
            "num_targets": len(modules_to_improve),
            "num_proposals": len(proposals),
            "experiment_ids": experiments,
        }

    # --------------------------------------------------
    # 内部：为单个 target 生成 PatchProposal
    # --------------------------------------------------
    def _generate_patch_proposal_for_target(
        self,
        target: PatchTarget,
        context: Dict[str, Any],
    ) -> PatchProposal:
        """
        用 LLM 为某个模块/技能生成 patch 提案。

        当前只生成“snippet” 类型的 patch：
          - 提供函数骨架或局部代码片段
          - 由后续的应用器决定如何真正应用到文件
        """

        # 先尝试读取当前文件内容（如果存在）
        file_path = self.repo_root / target.path
        current_code: str = ""
        if file_path.exists():
            try:
                current_code = file_path.read_text(encoding="utf-8")
            except Exception:
                current_code = ""

        # 构造 prompt
        prompt = self._build_patch_prompt(target, current_code, context)

        try:
            patch_snippet = self.llm(prompt)
        except Exception as e:
            patch_snippet = f"# LLM error while generating patch: {repr(e)}"

        # 简单的测试计划占位：后面可以由 LLM 自动生成更详细的计划
        test_plan = PatchTestPlan(
            unit_tests=[],
            regression_scenarios=[],
            static_checks=["mypy", "ruff"],
            fuzzing=[],
        )

        proposal_id = self._make_proposal_id(target)
        proposal = PatchProposal(
            proposal_id=proposal_id,
            created_at=now_iso(),
            created_by="self_programming_engine",
            target=target,
            rationale=context.get("rationale", "auto-generated by self-programming engine"),
            patch_type="snippet",
            patch_content=patch_snippet,
            test_plan=test_plan,
            meta={
                "source_metrics": context.get("metrics", {}),
                "notes": context.get("notes", ""),
            },
        )
        return proposal

    def _build_patch_prompt(
        self,
        target: PatchTarget,
        current_code: str,
        context: Dict[str, Any],
    ) -> str:
        """
        构造给 LLM 的 patch 生成 prompt。

        强调：
          - 只输出代码片段，不加解释
          - 遵守安全边界（不改 kernel / 风控 / schema）
        """
        problem_desc = context.get("problem_description", "")
        metrics_desc = json.dumps(context.get("metrics", {}), ensure_ascii=False, indent=2)

        symbol_hint = f"\n目标函数/类名: {target.symbol}\n" if target.symbol else ""

        prompt = (
            "你是一个严谨的 Python 重构工程师，任务是根据给定的模块和问题描述，"
            "为该模块生成一个改进后的代码片段（snippet），用于后续人工/自动集成到代码中。\n\n"
            "【重要约束】\n"
            "1. 你只能修改 /mutable/ 下允许 auto_updatable 的模块逻辑。\n"
            "2. 你不能修改安全内核（kernel）、风控逻辑和数据 schema。\n"
            "3. 输出必须是合法的 Python 代码，不要加任何解释文字。\n\n"
            f"目标模块(kind={target.kind}): {target.path}\n"
            f"{symbol_hint}"
            f"问题描述（可能来自指标/ImmuneLoop/ResearchLayer）:\n{problem_desc}\n\n"
            f"相关指标（JSON）:\n{metrics_desc}\n\n"
            "当前代码（如果为空字符串表示无法读取或文件尚未创建）:\n"
            "-------------------\n"
            f"{current_code}\n"
            "-------------------\n\n"
            "请基于上述信息，输出一个改进后的代码片段（可以是函数/类/若干辅助函数），"
            "用于替换或补充当前实现。只输出代码，不要解释。"
        )
        return prompt

    # --------------------------------------------------
    # 内部：落盘 PatchProposal
    # --------------------------------------------------
    def _persist_patch_proposal(self, proposal: PatchProposal) -> None:
        proposals_dir = self._patches_dir / "proposals"
        proposals_dir.mkdir(parents=True, exist_ok=True)

        path = proposals_dir / f"{proposal.proposal_id}.json"
        tmp = path.with_suffix(".tmp")

        with tmp.open("w", encoding="utf-8") as f:
            json.dump(proposal.to_dict(), f, ensure_ascii=False, indent=2)
        tmp.replace(path)

    def _make_proposal_id(self, target: PatchTarget) -> str:
        base = f"{target.kind}:{target.path}:{target.symbol or ''}:{now_iso()}"
        # 简单 hash，避免太长
        return f"patch-{abs(hash(base)) & 0xFFFFFFFF:08x}"

import json
from pathlib import Path
from typing import Dict, Any


class ExperimentOrchestrator:
    def __init__(self, repo_root: str | Path, variants_runtimes: Dict[str, Any]) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.variants_runtimes = variants_runtimes
        self._experiments_dir = self.repo_root / "data" / "experiments"
        self._experiments_dir.mkdir(parents=True, exist_ok=True)

    # ... 你已有的其它方法 ...

    def register_skill_experiment(self, skill_candidate: Dict[str, Any]) -> str:
        # 之前已经加过的占位方法，如果没有就一起补上
        exp_id = f"skill-exp-{hash(json.dumps(skill_candidate, sort_keys=True)) & 0xFFFFFFFF}"
        record = {
            "experiment_id": exp_id,
            "type": "skill_candidate",
            "skill_candidate": skill_candidate,
        }
        path = self._experiments_dir / "skill_candidates.jsonl"
        tmp = path.with_suffix(".tmp")
        with tmp.open("a", encoding="utf-8") as f:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")
        tmp.replace(path)
        return exp_id

    # 新增：Patch 实验登记
    def register_patch_experiment(self, patch_proposal: Dict[str, Any]) -> str:
        """
        注册一个“Patch 实验”。

        当前版本：
          - 不立刻执行，只是把 patch 提案写入 JSONL
          - 返回 experiment_id，后续 ResearchLayer / EvalHarness 可以根据这个 id 去跑实验
        """
        exp_id = f"patch-exp-{hash(json.dumps(patch_proposal, sort_keys=True)) & 0xFFFFFFFF}"

        record = {
            "experiment_id": exp_id,
            "type": "patch_proposal",
            "patch_proposal": patch_proposal,
        }

        path = self._experiments_dir / "patch_experiments.jsonl"
        tmp = path.with_suffix(".tmp")

        with tmp.open("a", encoding="utf-8") as f:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")
        tmp.replace(path)

        return exp_id

# quentin/research/layer.py
from __future__ import annotations

from pathlib import Path
from typing import Any, Dict, List, Optional

from quentin.self_programming.engine import SelfProgrammingEngine
from quentin.self_programming.schema import PatchTarget


class ResearchLayer:
    def __init__(
        self,
        *,
        repo_root: str | Path,
        experiment_orchestrator: Any,
        self_programming_engine: Optional[SelfProgrammingEngine] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.experiment_orchestrator = experiment_orchestrator
        self.self_programming_engine = self_programming_engine

    def attach_self_programming_engine(self, engine: SelfProgrammingEngine) -> None:
        self.self_programming_engine = engine

    # ... 你原来已有的其它方法，比如 run_blueprint_improvement_cycle ...

    def run_self_programming_cycle(
        self,
        modules_to_improve: List[Dict[str, Any]],
        context: Dict[str, Any],
    ) -> Dict[str, Any]:
        """
        对外提供的“自编程循环”入口：

          modules_to_improve: List[dict] 结构应符合 PatchTarget 的字段：
            {
              "kind": "...",
              "path": "mutable/xxx.py",
              "symbol": "SomeClass.some_method",
              "constraints": {...}
            }
        """
        if not self.self_programming_engine:
            return {
                "status": "no_self_programming_engine_attached",
                "num_targets": 0,
                "num_proposals": 0,
                "experiment_ids": [],
            }

        targets: List[PatchTarget] = [
            PatchTarget(
                kind=m["kind"],
                path=m["path"],
                symbol=m.get("symbol"),
                constraints=m.get("constraints", {}),
            )
            for m in modules_to_improve
        ]

        return self.self_programming_engine.run_cycle(
            modules_to_improve=targets,
            context=context,
        )
