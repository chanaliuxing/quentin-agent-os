# quentin/council/reflection.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, List, Optional
import logging

from quentin.council.manager import CouncilManager, CouncilQuestion, CouncilDecision

logger = logging.getLogger(__name__)


@dataclass
class ReflectionAdvice:
    """
    多 AI 反省后的建议。

    约定：
    - action:
        - "retry"           : 建议继续重试当前 goal（可以结合 new_plan 使用）
        - "abort"           : 建议停止本 goal
        - "adjust_goal"     : 建议调整目标（例如降低难度/范围）
        - "request_skill"   : 建议引入/实现新的技能
        - "other"           : 其他情况
    - new_plan        : 若提供，替代旧的 plan（结构与 PlannerResult.steps 兼容）
    - goal_adjustment : 对 goal 的修改建议（自然语言/结构化都可以）
    - skill_ideas     : 请求创造的新技能列表（后续可喂给 SkillManager / 自编程引擎）
    - notes           : 关键说明
    - raw             : 完整 CouncilDecision 用于 Trace / Research Layer
    """
    action: str
    new_plan: Optional[Dict[str, Any]]
    goal_adjustment: Optional[Dict[str, Any]]
    skill_ideas: List[Dict[str, Any]]
    notes: str
    raw: CouncilDecision

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        d["raw"] = self.raw.to_dict()
        return d
        

class ReflectionCouncil:
    """
    使用 CouncilManager 的“错误反省”封装。

    触发场景：
    - 同一个 goal 多次失败
    - 某次执行出现严重异常 / 指标异常（高 error_rate / 超预算等）
    """

    def __init__(self, council_manager: Optional[CouncilManager] = None) -> None:
        self.council_manager = council_manager or CouncilManager()

    def reflect(
        self,
        *,
        goal: Any,
        last_plan: Dict[str, Any],
        last_exec_result: Dict[str, Any],
        metrics: Dict[str, Any],
        failure_count: int,
        importance: str = "normal",
        recent_traces: Optional[List[Dict[str, Any]]] = None,
    ) -> ReflectionAdvice:
        """
        发起一次多 AI 反省。

        - goal            : 当前目标（对象或 dict）
        - last_plan       : 上一次使用的 plan（PlannerResult 序列化的 dict）
        - last_exec_result: Executor 返回的结构化结果
        - metrics         : 本 goal 的累计/最近指标（失败次数、error_rate 等）
        - failure_count   : 当前 goal 已经失败了几轮
        - recent_traces   : 可选，最近几次执行的 Trace 摘要
        """
        goal_dict: Dict[str, Any]
        if isinstance(goal, dict):
            goal_dict = goal
        else:
            goal_dict = {}
            for attr in ("id", "title", "description", "domain", "metadata"):
                if hasattr(goal, attr):
                    goal_dict[attr] = getattr(goal, attr)

        payload = {
            "goal": goal_dict,
            "last_plan": last_plan,
            "last_execution": last_exec_result,
            "metrics": metrics,
            "failure_count": failure_count,
            "recent_traces": recent_traces or [],
        }

        question = CouncilQuestion(
            topic="reflection",
            importance=importance,
            prompt=(
                "Analyze why the previous attempt(s) failed and propose a better strategy.\n"
                "You MUST:\n"
                "- diagnose the likely root causes,\n"
                "- decide what to do next (retry / abort / adjust_goal / request_skill / other),\n"
                "- if retry, optionally propose an IMPROVED PLAN compatible with the original plan schema,\n"
                "- if request_skill, describe the missing skills clearly.\n"
            ),
            payload=payload,
        )

        decision = self.council_manager.run_council("reflection", question)

        # 期望仲裁结果 final.answer 大致是：
        # {
        #   "action": "retry" | "abort" | "adjust_goal" | "request_skill" | "other",
        #   "new_plan": {...} | null,
        #   "goal_adjustment": {...} | null,
        #   "skill_ideas": [{...}, ...]
        # }
        answer = decision.final.get("answer") or {}
        if isinstance(answer, str):
            try:
                answer = json.loads(answer)
            except Exception:
                logger.exception("ReflectionCouncil: failed to parse answer string, using empty object.")
                answer = {}

        action = str(answer.get("action") or "retry")
        new_plan = answer.get("new_plan")
        goal_adjustment = answer.get("goal_adjustment")
        skill_ideas = answer.get("skill_ideas") or []
        if not isinstance(skill_ideas, list):
            skill_ideas = []

        notes = decision.final.get("reasoning") or ""

        return ReflectionAdvice(
            action=action,
            new_plan=new_plan if isinstance(new_plan, dict) else None,
            goal_adjustment=goal_adjustment if isinstance(goal_adjustment, dict) else None,
            skill_ideas=skill_ideas,
            notes=notes,
            raw=decision,
        )

    def run_goal(self, goal: "Goal", context: Dict[str, Any]) -> None:
        failure_count = 0
        metrics = {
            "total_attempts": 0,
            "success_attempts": 0,
            "error_attempts": 0,
            # 后面可以慢慢加
        }

        last_plan_dict: Dict[str, Any] = {}
        last_exec_result: Dict[str, Any] = {}
        recent_traces: List[Dict[str, Any]] = []

        while True:
            metrics["total_attempts"] += 1

            plan = self._run_planner(goal, context)
            # 假设 PlannerResult 有 .steps 等字段
            last_plan_dict = {
                "steps": getattr(plan, "steps", []),
                "raw_output": getattr(plan, "raw_output", None),
            }

            exec_result = self._run_executor(goal, plan, context)
            last_exec_result = exec_result

            # 从 exec_result 中提取简单指标
            if isinstance(exec_result, dict):
                success = bool(exec_result.get("success", False))
                step_records = exec_result.get("steps") or []
            else:
                success = False
                step_records = []

            if success:
                metrics["success_attempts"] += 1
            else:
                metrics["error_attempts"] += 1
                failure_count += 1

            # 这里可以把 step_records 转成简化 trace 放进 recent_traces
            recent_traces.append(
                {
                    "attempt_index": metrics["total_attempts"],
                    "steps": step_records,
                }
            )
            # 控制 recent_traces 长度，比如只保留最近 5 次
            if len(recent_traces) > 5:
                recent_traces = recent_traces[-5:]

            # 生成 execution_summary 给 DoneCouncil 和后续记忆用
            execution_summary = {
                "plan_steps": last_plan_dict.get("steps"),
                "executor_result": last_exec_result,
                "attempt_index": metrics["total_attempts"],
                "failure_count": failure_count,
            }

            done_decision = self.done_council.assess(
                goal=goal,
                execution_summary=execution_summary,
                metrics=metrics,
                importance="normal",
            )

            if done_decision.is_done:
                logger.info(
                    "CognitiveLoop: goal %s considered DONE by done_council (%s)",
                    getattr(goal, "id", None),
                    done_decision.verdict,
                )
                # TODO: 写入 Trace / Memory Fabric（已完成的 goal）
                break

            # 如果没 done，就先看要不要触发 reflection
            if self._should_trigger_reflection(goal, metrics, failure_count):
                advice = self.reflection_council.reflect(
                    goal=goal,
                    last_plan=last_plan_dict,
                    last_exec_result=last_exec_result,
                    metrics=metrics,
                    failure_count=failure_count,
                    importance="high",
                    recent_traces=recent_traces,
                )
                logger.info(
                    "CognitiveLoop: reflection advice for goal %s: action=%s",
                    getattr(goal, "id", None),
                    advice.action,
                )

                # 把 reflection 结果写入 Trace / Memory，为 SleepLoop / Research Layer 用
                self._record_reflection(goal, execution_summary, advice)

                # 根据反省建议调整下一步行为
                if advice.action == "abort":
                    logger.warning(
                        "CognitiveLoop: aborting goal %s based on reflection council advice.",
                        getattr(goal, "id", None),
                    )
                    break

                if advice.action == "adjust_goal" and advice.goal_adjustment:
                    # 简单示例：把调整建议塞到 goal.metadata 里
                    md = getattr(goal, "metadata", {}) or {}
                    md["adjustment_suggestion"] = advice.goal_adjustment
                    setattr(goal, "metadata", md)

                if advice.action in ("retry", "adjust_goal", "request_skill"):
                    # 如果给了 new_plan，就在下一轮使用它，而不是重新跑 planner
                    if advice.new_plan and isinstance(advice.new_plan.get("steps"), list):
                        logger.info("CognitiveLoop: using reflection-proposed plan for next attempt.")
                        plan = self._build_planner_result_from_reflection(goal, advice.new_plan)
                        # 这里可以直接执行 new_plan，或者在下一轮 loop 时判断有缓存的 plan
                        # 为了结构清晰，我们先把 new_plan 存起来，下一轮 _run_planner 里先检查缓存
                        self._cache_plan_override(goal, advice.new_plan)

                    # skill_ideas 可以作为 SkillCandidate，交给 SkillManager 或后续 SleepLoop
                    if advice.skill_ideas:
                        self._enqueue_skill_ideas_from_reflection(goal, advice.skill_ideas)

            # 否则不触发 reflection，直接进入下一轮 while 重试
上面 _build_planner_result_from_reflection / _cache_plan_override / _enqueue_skill_ideas_from_reflection 都是下面要补的小辅助函数。

3. Reflection 触发条件判断
    def _should_trigger_reflection(
        self,
        goal: "Goal",
        metrics: Dict[str, Any],
        failure_count: int,
    ) -> bool:
        """
        简单规则：
        - 连续失败次数 >= 2
        - 或总失败次数 >= 3
        后续可以接 Immune Loop / Anomaly Detector 的信号。
        """
        if failure_count >= 3:
            return True
        # 你可以根据 metrics 更精细地判断
        # 例如 error_attempts 占比 > 0.6 等
        return False
4. Reflection 结果的一些辅助处理函数
    def _build_planner_result_from_reflection(self, goal: "Goal", plan_obj: Dict[str, Any]) -> "PlannerResult":
        """
        当 reflection council 提供了一个 new_plan 时，
        将其转换为 PlannerResult，方便复用现有 Executor 逻辑。
        """
        raw_steps = plan_obj.get("steps") or []
        steps_list = []
        for idx, s in enumerate(raw_steps, start=1):
            steps_list.append(
                {
                    "step_id": s.get("step_id") or f"s{idx}",
                    "description": s.get("description") or "",
                    "skill_id": s.get("skill_id"),
                    "notes": s.get("notes"),
                }
            )

        return PlannerResult(
            goal_id=getattr(goal, "id", None),
            raw_output=json.dumps(plan_obj, ensure_ascii=False),
            steps=steps_list,
            used_skills=[],
        )

    def _cache_plan_override(self, goal: "Goal", plan_obj: Dict[str, Any]) -> None:
        """
        简单实现：在内存里放一个 override cache。
        你可以后面改成放到 Memory/DB。
        """
        if not hasattr(self, "_planner_overrides"):
            self._planner_overrides = {}
        self._planner_overrides[getattr(goal, "id", None)] = plan_obj

    def _get_plan_override_if_any(self, goal: "Goal") -> Optional[Dict[str, Any]]:
        if hasattr(self, "_planner_overrides"):
            return self._planner_overrides.get(getattr(goal, "id", None))
        return None

    def _enqueue_skill_ideas_from_reflection(self, goal: "Goal", ideas: List[Dict[str, Any]]) -> None:
        """
        将反省产生的 skill_ideas 进入“技能候选队列”。
        v1 简单处理：直接调用 SkillManager 的某个方法或写入一个临时 JSON 文件，
        后续由 SleepLoop 扫描处理。
        """
        try:
            if hasattr(self.skill_manager, "add_candidates_from_reflection"):
                self.skill_manager.add_candidates_from_reflection(goal, ideas)
                return
        except Exception:
            logger.exception("CognitiveLoop: failed to push reflection skill ideas to SkillManager.")

        # 如果 SkillManager 还没这接口，可以暂时写入一个本地文件，后续由其他组件消费
        try:
            buf = {
                "goal_id": getattr(goal, "id", None),
                "ideas": ideas,
            }
            out_dir = self.repo_root / "data" / "reflection_skill_ideas"
            out_dir.mkdir(parents=True, exist_ok=True)
            import uuid
            out_path = out_dir / f"{uuid.uuid4().hex}.json"
            out_path.write_text(json.dumps(buf, ensure_ascii=False, indent=2), encoding="utf-8")
        except Exception:
            logger.exception("CognitiveLoop: failed to persist reflection skill ideas to file.")

    def _record_reflection(
        self,
        goal: "Goal",
        execution_summary: Dict[str, Any],
        advice: "ReflectionAdvice",
    ) -> None:
        """
        把本次反省结果记入 Trace / Memory。
        v1 可以先写到 TraceStore，后续 SleepLoop 再统一处理。
        """
        try:
            if self.trace_store:
                self.trace_store.append_reflection(
                    goal=goal,
                    execution_summary=execution_summary,
                    advice=advice.to_dict(),
                )
        except Exception:
            logger.exception("CognitiveLoop: failed to record reflection in trace_store.")
trace_store.append_reflection 这个方法你可以在 ExecutionTraceStore 里加一个简单实现，比如追加到一个 JSONL 文件；目前先当成一个合理的扩展点。

四、现在「多 AI 反省 → 自我修正」这条链路的状态
到目前这个步骤，你的通用智能体在“失败处理”这块已经具备了一个 可跑的闭环：

CognitiveLoop 正常做：

planner council 多模型做计划 → Executor 执行 → DoneCouncil 判是否完成。

如果连续失败 / 指标异常：

CognitiveLoop._should_trigger_reflection 返回 True；

调用 ReflectionCouncil.reflect(...)：

把 goal / last_plan / exec_result / metrics / recent_traces 丢给 reflection council；

多个本地模型（deepseek + qwen 等）各自分析原因、给建议；

仲裁 LLM 综合，输出一个带 action/new_plan/skill_ideas 的 JSON。

CognitiveLoop 根据 ReflectionAdvice：

action == abort → 停止当前 goal；

action == retry / adjust_goal 且有 new_plan → 下次优先使用改良 plan；

skill_ideas → 写进技能候选队列，供 SleepLoop / 自编程引擎后续挖掘。

同时把整个反省的 CouncilDecision、执行 summary、advice 都写入 Trace / Memory：

给 SleepLoop 做「失败→成功」对照；

给 Research Layer 做研究数据（哪种反省策略最有效）。
