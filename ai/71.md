            patch_id=d["patch_id"],
            target_path=d.get("target_path", ""),

# quentin/selfprog/gate.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, Optional
import json
import logging

from quentin.llm.multi_council import MultiAICouncil, CouncilDecision
from quentin.llm.types import LLMConfig
from quentin.selfprog.models import Patch, PatchEvalReport

logger = logging.getLogger(__name__)


@dataclass
class PatchGateDecision:
    decision: str          # "approve" | "reject" | "needs_human"
    reason: str
    max_environment: str   # "staging" | "production"

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class PatchGateCouncil:
    """
    多 AI 开会审批补丁：

    - 输入：Patch + （可选）PatchEvalReport
    - 输出：PatchGateDecision（approve/reject/needs_human + reason + 最大允许环境）

    这里不直接改 Patch，只给出决策，由外层 PatchApplier 来更新 Patch 状态。
    """

    def __init__(self, council: Optional[MultiAICouncil] = None) -> None:
        if council is None:
            members = [
                LLMConfig(
                    provider="ollama",
                    model="qwen3-vl:latest",
                    role="governor",
                ),
                LLMConfig(
                    provider="ollama",
                    model="deepseek-r1:1.5b",
                    role="risk_analyst",
                ),
            ]
            council = MultiAICouncil(members=members)
        self.council = council

    def review_patch(
        self,
        patch: Patch,
        environment: str = "staging",
    ) -> PatchGateDecision:
        eval_report: Optional[PatchEvalReport] = patch.eval_report

        ctx_lines = [
            f"PATCH_ID        : {patch.patch_id}",
            f"TARGET_PATH     : {patch.target_path}",
            f"TARGET_SYMBOL   : {patch.target_symbol or '<unknown>'}",
            f"CHANGE_TYPE     : {patch.change_type}",
            f"CURRENT_STATUS  : {patch.status}",
            f"RISK_LEVEL      : {patch.risk_level}",
            f"REQUEST_ENV     : {environment}",
            "",
            "INTENT:",
            patch.intent,
            "",
            "PATCH_TEXT (truncated):",
            (patch.patch_text or "")[:3000],
        ]

        if eval_report is not None:
            ctx_lines.extend(
                [
                    "",
                    "EVAL_REPORT:",
                    f"- ok         : {eval_report.ok}",
                    f"- score      : {eval_report.score}",
                    f"- risk_level : {eval_report.risk_level}",
                    f"- issues     : {eval_report.issues}",
                    f"- warnings   : {eval_report.warnings}",
                ]
            )
        else:
            ctx_lines.append("\nEVAL_REPORT: <none>")

        context = "\n".join(ctx_lines)

        question = (
            "You are a PATCH GOVERNANCE COUNCIL of a self-evolving agent.\n"
            "You MUST decide whether this patch is allowed to be applied automatically.\n\n"
            "Possible decisions:\n"
            "- \"approve\": safe enough for automatic apply in the requested environment (staging/production).\n"
            "- \"reject\": too risky or clearly wrong; must NOT be applied.\n"
            "- \"needs_human\": ambiguous; human or external supervisor must review manually.\n\n"
            "You must be conservative:\n"
            "- If eval_report.ok is false or score < 0.7, you should usually reject or mark needs_human.\n"
            "- Prefer \"approve\" only when the risk seems low and the patch is well-tested.\n\n"
            "Return ONLY ONE JSON object with fields:\n"
            "{\n"
            "  \"decision\": \"approve\" | \"reject\" | \"needs_human\",\n"
            "  \"reason\": \"short explanation\",\n"
            "  \"max_environment\": \"staging\" | \"production\"\n"
            "}\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="patch_governance",
        )
        ans = decision.final_answer or "{}"

        try:
            s = ans.strip()
            first = s.find("{")
            last = s.rfind("}")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)
            dec = str(data.get("decision", "needs_human"))
            reason = str(data.get("reason", "") or "no reason provided")
            max_env = str(data.get("max_environment", "staging"))
        except Exception:
            logger.exception("PatchGateCouncil: failed to parse council answer: %r", ans)
            dec = "needs_human"
            reason = "Failed to parse council JSON answer."
            max_env = "staging"

        if dec not in ("approve", "reject", "needs_human"):
            dec = "needs_human"

        if max_env not in ("staging", "production"):
            max_env = "staging"

        return PatchGateDecision(
            decision=dec,
            reason=reason,
            max_environment=max_env,
        )

# quentin/selfprog/apply.py
from __future__ import annotations

from pathlib import Path
from typing import Optional
from datetime import datetime
import shutil
import logging

from quentin.kernel.safety import SafetyKernel
from quentin.selfprog.models import Patch
from quentin.selfprog.registry import FileBackedPatchRegistry
from quentin.selfprog.gate import PatchGateCouncil, PatchGateDecision
from quentin.research.experiments import ResearchManager, Experiment

logger = logging.getLogger(__name__)


class PatchApplier:
    """
    负责：

    - gate_patch: 调多 AI 审批补丁，并更新 Patch.gate_decision / gate_reason / status
    - apply_patch: 在通过 gate + safety 后，备份文件，写入 patch_text，更新状态
    - rollback_patch: 用备份文件还原

    v1 简化策略：
      - 把 patch.patch_text 当作 “整个目标文件的新内容” 来写入
      - 不解析 diff（统一 diff 解析可以后面加）
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        registry: Optional[FileBackedPatchRegistry] = None,
        safety_kernel: Optional[SafetyKernel] = None,
        gate_council: Optional[PatchGateCouncil] = None,
        research_manager: Optional[ResearchManager] = None,
        backup_dir: str | Path = "data/patch_backups",
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.registry = registry or FileBackedPatchRegistry()
        self.safety_kernel = safety_kernel or SafetyKernel(repo_root=self.repo_root)
        self.gate_council = gate_council or PatchGateCouncil()
        self.research_manager = research_manager or ResearchManager()
        self.backup_dir = self.repo_root / backup_dir
        self.backup_dir.mkdir(parents=True, exist_ok=True)

    # ---------- 补丁审批（多 AI 开会） ----------

    def gate_patch(
        self,
        patch_id: str,
        environment: str = "staging",
    ) -> Optional[Patch]:
        patch = self.registry.get_patch(patch_id)
        if not patch:
            logger.warning("PatchApplier.gate_patch: patch not found: %s", patch_id)
            return None

        decision: PatchGateDecision = self.gate_council.review_patch(patch, environment=environment)

        patch.gate_decision = decision.decision
        patch.gate_reason = decision.reason
        patch.updated_at = datetime.utcnow().isoformat()

        # 更新 patch.status
        if decision.decision == "approve":
            patch.status = "approved"
        elif decision.decision == "reject":
            patch.status = "rejected"
        else:
            # "needs_human"
            patch.status = "evaluated" if patch.status == "evaluated" else patch.status

        self.registry.upsert_patch(patch)
        self._record_gate_experiment(patch, decision)

        return patch

    # ---------- 应用补丁 ----------

    def apply_patch(
        self,
        patch_id: str,
        environment: str = "staging",
        *,
        require_gate: bool = True,
        min_score: float = 0.7,
    ) -> Optional[Patch]:
        """
        应用补丁（写文件）：

        安全逻辑（v1）：
          1. 如果 require_gate=True，且 gate_decision 不是 approve -> 拒绝
          2. SafetyKernel.check_self_programming_target(target_path) 决定是否允许
          3. 如果 eval_report 存在但 ok=False 或 score<min_score -> 拒绝
          4. 备份原文件（若存在）
          5. 把 patch_text 写到目标文件（整体覆盖）
        """
        patch = self.registry.get_patch(patch_id)
        if not patch:
            logger.warning("PatchApplier.apply_patch: patch not found: %s", patch_id)
            return None

        # gate 检查
        if require_gate:
            if patch.gate_decision != "approve":
                logger.warning(
                    "PatchApplier.apply_patch: patch %s is not approved (gate_decision=%r).",
                    patch_id,
                    patch.gate_decision,
                )
                return patch

        # safety kernel 检查
        policy = self.safety_kernel.check_self_programming_target(patch.target_path)
        if policy.decision == "deny":
            logger.warning(
                "PatchApplier.apply_patch: safety kernel denies patch %s: %s",
                patch_id,
                policy.reason,
            )
            patch.status = "rejected"
            patch.gate_decision = patch.gate_decision or "reject"
            patch.gate_reason = (patch.gate_reason or "") + f" | safety_kernel: {policy.reason}"
            patch.updated_at = datetime.utcnow().isoformat()
            self.registry.upsert_patch(patch)
            return patch

        # 评估结果检查
        if patch.eval_report is not None:
            if not patch.eval_report.ok or patch.eval_report.score < min_score:
                logger.warning(
                    "PatchApplier.apply_patch: patch %s eval_report not ok or score<%s.",
                    patch_id,
                    min_score,
                )
                return patch

        # 文件路径
        target_file = (self.repo_root / patch.target_path).resolve()
        target_file.parent.mkdir(parents=True, exist_ok=True)

        # 备份原文件（如果存在且还没备份）
        backup_path: Optional[Path] = None
        if target_file.exists():
            backup_path = self._backup_file(patch, target_file)
            patch.backup_path = str(backup_path) if backup_path else None

        # 写入 patch_text
        target_file.write_text(patch.patch_text or "", encoding="utf-8")

        # 更新 patch 状态
        now = datetime.utcnow().isoformat()
        patch.applied_at = now
        patch.applied_environment = environment
        patch.status = "applied"
        patch.updated_at = now

        # 记录一些元信息
        patch.metadata.setdefault("apply_history", []).append(
            {
                "applied_at": now,
                "environment": environment,
                "backup_path": patch.backup_path,
            }
        )

        self.registry.upsert_patch(patch)
        self._record_apply_experiment(patch, environment, backup_path is not None)

        return patch

    # ---------- 回滚补丁 ----------

    def rollback_patch(self, patch_id: str) -> Optional[Patch]:
        """
        用备份文件回滚（如果有备份）。
        """
        patch = self.registry.get_patch(patch_id)
        if not patch:
            logger.warning("PatchApplier.rollback_patch: patch not found: %s", patch_id)
            return None

        if not patch.backup_path:
            logger.warning("PatchApplier.rollback_patch: patch %s has no backup_path.", patch_id)
            return patch

        backup_file = Path(patch.backup_path)
        target_file = (self.repo_root / patch.target_path).resolve()

        if not backup_file.exists():
            logger.warning(
                "PatchApplier.rollback_patch: backup file not found: %s",
                backup_file,
            )
            return patch

        # 还原
        target_file.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(backup_file, target_file)

        now = datetime.utcnow().isoformat()
        patch.status = "rolled_back"
        patch.updated_at = now
        patch.metadata.setdefault("rollback_history", []).append(
            {
                "rolled_back_at": now,
                "backup_path": patch.backup_path,
            }
        )
        self.registry.upsert_patch(patch)

        self._record_rollback_experiment(patch)
        return patch

    # ---------- 内部：备份 ----------

    def _backup_file(self, patch: Patch, target_file: Path) -> Optional[Path]:
        try:
            backup_name = f"{patch.patch_id}.orig"
            backup_path = self.backup_dir / backup_name
            shutil.copy2(target_file, backup_path)
            return backup_path
        except Exception:
            logger.exception("PatchApplier._backup_file: failed to backup %s", target_file)
            return None

    # ---------- 内部：Research 记录 ----------

    def _get_or_create_experiment(
        self,
        capability: str,
        requirement_id: str,
        purpose: str,
        tags: list[str],
    ) -> Experiment:
        rm = self.research_manager
        exps = rm.list_experiments()
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == purpose
            ):
                return e
        return rm.create_experiment(
            name=purpose,
            purpose=purpose,
            capability=capability,
            requirement_id=requirement_id,
            tags=tags,
        )

    def _record_gate_experiment(self, patch: Patch, decision: PatchGateDecision) -> None:
        exp = self._get_or_create_experiment(
            capability="selfprog:patch_gate",
            requirement_id="patch_governance_quality",
            purpose="patch_gate_council",
            tags=["selfprog", "gate"],
        )
        ok = 1 if decision.decision == "approve" else 0
        self.research_manager.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics={"approved": ok},
            context={
                "patch_id": patch.patch_id,
                "decision": decision.decision,
                "max_environment": decision.max_environment,
            },
        )

    def _record_apply_experiment(
        self,
        patch: Patch,
        environment: str,
        had_backup: bool,
    ) -> None:
        exp = self._get_or_create_experiment(
            capability="selfprog:patch_apply",
            requirement_id="patch_apply_success_rate",
            purpose="patch_apply",
            tags=["selfprog", "apply"],
        )
        self.research_manager.record_run(
            experiment_id=exp.experiment_id,
            variant_id=environment,
            metrics={"applied": 1, "had_backup": int(had_backup)},
            context={
                "patch_id": patch.patch_id,
                "target_path": patch.target_path,
                "environment": environment,
            },
        )

    def _record_rollback_experiment(self, patch: Patch) -> None:
        exp = self._get_or_create_experiment(
            capability="selfprog:patch_rollback",
            requirement_id="patch_rollback_rate",
            purpose="patch_rollback",
            tags=["selfprog", "rollback"],
        )
        self.research_manager.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics={"rolled_back": 1},
            context={
                "patch_id": patch.patch_id,
                "target_path": patch.target_path,
            },
        )
