# ============================================
# 自编程 / Patch 相关 Schema
# ============================================

class PatchStatus(str, Enum):
    pending = "pending"
    running = "running"
    succeeded = "succeeded"
    failed = "failed"
    discarded = "discarded"


class PatchCandidate(BaseModel):
    """
    LLM 生成的“补丁候选”：
    - module_path: 要修改的模块（相对于 mutable 根目录）
    - diff: 统一 diff（MVP 中可以先放占位文本）
    """
    patch_id: str
    module_path: str
    description: str

    base_version: Optional[str] = None  # 可以存 git commit / 文件 hash
    diff: str

    created_at: datetime = Field(default_factory=datetime.utcnow)
    created_by_model: Optional[str] = None
    suggested_tests: List[str] = Field(default_factory=list)

    meta: Dict[str, Any] = Field(default_factory=dict)


class PatchEvaluation(BaseModel):
    """
    在 sandbox 中跑测试的结果
    """
    patch_id: str
    run_id: str

    status: PatchStatus = PatchStatus.pending

    test_command: str = "pytest"
    exit_code: int = 0
    passed: bool = False

    started_at: datetime = Field(default_factory=datetime.utcnow)
    finished_at: Optional[datetime] = None

    stdout: str = ""
    stderr: str = ""

    metrics: Dict[str, Any] = Field(default_factory=dict)


class PatchReport(BaseModel):
    """
    聚合视角：一个 patch + 多次评估的完整报告
    """
    patch_id: str
    module_path: str
    description: str

    status: PatchStatus
    candidate: PatchCandidate
    evaluations: List[PatchEvaluation] = Field(default_factory=list)

    selected: bool = False            # 是否被选为“推荐方案”
    deployed_to: List[str] = Field(default_factory=list)  # ["sandbox", "staging", ...]

    created_at: datetime = Field(default_factory=datetime.utcnow)
    meta: Dict[str, Any] = Field(default_factory=dict)


class ImprovementCandidate(BaseModel):
    """
    来自 Meta / Immune / Sleep 等环节的“需要改进的模块”提案
    """
    candidate_id: str
    module_path: str                # 相对于 mutable 根目录
    reason: str                     # 为什么要改

    evidence: Dict[str, Any] = Field(default_factory=dict)
    suggested_tests: List[str] = Field(default_factory=list)

    created_at: datetime = Field(default_factory=datetime.utcnow)
    meta: Dict[str, Any] = Field(default_factory=dict)
这样就把蓝皮书里的 “需要改进的模块 → PatchCandidate → PatchReport” 这条链的结构先定死了。

二、自编程引擎 SelfProgrammingEngine v1
文件：quentin/self_programming/engine.py（如果已经有，就用下面的版本覆盖 / 合并）

# ============================================
# quentin/self_programming/engine.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass
from typing import List, Dict, Any, Optional
from pathlib import Path
import uuid
import asyncio
from datetime import datetime

from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.runtime.settings import Settings
from quentin.runtime.model_router import ModelRouter
from quentin.kernel.safety import SafetyKernel
from quentin.kernel.schemas import (
    PatchCandidate,
    PatchEvaluation,
    PatchReport,
    PatchStatus,
    ImprovementCandidate,
)


@dataclass
class SelfProgrammingEngine:
    """
    自编程引擎 v1（MVP）：

    - 只在 /mutable 下工作的“代码医生”
    - 当前版本不真正改源文件，只在 sandbox 目录复制代码并跑测试
    - patch diff 先用占位文本，后续可以接真实 LLM 生成 diff
    """
    tracer: Tracer
    kv: KVStore
    settings: Settings
    model_router: ModelRouter
    safety_kernel: SafetyKernel

    patch_candidate_prefix: str = "patch:candidate:"
    patch_eval_prefix: str = "patch:eval:"
    patch_eval_index_prefix: str = "patch:eval_index:"
    patch_report_prefix: str = "patch:report:"
    patch_report_all_key: str = "patch:report:all"

    # ---------- 基础路径工具 ----------

    def _repo_root(self) -> Path:
        root = getattr(self.settings, "repo_root", None) or "."
        return Path(root).resolve()

    def _mutable_root(self) -> Path:
        repo = self._repo_root()
        mutable_dir = getattr(self.settings, "mutable_dir", "mutable")
        return (repo / mutable_dir).resolve()

    def _sandbox_root(self) -> Path:
        repo = self._repo_root()
        return (repo / "sandbox_patches").resolve()

    def _ensure_mutable_module(self, module_path: str) -> Path:
        """
        module_path 按“相对于 mutable 根目录”的路径解释，
        并确保不会越界。
        """
        mutable_root = self._mutable_root()
        p = (mutable_root / module_path).resolve()
        if not str(p).startswith(str(mutable_root)):
            raise ValueError(f"Self-programming only allowed under {mutable_root}, got {p}")
        if not p.exists():
            raise FileNotFoundError(f"Module file not found: {p}")
        return p

    # ---------- KV 存取工具 ----------

    async def _save_candidate(self, cand: PatchCandidate) -> None:
        key = f"{self.patch_candidate_prefix}{cand.patch_id}"
        await self.kv.save(key, cand.model_dump())

    async def _load_candidate(self, patch_id: str) -> Optional[PatchCandidate]:
        data = await self.kv.get(f"{self.patch_candidate_prefix}{patch_id}")
        return PatchCandidate(**data) if data else None

    async def _save_evaluation(self, ev: PatchEvaluation) -> None:
        key = f"{self.patch_eval_prefix}{ev.patch_id}:{ev.run_id}"
        await self.kv.save(key, ev.model_dump())
        idx_key = f"{self.patch_eval_index_prefix}{ev.patch_id}"
        await self.kv.append_list(idx_key, ev.run_id)

    async def _list_evaluations(self, patch_id: str, limit: int = 20) -> List[PatchEvaluation]:
        idx_key = f"{self.patch_eval_index_prefix}{patch_id}"
        run_ids = await self.kv.get(idx_key) or []
        run_ids = list(dict.fromkeys(run_ids))[-limit:]
        res: List[PatchEvaluation] = []
        for rid in run_ids:
            data = await self.kv.get(f"{self.patch_eval_prefix}{patch_id}:{rid}")
            if data:
                res.append(PatchEvaluation(**data))
        return res

    async def _save_report(self, report: PatchReport) -> None:
        key = f"{self.patch_report_prefix}{report.patch_id}"
        await self.kv.save(key, report.model_dump())
        await self.kv.append_list(self.patch_report_all_key, report.patch_id)

    async def get_patch_report(self, patch_id: str) -> Optional[PatchReport]:
        data = await self.kv.get(f"{self.patch_report_prefix}{patch_id}")
        return PatchReport(**data) if data else None

    async def list_patch_reports(self, limit: int = 50) -> List[PatchReport]:
        ids = await self.kv.get(self.patch_report_all_key) or []
        ids = list(dict.fromkeys(ids))[-limit:]
        res: List[PatchReport] = []
        for pid in ids:
            data = await self.kv.get(f"{self.patch_report_prefix}{pid}")
            if data:
                res.append(PatchReport(**data))
        return res

    # ---------- 占位：LLM 生成 diff ----------

    async def _generate_diff_placeholder(
        self,
        original_code: str,
        problem_description: str,
    ) -> str:
        """
        MVP 版：暂时不真正生成 diff，只写一段注释。
        将来你可以在这里调用 model_router，把 original_code + 描述交给 LLM 产出统一 diff。
        """
        note = (
            "# NOTE: SelfProgrammingEngine patch generation not implemented yet.\n"
            f"# Problem: {problem_description}\n"
        )
        return note

    # ---------- 公开 API：为模块生成补丁候选 ----------

    async def propose_patch_for_module(
        self,
        module_path: str,
        problem_description: str,
        suggested_tests: Optional[List[str]] = None,
    ) -> PatchCandidate:
        """
        只生成 PatchCandidate，并不跑测试。
        module_path：相对于 mutable 根目录的路径，例如 "skills/planner.py"。
        """
        file_path = self._ensure_mutable_module(module_path)
        original_code = file_path.read_text(encoding="utf-8")

        diff_text = await self._generate_diff_placeholder(original_code, problem_description)

        patch_id = f"patch_{uuid.uuid4().hex}"
        cand = PatchCandidate(
            patch_id=patch_id,
            module_path=module_path,
            description=problem_description,
            base_version=None,
            diff=diff_text,
            created_by_model=None,
            suggested_tests=suggested_tests or [],
        )
        await self._save_candidate(cand)

        self.tracer.span(
            "L3",
            "patch_candidate_created",
            module="self_programming",
            input={"patch_id": patch_id, "module_path": module_path},
        )
        return cand

    # ---------- 在 sandbox 中评估 patch ----------

    async def evaluate_patch(
        self,
        candidate: PatchCandidate,
        test_command: str = "pytest",
    ) -> PatchEvaluation:
        """
        在 sandbox 目录中复制 mutable 模块，暂不真正应用 diff，
        然后执行 test_command，记录 stdout/stderr/exit_code。
        """
        repo_root = self._repo_root()
        mutable_root = self._mutable_root()
        sandbox_root = self._sandbox_root() / candidate.patch_id

        # 准备 sandbox
        sandbox_root.mkdir(parents=True, exist_ok=True)
        src_file = self._ensure_mutable_module(candidate.module_path)
        dst_file = (sandbox_root / candidate.module_path).resolve()
        dst_file.parent.mkdir(parents=True, exist_ok=True)
        dst_file.write_text(src_file.read_text(encoding="utf-8"), encoding="utf-8")

        # TODO：将来在这里真正应用 candidate.diff 到 dst_file

        started_at = datetime.utcnow()
        stdout_text = ""
        stderr_text = ""
        exit_code = -1

        try:
            proc = await asyncio.create_subprocess_shell(
                test_command,
                cwd=str(sandbox_root),
                stdout=asyncio.subprocess.PIPE,
                stderr=asyncio.subprocess.PIPE,
            )
            stdout_b, stderr_b = await proc.communicate()
            exit_code = proc.returncode or 0
            stdout_text = stdout_b.decode("utf-8", errors="ignore")
            stderr_text = stderr_b.decode("utf-8", errors="ignore")
        except Exception as e:
            stderr_text = repr(e)
            exit_code = -1

        finished_at = datetime.utcnow()
        passed = (exit_code == 0)
        status = PatchStatus.succeeded if passed else PatchStatus.failed

        ev = PatchEvaluation(
            patch_id=candidate.patch_id,
            run_id=f"eval_{uuid.uuid4().hex}",
            status=status,
            test_command=test_command,
            exit_code=exit_code,
            passed=passed,
            started_at=started_at,
            finished_at=finished_at,
            stdout=stdout_text,
            stderr=stderr_text,
            metrics={"exit_code": exit_code},
        )
        await self._save_evaluation(ev)

        self.tracer.span(
            "L3",
            "patch_evaluated",
            module="self_programming",
            input={"patch_id": candidate.patch_id},
            output={"status": status.value, "exit_code": exit_code},
        )
        return ev

    # ---------- 高层：自动处理 ImprovementCandidate ----------

    async def auto_patch_for_improvement(
        self,
        improvement: ImprovementCandidate,
    ) -> Optional[PatchReport]:
        """
        从 ImprovementCandidate 出发：
        - 检查 SafetyKernel 是否允许自编程
        - 生成 PatchCandidate
        - 在 sandbox 里跑一次测试
        - 生成 PatchReport 并持久化
        """
        state = await self.safety_kernel.get_state()
        if not state.self_programming_enabled:
            self.tracer.span(
                "L3",
                "self_programming_skipped",
                module="self_programming",
                input={
                    "reason": "disabled_by_safety_kernel",
                    "candidate_id": improvement.candidate_id,
                },
            )
            return None

        test_cmd = improvement.suggested_tests[0] if improvement.suggested_tests else "pytest"

        cand = await self.propose_patch_for_module(
            module_path=improvement.module_path,
            problem_description=improvement.reason,
            suggested_tests=improvement.suggested_tests,
        )
        ev = await self.evaluate_patch(cand, test_command=test_cmd)

        status = ev.status
        report = PatchReport(
            patch_id=cand.patch_id,
            module_path=cand.module_path,
            description=cand.description,
            status=status,
            candidate=cand,
            evaluations=[ev],
            selected=False,
            deployed_to=["sandbox"],
            meta={
                "improvement_candidate_id": improvement.candidate_id,
                "test_command": test_cmd,
            },
        )
        await self._save_report(report)

        self.tracer.span(
            "L3",
            "patch_report_created",
            module="self_programming",
            input={"patch_id": cand.patch_id},
            output={"status": status.value},
        )

        return report

    # ---------- 辅助：更通用的自动 patch 接口（给 CLI 用） ----------

    async def auto_patch_for_module(
        self,
        module_path: str,
        reason: str,
        test_command: str = "pytest",
    ) -> Optional[PatchReport]:
        """
        CLI 友好的入口：不需要先构建 ImprovementCandidate。
        """
        ic = ImprovementCandidate(
            candidate_id=f"imp_{uuid.uuid4().hex}",
            module_path=module_path,
            reason=reason,
            suggested_tests=[test_command],
        )
        return await self.auto_patch_for_improvement(ic)

@cli.command("list-patches")
@click.option("--limit", default=20, show_default=True)
def list_patches_cmd(limit: int):
    """
    列出最近的 PatchReport。
    """
    async def _run():
        settings = Settings()
        runtime = AgentRuntime.create_from_settings(settings)
        reports = await runtime.self_programming.list_patch_reports(limit=limit)
        if not reports:
            print("No patches recorded yet.")
            return
        for r in reports:
            print(
                f"- {r.patch_id} | {r.module_path} | "
                f"status={r.status.value} | created_at={r.created_at.isoformat()}"
            )

    asyncio.run(_run())

6.3 手动对某个模块触发自编程
@cli.command("auto-patch-module")
@click.option("--path", "module_path", required=True, help="Module path relative to mutable root.")
@click.option("--reason", required=True, help="Why this module should be improved.")
@click.option("--test", "test_command", default="pytest", show_default=True, help="Test command to run in sandbox.")
def auto_patch_module_cmd(module_path: str, reason: str, test_command: str):
    """
    手动对某个 mutable 模块触发一次自编程：
    - 生成 PatchCandidate（当前为占位 diff）
    - 在 sandbox 里运行 test_command
    - 生成 PatchReport
    """
    async def _run():
        settings = Settings()
        runtime = AgentRuntime.create_from_settings(settings)
        engine = runtime.self_programming
        report = await engine.auto_patch_for_module(
            module_path=module_path,
            reason=reason,
            test_command=test_command,
        )
        if not report:
            print("Self-programming disabled by SafetyKernel or patch generation failed.")
            return

        print(f"Patch {report.patch_id} created for {report.module_path}")
        print(f"  status    : {report.status.value}")
        print(f"  evaluations: {len(report.evaluations)}")

    asyncio.run(_run())

✅ 安全边界

✅ Research-Friendly
