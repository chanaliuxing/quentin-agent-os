# quentin/models/ollama_client.py
from __future__ import annotations

import json
import logging
from dataclasses import dataclass
from typing import Dict, Any, Optional

import requests

logger = logging.getLogger(__name__)


@dataclass
class OllamaConfig:
    base_url: str = "http://localhost:11434"
    timeout: int = 120  # seconds


class OllamaClient:
    """
    统一封装 Ollama 的 /api/generate 接口。
    支持简单的单轮对话：把系统提示 + 会议上下文 + 问题拼成 prompt，让不同模型各自回答。
    """

    def __init__(self, config: Optional[OllamaConfig] = None) -> None:
        self.config = config or OllamaConfig()

    def generate(
        self,
        model: str,
        prompt: str,
        temperature: float = 0.3,
        max_tokens: int = 2048,
    ) -> str:
        """
        调用 Ollama /api/generate。
        注意：这里假设你本地 Ollama 支持 stream=false 一次性返回。
        """
        url = f"{self.config.base_url}/api/generate"
        payload: Dict[str, Any] = {
            "model": model,
            "prompt": prompt,
            "stream": False,
            "options": {
                "temperature": temperature,
                "num_predict": max_tokens,
            },
        }

        logger.debug("Calling Ollama: model=%s, prompt_len=%d", model, len(prompt))

        resp = requests.post(url, json=payload, timeout=self.config.timeout)
        resp.raise_for_status()

        data = resp.json()
        # Ollama /api/generate 返回形如 {"model": "...", "created_at": "...", "response": "...", ...}
        text = data.get("response", "")
        logger.debug("Ollama response (truncated): %s", text[:200])
        return text
说明：

现在只用 /api/generate 单轮调用，协议最稳定。

多轮对话我们自己在上层通过“拼接会议纪录文本”来实现，不依赖 /api/chat。

五、“多 AI 开会”核心类：MultiLLMCouncil
文件：quentin/agents/multi_llm_council.py

这个类做三件事：

第一轮：每个角色独立提出方案

第二轮：由一个“主持人模型”看完所有方案，做批判 + 合并

可选第三轮：让另一个模型对主持人方案做“风险复审”

# quentin/agents/multi_llm_council.py
from __future__ import annotations

from dataclasses import dataclass
from typing import List, Dict, Any, Optional

from quentin.models.ollama_client import OllamaClient
from quentin.agents.multi_llm_roles import (
    LLMRoleConfig,
    MeetingType,
    default_roles_for_local_ollama,
)


@dataclass
class Proposal:
    role_id: str
    role_name: str
    model: str
    content: str


@dataclass
class CouncilResult:
    meeting_type: MeetingType
    problem: str
    context: Dict[str, Any]
    proposals: List[Proposal]
    merged_answer: str
    risks: str
    notes: Dict[str, Any]


class MultiLLMCouncil:
    """
    多 LLM 会议模式：
    - 多个角色各自给出 proposal
    - 由一个主持人模型总结 + 合并 + 列出风险
    """

    def __init__(
        self,
        client: Optional[OllamaClient] = None,
        roles_by_meeting: Optional[Dict[MeetingType, List[LLMRoleConfig]]] = None,
        moderator_model: str = "qwen3-vl:latest",
        reviewer_model: str = "deepseek-r1:1.5b",
    ) -> None:
        self.client = client or OllamaClient()
        self.roles_by_meeting = roles_by_meeting or default_roles_for_local_ollama()
        self.moderator_model = moderator_model
        self.reviewer_model = reviewer_model

    def _build_role_prompt(
        self,
        role: LLMRoleConfig,
        meeting_type: MeetingType,
        problem: str,
        context: Dict[str, Any],
    ) -> str:
        """
        给单个角色的完整 prompt：
        - 角色职责 & 风格
        - 当前要解决的问题
        - 上下文（如：失败记录 / 相关限制）
        """
        context_str = ""
        if context:
            context_str = f"\n[Context]\n{self._format_context(context)}\n"

        return f"""You are participating in a multi-AI council.

[Your Role]
Name: {role.name}
Purpose: {role.purpose}
Style: {role.style}

[Meeting Type]
{meeting_type}

[Problem to solve]
{problem}
{context_str}

[Instruction]
1. Think independently, do not try to agree with other models (you will not see them).
2. Provide a clear, structured answer.
3. If you are unsure, say what you are unsure about and list assumptions.
4. Output in Chinese when user is Chinese, but keep structure very clear.

现在请根据你的角色给出你的完整建议。
"""

    @staticmethod
    def _format_context(context: Dict[str, Any]) -> str:
        # 简单地 json 化，后续可以更精细
        try:
            import json

            return json.dumps(context, ensure_ascii=False, indent=2)
        except Exception:
            return str(context)

    def deliberate(
        self,
        meeting_type: MeetingType,
        problem: str,
        context: Optional[Dict[str, Any]] = None,
    ) -> CouncilResult:
        """
        对外统一入口。
        1. 选出对应 MeetingType 的角色
        2. 每个角色独立生成 proposal
        3. 让主持人模型阅读所有 proposal 并合并 + 批判
        4. 让另一模型对主持人结果做风险复审（可选）
        """
        ctx = context or {}
        roles = self.roles_by_meeting.get(meeting_type) or self.roles_by_meeting["generic"]

        # 1) Proposal Round
        proposals: List[Proposal] = []
        for role in roles:
            prompt = self._build_role_prompt(role, meeting_type, problem, ctx)
            answer = self.client.generate(
                model=role.model,
                prompt=prompt,
                temperature=role.default_temperature,
            )
            proposals.append(
                Proposal(
                    role_id=role.id,
                    role_name=role.name,
                    model=role.model,
                    content=answer.strip(),
                )
            )

        # 2) Critique & Merge Round 由 moderator_model 负责
        merged_answer, risks, notes = self._moderate_and_merge(
            meeting_type=meeting_type,
            problem=problem,
            context=ctx,
            proposals=proposals,
        )

        return CouncilResult(
            meeting_type=meeting_type,
            problem=problem,
            context=ctx,
            proposals=proposals,
            merged_answer=merged_answer,
            risks=risks,
            notes=notes,
        )

    def _moderate_and_merge(
        self,
        meeting_type: MeetingType,
        problem: str,
        context: Dict[str, Any],
        proposals: List[Proposal],
    ) -> (str, str, Dict[str, Any]):
        """
        主持人模型：
        - 总结共识
        - 指出分歧
        - 给出综合方案
        - 列出风险 & TODO
        """
        proposals_text = ""
        for i, p in enumerate(proposals, start=1):
            proposals_text += f"\n=== Proposal {i} ({p.role_name} / {p.model}) ===\n{p.content}\n"

        context_str = self._format_context(context)

        moderator_prompt = f"""You are the moderator of a multi-AI council.

[Meeting Type]
{meeting_type}

[Problem]
{problem}

[Context]
{context_str}

[Proposals from different AI roles]
{proposals_text}

[Your task]
1. Summarize the common consensus across proposals.
2. Highlight key disagreements or conflicting assumptions.
3. Propose a merged, concrete plan or answer that:
   - is as safe and robust as possible
   - keeps useful diversity from different proposals
4. Explicitly list potential risks, uncertainties, or TODO items.
5. Answer in Chinese, with clear structure and headings.

请现在给出：
- 《综合方案》
- 《共识点》
- 《分歧与假设》
- 《风险与待确认事项》
"""

        merged = self.client.generate(
            model=self.moderator_model,
            prompt=moderator_prompt,
            temperature=0.25,
        )

        # 3) 可选：让 reviewer_model 做一次“风险复审”
        reviewer_prompt = f"""下面是一份由主持人总结的《综合方案》。

请你扮演「风险复审官」，任务是：
1. 找出此方案中可能存在的风险、遗漏和不合理假设。
2. 不要重复明显内容，重点补充新的视角。
3. 用简洁的要点形式列出。

《综合方案》如下：
{merged}
"""

        risks = self.client.generate(
            model=self.reviewer_model,
            prompt=reviewer_prompt,
            temperature=0.3,
        )

        notes: Dict[str, Any] = {
            "moderator_model": self.moderator_model,
            "reviewer_model": self.reviewer_model,
            "proposal_count": len(proposals),
        }
        return merged.strip(), risks.strip(), notes
