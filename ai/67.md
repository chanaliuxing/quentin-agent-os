# quentin/core/skill_evolution.py
from __future__ import annotations

import json
import logging
from dataclasses import dataclass, asdict
from datetime import datetime
from typing import Any, Dict, List, Optional

from quentin.memory.fabric import FileBackedMemoryFabric, MemoryItem
from quentin.skills.models import SkillCandidate
from quentin.skills.registry import FileBackedSkillRegistry
from quentin.skills.promotion import SkillPromotionEngine, SkillPromotionResult
from quentin.llm.multi_council import MultiAICouncil, CouncilDecision
from quentin.llm.types import LLMConfig
from quentin.research.experiments import ResearchManager, Experiment

logger = logging.getLogger(__name__)


@dataclass
class SkillEvolutionReport:
    run_id: str
    started_at: str
    finished_at: str

    seeds_used: int         # 用于挖技能的记忆条目数量
    candidates_proposed: int
    candidates_promoted: int

    notes: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class SkillEvolutionLoop:
    """
    技能进化循环 v1：

    角色：夜间（或定期）扫描 MemoryFabric 中的“可学习经验”，自动挖掘 SkillCandidate，
    调用 SkillPromotionEngine 评估 & 晋升，并把整个过程记录到 ResearchManager。

    v1 策略（刻意简单但可工作）：

      1. 从 MemoryFabric 里找：
         - 最近的 MTS semantic_summary（kind="semantic_summary"）
         - 最近的 STS skill_result / task_lesson（kind in ["skill_result", "task_lesson"]）
      2. 拼成一个上下文给 MultiAICouncil，问题是：
         “请基于这些经验提炼出 0~N 个可复用技能（Skill），用 JSON 返回。”
      3. 解析 JSON → SkillCandidate 列表，写入 SkillRegistry
      4. 对每个新 Candidate 调用 SkillPromotionEngine.promote_candidate(auto=True)
      5. 整个过程记录到 ResearchManager（purpose="skill_evolution"）
    """

    def __init__(
        self,
        *,
        memory_fabric: Optional[FileBackedMemoryFabric] = None,
        registry: Optional[FileBackedSkillRegistry] = None,
        council: Optional[MultiAICouncil] = None,
        promotion_engine: Optional[SkillPromotionEngine] = None,
        research_manager: Optional[ResearchManager] = None,
        max_seed_items: int = 50,
    ) -> None:
        self.memory_fabric = memory_fabric or FileBackedMemoryFabric()
        self.registry = registry or FileBackedSkillRegistry()

        if council is None:
            default_members = [
                LLMConfig(
                    provider="ollama",
                    model="qwen3-vl:latest",
                    role="skill_miner",
                ),
                LLMConfig(
                    provider="ollama",
                    model="deepseek-r1:1.5b",
                    role="skill_critic",
                ),
            ]
            council = MultiAICouncil(members=default_members)
        self.council = council

        self.promotion_engine = promotion_engine or SkillPromotionEngine(
            registry=self.registry,
        )
        self.research_manager = research_manager or ResearchManager()
        self.max_seed_items = max_seed_items

    # ---------- 主入口 ----------

    def run_once(self) -> SkillEvolutionReport:
        started_at = datetime.utcnow().isoformat()

        # 1) 收集“种子”记忆
        seeds = self._collect_seeds(limit=self.max_seed_items)
        if not seeds:
            finished = datetime.utcnow().isoformat()
            report = SkillEvolutionReport(
                run_id=started_at,
                started_at=started_at,
                finished_at=finished,
                seeds_used=0,
                candidates_proposed=0,
                candidates_promoted=0,
                notes="no seeds for skill evolution",
            )
            self._record_research(report)
            return report

        # 2) 用 MultiAICouncil 提出技能候选 JSON
        council_answer = self._ask_council_for_skills(seeds)
        candidates_data = self._parse_candidates_json(council_answer)

        # 3) 写入 SkillRegistry 作为 SkillCandidate
        candidates: List[SkillCandidate] = []
        for cdata in candidates_data:
            try:
                cand = self.registry.create_candidate_from_dict(cdata)
                candidates.append(cand)
            except Exception:
                logger.exception("SkillEvolutionLoop: failed to create candidate from %r", cdata)

        # 4) 调用 SkillPromotionEngine 自动晋升
        promoted_count = 0
        for cand in candidates:
            try:
                res: SkillPromotionResult = self.promotion_engine.promote_candidate(
                    candidate_id=cand.candidate_id,
                    auto=True,
                )
                if res.status == "promoted":
                    promoted_count += 1
            except Exception:
                logger.exception(
                    "SkillEvolutionLoop: promotion failed for candidate_id=%s",
                    cand.candidate_id,
                )

        finished_at = datetime.utcnow().isoformat()
        report = SkillEvolutionReport(
            run_id=started_at,
            started_at=started_at,
            finished_at=finished_at,
            seeds_used=len(seeds),
            candidates_proposed=len(candidates),
            candidates_promoted=promoted_count,
            notes=f"evolved {len(candidates)} candidates, promoted {promoted_count}",
        )

        # 5) 写入 ResearchManager
        self._record_research(report)
        return report

    # ---------- 收集种子记忆 ----------

    def _collect_seeds(self, limit: int) -> List[MemoryItem]:
        """
        v1 策略：

          - 优先拿最近的 MTS semantic_summary（长期/项目级总结）
          - 再拿 STS skill_result / task_lesson

        目的是给 LLM 足够的语义上下文，让它能提炼出通用技能。
        """
        seeds: List[MemoryItem] = []

        # 先 MTS semantic_summary
        mts_summaries = self.memory_fabric.list_items(
            tier="MTS",
            kind="semantic_summary",
            limit=limit,
        )
        seeds.extend(mts_summaries)

        if len(seeds) < limit:
            remaining = limit - len(seeds)
            # 再从 STS 中拿 skill_result / task_lesson
            skill_results = self._list_sts_by_kind("skill_result", remaining)
            seeds.extend(skill_results)

        if len(seeds) < limit:
            remaining = limit - len(seeds)
            task_lessons = self._list_sts_by_kind("task_lesson", remaining)
            seeds.extend(task_lessons)

        return seeds

    def _list_sts_by_kind(self, kind: str, limit: int) -> List[MemoryItem]:
        if limit <= 0:
            return []
        items = self.memory_fabric.list_items(
            tier="STS",
            kind=kind,
            limit=limit,
        )
        return items

    # ---------- 向 Council 请求技能候选 ----------

    def _ask_council_for_skills(self, seeds: List[MemoryItem]) -> str:
        lines: List[str] = []
        for m in seeds:
            # 简单展平，未来可以更结构化
            lines.append(f"[{m.tier}/{m.kind}/{m.domain}] {m.content[:400]}")
        context = "\n".join(lines)

        question = (
            "You are the SKILL EVOLUTION loop of a self-improving agent.\n"
            "Given the following episodic and semantic memories from recent tasks,\n"
            "you must propose 0~N NEW skills that would help the agent solve similar tasks\n"
            "in the future more reliably and efficiently.\n\n"
            "Each skill should be domain-agnostic or clearly scoped to a domain.\n"
            "Return ONLY a JSON array. No extra text.\n\n"
            "JSON SCHEMA (informal):\n"
            "[\n"
            "  {\n"
            "    \"name\": \"string\",\n"
            "    \"domain\": \"general | desktop | legal | trading | ...\",\n"
            "    \"description\": \"what this skill does and when to use it\",\n"
            "    \"code\": \"pseudo-code or implementation in Python-like syntax\",\n"
            "    \"tests\": \"how to test this skill (scenarios, assertions)\"\n"
            "  },\n"
            "  ...\n"
            "]\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="skill_evolution",
        )
        return decision.final_answer or "[]"

    def _parse_candidates_json(self, text: str) -> List[Dict[str, Any]]:
        """
        尝试从 council 的回答中取出 JSON 数组。
        """
        try:
            s = text.strip()
            first = s.find("[")
            last = s.rfind("]")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)
            if not isinstance(data, list):
                return []
            normalized: List[Dict[str, Any]] = []
            for item in data:
                if not isinstance(item, dict):
                    continue
                # 简单规范化字段名
                name = str(item.get("name", "")).strip()
                if not name:
                    continue
                domain = str(item.get("domain", "general")).strip() or "general"
                description = str(item.get("description", "")).strip()
                code = str(item.get("code", "")).strip()
                tests = str(item.get("tests", "")).strip()
                normalized.append(
                    {
                        "name": name,
                        "domain": domain,
                        "description": description,
                        "code": code,
                        "tests": tests,
                    }
                )
            return normalized
        except Exception:
            logger.exception("SkillEvolutionLoop: failed to parse JSON from answer=%r", text)
            return []

    # ---------- Research 记录 ----------

    def _record_research(self, report: SkillEvolutionReport) -> None:
        rm = self.research_manager

        capability = "skills:evolution"
        requirement_id = "skill_discovery_rate"

        exps = rm.list_experiments()
        exp: Optional[Experiment] = None
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "skill_evolution"
            ):
                exp = e
                break

        if exp is None:
            exp = rm.create_experiment(
                name="SkillEvolutionLoop",
                purpose="skill_evolution",
                capability=capability,
                requirement_id=requirement_id,
                tags=["skills", "evolution"],
            )

        metrics = {
            "seeds_used": report.seeds_used,
            "candidates_proposed": report.candidates_proposed,
            "candidates_promoted": report.candidates_promoted,
        }
        context = {
            "run_id": report.run_id,
            "notes": report.notes,
            "started_at": report.started_at,
            "finished_at": report.finished_at,
        }

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics=metrics,
            context=context,
        )
注意这里我假设 FileBackedSkillRegistry 有一个 create_candidate_from_dict 方法。

蓝皮书 3.4 要求的几个步骤，对照一下：

输入： 高成功率的 Trace & Immune 标记的差模块

从 Trace 中挖模式

输出： 新技能定义 + supersedes 记录 + Patch 报告

目前 repo 侧已经有：

✅ 认知层挖技能（在线 / 任务内）

SkillMiner.mine_from_task(...)（之前已经落地）

用 MultiAICouncil 从 Task + Trace 直接抽一个 SkillCandidate

✅ 夜间技能进化（离线 / 任务间）

SleepLoop.run_once() 把 STS → MTS semantic_summary

SkillEvolutionLoop.run_once() 从 MTS/STS 记忆中总结模式 → 自动发明新 SkillCandidate → SkillPromotionEngine 晋升

✅ SkillPromotionEngine + SkillEvalHarness

SkillEvalHarness：检查 description / code / tests 完整性，评分

SkillPromotionEngine：把 Candidate 晋升为 SkillDefinition，写 ResearchManager

✅ 技能执行闭环

Planner 在 plan 里引用 CALL_SKILL

SkillAwareExecutor 解析 plan，调用 SkillManager.run_skill（多模型解释执行）

结果写 Trace + STS memory（kind="skill_result"）→ 为后续 SleepLoop & SkillEvolutionLoop 提供数据

✅ 研究记录

ResearchManager + Experiments：SkillPromotion / SleepLoop / SkillEvolution 都会写 experiment_run，方便后面分析

还缺（后面几步我们会再落地）：

ImmuneLoop 真正标记「表现差的模块」→ 输入 SelfProgrammingEngine

真正的 Sandbox 代码执行 + 回归测试框架（现在 Skill 执行是多 AI 解释执行）

supersedes 关联（部分我们在 SkillDefinition.metadata 里已经给了 source_candidate_id，占位）

Patch Engine（针对 repo 真实代码模块的自编程 & 测试 &回滚）
