# ============================================
# Memory tiers & update policy
# ============================================

class MemoryTier(str, Enum):
    STS = "STS"   # Short-Term Store  - 会话 / 任务级（分钟 / 小时）
    MTS = "MTS"   # Mid-Term Store    - 项目 / 版本级（天 / 周）
    LTS = "LTS"   # Long-Term Store   - 跨项目稳定知识（周 / 月 / 年）


class UpdatePolicy(BaseModel):
    """
    记忆的更新策略（目前更多是描述性字段，SleepLoop/MemoryFabric
    可以根据 tier 使用硬编码规则，后续再真正解析这些 string）。
    """
    promotion_rule: Optional[str] = None      # 例如 "access>=3 || age>3600"
    decay_rule: Optional[str] = None          # 例如 "age>86400 && access==0"
    consolidation_batch: str = "nightly"      # nightly | weekly | manual

# ============================================
# quentin/memory/fabric.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Dict, Any, List, Optional
from datetime import datetime, timedelta

from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.memory.schemas import MemoryItem, MemoryTier


@dataclass
class MemoryFabric:
    """
    Memory Fabric v1：
    - 统一管理 STS / MTS / LTS 三层记忆
    - 提供简单的写入 / 读取 / 统计
    - 提供 run_consolidation：按照简单规则做晋升 / 遗忘
    """
    kv: KVStore
    tracer: Tracer

    item_prefix: str = "memory:item:"
    index_all_key: str = "memory:index:all"
    index_tier_prefix: str = "memory:index:tier:"

    # ========= 基础存取 =========

    async def _save_item(self, item: MemoryItem) -> None:
        key = f"{self.item_prefix}{item.memory_id}"
        await self.kv.save(key, item.model_dump())
        await self._index_item(item)

    async def _load_item(self, memory_id: str) -> Optional[MemoryItem]:
        key = f"{self.item_prefix}{memory_id}"
        data = await self.kv.get(key)
        if not data:
            return None
        return MemoryItem(**data)

    async def _index_item(self, item: MemoryItem) -> None:
        # 全局索引
        await self.kv.append_list(self.index_all_key, item.memory_id)
        # 分 tier 索引
        tier_key = f"{self.index_tier_prefix}{item.memory_tier.value}"
        await self.kv.append_list(tier_key, item.memory_id)

    async def _reindex_item_tier(self, memory_id: str, old_tier: MemoryTier, new_tier: MemoryTier) -> None:
        # 简单实现：把 old_tier 列表中的 id 过滤掉再保存；量大时可以优化
        old_key = f"{self.index_tier_prefix}{old_tier.value}"
        ids = await self.kv.get(old_key) or []
        ids = [i for i in ids if i != memory_id]
        await self.kv.save(old_key, ids)

        new_key = f"{self.index_tier_prefix}{new_tier.value}"
        await self.kv.append_list(new_key, memory_id)

    async def _delete_item(self, memory_id: str) -> None:
        # 删除本体
        await self.kv.delete(f"{self.item_prefix}{memory_id}")
        # 从全局索引中移除
        ids = await self.kv.get(self.index_all_key) or []
        ids = [i for i in ids if i != memory_id]
        await self.kv.save(self.index_all_key, ids)
        # 从各 tier 索引中移除
        for tier in MemoryTier:
            key = f"{self.index_tier_prefix}{tier.value}"
            lst = await self.kv.get(key) or []
            lst = [i for i in lst if i != memory_id]
            await self.kv.save(key, lst)

    # ========= 写入 / 读取 / 搜索 =========

    async def add_memory(self, item: MemoryItem) -> MemoryItem:
        """
        写入新记忆（默认 STS）。
        """
        now = datetime.utcnow()
        item.created_at = item.created_at or now
        item.updated_at = now
        item.last_access_at = item.last_access_at or now
        item.access_count = item.access_count or 0
        await self._save_item(item)

        self.tracer.span(
            "L1",
            "memory_add",
            module="memory_fabric",
            input={"memory_id": item.memory_id, "tier": item.memory_tier.value},
        )
        return item

    async def get_memory(self, memory_id: str, touch: bool = True) -> Optional[MemoryItem]:
        """
        按 ID 获取记忆，如果 touch=True 会更新访问次数和时间。
        """
        item = await self._load_item(memory_id)
        if not item:
            return None
        if touch:
            item.access_count += 1
            item.last_access_at = datetime.utcnow()
            item.updated_at = item.last_access_at
            await self._save_item(item)
        return item

    async def sample_items_by_tier(self, tier: MemoryTier, limit: int = 1000) -> List[MemoryItem]:
        """
        从某一层采样一批记忆（最近写入的若干条）。
        """
        key = f"{self.index_tier_prefix}{tier.value}"
        ids = await self.kv.get(key) or []
        ids = list(dict.fromkeys(ids))[-limit:]  # 去重 + 取最近的
        res: List[MemoryItem] = []
        for mid in ids:
            item = await self._load_item(mid)
            if item:
                res.append(item)
        return res

    # ========= 统计信息 =========

    async def get_stats(self) -> Dict[str, Any]:
        all_ids = await self.kv.get(self.index_all_key) or []
        all_ids = list(dict.fromkeys(all_ids))
        total = len(all_ids)

        per_tier: Dict[str, int] = {}
        for tier in MemoryTier:
            key = f"{self.index_tier_prefix}{tier.value}"
            ids = await self.kv.get(key) or []
            ids = list(dict.fromkeys(ids))
            per_tier[tier.value] = len(ids)

        return {
            "total": total,
            "per_tier": per_tier,
        }

    # ========= 晋升 / 遗忘（Consolidation） =========

    async def promote_memory(self, item: MemoryItem, target_tier: MemoryTier) -> MemoryItem:
        """
        把某条记忆从当前 tier 晋升到更高层。
        """
        now = datetime.utcnow()
        old_tier = item.memory_tier
        if old_tier == target_tier:
            return item

        item.memory_tier = target_tier
        item.last_promoted_at = now
        item.updated_at = now
        await self._save_item(item)
        await self._reindex_item_tier(item.memory_id, old_tier, target_tier)

        self.tracer.span(
            "L2",
            "memory_promote",
            module="memory_fabric",
            input={
                "memory_id": item.memory_id,
                "from": old_tier.value,
                "to": target_tier.value,
            },
        )
        return item

    async def decay_memory(self, item: MemoryItem) -> None:
        """
        对某条记忆执行“遗忘”：当前实现为直接删除。
        """
        now = datetime.utcnow()
        item.last_decay_at = now
        await self._delete_item(item.memory_id)

        self.tracer.span(
            "L2",
            "memory_decay",
            module="memory_fabric",
            input={"memory_id": item.memory_id, "tier": item.memory_tier.value},
        )

    async def run_consolidation(self, max_items_per_tier: int = 1000) -> Dict[str, Any]:
        """
        记忆合并 / 晋升 / 遗忘的核心逻辑：

        简化版规则（你以后可以根据论文/实验再调参）：
        - STS:
            - 如果 access_count >= 3 或 age > 1 小时 -> 晋升到 MTS
            - 如果 age > 24 小时 且 access_count == 0 -> 遗忘
        - MTS:
            - 如果 access_count >= 5 或 age > 7 天 -> 晋升到 LTS
            - 如果 age > 90 天 且 access_count < 2 -> 遗忘
        - LTS:
            - 暂不自动遗忘（只通过 supersedes 替换）
        """
        now = datetime.utcnow()
        promoted: Dict[str, int] = {"STS->MTS": 0, "MTS->LTS": 0}
        decayed: Dict[str, int] = {"STS": 0, "MTS": 0, "LTS": 0}

        # 处理 STS
        sts_items = await self.sample_items_by_tier(MemoryTier.STS, limit=max_items_per_tier)
        for item in sts_items:
            age = (now - item.created_at).total_seconds()
            if item.access_count >= 3 or age > 3600:
                await self.promote_memory(item, MemoryTier.MTS)
                promoted["STS->MTS"] += 1
            elif age > 86400 and item.access_count == 0:
                await self.decay_memory(item)
                decayed["STS"] += 1

        # 处理 MTS
        mts_items = await self.sample_items_by_tier(MemoryTier.MTS, limit=max_items_per_tier)
        for item in mts_items:
            age_days = (now - item.created_at).days
            if item.access_count >= 5 or age_days > 7:
                await self.promote_memory(item, MemoryTier.LTS)
                promoted["MTS->LTS"] += 1
            elif age_days > 90 and item.access_count < 2:
                await self.decay_memory(item)
                decayed["MTS"] += 1

        # LTS 目前只统计，不做自动遗忘
        lts_items = await self.sample_items_by_tier(MemoryTier.LTS, limit=max_items_per_tier)
        decayed["LTS"] += 0  # 占位，方便以后扩展

        result = {
            "promoted": promoted,
            "decayed": decayed,
            "snapshot": await self.get_stats(),
        }

        self.tracer.span(
            "L2",
            "memory_consolidation_run",
            module="memory_fabric",
            output=result,
        )
        return result
这就是蓝皮书里说的 “Sleep Loop 输出之一：记忆压缩与升级” 在工程上的第一版实现。

def memory_consolidate_cmd(limit_per_tier: int):

    手动触发一次 MemoryFabric 的合并/晋升/遗忘逻辑。
