# quentin/llm/ollama_client.py
from __future__ import annotations

import logging
from dataclasses import dataclass
from typing import Dict, Any, Optional

import requests  # 需要 pip install requests

logger = logging.getLogger(__name__)


@dataclass
class OllamaClientConfig:
    base_url: str = "http://localhost:11434"
    timeout_sec: int = 120
    default_temperature: float = 0.3


class OllamaClient:
    """
    一个最小可用的 Ollama HTTP 客户端。

    只实现一个 generate()：
    - 不做流式，stream=false
    - 返回字符串结果（response 字段）
    """

    def __init__(self, config: Optional[OllamaClientConfig] = None) -> None:
        self.config = config or OllamaClientConfig()
        self._base_generate_url = self.config.base_url.rstrip("/") + "/api/generate"

    def generate(
        self,
        model: str,
        prompt: str,
        *,
        system: Optional[str] = None,
        temperature: Optional[float] = None,
        extra_params: Optional[Dict[str, Any]] = None,
    ) -> str:
        temperature = (
            self.config.default_temperature if temperature is None else temperature
        )
        payload: Dict[str, Any] = {
            "model": model,
            "prompt": prompt,
            "stream": False,
            "options": {
                "temperature": temperature,
            },
        }
        if system:
            # Ollama 支持在 prompt 里直接拼 system，也可以用 messages 格式
            # 这里先简单拼接
            payload["prompt"] = f"<<SYS>>\n{system}\n<</SYS>>\n\n{prompt}"

        if extra_params:
            payload.update(extra_params)

        try:
            resp = requests.post(
                self._base_generate_url,
                json=payload,
                timeout=self.config.timeout_sec,
            )
            resp.raise_for_status()
            data = resp.json()
        except Exception:
            logger.exception("Ollama generate failed for model=%s", model)
            raise

        # /api/generate 的非流式返回里有 "response" 字段
        text = data.get("response", "")
        return text

# quentin/llm/multi_council.py
from __future__ import annotations

import logging
from dataclasses import dataclass, asdict
from typing import Dict, Any, List, Optional

from quentin.llm.types import LLMConfig
from quentin.llm.ollama_client import OllamaClient

logger = logging.getLogger(__name__)


@dataclass
class CouncilMemberOpinion:
    model: str
    role: str
    answer: str
    notes: str = ""  # 可以记录“偏向速度”、“偏向安全”等描述

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class CouncilDecision:
    final_answer: str
    reasoning: str
    opinions: List[CouncilMemberOpinion]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "final_answer": self.final_answer,
            "reasoning": self.reasoning,
            "opinions": [o.to_dict() for o in self.opinions],
        }


class MultiAICouncil:
    """
    多模型“开会”模式 v1：

    流程：
      1. 每个 LLM 成员单独给出自己的 proposal。
      2. 选一个模型做“主持人/整合者”，看到所有 proposal 后给出综合结论。
    """

    def __init__(
        self,
        members: List[LLMConfig],
        ollama_client: Optional[OllamaClient] = None,
    ) -> None:
        if not members:
            raise ValueError("MultiAICouncil requires at least one LLM member.")
        self.members = members
        self.client = ollama_client or OllamaClient()

    # ---------- 主 API ----------

    def deliberate(
        self,
        *,
        question: str,
        context: Optional[str] = None,
        mode: str = "general",
    ) -> CouncilDecision:
        """
        - question : 要解决的核心问题（如“如何修这个代码？”）
        - context  : 背景（蓝皮书 requirement、现有代码路径等）
        - mode     : "plan" / "reflection" / "coding" 等，用于提示角色
        """
        # 1) 每个成员给出意见
        opinions: List[CouncilMemberOpinion] = []
        for m in self.members:
            if m.provider != "ollama":
                raise ValueError(f"Unsupported provider: {m.provider}")

            prompt = self._build_member_prompt(
                role=m.role,
                mode=mode,
                question=question,
                context=context,
            )
            answer = self.client.generate(model=m.model, prompt=prompt)
            opinions.append(
                CouncilMemberOpinion(
                    model=m.model,
                    role=m.role,
                    answer=answer,
                    notes="",
                )
            )

        # 2) 选一个“主持人”，整合大家意见
        aggregator = self.members[0]
        agg_prompt = self._build_aggregator_prompt(
            mode=mode,
            question=question,
            context=context,
            opinions=opinions,
        )
        final_answer = self.client.generate(model=aggregator.model, prompt=agg_prompt)

        reasoning = (
            "Aggregated by model "
            f"{aggregator.model} in role '{aggregator.role}' "
            "based on the above member opinions."
        )

        return CouncilDecision(
            final_answer=final_answer,
            reasoning=reasoning,
            opinions=opinions,
        )

    # ---------- Prompt 构造 ----------

    def _build_member_prompt(
        self,
        *,
        role: str,
        mode: str,
        question: str,
        context: Optional[str],
    ) -> str:
        ctx = context or ""
        return (
            f"You are an AI committee member with role '{role}'.\n"
            f"Task mode: {mode}.\n\n"
            "Question:\n"
            f"{question}\n\n"
            "Context (if any):\n"
            f"{ctx}\n\n"
            "Please provide your best proposal. "
            "Focus on clarity, concrete steps, and avoid vague language. "
            "Output plain text, no markdown unless needed for code blocks."
        )

    def _build_aggregator_prompt(
        self,
        *,
        mode: str,
        question: str,
        context: Optional[str],
        opinions: List[CouncilMemberOpinion],
    ) -> str:
        ctx = context or ""
        opinions_text = ""
        for idx, op in enumerate(opinions, start=1):
            opinions_text += (
                f"--- Opinion {idx} ---\n"
                f"model: {op.model}\n"
                f"role : {op.role}\n"
                f"text : {op.answer}\n\n"
            )

        return (
            "You are the chair of an AI committee.\n"
            f"Task mode: {mode}.\n\n"
            "You are given a question, some context, and multiple member opinions. "
            "Your job is to:\n"
            "1) Compare the opinions.\n"
            "2) Resolve contradictions.\n"
            "3) Produce a single, coherent final answer.\n"
            "4) If appropriate, mention which ideas you adopted from which opinions.\n\n"
            "Question:\n"
            f"{question}\n\n"
            "Context:\n"
            f"{ctx}\n\n"
            "Member opinions:\n"
            f"{opinions_text}\n"
            "Now write your final consolidated answer. "
            "Be specific and practical. "
            "If the task is about CODE, prefer to output a single code block or patch."
        )

import json

logger = logging.getLogger(__name__)

PatchStatus = Literal["proposed", "evaluated", "accepted", "rejected", "applied", "superseded"]
PatchType = Literal["diff", "file_replace", "snippet", "other"]


@dataclass
class CodePatch:
    """
    自编程引擎生成的“补丁”抽象，暂时只是结构定义：

    - patch_id       : 唯一 ID
    - task_id        : 来源的 ImprovementTask ID（如 "bp::memory_fabric::mf.storage"）
    - capability     : 对应 capability_id（如 "memory_fabric"）
    - requirement_id : 对应 requirement_id（如 "mf.storage"）

    - target_module  : 目标 Python 模块名（如 "quentin.memory.store"）
    - target_file    : 目标文件相对路径（如 "quentin/memory/store.py"）

    - patch_type     : "diff" / "file_replace" / "snippet" / ...
    - patch_content  : 补丁内容（可以是 unified diff / 新文件源码 / 片段）

    - created_by     : 谁生成的（如 "MultiAICouncil"）
    - created_at     : ISO 时间
    - updated_at     : ISO 时间
    - status         : 当前状态（proposed / evaluated / accepted / ...）

    - metadata       : 额外信息（比如参与的模型、会议记录等）
    - eval_summary   : 测试/评估结果摘要（由 EvalHarness 填充）
    """
    patch_id: str
    task_id: str
    capability: str
    requirement_id: str

    target_module: str
    target_file: str

    patch_type: PatchType
    patch_content: str

    created_by: str = "unknown"
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    status: PatchStatus = "proposed"

    metadata: Dict[str, Any] = field(default_factory=dict)
    eval_summary: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "CodePatch":
        return cls(
            patch_id=data["patch_id"],
            task_id=data["task_id"],
            capability=data.get("capability", ""),
            requirement_id=data.get("requirement_id", ""),
            target_module=data.get("target_module", ""),
            target_file=data.get("target_file", ""),
            patch_type=data.get("patch_type", "other"),
            patch_content=data.get("patch_content", ""),
            created_by=data.get("created_by", "unknown"),
            created_at=data.get("created_at", ""),
            updated_at=data.get("updated_at", ""),
            status=data.get("status", "proposed"),
            metadata=data.get("metadata", {}) or {},
            eval_summary=data.get("eval_summary", {}) or {},
        )

class FileBackedPatchStore:
    """
    最简单的 Patch 存储：一个 JSON 文件 {patch_id: CodePatchDict}
    """

    def __init__(self, path: str | Path = "data/code_patches.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._cache: Dict[str, Dict[str, Any]] = {}
        self._loaded = False

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                raw = json.loads(self.path.read_text(encoding="utf-8") or "{}")
                self._cache = raw
            except Exception:
                logger.exception("Failed to load patches from %s", self.path)
                self._cache = {}
        else:
            self._cache = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.path.write_text(
            json.dumps(self._cache, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    # ---------- CRUD ----------

    def get(self, patch_id: str) -> Optional[CodePatch]:
        self._load()
        d = self._cache.get(patch_id)
        return CodePatch.from_dict(d) if d else None

    def upsert(self, patch: CodePatch) -> None:
        self._load()
        patch.updated_at = datetime.utcnow().isoformat()
        self._cache[patch.patch_id] = patch.to_dict()
        self._save()

    def delete(self, patch_id: str) -> None:
        self._load()
        if patch_id in self._cache:
            del self._cache[patch_id]
            self._save()

    def list_all(self) -> List[CodePatch]:
        self._load()
        return [CodePatch.from_dict(d) for d in self._cache.values()]

    def list_by_status(self, status: PatchStatus) -> List[CodePatch]:
        return [p for p in self.list_all() if p.status == status]

import logging
import subprocess
import time

from pathlib import Path

from quentin.self_programming.patches import CodePatch

@dataclass
class PatchEvalMetrics:
    """
    一次 Patch 评估的结果摘要：

    - success      : 测试是否通过
    - test_command : 运行的测试命令（如 "pytest"）
    - exit_code    : 进程退出码
    - duration_sec : 耗时
    - stdout_tail  : stdout 末尾截断（防止太长）
    - stderr_tail  : stderr 末尾截断
    - notes        : 额外说明
    """
    success: bool
    test_command: str
    exit_code: int
    duration_sec: float
    stdout_tail: str
    stderr_tail: str
    notes: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class EvalHarness:
    """
    v1：非常简单的 Patch 测试壳子。

    设计原则：
    - 不负责实际“应用 patch”到代码文件中（那是另一个模块的职责）
    - 假设调用方已经在某个工作目录里应用了 patch
    - EvalHarness 只负责：
        1) 在 repo_root 下执行 test_command（默认 pytest）
        2) 收集 exit_code / stdout / stderr 等信息
    """

    def __init__(
        self,
        repo_root: str | Path = ".",
        test_command: str = "pytest",
        max_output_chars: int = 4000,
    ) -> None:
        self.repo_root = Path(repo_root)
        self.test_command = test_command
        self.max_output_chars = max_output_chars

    def evaluate_patch(
        self,
        patch: CodePatch,
        *,
        apply_notes: Optional[str] = None,
    ) -> PatchEvalMetrics:
        """
        说明：
        - 假定 patch 已经被应用到 self.repo_root 所指目录下
        - 这里只是运行 test_command。未来可以扩展为多阶段测试（静态分析 + 单测 + 回归等）。
        """
        cmd = self.test_command
        t0 = time.time()
        try:
            proc = subprocess.run(
                cmd,
                cwd=str(self.repo_root),
                shell=True,
                capture_output=True,
                text=True,
            )
            duration = time.time() - t0
        except Exception as e:
            logger.exception("Failed to run tests for patch %s", patch.patch_id)
            return PatchEvalMetrics(
                success=False,
                test_command=cmd,
                exit_code=-1,
                duration_sec=time.time() - t0,
                stdout_tail="",
                stderr_tail=f"Exception when running tests: {e}",
                notes="test_command failed to execute",
            )

        stdout = proc.stdout or ""
        stderr = proc.stderr or ""

        def _tail(s: str) -> str:
            if len(s) <= self.max_output_chars:
                return s
            return s[-self.max_output_chars :]

        success = proc.returncode == 0

        metrics = PatchEvalMetrics(
            success=success,
            test_command=cmd,
            exit_code=proc.returncode,
            duration_sec=duration,
            stdout_tail=_tail(stdout),
            stderr_tail=_tail(stderr),
            notes=apply_notes or "",
        )
        return metrics

# quentin/self_programming/engine.py
from __future__ import annotations

import logging
import uuid
from dataclasses import asdict
from typing import Dict, Any, List, Optional

from quentin.blueprint.gap_tracker import BlueprintGapTracker, BlueprintGap
from quentin.evaluation.harness import EvalHarness
from quentin.llm.multi_council import MultiAICouncil, CouncilDecision
from quentin.llm.types import LLMConfig
from quentin.self_improvement.manager import SelfImprovementManager
from quentin.self_improvement.schemas import ImprovementTask
from quentin.self_programming.patches import CodePatch, FileBackedPatchStore

logger = logging.getLogger(__name__)


class SelfProgrammingEngine:
    """
    自编程引擎 v1 骨架：

    - 输入：一个 ImprovementTask（通常来自蓝皮书的 gap）
    - 流程：
        1) 找到对应的 BlueprintGap（拿到 impl.module / impl.symbol）
        2) 用 MultiAICouncil 开会，生成 patch 文本（可以是 diff/snippet）
        3) 把 patch 存入 FileBackedPatchStore
        4) 可选：调用 EvalHarness 运行测试，并更新 patch.eval_summary
    """

    def __init__(
        self,
        *,
        blueprint_tracker: Optional[BlueprintGapTracker] = None,
        si_manager: Optional[SelfImprovementManager] = None,
        patch_store: Optional[FileBackedPatchStore] = None,
        eval_harness: Optional[EvalHarness] = None,
        council: Optional[MultiAICouncil] = None,
    ) -> None:
        self.blueprint_tracker = blueprint_tracker or BlueprintGapTracker()
        self.si_manager = si_manager or SelfImprovementManager()
        self.patch_store = patch_store or FileBackedPatchStore()
        self.eval_harness = eval_harness or EvalHarness()

        if council is None:
            # 默认给一个简单的委员会配置：你可以在实际运行时换成自己的模型组合
            default_members = [
                LLMConfig(provider="ollama", model="qwen3-vl:latest", role="planner"),
                LLMConfig(provider="ollama", model="deepseek-r1:1.5b", role="critic"),
            ]
            self.council = MultiAICouncil(members=default_members)
        else:
            self.council = council

    # ---------- 工具：找到与 ImprovementTask 对应的 BlueprintGap ----------

    def _find_gap_for_task(self, task: ImprovementTask) -> Optional[BlueprintGap]:
        gaps = self.blueprint_tracker.scan()
        for g in gaps:
            if (
                getattr(task, "capability", None) == g.capability_id
                and getattr(task, "requirement_id", None) == g.requirement_id
            ):
                return g
        return None

    # ---------- 生成 patch ----------

    def propose_patches_for_task(
        self,
        task_id: str,
        *,
        num_candidates: int = 1,
    ) -> List[CodePatch]:
        """
        面向 CLI 的入口：
        - 给定一个 ImprovementTask ID
        - 使用 MultiAICouncil 生成若干个 patch 提案
        """
        task = self.si_manager.get_task(task_id)
        if not task:
            raise ValueError(f"ImprovementTask not found: {task_id}")

        gap = self._find_gap_for_task(task)
        if not gap:
            logger.warning(
                "No BlueprintGap found for task_id=%s (cap=%s, req=%s)",
                task.task_id,
                getattr(task, "capability", None),
                getattr(task, "requirement_id", None),
            )

        patches: List[CodePatch] = []
        for _ in range(max(1, num_candidates)):
            patch = self._propose_single_patch(task, gap)
            self.patch_store.upsert(patch)
            patches.append(patch)

        return patches

    def _propose_single_patch(
        self,
        task: ImprovementTask,
        gap: Optional[BlueprintGap],
    ) -> CodePatch:
        """
        真正调用 MultiAICouncil 的地方。
        现在只做“生成 patch 文本并存储”，不直接应用到文件系统。
        """
        capability = getattr(task, "capability", "")
        requirement_id = getattr(task, "requirement_id", "")
        desc = getattr(task, "description", "")

        impl_info = gap.impl if gap else {}
        target_module = impl_info.get("module", "")
        target_symbol = impl_info.get("symbol", "")

        question = (
            "You are part of a self-improving AI agent system. "
            "Your job is to propose a CODE PATCH that helps implement or improve "
            "the specified requirement in the codebase.\n\n"
            f"Requirement (from blueprint):\n{desc}\n\n"
            f"Target module (if provided): {target_module}\n"
            f"Target symbol (if provided): {target_symbol}\n\n"
            "Please propose a concrete code change that moves the system closer "
            "to satisfying this requirement."
        )

        context = (
            "You are working inside a Python project named 'quentin'. "
            "The codebase is organized roughly as:\n"
            "- quentin/memory/*   : Memory Fabric & Sleep Loop\n"
            "- quentin/core/*     : Cognitive loop, Task, Trace\n"
            "- quentin/self_*/*   : Self-improvement and self-programming modules\n"
            "- quentin/cli/qctl.py: Command line entry point\n\n"
            "Constraints:\n"
            "- Prefer minimal, incremental changes.\n"
            "- Preserve existing public APIs where possible.\n"
            "- If you output a diff, use unified diff format against the target file.\n"
            "- If you output full file contents, clearly indicate file path and content.\n"
            "- Avoid changing kernel/safety-related modules.\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="coding",
        )

        patch_id = str(uuid.uuid4())
        target_file = self._guess_target_file_from_module(target_module)

        metadata = {
            "council_reasoning": decision.reasoning,
            "member_opinions": [asdict(op) for op in decision.opinions],
            "gap": asdict(gap) if gap else None,
        }

        patch = CodePatch(
            patch_id=patch_id,
            task_id=task.task_id,
            capability=capability,
            requirement_id=requirement_id,
            target_module=target_module,
            target_file=target_file,
            patch_type="other",  # 先标记为 "other"，真正类型可以以后自动识别
            patch_content=decision.final_answer,
            created_by="SelfProgrammingEngine.MultiAICouncil",
            metadata=metadata,
        )
        return patch

    def _guess_target_file_from_module(self, module_name: str) -> str:
        """
        简单从模块名猜文件路径：
        e.g. 'quentin.memory.store' -> 'quentin/memory/store.py'
        如果拿不到模块名，则返回空字符串，让后续流程/人来补。
        """
        if not module_name:
            return ""
        parts = module_name.split(".")
        return "/".join(parts) + ".py"

    # ---------- 对 patch 做评估 ----------

    def evaluate_patch(self, patch_id: str) -> CodePatch:
        """
        v1：假设 patch 已经由人工或其他工具应用到了 repo_root。

        TODO（未来版本）：
        - 提供自动 apply/rollback 机制
        """
        patch = self.patch_store.get(patch_id)
        if not patch:
            raise ValueError(f"Patch not found: {patch_id}")

        metrics = self.eval_harness.evaluate_patch(
            patch,
            apply_notes="Assumes patch was applied manually or by another tool.",
        )
        patch.eval_summary = metrics.to_dict()
        patch.status = "evaluated" if metrics.success else "rejected"
        self.patch_store.upsert(patch)
        return patch
