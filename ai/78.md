# quentin/blueprint/tracker.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging
from datetime import datetime

from quentin.core.traces import ExecutionTraceStore
from quentin.research.experiments import ResearchManager

logger = logging.getLogger(__name__)


# ---------- 数据结构 ----------

@dataclass
class BlueprintItemSpec:
    id: str
    name: str
    layer: str
    phase: str
    category: str
    description: str
    required_capabilities: List[str] = field(default_factory=list)
    hints: Dict[str, Any] = field(default_factory=dict)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "BlueprintItemSpec":
        return cls(
            id=data["id"],
            name=data.get("name", data["id"]),
            layer=data.get("layer", ""),
            phase=data.get("phase", ""),
            category=data.get("category", ""),
            description=data.get("description", ""),
            required_capabilities=list(data.get("required_capabilities") or []),
            hints=data.get("hints") or {},
        )

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class BlueprintItemStatus:
    """
    一条蓝皮书条目的当前状态快照：
    - file_implemented : 对应模块文件是否存在
    - has_traces       : ExecutionTrace 中是否出现过该 capability_id
    - has_experiments  : ResearchManager 中是否有对应 experiment
    - score            : 一个 0~1 的粗略完成度（v1 简单平均）
    """

    spec: BlueprintItemSpec
    file_implemented: bool
    has_traces: bool
    has_experiments: bool
    score: float

    def to_dict(self) -> Dict[str, Any]:
        d = self.spec.to_dict()
        d.update(
            {
                "file_implemented": self.file_implemented,
                "has_traces": self.has_traces,
                "has_experiments": self.has_experiments,
                "score": self.score,
            }
        )
        return d


@dataclass
class BlueprintScanReport:
    version: str
    scanned_at: str
    total_items: int
    avg_score: float
    items: List[BlueprintItemStatus]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "version": self.version,
            "scanned_at": self.scanned_at,
            "total_items": self.total_items,
            "avg_score": self.avg_score,
            "items": [it.to_dict() for it in self.items],
        }


# ---------- Tracker 主体 ----------

class BlueprintGapTracker:
    """
    BlueprintGapTracker v1:

    - 读取 blueprint/spec_v3.json
    - 对每个 item 做 3 个维度的静态检查：
        1) 对应 modules 文件是否存在（粗略视为“已经动工”）
        2) ExecutionTrace 是否有该 capability 的运行记录（视为“跑通过”）
        3) ResearchManager 是否有该 capability 的实验记录（视为“有指标监控”）
    - 根据 3 项给出一个 0~1 的 completion score：
        score = (file_implemented + has_traces + has_experiments) / 3.0

    这只是 v1 的“工程可见度雷达图”，以后可以接入多 AI 评审：
    - 判断“实现质量是否达标”
    - 自动写 TODO 文本 / issue
    - 驱动 SelfProgrammingEngine 去补齐缺口
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        spec_path: str | Path = "blueprint/spec_v3.json",
        traces_store: Optional[ExecutionTraceStore] = None,
        research_manager: Optional[ResearchManager] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.spec_path = self.repo_root / spec_path
        self.traces_store = traces_store or ExecutionTraceStore()
        self.research_manager = research_manager or ResearchManager()

        self._items: List[BlueprintItemSpec] = []
        self._version: str = "unknown"

        self._load_spec()

    # ---------- Public API ----------

    def scan(self) -> BlueprintScanReport:
        statuses: List[BlueprintItemStatus] = []

        for item in self._items:
            file_implemented = self._check_files(item)
            has_traces = self._check_traces(item)
            has_experiments = self._check_experiments(item)

            score = (int(file_implemented) + int(has_traces) + int(has_experiments)) / 3.0

            statuses.append(
                BlueprintItemStatus(
                    spec=item,
                    file_implemented=file_implemented,
                    has_traces=has_traces,
                    has_experiments=has_experiments,
                    score=score,
                )
            )

        avg_score = 0.0
        if statuses:
            avg_score = sum(s.score for s in statuses) / len(statuses)

        return BlueprintScanReport(
            version=self._version,
            scanned_at=datetime.utcnow().isoformat(),
            total_items=len(statuses),
            avg_score=avg_score,
            items=statuses,
        )

    def find_item(self, item_id: str) -> Optional[BlueprintItemStatus]:
        """
        简易查询：扫描一次后按 id 找状态。
        """
        report = self.scan()
        for it in report.items:
            if it.spec.id == item_id:
                return it
        return None

    # ---------- Internal: spec ----------

    def _load_spec(self) -> None:
        if not self.spec_path.exists():
            logger.warning("BlueprintGapTracker: spec file not found: %s", self.spec_path)
            self._items = []
            self._version = "missing"
            return

        try:
            raw = json.loads(self.spec_path.read_text(encoding="utf-8") or "{}")
        except Exception:
            logger.exception("BlueprintGapTracker: failed to read spec: %s", self.spec_path)
            raw = {}

        self._version = str(raw.get("version", "unknown"))
        items_raw = raw.get("items") or []
        self._items = [BlueprintItemSpec.from_dict(x) for x in items_raw]

    # ---------- Internal: checks ----------

    def _check_files(self, item: BlueprintItemSpec) -> bool:
        hints = item.hints or {}
        modules = hints.get("modules") or []
        for rel in modules:
            path = (self.repo_root / rel).resolve()
            if path.exists():
                return True
        return False

    def _check_traces(self, item: BlueprintItemSpec) -> bool:
        hints = item.hints or {}
        cap_id = hints.get("trace_capability_id")
        if not cap_id:
            return False

        # ExecutionTraceStore v1 假设有 .iter_traces() 生成器
        try:
            for tr in self.traces_store.iter_traces(limit=2000):
                if tr.blueprint_capability_id == cap_id:
                    return True
        except Exception:
            logger.exception("BlueprintGapTracker._check_traces: error while scanning traces")
        return False

    def _check_experiments(self, item: BlueprintItemSpec) -> bool:
        hints = item.hints or {}
        cap = hints.get("experiment_capability")
        if not cap:
            return False

        try:
            exps = self.research_manager.list_experiments()
        except Exception:
            logger.exception("BlueprintGapTracker._check_experiments: failed to list experiments")
            return False

        for e in exps:
            if e.capability == cap:
                return True
        return False
✅ 注意：

这里假设你之前已经有 ExecutionTraceStore.iter_traces() 以及 ResearchManager.list_experiments()（我们之前设计过骨架）。如果还没有，后面可以补。

def cmd_blueprint_scan(args: argparse.Namespace) -> None:
    tracker = BlueprintGapTracker()
    report = tracker.scan()

    print(f"Blueprint version : {report.version}")
    print(f"Scanned at        : {report.scanned_at}")
    print(f"Total items       : {report.total_items}")
    print(f"Average score     : {report.avg_score:.3f}")
    print("")
    print("Items:")
    for it in report.items:
        status_flag = "✅" if it.score >= 0.9 else ("⚠️" if it.score > 0.0 else "❌")
        print(
            f"{status_flag} {it.spec.id:<30} "
            f"layer={it.spec.layer:<18} "
            f"phase={it.spec.phase:<24} "
            f"score={it.score:.2f} "
            f"[files={it.file_implemented}, traces={it.has_traces}, exp={it.has_experiments}]"
        )


def cmd_blueprint_item(args: argparse.Namespace) -> None:
    tracker = BlueprintGapTracker()
    status = tracker.find_item(args.item_id)
    if not status:
        print(f"No blueprint item found with id: {args.item_id}")
        return

    it = status
    print(f"ID         : {it.spec.id}")
    print(f"Name       : {it.spec.name}")
    print(f"Layer      : {it.spec.layer}")
    print(f"Phase      : {it.spec.phase}")
    print(f"Category   : {it.spec.category}")
    print(f"Score      : {it.score:.2f}")
    print(f"Files      : {it.file_implemented}")
    print(f"Traces     : {it.has_traces}")
    print(f"Experiments: {it.has_experiments}")
    print("")
    print("Description:")
    print(it.spec.description)
    print("")
    print("Required capabilities:")
    for cap in it.spec.required_capabilities:
        print(f"- {cap}")
    print("")
    print("Hints:")
    for k, v in (it.spec.hints or {}).items():
        print(f"- {k}: {v}")
