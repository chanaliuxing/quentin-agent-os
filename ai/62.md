# quentin/skills/models.py
from __future__ import annotations

from dataclasses import dataclass, field, asdict
from datetime import datetime
from typing import Dict, Any, List, Optional


@dataclass
class SkillStats:
    """
    蓝皮书里提到的 performance_stats 落地：
      - 调用次数 / 成功次数
      - 最近调用时间
      - 最近几次调用的结果（可选）
    """
    call_count: int = 0
    success_count: int = 0
    last_called_at: Optional[str] = None
    last_success_at: Optional[str] = None

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "SkillStats":
        if d is None:
            return cls()
        return cls(
            call_count=int(d.get("call_count", 0)),
            success_count=int(d.get("success_count", 0)),
            last_called_at=d.get("last_called_at"),
            last_success_at=d.get("last_success_at"),
        )


@dataclass
class SkillDefinition:
    """
    稳定的技能定义（被真正“注册”后使用）：

    - skill_id      : 唯一 ID
    - name          : 简短可读的名字
    - domain        : 使用场景（"general" / "desktop" / "legal" / "trading" / ...）
    - description   : 自然语言描述
    - input_schema  : 输入字段说明（纯 metadata，后续可扩展成 JSON Schema）
    - output_schema : 输出字段说明
    - code          : 代码实现（Python 函数草稿 / 伪代码均可）
    - tests         : 测试用例（文本或简单结构）
    - tags          : 标签（["logging", "retry", ...]）

    - created_from_task_id : 来源 Task（如果是从 Trace 挖掘出来）
    - supersedes           : 替代了哪些旧 skill_id
    - stats                : performance_stats
    """
    skill_id: str
    name: str
    domain: str
    description: str
    input_schema: Dict[str, Any] = field(default_factory=dict)
    output_schema: Dict[str, Any] = field(default_factory=dict)
    code: str = ""
    tests: str = ""
    tags: List[str] = field(default_factory=list)

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    created_from_task_id: Optional[str] = None
    supersedes: List[str] = field(default_factory=list)

    stats: SkillStats = field(default_factory=SkillStats)
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        d["stats"] = self.stats.to_dict()
        return d

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "SkillDefinition":
        return cls(
            skill_id=d["skill_id"],
            name=d.get("name", ""),
            domain=d.get("domain", "general"),
            description=d.get("description", ""),
            input_schema=d.get("input_schema", {}) or {},
            output_schema=d.get("output_schema", {}) or {},
            code=d.get("code", ""),
            tests=d.get("tests", ""),
            tags=list(d.get("tags", []) or []),
            created_at=d.get("created_at", ""),
            created_from_task_id=d.get("created_from_task_id"),
            supersedes=list(d.get("supersedes", []) or []),
            stats=SkillStats.from_dict(d.get("stats") or {}),
            metadata=d.get("metadata", {}) or {},
        )


@dataclass

    - code         : 建议代码实现（可以是伪代码）

# quentin/skills/registry.py
from __future__ import annotations

import json
import logging
from pathlib import Path
from typing import Dict, Any, List, Optional

from quentin.skills.models import SkillDefinition, SkillCandidate, SkillStats

logger = logging.getLogger(__name__)


class FileBackedSkillRegistry:
    """
    Skill & SkillCandidate 的文件存储：

    结构：
      {
        "skills": {
          "<skill_id>": { ...SkillDefinition... },
          ...
        },
        "candidates": {
          "<candidate_id>": { ...SkillCandidate... },
          ...
        }
      }
    """

    def __init__(self, path: str | Path = "data/skills.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._loaded = False
        self._skills: Dict[str, SkillDefinition] = {}
        self._candidates: Dict[str, SkillCandidate] = {}

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                raw = json.loads(self.path.read_text(encoding="utf-8") or "{}")
                skills_raw = raw.get("skills", {}) or {}
                cands_raw = raw.get("candidates", {}) or {}
                self._skills = {
                    sid: SkillDefinition.from_dict(sd) for sid, sd in skills_raw.items()
                }
                self._candidates = {
                    cid: SkillCandidate.from_dict(cd)
                    for cid, cd in cands_raw.items()
                }
            except Exception:
                logger.exception("SkillRegistry: failed to load, using empty state.")
                self._skills = {}
                self._candidates = {}
        else:
            self._skills = {}
            self._candidates = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        raw = {
            "skills": {sid: s.to_dict() for sid, s in self._skills.items()},
            "candidates": {cid: c.to_dict() for cid, c in self._candidates.items()},
        }
        self.path.write_text(
            json.dumps(raw, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    # ---------- SkillDefinition CRUD ----------

    def list_skills(self) -> List[SkillDefinition]:
        self._load()
        return list(self._skills.values())

    def get_skill(self, skill_id: str) -> Optional[SkillDefinition]:
        self._load()
        return self._skills.get(skill_id)

    def upsert_skill(self, skill: SkillDefinition) -> None:
        self._load()
        self._skills[skill.skill_id] = skill
        self._save()

    def delete_skill(self, skill_id: str) -> None:
        self._load()
        if skill_id in self._skills:
            del self._skills[skill_id]
            self._save()

    # ---------- SkillCandidate CRUD ----------

    def list_candidates(self) -> List[SkillCandidate]:
        self._load()
        return list(self._candidates.values())

    def get_candidate(self, candidate_id: str) -> Optional[SkillCandidate]:
        self._load()
        return self._candidates.get(candidate_id)

    def upsert_candidate(self, cand: SkillCandidate) -> None:
        self._load()
        self._candidates[cand.candidate_id] = cand
        self._save()

    def delete_candidate(self, candidate_id: str) -> None:
        self._load()
        if candidate_id in self._candidates:
            del self._candidates[candidate_id]
            self._save()

    # ---------- stats 更新（以后给 Tool & Skill Manager 用） ----------

    def record_skill_call(self, skill_id: str, success: bool) -> None:
        skill = self.get_skill(skill_id)
        if not skill:
            return
        stats: SkillStats = skill.stats
        stats.call_count += 1
        stats.last_called_at = datetime.utcnow().isoformat()
        if success:
            stats.success_count += 1
            stats.last_success_at = stats.last_called_at
        skill.stats = stats
        self.upsert_skill(skill)

三、SkillMiner：从 Task + Trace + Lessons 自动挖技能候选
新文件：quentin/skills/miner.py

# quentin/skills/miner.py
from __future__ import annotations

import logging
import uuid
from typing import List, Optional

from quentin.core.task import TaskSpec, TaskResult
from quentin.core.trace import TraceEvent
from quentin.core.critic import CriticFeedback
from quentin.llm.multi_council import MultiAICouncil, CouncilDecision
from quentin.llm.types import LLMConfig
from quentin.skills.models import SkillCandidate
from quentin.skills.registry import FileBackedSkillRegistry

logger = logging.getLogger(__name__)


class SkillMiner:
    """
    Skill & Code Evolution Loop 的“前半段”：从成功 Trace 中挖“可复用技能”。

    v1 策略：
      - 只在任务成功 & critic_confidence 高时才考虑挖技能；
      - 把 Task goal / plan / lessons / Trace 压成上下文；
      - 让 MultiAICouncil 生成一个 SkillCandidate（name + code + tests）；
      - 写入 SkillRegistry 的 candidates 区域，等后续人工/AI 审核再注册为正式 Skill。

    未来扩展：
      - 用 ResearchManager 的统计先过滤高价值任务；
      - 对同一类型任务做 pattern mining（非单次任务）；
      - 自动评估 candidate 的效果后，自动晋升为正式 SkillDefinition。
    """

    def __init__(
        self,
        *,
        registry: Optional[FileBackedSkillRegistry] = None,
        council: Optional[MultiAICouncil] = None,
    ) -> None:
        self.registry = registry or FileBackedSkillRegistry()

        if council is None:
            default_members = [
                LLMConfig(provider="ollama", model="qwen3-vl:latest", role="designer"),
                LLMConfig(provider="ollama", model="deepseek-r1:1.5b", role="critic"),
            ]
            council = MultiAICouncil(members=default_members)
        self.council = council

    def mine_from_task(
        self,
        *,
        task: TaskSpec,
        result: TaskResult,
        trace_events: List[TraceEvent],
        feedback: Optional[CriticFeedback] = None,
    ) -> Optional[SkillCandidate]:
        if not result.success:
            logger.info(
                "SkillMiner: skip task_id=%s because result.success=False",
                task.task_id,
            )
            return None

        if feedback and feedback.confidence < 0.7:
            logger.info(
                "SkillMiner: skip task_id=%s due to low critic_confidence=%.3f",
                task.task_id,
                feedback.confidence,
            )
            return None

        prompt, context = self._build_prompt_context(task, result, trace_events, feedback)
        decision: CouncilDecision = self.council.deliberate(
            question=prompt,
            context=context,
            mode="skill_mining",
        )

        cand = self._parse_candidate(decision.final_answer, task)
        if cand is None:
            logger.info(
                "SkillMiner: council output could not be parsed as SkillCandidate for task_id=%s",
                task.task_id,
            )
            return None

        self.registry.upsert_candidate(cand)
        logger.info(
            "SkillMiner: created SkillCandidate=%s from task_id=%s",
            cand.candidate_id,
            task.task_id,
        )
        return cand

    # ---------- prompt & parse ----------

    def _build_prompt_context(
        self,
        task: TaskSpec,
        result: TaskResult,
        trace_events: List[TraceEvent],
        feedback: Optional[CriticFeedback],
    ) -> tuple[str, str]:
        prompt = (
            "You are helping a self-improving AI system mine reusable SKILLS "
            "from successful task executions.\n"
            "Given the task, final answer, trace, and critic feedback, extract "
            "ONE generalizable skill that could be re-used in similar tasks.\n\n"
            "Output in the following strict format:\n"
            "NAME: <short skill name>\n"
            "DOMAIN: <domain>\n"
            "DESCRIPTION:\n"
            "<description>\n"
            "CODE:\n"
            "<pseudo-code or python-like implementation>\n"
            "TESTS:\n"
            "<how to test this skill>\n"
            "QUALITY_NOTES:\n"
            "<notes on when this skill works or fails>\n"
        )

        trace_lines: List[str] = []
        for e in trace_events[-20:]:
            trace_lines.append(
                f"[{e.step}/{e.role}/{e.kind}] keys={list((e.payload or {}).keys())}"
            )

        context_parts: List[str] = [
            f"Task goal:\n{task.goal}",
            f"Domain: {task.domain}",
            f"Final answer (truncated):\n{result.final_answer[:1500]}",
        ]

        if feedback:
            context_parts.append(
                f"Lessons learned (truncated):\n{feedback.lessons_learned[:1500]}"
            )

        context_parts.append("Trace events summary:\n" + "\n".join(trace_lines))

        context = "\n\n".join(context_parts)
        return prompt, context

    def _parse_candidate(
        self,
        text: str,
        task: TaskSpec,
    ) -> Optional[SkillCandidate]:
        """
        v1：简单基于前缀解析，后续可以改为 JSON 输出。
        """
        sections = {
            "NAME:": "",
            "DOMAIN:": "",
            "DESCRIPTION:": "",
            "CODE:": "",
            "TESTS:": "",
            "QUALITY_NOTES:": "",
        }
        current = None
        for line in text.splitlines():
            stripped = line.strip()
            if stripped in sections:
                current = stripped
                continue
            if current:
                sections[current] += line + "\n"

        name = sections["NAME:"].strip() or f"skill_from_{task.domain}"
        domain = sections["DOMAIN:"].strip() or task.domain
        description = sections["DESCRIPTION:"].strip()
        code = sections["CODE:"].strip()
        tests = sections["TESTS:"].strip()
        notes = sections["QUALITY_NOTES:"].strip()

        if not description and not code:
            return None

        cand = SkillCandidate(
            candidate_id=str(uuid.uuid4()),
            name=name,
            domain=domain,
            description=description,
            code=code,
            tests=tests,
            source="cognitive_trace",
            from_task_id=task.task_id,
            quality_notes=notes,
            metadata={
                "goal": task.goal[:200],
            },
        )
        return cand

def cmd_skill_mine_from_goal(args: argparse.Namespace) -> None:
    """
    组合 CognitiveLoop + SkillMiner：
      1) 用 CognitiveLoop 跑一次任务
      2) 用同一个 council + trace，从这次执行挖一个 SkillCandidate
    """
    loop = CognitiveLoop()
    result = loop.run_oneoff(goal=args.goal, domain=args.domain)

    # 这里我们需要拿到刚才的 TaskSpec/TaskResult/Feedback
    # TaskSpec 在 run_oneoff 内部是临时的，我们可以通过 trace 中的 task_id 拿到
    # 简化：直接复用 run_oneoff 里生成的 task_id 和 goal/domain 构造 TaskSpec，
    #       Feedback 在 run_task 中已经计算过，但那里没有返回，我们就再算一遍最小 feedback（也可以接受）。
    # 为了不改 CognitiveLoop 签名，这里只做一个“弱版本挖掘”：只用 TaskSpec + TaskResult，不依赖 feedback 细节。

    # 这里的做法：重新构造 TaskSpec（goal/domain）和一个空反馈
    dummy_task = TaskSpec(
        task_id=result.task_id,
        goal=args.goal,
        domain=args.domain,
    )

    trace_store = FileBackedTraceStore()
    trace_events = trace_store.list_by_task(result.task_id)

    # 最小 feedback：只提供 confidence=0.8
    dummy_feedback = CriticFeedback(
        ok=True,
        confidence=0.8,
        issues=[],
        suggestions=result.lessons or "",
        lessons_learned=result.lessons or "",
    )

    miner = SkillMiner()
    cand = miner.mine_from_task(
        task=dummy_task,
        result=result,
        trace_events=trace_events,
        feedback=dummy_feedback,
    )

    print(f"Task finished. task_id={result.task_id}")
    print("\n=== Final Answer ===\n")
    print(result.final_answer)

    if result.lessons:
        print("\n=== Lessons Learned ===\n")
        print(result.lessons)

    if cand:
        print("\n=== SkillCandidate Created ===\n")
        print(f"candidate_id: {cand.candidate_id}")
        print(f"name        : {cand.name}")
        print(f"domain      : {cand.domain}")
    else:
        print("\n(no SkillCandidate created from this run)")
