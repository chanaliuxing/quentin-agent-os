# quentin/self_programming/patches.py
from __future__ import annotations

import json
import logging
from dataclasses import dataclass, field, asdict
from datetime import datetime
from pathlib import Path
from typing import Dict, Any, List, Literal, Optional

logger = logging.getLogger(__name__)

PatchType = Literal["unified_diff", "full_file"]

PATCH_STATUS_GENERATED = "generated"      # 已生成，但未评审/未测试
PATCH_STATUS_REVIEWED = "reviewed"        # 已有评审报告
PATCH_STATUS_APPROVED = "approved"        # 通过（可以进入后续应用环节）
PATCH_STATUS_REJECTED = "rejected"        # 拒绝（设计不合格/风险过高）
PATCH_STATUS_APPLIED = "applied"          # 已应用到代码库（后续可以加 git 钩子）


@dataclass
class FilePatch:
    """
    针对单个文件的补丁。

    两种模式：
    - unified_diff: 生成 GNU unified diff（适合后面直接打 patch）
    - full_file:    直接给出完整文件内容（适合新文件或重写较简单文件）
    """
    path: str
    patch_type: PatchType
    content: str


@dataclass
class PatchReview:
    """
    多 AI 评审后对补丁的总结。
    v1 先只存文本，后续可以结构化 decision / risk_level / required_changes。
    """
    review_id: str
    patch_id: str
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    reviewer: str = "multi_llm_council"
    summary: str = ""      # 综合评审意见（例如：可行/建议调整/高风险点）
    risks: str = ""        # 风险与不确定性列表（直接用 council 的 risks 文本）
    raw_council_output: Dict[str, Any] = field(default_factory=dict)


@dataclass
class CodePatch:
    """
    针对某个 ImprovementTask + DesignDraft 的完整补丁草案。
    """
    patch_id: str
    task_id: str
    draft_id: str
    capability: str
    requirement_id: Optional[str]

    status: str = PATCH_STATUS_GENERATED
    summary: str = ""           # 模型生成的简要说明（做了什么、为什么这么做）
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    files: List[FilePatch] = field(default_factory=list)

    review: Optional[PatchReview] = None  # 多 AI 评审结果（可选）
    meta: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        # dataclass 嵌套已经展开了，不需要特殊处理
        return d

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "CodePatch":
        # 手动恢复嵌套类型
        files = [FilePatch(**f) for f in data.get("files", [])]
        review_data = data.get("review")
        review = PatchReview(**review_data) if review_data else None

        return cls(
            patch_id=data["patch_id"],
            task_id=data["task_id"],
            draft_id=data["draft_id"],
            capability=data["capability"],
            requirement_id=data.get("requirement_id"),
            status=data.get("status", PATCH_STATUS_GENERATED),
            summary=data.get("summary", ""),
            created_at=data.get("created_at", ""),
            updated_at=data.get("updated_at", ""),
            files=files,
            review=review,
            meta=data.get("meta", {}),
        )


class FileBackedPatchStore:
    """
    CodePatch 的最简单存储：一个 JSON 文件 {patch_id: patch_dict}
    """

    def __init__(self, path: str | Path = "data/improvement_patches.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._cache: Dict[str, Dict[str, Any]] = {}
        self._loaded = False

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                with self.path.open("r", encoding="utf-8") as f:
                    self._cache = json.load(f)
            except Exception as e:
                logger.exception("Failed to load patch store from %s", self.path)
                self._cache = {}
        else:
            self._cache = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self.path.open("w", encoding="utf-8") as f:
            json.dump(self._cache, f, ensure_ascii=False, indent=2)

    def get_all(self) -> Dict[str, Dict[str, Any]]:
        self._load()
        return dict(self._cache)

    def get(self, patch_id: str) -> Optional[CodePatch]:
        self._load()
        d = self._cache.get(patch_id)
        if not d:
            return None
        return CodePatch.from_dict(d)

    def upsert(self, patch: CodePatch) -> None:
        self._load()
        self._cache[patch.patch_id] = patch.to_dict()
        self._save()

# quentin/self_programming/engine.py
from __future__ import annotations

import json
import logging
from datetime import datetime
from typing import Dict, Any, Optional, List

from quentin.models.ollama_client import OllamaClient
from quentin.self_programming.patches import (
    CodePatch,
    FilePatch,
    PatchReview,
    FileBackedPatchStore,
    PATCH_STATUS_GENERATED,
    PATCH_STATUS_REVIEWED,
)
from quentin.self_improvement.design import (
    ImprovementDesignPlanner,
    FileBackedDesignStore,
    DesignDraft,
)
from quentin.self_improvement.manager import SelfImprovementManager
from quentin.self_improvement.schemas import ImprovementTask
from quentin.agents.multi_llm_council import MultiLLMCouncil
from quentin.agents.multi_llm_roles import MeetingType

logger = logging.getLogger(__name__)


class SelfProgrammingEngine:
    """
    自编程引擎 v1（纯草案版本）：

    - 输入：DesignDraft（由多 AI 会议产生的设计草案）
    - 输出：CodePatch（代码补丁草案，包含待应用的文件 patch）
    - 评审：使用 MultiLLMCouncil 对 CodePatch 做一次 patch_review，生成 PatchReview

    v1 不负责真正执行测试和打 patch，只负责：
    - 生成补丁文本
    - 产出评审文本
    - 方便后续手动/自动集成 CI 流程
    """

    def __init__(
        self,
        blueprint_path: str = "quentin/blueprint/meta_agent_v3.yaml",
        coder_model: str = "qwen3-vl:latest",
        ollama_client: Optional[OllamaClient] = None,
        patch_store: Optional[FileBackedPatchStore] = None,
        design_store: Optional[FileBackedDesignStore] = None,
        council: Optional[MultiLLMCouncil] = None,
    ) -> None:
        self.blueprint_path = blueprint_path
        self.coder_model = coder_model
        self.client = ollama_client or OllamaClient()
        self.patch_store = patch_store or FileBackedPatchStore()
        self.design_store = design_store or FileBackedDesignStore()
        self.si_manager = SelfImprovementManager(blueprint_path=blueprint_path)
        self.council = council or MultiLLMCouncil()

    # -------- Patch 生成主入口 --------

    def propose_patch_for_design(
        self,
        draft_id: str,
    ) -> CodePatch:
        """
        根据指定的 DesignDraft 生成一个 CodePatch 草案。
        默认策略：
        - 使用 coder_model（如 qwen3-vl:latest）生成 JSON 格式的补丁描述
        - 如果 JSON 解析失败，则退化为 single full_file patch，把模型输出原样存入 content
        """
        draft = self._get_design_by_id(draft_id)
        if not draft:
            raise ValueError(f"DesignDraft not found: {draft_id}")

        task = self._get_task_by_id(draft.task_id)
        if not task:
            raise ValueError(f"Related ImprovementTask not found: {draft.task_id}")

        prompt = self._build_coder_prompt(draft, task)

        raw_answer = self.client.generate(
            model=self.coder_model,
            prompt=prompt,
            temperature=0.25,
            max_tokens=4096,
        )

        patch_data = self._parse_patch_answer(raw_answer)
        patch = self._build_code_patch_from_data(
            draft=draft,
            task=task,
            patch_data=patch_data,
            raw_answer=raw_answer,
        )

        self.patch_store.upsert(patch)
        logger.info("CodePatch created: %s for draft %s", patch.patch_id, draft.draft_id)
        return patch

    # -------- Patch 评审主入口（多 AI 会议 patch_review） --------

    def review_patch_with_council(
        self,
        patch_id: str,
        meeting_type: MeetingType = "patch_review",
    ) -> CodePatch:
        """
        使用 MultiLLMCouncil 对补丁做一次 patch_review：
        - 问题：这个补丁是否合理？风险在哪里？有无改进建议？
        - 输出：PatchReview，附加到 CodePatch 上，状态改为 reviewed
        """
        patch = self.patch_store.get(patch_id)
        if not patch:
            raise ValueError(f"CodePatch not found: {patch_id}")

        task = self._get_task_by_id(patch.task_id)
        if not task:
            raise ValueError(f"Related ImprovementTask not found: {patch.task_id}")

        draft = self._get_design_by_id(patch.draft_id)
        if not draft:
            raise ValueError(f"Related DesignDraft not found: {patch.draft_id}")

        problem, context = self._build_patch_review_problem_and_context(
            patch=patch,
            task=task,
            draft=draft,
        )

        council_result = self.council.deliberate(
            meeting_type=meeting_type,
            problem=problem,
            context=context,
        )

        review_id = f"{patch.patch_id}::review:{int(datetime.utcnow().timestamp())}"
        review = PatchReview(
            review_id=review_id,
            patch_id=patch.patch_id,
            summary=council_result.merged_answer,
            risks=council_result.risks,
            raw_council_output={
                "notes": council_result.notes,
                "proposals": [
                    {
                        "role_id": p.role_id,
                        "role_name": p.role_name,
                        "model": p.model,
                        "content": p.content,
                    }
                    for p in council_result.proposals
                ],
            },
        )

        patch.review = review
        patch.status = PATCH_STATUS_REVIEWED
        patch.updated_at = datetime.utcnow().isoformat()

        self.patch_store.upsert(patch)
        logger.info("CodePatch reviewed: %s", patch.patch_id)
        return patch

    # -------- 查询/辅助 --------

    def list_patches(self) -> List[CodePatch]:
        raw = self.patch_store.get_all()
        patches = [CodePatch.from_dict(d) for d in raw.values()]
        patches.sort(key=lambda p: p.created_at)
        return patches

    def _get_design_by_id(self, draft_id: str) -> Optional[DesignDraft]:
        all_drafts = self.design_store.get_all()
        d = all_drafts.get(draft_id)
        if not d:
            return None
        return DesignDraft.from_dict(d)

    def _get_task_by_id(self, task_id: str) -> Optional[ImprovementTask]:
        tasks = self.si_manager.list_tasks()
        for t in tasks:
            if t.task_id == task_id:
                return t
        return None

    # -------- Prompt 构造 --------

    def _build_coder_prompt(
        self,
        draft: DesignDraft,
        task: ImprovementTask,
    ) -> str:
        """
        给“写代码的模型”的提示词。
        目标：让它根据 DesignDraft 生成 JSON 结构的补丁。
        """
        cap_spec = draft.blueprint_snapshot
        design = draft.merged_answer
        risks = draft.risks

        return f"""你是一名资深 Python 工程师，正在为一个 Meta-Agent v3 系统编写代码补丁。

你的任务：
- 针对下面的“自我改进任务”和“设计草案”，生成一份代码补丁方案。
- 输出必须是严格的 JSON 格式，方便程序解析。

[自我改进任务]
{json.dumps(task.to_dict(), ensure_ascii=False, indent=2)}

[对应蓝皮书能力说明]
{json.dumps(cap_spec, ensure_ascii=False, indent=2)}

[设计草案 - 综合方案]
{design}

[设计草案 - 风险与注意事项]
{risks}

[现有组件 Gap 信息]
{json.dumps(task.meta.get("components", []), ensure_ascii=False, indent=2)}

输出要求（非常重要）：
1. 严格输出一个 JSON 对象，不要有任何解释性文字。
2. JSON 结构必须为：
{{
  "summary": "用中文简要说明补丁做了什么、修改哪些模块，以及为什么这样设计。",
  "files": [
    {{
      "path": "quentin/xxx/yyy.py",
      "patch_type": "unified_diff" 或 "full_file",
      "content": "补丁内容。对于 unified_diff 请使用标准 diff 语法；full_file 则提供完整文件文本。"
    }}
  ]
}}

3. 如果你不确定某些文件的精确现有内容，可以：
   - 使用 full_file 给出你认为合理的完整文件实现；
   - 或在 unified_diff 中以注释形式标明“假设原始内容为...”。

现在请只输出 JSON，不要使用 markdown，不要加解释。
"""

    def _build_patch_review_problem_and_context(
        self,
        patch: CodePatch,
        task: ImprovementTask,
        draft: DesignDraft,
    ) -> tuple[str, Dict[str, Any]]:
        """
        给多 AI 会议的 patch_review 问题+上下文。
        """
        problem = f"""请对下面这份 CodePatch 做全面的代码评审，重点关注：
1. 设计是否和 DesignDraft 一致？
2. 是否有明显的错误风险（如状态不一致、异常未处理、边界条件遗漏）？
3. 对 Meta-Agent v3 的安全内核/数据协议是否有潜在破坏？
4. 建议的改进方向。

注意：你们不需要输出具体代码，只需要给出评审结论。
"""

        context: Dict[str, Any] = {
            "task": task.to_dict(),
            "design_draft": draft.to_dict(),
            "code_patch": patch.to_dict(),
        }
        return problem, context

    # -------- 解析 LLM 输出为 patch_data --------

    @staticmethod
    def _parse_patch_answer(raw_answer: str) -> Dict[str, Any]:
        """
        尝试从 LLM 输出中解析 JSON。
        如果失败，则退化为一个单文件 full_file patch，把原始输出存进去。
        """
        text = raw_answer.strip()
        # 有些模型可能会前后加解释性文字，这里简单 heuristic 提取 { ... } 段落
        start = text.find("{")
        end = text.rfind("}")
        if start != -1 and end != -1 and end > start:
            candidate = text[start : end + 1]
        else:
            candidate = text

        try:
            data = json.loads(candidate)
            if not isinstance(data, dict):
                raise ValueError("Top-level JSON is not an object")
            return data
        except Exception as e:
            logger.warning("Failed to parse patch JSON, fallback to raw full_file patch: %s", e)
            return {
                "summary": "LLM 输出未能解析为结构化 JSON，已使用 raw 输出作为单文件补丁。",
                "files": [
                    {
                        "path": "NOT_APPLIED/raw_patch.txt",
                        "patch_type": "full_file",
                        "content": raw_answer,
                    }
                ],
            }

    @staticmethod
    def _build_code_patch_from_data(
        draft: DesignDraft,
        task: ImprovementTask,
        patch_data: Dict[str, Any],
        raw_answer: str,
    ) -> CodePatch:
        """
        把 JSON 转成 CodePatch 对象。
        """
        summary = patch_data.get("summary", "")
        files_data = patch_data.get("files", [])

        files: List[FilePatch] = []
        for f in files_data:
            try:
                fp = FilePatch(
                    path=str(f.get("path", "")),
                    patch_type=f.get("patch_type", "full_file"),  # type: ignore
                    content=str(f.get("content", "")),
                )
                files.append(fp)
            except Exception as e:
                logger.warning("Failed to parse file patch entry %s: %s", f, e)

        if not files:
            # 安全兜底：至少有一个 raw 文件
            files = [
                FilePatch(
                    path="NOT_APPLIED/raw_patch.txt",
                    patch_type="full_file",
                    content=raw_answer,
                )
            ]

        patch_id = f"{task.task_id}::draft={draft.draft_id}::ts={int(datetime.utcnow().timestamp())}"

        meta: Dict[str, Any] = {
            "raw_llm_output": raw_answer[:2000],  # 截断存一下，防止太大
        }

        return CodePatch(
            patch_id=patch_id,
            task_id=task.task_id,
            draft_id=draft.draft_id,
            capability=task.capability,
            requirement_id=task.requirement_id,
            status=PATCH_STATUS_GENERATED,
            summary=summary,
            files=files,
            meta=meta,
        )
