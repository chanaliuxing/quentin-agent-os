# ============================================
# quentin/self_programming/schemas.py
# ============================================

from __future__ import annotations

from typing import Dict, Any, Optional
from datetime import datetime
from enum import Enum
from pydantic import BaseModel, Field


class PatchKind(str, Enum):
    code = "code"
    config = "config"
    docs = "docs"


class PatchStatus(str, Enum):
    proposed = "proposed"       # 刚提出，还没测
    testing = "testing"         # 正在跑测试
    ready = "ready"             # 测试通过，可候选上线
    applied = "applied"         # 已经应用到代码
    rolled_back = "rolled_back" # 应用了又撤回
    rejected = "rejected"       # 不合格，被废弃


class PatchTestSummary(BaseModel):
    """
    测试结果摘要：
    v1 只做语法检查，将来可以扩展为 pytest / 回归测试。
    """
    ran_tests: bool = False
    passed: bool = False

    syntax_ok: bool = False
    test_command: Optional[str] = None  # 将来可记录 pytest 命令
    logs: str = ""

    started_at: Optional[datetime] = None
    finished_at: Optional[datetime] = None


class PatchEvalSummary(BaseModel):
    """
    Patch 在“业务指标上的评估结果”：
    - 比如通过 ExperimentOrchestrator 跑出来的 A/B 结果。
    """
    evaluated: bool = False
    metric: str = "score"

    baseline_score: Optional[float] = None
    patch_score: Optional[float] = None
    improvement: Optional[float] = None

    experiment_id: Optional[str] = None
    notes: str = ""


class PatchRecord(BaseModel):
    """
    一条完整的“自编程补丁”记录：
    - 元信息 + 简单的 text_replace patch（v1）
    """
    patch_id: str

    kind: PatchKind = PatchKind.code
    target_file: str                 # 相对仓库根目录的路径
    description: str

    created_at: datetime = Field(default_factory=datetime.utcnow)
    created_by: str = "self_programming"

    # v1 的 patch 形式：简单的文本替换
    apply_mode: str = "text_replace"  # 预留扩展 file_overwrite / diff 等
    search: str                       # 要查找的片段
    replace: str                      # 替换后的片段

    status: PatchStatus = PatchStatus.proposed
    risk_score: float = 0.0
    allow_auto_apply: bool = False     # 是否允许自动上线（默认不允许）

    test_summary: PatchTestSummary = Field(default_factory=PatchTestSummary)
    eval_summary: PatchEvalSummary = Field(default_factory=PatchEvalSummary)

    meta: Dict[str, Any] = Field(default_factory=dict)
这里我们用的是 最简单但完全通用 的 patch 表示：search → replace。
将来你要升级成 unified diff / AST patch，只要扩这个 schema，不动上层逻辑。

二、自编程引擎：SelfProgrammingEngine v1
新建文件：quentin/self_programming/engine.py

核心职责（对齐蓝皮书 4.1）：

只在允许的目录下创建补丁（/mutable 等）

生成 PatchRecord（现在是手工/外部调用，未来可以接 LLM）

在“沙箱模式”下对 patch 做语法测试（compile 一下）

测试通过后，允许手动（或在 allow_auto_apply=True 时自动）应用补丁

备份原文件、支持简单回滚

所有状态写 KV，可审计

# ============================================
# quentin/self_programming/engine.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Dict, Any, List, Optional, Tuple
from datetime import datetime
from pathlib import Path
import uuid
import traceback

from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.kernel.safety import SafetyKernel
from quentin.self_programming.schemas import (
    PatchRecord,
    PatchStatus,
    PatchTestSummary,
    PatchEvalSummary,
)


@dataclass
class SelfProgrammingEngine:
    """
    Self-Programming Engine v1：

    - 目前只支持“手动 / 外部调用”提出 text_replace 补丁
    - 自动跑语法测试（compile）
    - 通过 SafetyKernel 权限检查后，才能应用补丁
    - 应用时自动备份原文件，提供 rollback 能力
    """
    tracer: Tracer
    kv: KVStore
    safety_kernel: SafetyKernel

    repo_root: Path                     # 仓库根目录
    allowed_roots: List[str] = field(
        default_factory=lambda: ["mutable", "quentin/skills"]
    )

    patch_prefix: str = "patch:record:"
    patch_index_key: str = "patch:index:all"
    patch_applied_index_key: str = "patch:index:applied"

    # ---------- KV 工具 ----------

    async def _save_patch(self, record: PatchRecord) -> None:
        key = f"{self.patch_prefix}{record.patch_id}"
        await self.kv.save(key, record.model_dump())
        await self.kv.append_list(self.patch_index_key, record.patch_id)
        if record.status == PatchStatus.applied:
            await self.kv.append_list(self.patch_applied_index_key, record.patch_id)

    async def get_patch(self, patch_id: str) -> Optional[PatchRecord]:
        data = await self.kv.get(f"{self.patch_prefix}{patch_id}")
        if not data:
            return None
        return PatchRecord(**data)

    async def list_patches(self, limit: int = 100) -> List[PatchRecord]:
        ids = await self.kv.get(self.patch_index_key) or []
        ids = list(dict.fromkeys(ids))[-limit:]
        res: List[PatchRecord] = []
        for pid in ids:
            data = await self.kv.get(f"{self.patch_prefix}{pid}")
            if data:
                res.append(PatchRecord(**data))
        return res

    # ---------- 路径安全检查 ----------

    def _ensure_allowed_path(self, rel_path: str) -> Path:
        """
        确保 patch 的目标文件在 allowed_roots 目录之下。
        """
        rel = Path(rel_path)
        if rel.is_absolute():
            # 一律不允许给绝对路径打 patch
            raise ValueError(f"Absolute path not allowed: {rel_path}")

        abs_path = (self.repo_root / rel).resolve()

        # 必须在 repo_root 之下
        if self.repo_root not in abs_path.parents and abs_path != self.repo_root:
            raise ValueError(f"Target file outside repo_root: {abs_path}")

        # 必须匹配 allowed_roots 之一
        if self.allowed_roots:
            norm_rel_str = str(abs_path.relative_to(self.repo_root)).replace("\\", "/")
            if not any(norm_rel_str.startswith(prefix) for prefix in self.allowed_roots):
                raise ValueError(
                    f"Target file '{norm_rel_str}' is not under allowed roots: {self.allowed_roots}"
                )

        return abs_path

    async def _build_patched_source(self, patch: PatchRecord) -> Tuple[str, str, Path]:
        """
        从磁盘读取原文件，应用 text_replace，返回 (original, patched, abs_path)。
        """
        abs_path = self._ensure_allowed_path(patch.target_file)
        original = abs_path.read_text(encoding="utf-8")

        if patch.search not in original:
            raise ValueError("search snippet not found in target file")

        patched = original.replace(patch.search, patch.replace, 1)
        return original, patched, abs_path

    # ---------- 语法测试 ----------

    def _syntax_check(self, source: str, filename: str) -> Tuple[bool, str]:
        """
        v1：仅做 Python 语法编译检查。
        将来可以扩展为 pytest / 静态分析 / fuzz。
        """
        try:
            compile(source, filename, "exec")
            return True, ""
        except SyntaxError:
            return False, traceback.format_exc()
        except Exception:
            # 其他错误也视作语法测试失败
            return False, traceback.format_exc()

    # ---------- 对外 API：创建补丁 ----------

    async def propose_manual_text_patch(
        self,
        target_file: str,
        description: str,
        search: str,
        replace: str,
        allow_auto_apply: bool = False,
        meta: Optional[Dict[str, Any]] = None,
    ) -> PatchRecord:
        """
        v1 的补丁入口：手动/外部创建一个 text_replace patch。
        未来可以在这里增加：由 LLM 自动生成 patch 的入口。
        """
        # 先做路径安全检查
        self._ensure_allowed_path(target_file)

        patch_id = f"patch_{uuid.uuid4().hex}"
        record = PatchRecord(
            patch_id=patch_id,
            target_file=target_file,
            description=description,
            search=search,
            replace=replace,
            allow_auto_apply=allow_auto_apply,
            meta=meta or {},
        )

        await self._save_patch(record)

        self.tracer.span(
            "L2",
            "patch_proposed_manual",
            module="self_programming",
            input={
                "patch_id": patch_id,
                "target_file": target_file,
                "allow_auto_apply": allow_auto_apply,
            },
        )

        return record

    # ---------- 对外 API：跑测试 ----------

    async def run_tests(self, patch_id: str) -> PatchRecord:
        """
        对指定 patch 运行测试（当前仅语法检查）。
        """
        record = await self.get_patch(patch_id)
        if not record:
            raise ValueError(f"Patch not found: {patch_id}")

        record.status = PatchStatus.testing
        await self._save_patch(record)

        started = datetime.utcnow()
        try:
            original, patched, abs_path = await self._build_patched_source(record)
            ok, logs = self._syntax_check(patched, record.target_file)
            passed = ok
        except Exception as e:
            ok = False
            passed = False
            logs = f"Exception while preparing patch:\n{traceback.format_exc()}"

        finished = datetime.utcnow()

        record.test_summary = PatchTestSummary(
            ran_tests=True,
            passed=passed,
            syntax_ok=ok,
            logs=logs,
            started_at=started,
            finished_at=finished,
            test_command="python compile() syntax-check",
        )

        # 状态更新
        record.status = PatchStatus.ready if passed else PatchStatus.rejected
        await self._save_patch(record)

        self.tracer.span(
            "L2",
            "patch_tested",
            module="self_programming",
            input={"patch_id": patch_id},
            output={"passed": passed},
        )

        return record

    # ---------- 对外 API：应用补丁 ----------

    async def apply_patch(self, patch_id: str, auto: bool = False) -> PatchRecord:
        """
        在通过 SafetyKernel 检查后，真正修改代码文件：
        - 写入补丁
        - 备份原文件
        - 更新状态为 applied
        """
        state = await self.safety_kernel.get_state()
        if not state.self_programming_enabled or state.degraded:
            raise RuntimeError(
                "Self-programming is disabled or system is in degraded mode."
            )

        record = await self.get_patch(patch_id)
        if not record:
            raise ValueError(f"Patch not found: {patch_id}")

        if auto and not record.allow_auto_apply:
            raise RuntimeError("Patch is not marked as allow_auto_apply; manual review required.")

        if record.status not in (PatchStatus.ready, PatchStatus.applied):
            # 不允许直接对未测试的 patch 自动 apply
            raise RuntimeError(f"Patch is not ready to apply. Current status: {record.status}")

        original, patched, abs_path = await self._build_patched_source(record)

        # 备份
        backup_suffix = datetime.utcnow().strftime("%Y%m%d%H%M%S")
        backup_path = abs_path.with_suffix(abs_path.suffix + f".bak.{backup_suffix}")
        backup_path.write_text(original, encoding="utf-8")

        # 真正写入补丁
        abs_path.write_text(patched, encoding="utf-8")

        record.status = PatchStatus.applied
        record.meta.setdefault("backup_file", str(backup_path))
        await self._save_patch(record)

        self.tracer.span(
            "L2",
            "patch_applied",
            module="self_programming",
            input={"patch_id": patch_id},
            output={"target_file": record.target_file, "backup_file": str(backup_path)},
        )

        return record

    # ---------- 对外 API：回滚补丁 ----------

    async def rollback_patch(self, patch_id: str) -> PatchRecord:
        """
        根据 backup_file 做简单回滚。
        """
        record = await self.get_patch(patch_id)
        if not record:
            raise ValueError(f"Patch not found: {patch_id}")

        backup_file = record.meta.get("backup_file")
        if not backup_file:
            raise RuntimeError("No backup_file recorded for this patch; cannot rollback automatically.")

        abs_target = self._ensure_allowed_path(record.target_file)
        backup_path = Path(backup_file)

        if not backup_path.exists():
            raise RuntimeError(f"Backup file not found: {backup_path}")

        content = backup_path.read_text(encoding="utf-8")
        abs_target.write_text(content, encoding="utf-8")

        record.status = PatchStatus.rolled_back
        await self._save_patch(record)

        self.tracer.span(
            "L2",
            "patch_rolled_back",
            module="self_programming",
            input={"patch_id": patch_id},
            output={"target_file": record.target_file, "backup_file": backup_file},
        )

        return record

    # ---------- 评估摘要写入（给实验层调用的接口占位） ----------

    async def update_eval_summary(
        self,
        patch_id: str,
        eval_summary: PatchEvalSummary,
    ) -> PatchRecord:
        record = await self.get_patch(patch_id)
        if not record:
            raise ValueError(f"Patch not found: {patch_id}")

        record.eval_summary = eval_summary
        await self._save_patch(record)

        self.tracer.span(
            "L3",
            "patch_eval_updated",
            module="self_programming",
            input={"patch_id": patch_id},
            output={
                "metric": eval_summary.metric,
                "improvement": eval_summary.improvement,
                "experiment_id": eval_summary.experiment_id,
            },
        )

        return record

# ============================================
# quentin/meta/self_improvement.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass
from typing import Dict, Any, List, Optional

from quentin.infra.logging import Tracer
from quentin.immune.loop import ImmuneLoop
from quentin.immune.schemas import ImmuneCase, ImmuneCaseStatus
from quentin.self_programming.engine import SelfProgrammingEngine
from quentin.self_programming.schemas import PatchRecord
from quentin.experiments.orchestrator import ExperimentOrchestrator
from quentin.experiments.schemas import (
    ExperimentConfig,
    VariantConfig,
)


@dataclass
class SelfImprovementManager:
    """
    Self-Improvement Manager v1：

    - 从 ImmuneLoop 获取 “最近的异常案例” 作为改进线索
    - 帮助为某个 ImmuneCase 创建一个补丁任务（仍然是手动给 search/replace）
    - 为 patch 创建一个简单的 Experiment，用于后续对比效果
    """
    tracer: Tracer
    immune_loop: ImmuneLoop
    self_programming: SelfProgrammingEngine
    experiment_orchestrator: ExperimentOrchestrator

    async def suggest_hotspots(self, limit_cases: int = 10) -> List[ImmuneCase]:
        """
        返回最近的 open ImmuneCase，作为“重点优化对象”。
        """
        cases = await self.immune_loop.list_cases(limit=limit_cases)
        # 只看 open / investigating
        focus = [
            c for c in cases
            if c.status in (ImmuneCaseStatus.open, ImmuneCaseStatus.investigating)
        ]
        self.tracer.span(
            "L2",
            "self_improvement_suggest_hotspots",
            module="self_improvement",
            output={"cases": [c.case_id for c in focus]},
        )
        return focus

    async def open_patch_for_case(
        self,
        case_id: str,
        target_file: str,
        description: str,
        search: str,
        replace: str,
        allow_auto_apply: bool = False,
    ) -> PatchRecord:
        """
        以 ImmuneCase 为线索，创建一条 patch。
        """
        case = await self.immune_loop.get_case(case_id)
        if not case:
            raise ValueError(f"Immune case not found: {case_id}")

        meta = {
            "immune_case_id": case_id,
            "window_start": case.window_start.isoformat(),
            "window_end": case.window_end.isoformat(),
        }

        patch = await self.self_programming.propose_manual_text_patch(
            target_file=target_file,
            description=description,
            search=search,
            replace=replace,
            allow_auto_apply=allow_auto_apply,
            meta=meta,
        )

        self.tracer.span(
            "L3",
            "self_improvement_open_patch_for_case",
            module="self_improvement",
            input={"case_id": case_id, "patch_id": patch.patch_id},
        )

        return patch

    async def create_experiment_for_patch(
        self,
        patch: PatchRecord,
        baseline_label: str = "baseline",
        patched_label: str = "with_patch",
    ) -> str:
        """
        为某个 patch 创建一个简单的 Experiment：
        - 两个 variant：baseline / patched
        - 具体如何在任务执行时切换 variant，留给 domain/业务代码实现
        """
        cfg = ExperimentConfig(
            name=f"patch_{patch.patch_id}",
            description=f"Experiment for patch {patch.patch_id} on {patch.target_file}",
            eval_kind="offline",
            dataset=None,
            variants=[
                VariantConfig(
                    variant_id=f"{patch.patch_id}_baseline",
                    label=baseline_label,
                    kind="patch_toggle",
                    params={"use_patch": False, "patch_id": patch.patch_id},
                ),
                VariantConfig(
                    variant_id=f"{patch.patch_id}_patched",
                    label=patched_label,
                    kind="patch_toggle",
                    params={"use_patch": True, "patch_id": patch.patch_id},
                ),
            ],
            meta={"patch_id": patch.patch_id},
        )

        exp = await self.experiment_orchestrator.create_experiment(cfg)
        self.tracer.span(
            "L3",
            "self_improvement_experiment_for_patch",
            module="self_improvement",
            input={"patch_id": patch.patch_id},
            output={"experiment_id": exp.experiment_id},
        )

        return exp.experiment_id

@cli.command("patch-propose-text")
@click.option("--file", "target_file", required=True, help="Target file relative to repo root.")
@click.option("--desc", "description", required=True, help="Human-readable description of the patch.")
@click.option("--search", required=True, help="Snippet to search for.")
@click.option("--replace", required=True, help="Snippet to replace with.")
@click.option("--auto-apply", is_flag=True, default=False, show_default=True)
def patch_propose_text_cmd(target_file: str, description: str, search: str, replace: str, auto_apply: bool):
    """
    手动创建一个 text_replace patch（不立即应用）。
    """
    async def _run():
        import json
        settings = Settings()
        runtime = AgentRuntime.create_from_settings(settings)

        patch = await runtime.self_programming.propose_manual_text_patch(
            target_file=target_file,
            description=description,
            search=search,
            replace=replace,
            allow_auto_apply=auto_apply,
        )

        print("Patch proposed:")
        print(json.dumps(patch.model_dump(), ensure_ascii=False, indent=2))

    asyncio.run(_run())

@cli.command("patch-list")
@click.option("--limit", default=20, show_default=True)
def patch_list_cmd(limit: int):
    """
    列出最近的 patch。
    """
    async def _run():
        settings = Settings()
        runtime = AgentRuntime.create_from_settings(settings)

        patches = await runtime.self_programming.list_patches(limit=limit)
        if not patches:
            print("No patches found.")
            return

        for p in patches:
            print(f"- {p.patch_id} | {p.target_file} | {p.status.value}")

    asyncio.run(_run())
