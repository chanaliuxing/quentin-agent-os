# quentin/eval/spec.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional


@dataclass
class EvalTaskSpec:
    """
    单个评测任务（一个“样本”）。

    举例：
    - 对 planner：给定 goal / context，期望拆成多少步、大致结构
    - 对 model_router：给定任务特征，期望选择哪个模型、成本范围等
    """
    task_id: str
    description: str
    component: str              # "planner" / "model_router" / "skill_evolution" 等
    input_payload: Dict[str, Any] = field(default_factory=dict)
    expected_output: Dict[str, Any] = field(default_factory=dict)
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class EvalDatasetSpec:
    """
    一组 EvalTask 的集合（一个评测数据集）。
    """
    dataset_id: str
    name: str
    description: str
    component: str                      # 针对哪个组件
    tasks: List[EvalTaskSpec] = field(default_factory=list)
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        d["tasks"] = [t.to_dict() for t in self.tasks]
        return d


@dataclass
class EvalCaseResult:
    """
    单个任务的评测结果。
    """
    task_id: str
    variant_id: str
    success: bool
    score: float
    raw_output: Dict[str, Any] = field(default_factory=dict)
    notes: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)
二、EvalHarness：统一评测引擎 quentin/eval/harness.py
# quentin/eval/harness.py
from __future__ import annotations

from typing import Any, Callable, Dict, List, Optional
import logging

from quentin.eval.spec import EvalDatasetSpec, EvalTaskSpec, EvalCaseResult
from quentin.research.experiments import VariantConfig, VariantResult, ExperimentConfig

logger = logging.getLogger(__name__)

EvalFn = Callable[[EvalTaskSpec, VariantConfig, Dict[str, Any]], Dict[str, Any]]


class EvalHarness:
    """
    通用评测引擎：

    - 给定 ExperimentConfig + VariantConfig + EvalDataset + eval_fn
    - 依次跑每个任务，收集 per-case 结果
    - 聚合得到 VariantResult（success_rate / avg_score / score）
    """

    def __init__(self, *, max_cases_in_extra: int = 100) -> None:
        self.max_cases_in_extra = max_cases_in_extra

    def run_dataset(
        self,
        experiment: ExperimentConfig,
        variant: VariantConfig,
        dataset: EvalDatasetSpec,
        eval_fn: EvalFn,
        *,
        context: Optional[Dict[str, Any]] = None,
    ) -> VariantResult:
        """对某个 variant 在 dataset 上跑一次完整评测。"""
        ctx = dict(context or {})
        ctx.setdefault("experiment_id", experiment.experiment_id)
        ctx.setdefault("variant_id", variant.variant_id)
        ctx.setdefault("dataset_id", dataset.dataset_id)

        case_results: List[EvalCaseResult] = []
        success_count = 0
        score_sum = 0.0
        score_count = 0

        for task in dataset.tasks:
            try:
                raw = eval_fn(task, variant, ctx) or {}
            except Exception as e:
                logger.exception(
                    "EvalHarness: eval_fn failed for experiment=%s variant=%s task=%s",
                    experiment.experiment_id,
                    variant.variant_id,
                    task.task_id,
                )
                raw = {"success": False, "score": 0.0, "error": str(e)}

            success = bool(raw.get("success", False))
            score = float(raw.get("score", 0.0))

            if success:
                success_count += 1
            score_sum += score
            score_count += 1

            case_results.append(
                EvalCaseResult(
                    task_id=task.task_id,
                    variant_id=variant.variant_id,
                    success=success,
                    score=score,
                    raw_output=raw,
                    notes=raw.get("notes", ""),
                )
            )

        n = len(dataset.tasks) or 1
        success_rate = success_count / n
        avg_score = (score_sum / score_count) if score_count > 0 else 0.0

        metrics = {
            "success_rate": float(success_rate),
            "avg_score": float(avg_score),
            # 为方便 Orchestrator 默认比较，提供一个总评分字段
            "score": float(avg_score),
        }

        # 为了不让结果 explode，只保留前 max_cases_in_extra 个样本的详情
        extra = {
            "dataset_id": dataset.dataset_id,
            "num_tasks": len(dataset.tasks),
            "case_results_preview": [
                cr.to_dict() for cr in case_results[: self.max_cases_in_extra]
            ],
        }

        return VariantResult(
            variant_id=variant.variant_id,
            metrics=metrics,
            extra=extra,
        )
现在任何组件的 ExperimentRunner，只要把“如何在单个 EvalTask 上运行这个 variant”表达成一个 eval_fn，就能直接复用 EvalHarness。

# quentin/research/runners/planner_offline_runner.py
from __future__ import annotations

from typing import Any, Dict, Optional, Callable
import logging

from quentin.research.orchestrator import ExperimentRunner
from quentin.research.experiments import (
    ExperimentConfig,
    VariantConfig,
    VariantResult,
)
from quentin.eval.spec import EvalDatasetSpec, EvalTaskSpec
from quentin.eval.harness import EvalHarness

logger = logging.getLogger(__name__)

# planner_fn: (task, variant, context) -> dict with keys: success, score, notes, etc.
PlannerFn = Callable[[EvalTaskSpec, VariantConfig, Dict[str, Any]], Dict[str, Any]]


class PlannerOfflineRunner(ExperimentRunner):
    """
    针对 planner 组件的离线评测 Runner。

    设计理念：
    - 不直接绑死到某个 Planner 实现；
    - 将“如何从 EvalTask 调用具体 planner”的逻辑放在 planner_fn 里；
    - 默认提供一个 stub_planner_fn，至少 pipeline 可以跑通。
    """

    def __init__(
        self,
        *,
        eval_harness: Optional[EvalHarness] = None,
        planner_fn: Optional[PlannerFn] = None,
    ) -> None:
        self.eval_harness = eval_harness or EvalHarness()
        self.planner_fn = planner_fn or self._stub_planner_fn

    def run_variant(
        self,
        experiment: ExperimentConfig,
        variant: VariantConfig,
        context: Optional[Dict[str, Any]] = None,
    ) -> VariantResult:
        ctx = dict(context or {})

        dataset: Optional[EvalDatasetSpec] = ctx.get("planner_dataset") or ctx.get("eval_dataset")
        if dataset is None:
            logger.warning(
                "PlannerOfflineRunner: no EvalDatasetSpec provided in context "
                "(planner_dataset / eval_dataset missing). Using empty dataset."
            )
            dataset = EvalDatasetSpec(
                dataset_id="empty",
                name="empty",
                description="Empty dataset (no tasks).",
                component="planner",
                tasks=[],
            )

        if dataset.component != "planner":
            logger.warning(
                "PlannerOfflineRunner: dataset.component=%s but runner is for planner.",
                dataset.component,
            )

        def _eval_fn(task: EvalTaskSpec, variant_cfg: VariantConfig, local_ctx: Dict[str, Any]) -> Dict[str, Any]:
            return self.planner_fn(task, variant_cfg, local_ctx)

        return self.eval_harness.run_dataset(
            experiment=experiment,
            variant=variant,
            dataset=dataset,
            eval_fn=_eval_fn,
            context=ctx,
        )

    # ---------- 默认 stub planner ---------- #

    def _stub_planner_fn(
        self,
        task: EvalTaskSpec,
        variant: VariantConfig,
        context: Dict[str, Any],
    ) -> Dict[str, Any]:
        """
        一个简单的 stub 实现，目的是让 pipeline 能跑通。

        行为：
        - 如果 expected_output 里有非空 "reference_plan"，则认为 success=True, score=1.0；
        - 否则 success=False, score=0.0。

        实际项目中，你应该替换为：
        - 调用真正的 CognitiveLoop._run_planner 或 planner 策略；
        - 解析 planner 输出，将其与 expected_output 对比打分。
        """
        expected = task.expected_output or {}
        has_ref_plan = bool(expected.get("reference_plan"))

        success = has_ref_plan
        score = 1.0 if has_ref_plan else 0.0

        return {
            "success": success,
            "score": score,
            "notes": "stub_planner_fn used; replace with real planner integration.",
            "used_stub": True,
            "task_id": task.task_id,
            "variant_id": variant.variant_id,
        }
以后要接真实 planner，很简单：

写一个函数 real_planner_fn(task, variant, ctx)，里面调用 CognitiveLoop / PlannerCouncil；

初始化 PlannerOfflineRunner(planner_fn=real_planner_fn) 并注册到 Orchestrator。
