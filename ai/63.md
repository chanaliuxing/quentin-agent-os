一、技能评估器：SkillEvalHarness（最小可运行版）
新文件：quentin/skills/eval.py

# quentin/skills/eval.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Dict, Any, List

from quentin.skills.models import SkillCandidate


@dataclass
class SkillEvalReport:
    """
    对一个 SkillCandidate 的评估结果。

    - ok       : 是否通过最小要求
    - score    : 0.0 ~ 1.0 的粗略评分
    - issues   : 发现的问题列表
    - warnings : 警告（不致命）
    """
    ok: bool
    score: float
    issues: List[str]
    warnings: List[str]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class SkillEvalHarness:
    """
    Skill 评估器 v1（非常简单，但接口清晰、可替换）：

    现在只做：
      - 检查 description / code / tests 是否为空
      - 根据完整度给一个粗略分数

    将来可以替换为：
      - 在 Sandbox 中真正 import/执行代码
      - 对照一组固定样例跑用例
      - 多模型一起做 robustness 检查
    """

    def evaluate_candidate(self, cand: SkillCandidate) -> SkillEvalReport:
        issues: List[str] = []
        warnings: List[str] = []

        score = 1.0

        if not cand.description.strip():
            issues.append("Missing DESCRIPTION.")
            score -= 0.3

        if not cand.code.strip():
            issues.append("Missing CODE implementation.")
            score -= 0.4

        if not cand.tests.strip():
            warnings.append("Missing TESTS description.")
            score -= 0.2

        if score < 0:
            score = 0.0

        ok = score >= 0.5 and not issues  # v1：有致命问题就不通过

        return SkillEvalReport(
            ok=ok,
            score=score,
            issues=issues,
            warnings=warnings,
        )
这就是蓝皮书里「Eval Harness」在技能层面的一个具体落点：

有清晰的输入（SkillCandidate）和输出（SkillEvalReport），

将来你可以很容易把内部实现替换成真 Sandbox 测试、A/B、甚至多模型评估。

二、SkillPromotionEngine：候选技能 → 正式技能
新文件：quentin/skills/promotion.py

# quentin/skills/promotion.py
from __future__ import annotations

import logging
import uuid
from dataclasses import dataclass, asdict
from datetime import datetime
from typing import Optional, Dict, Any, List

from quentin.skills.models import SkillCandidate, SkillDefinition
from quentin.skills.registry import FileBackedSkillRegistry
from quentin.skills.eval import SkillEvalHarness, SkillEvalReport
from quentin.research.experiments import ResearchManager, Experiment

logger = logging.getLogger(__name__)


@dataclass
class SkillPromotionResult:
    """
    一次晋升尝试的结果。

    - candidate_id : 候选技能 ID
    - status       : "promoted" / "skipped" / "rejected" / "error"
    - skill_id     : 若晋升成功，则为新 SkillDefinition 的 ID
    - eval_report  : SkillEvalReport.to_dict()
    - reasons      : 说明（列表）
    """
    candidate_id: str
    status: str
    skill_id: Optional[str]
    eval_report: Optional[Dict[str, Any]]
    reasons: List[str]
    created_at: str = datetime.utcnow().isoformat()

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class SkillPromotionEngine:
    """
    SkillCandidate -> SkillDefinition 的推广引擎。

    v1 设计：

      - 从 SkillRegistry 读取候选技能
      - 用 SkillEvalHarness 做一个最小评估（完整度检查）
      - 若 eval.ok，则注册为正式 SkillDefinition，并删除 Candidate
      - 将结果写入 ResearchManager（方便做研究统计）

    后续可以挂接：
      - SafetyKernel（如果真正改动 Python 源文件时）
      - MultiAICouncil 做 promotion 审核
      - EvalHarness 做更复杂的回归与对比
    """

    def __init__(
        self,
        *,
        registry: Optional[FileBackedSkillRegistry] = None,
        eval_harness: Optional[SkillEvalHarness] = None,
        research_manager: Optional[ResearchManager] = None,
    ) -> None:
        self.registry = registry or FileBackedSkillRegistry()
        self.eval_harness = eval_harness or SkillEvalHarness()
        self.research_manager = research_manager or ResearchManager()

    # ---------- 对外主入口 ----------

    def promote_candidate(
        self,
        candidate_id: str,
        *,
        auto: bool = False,
    ) -> SkillPromotionResult:
        """
        尝试晋升一个指定的 SkillCandidate。

        auto=True 时可以用来做批量自动晋升（未来只允许低风险、评分很高的候选）。
        """
        cand = self.registry.get_candidate(candidate_id)
        if not cand:
            return SkillPromotionResult(
                candidate_id=candidate_id,
                status="error",
                skill_id=None,
                eval_report=None,
                reasons=[f"SkillCandidate not found: {candidate_id}"],
            )

        eval_report: SkillEvalReport = self.eval_harness.evaluate_candidate(cand)

        reasons: List[str] = []
        if not eval_report.ok:
            reasons.append("Evaluation failed; candidate not promoted.")
            res = SkillPromotionResult(
                candidate_id=candidate_id,
                status="rejected",
                skill_id=None,
                eval_report=eval_report.to_dict(),
                reasons=reasons + eval_report.issues,
            )
            # 记录到 ResearchManager
            self._record_promotion_run(cand, res)
            return res

        # v1：通过评估就直接晋升为正式 SkillDefinition
        skill_id = str(uuid.uuid4())
        skill = SkillDefinition(
            skill_id=skill_id,
            name=cand.name,
            domain=cand.domain,
            description=cand.description,
            input_schema={},      # 可以后续手工/AI补充
            output_schema={},
            code=cand.code,
            tests=cand.tests,
            tags=["auto-mined"],
            created_from_task_id=cand.from_task_id,
            supersedes=[],
            metadata={
                "source_candidate_id": cand.candidate_id,
                "promotion_auto": auto,
            },
        )

        self.registry.upsert_skill(skill)
        # v1：晋升成功后保留 candidate 还是删掉？
        #   - 这里选择保留，以便将来可以重评 / 对比。
        #   - 如果你想删，可以开启这行：
        # self.registry.delete_candidate(cand.candidate_id)

        reasons.append("Evaluation passed; candidate promoted to SkillDefinition.")

        res = SkillPromotionResult(
            candidate_id=cand.candidate_id,
            status="promoted",
            skill_id=skill_id,
            eval_report=eval_report.to_dict(),
            reasons=reasons,
        )

        self._record_promotion_run(cand, res)
        return res

    # ---------- ResearchManager 集成 ----------

    def _record_promotion_run(
        self,
        cand: SkillCandidate,
        res: SkillPromotionResult,
    ) -> None:
        """
        把这次晋升尝试记录为一个 experiment run：
          - capability   : skills:<domain>
          - requirement  : skill_quality
          - purpose      : skill_promotion
          - metrics      : eval_score / promoted / rejected
        """
        capability = f"skills:{cand.domain}"
        requirement_id = "skill_quality"

        exp = self._get_or_create_experiment(
            capability=capability,
            requirement_id=requirement_id,
        )

        eval_report = res.eval_report or {}
        score = float(eval_report.get("score") or 0.0)
        promoted = int(res.status == "promoted")
        rejected = int(res.status == "rejected")

        metrics = {
            "eval_score": score,
            "promoted": promoted,
            "rejected": rejected,
        }
        context = {
            "candidate_id": cand.candidate_id,
            "name": cand.name[:100],
            "domain": cand.domain,
            "status": res.status,
        }

        self.research_manager.record_run(
            experiment_id=exp.experiment_id,
            variant_id="auto" if res.eval_report else "manual",
            metrics=metrics,
            context=context,
        )

    def _get_or_create_experiment(
        self,
        *,
        capability: str,
        requirement_id: str,
    ) -> Experiment:
        exps = self.research_manager.list_experiments()
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "skill_promotion"
            ):
                return e

        return self.research_manager.create_experiment(
            name=f"SkillPromotion-{capability}-{requirement_id}",
            purpose="skill_promotion",
            capability=capability,
            requirement_id=requirement_id,
            tags=["skills", "promotion"],
        )
这里我们刻意把 “评估” 和 “晋升” 分开两层：

SkillEvalHarness：只是看候选技能本身完整不完整、有没有明显坑；

SkillPromotionEngine：决定是否晋升，并把结果写进 ResearchManager，方便后面做统计研究。

未来如果你要引入多 AI 审核、安全内核等，就在 SkillPromotionEngine.promote_candidate 里再多插几步即可。

def cmd_skill_list(args: argparse.Namespace) -> None:
    registry = FileBackedSkillRegistry()
    skills = registry.list_skills()
    if not skills:
        print("No Skills registered yet.")
        return
    print(f"Found {len(skills)} Skills:")
    for s in skills:
        print(
            f"- {s.skill_id} | {s.name} | domain={s.domain} | "
            f"calls={s.stats.call_count} succ={s.stats.success_count}"
        )


def cmd_skill_promote_candidate(args: argparse.Namespace) -> None:
    engine = SkillPromotionEngine()
    res = engine.promote_candidate(
        candidate_id=args.candidate_id,
        auto=args.auto,
    )

    data = res.to_dict()
    print(f"SkillPromotionResult for candidate_id={data['candidate_id']}")
    print(f"- status : {data['status']}")
    print(f"- skill_id : {data.get('skill_id')}")
    print("- reasons:")
    for r in data.get("reasons", []):
        print(f"  * {r}")

    if data.get("eval_report"):
        er = data["eval_report"]
        print("\nEvalReport:")
        print(f"- ok    : {er['ok']}")
        print(f"- score : {er['score']}")
        if er.get("issues"):
            print("- issues:")
            for i in er["issues"]:
                print(f"  * {i}")
        if er.get("warnings"):
            print("- warnings:")
            for w in er["warnings"]:
                print(f"  * {w}")
现在整个 Skill Evolution Loop 到哪一步了？
从蓝皮书的 3.4 来看，你现在已经有：

✅ 从成功 Trace 中挖技能模式

SkillMiner.mine_from_task：MultiAICouncil 负责从 Task + Trace + Lessons 输出一个 SkillCandidate（含 NAME/DESCRIPTION/CODE/TESTS）。

✅ 生成技能代码 + 测试

保存在 SkillCandidate.code / SkillCandidate.tests。

✅ 评估候选技能

SkillEvalHarness.evaluate_candidate：给出 ok/score/issues/warnings。

能被任何更强的实现替换（比如真 Sandbox 测试）。

✅ 晋升为正式技能 / 保留候选

SkillPromotionEngine.promote_candidate：

评估通过 → 创建 SkillDefinition，写入 registry；

评估不通过 → 标记为 rejected，保留 candidate；

所有结果 → 写入 ResearchManager（供后续研究）。

✅ CLI 操作链路

qctl skill-mine-from-goal --goal "..."：一键跑 CognitiveLoop & 挖一个 SkillCandidate。

qctl skill-list-candidates / skill-show-candidate：浏览候选技能。

qctl skill-promote-candidate --candidate-id ...：晋升候选技能。

qctl skill-list：查看当前 Skill 库。

这已经构成了一个完整的 “认知 → 挖技能 → 评估 → 晋升 → 研究记录” 小闭环。
