# quentin/traces/schemas.py

from __future__ import annotations

from datetime import datetime
from typing import Any, Dict, List, Optional

from pydantic import BaseModel, Field


class TraceSpan(BaseModel):
    """
    系统内部统一的 span 结构。

    设计目标：
    - 覆盖 Tracer 写入的所有字段
    - 不限制具体存储格式（KV / DB / 文件都行）
    - 方便后续 EpisodeBuilder 做聚合

    约定：
    - task_id / episode_id 等“归属于哪条任务”的信息，放在 meta 或 input 里
    """

    span_id: str
    ts: datetime = Field(default_factory=datetime.utcnow)

    layer: str               # "L0" / "L1" / "L2" / "L3"
    event: str               # "task_start" / "task_end" / "sleep_loop" / ...
    module: str              # "cognitive" / "sleep" / "skills" / ...

    input: Dict[str, Any] = Field(default_factory=dict)
    output: Dict[str, Any] = Field(default_factory=dict)
    meta: Dict[str, Any] = Field(default_factory=dict)


class TraceEpisode(BaseModel):
    """
    由多条 TraceSpan 聚合而成的一条 Episode（一次完整尝试 / 一次任务运行）。

    注意：这和 SkillMiner 里的 TraceEpisode 结构保持一致：
    {
      "episode_id": "...",
      "domain": "demo",
      "steps": [ {...}, {...} ],
      "meta": {...}
    }
    """

    episode_id: str
    domain: Optional[str] = None

    # 一个 episode 下的“步骤序列”
    # 每一步都可以是任意结构，但推荐：
    # {
    #   "event": "...",
    #   "module": "...",
    #   "layer": "L1",
    #   "tool": "add" / "planner" / "...",
    #   "input": {...},
    #   "output": {...},
    #   "span_id": "...",
    #   "ts": "..."
    # }
    steps: List[Dict[str, Any]] = Field(default_factory=list)

    # 其他元信息：task_id / trace_ids / 成功与否等
    meta: Dict[str, Any] = Field(default_factory=dict)
二、EpisodeBuilder：从 spans 聚合成 episodes
新建文件：quentin/traces/episode_builder.py

# quentin/traces/episode_builder.py

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Optional, Iterable
from collections import defaultdict

from quentin.traces.schemas import TraceSpan, TraceEpisode


@dataclass
class TraceEpisodeBuilder:
    """
    把 TraceSpan 列表聚合成 TraceEpisode 列表的工具类。

    v1 策略（非常简单版）：
    - 用 task_id 作为 episode_id（如果有的话）
    - 如果没有 task_id，则用 span.meta["episode_id"]，还没有就丢弃
    - 一个 episode 内的 steps 按时间排序

    你以后可以在这里玩更复杂的 episode 划分逻辑：
    - 按 trace window（时间窗口）
    - 按 domain / module 分类
    """

    # 从 span 中提取 episode_id 的逻辑
    def _extract_episode_id(self, span: TraceSpan) -> Optional[str]:
        # 优先 meta.episode_id
        ep = span.meta.get("episode_id")
        if ep:
            return str(ep)

        # 其次 meta.task_id
        ep = span.meta.get("task_id")
        if ep:
            return str(ep)

        # 再其次 input.task_id
        inp = span.input or {}
        ep = inp.get("task_id")
        if ep:
            return str(ep)

        # 最后 output.task_id（一般用不到）
        out = span.output or {}
        ep = out.get("task_id")
        if ep:
            return str(ep)

        return None

    def _extract_domain(self, span: TraceSpan) -> Optional[str]:
        # 优先 meta.domain
        dom = span.meta.get("domain")
        if dom:
            return str(dom)

        # 其次 input.domain
        dom = (span.input or {}).get("domain")
        if dom:
            return str(dom)

        return None

    def build_episodes_from_spans(
        self,
        spans: Iterable[TraceSpan],
        limit_episodes: Optional[int] = None,
    ) -> List[TraceEpisode]:
        """
        给定一堆 spans，按 episode_id 聚合成 episodes。
        """

        # 1) 分组：episode_id -> spans[]
        groups: Dict[str, List[TraceSpan]] = defaultdict(list)

        for span in spans:
            ep_id = self._extract_episode_id(span)
            if not ep_id:
                # 没有 episode_id 的 span 暂时丢弃
                continue
            groups[ep_id].append(span)

        # 2) 每组内部按 ts 排序
        episodes: List[TraceEpisode] = []

        for ep_id, sp_list in groups.items():
            sp_list_sorted = sorted(sp_list, key=lambda s: s.ts)

            # 选一个 domain（目前简单用第一条 span 提取）
            domain = self._extract_domain(sp_list_sorted[0])

            # 构造 steps
            steps: List[Dict[str, Any]] = []
            span_ids: List[str] = []

            for s in sp_list_sorted:
                step = {
                    "event": s.event,
                    "module": s.module,
                    "layer": s.layer,
                    "tool": s.meta.get("tool") or s.input.get("tool"),
                    "input": s.input,
                    "output": s.output,
                    "span_id": s.span_id,
                    "ts": s.ts.isoformat(),
                }
                steps.append(step)
                span_ids.append(s.span_id)

            ep = TraceEpisode(
                episode_id=ep_id,
                domain=domain,
                steps=steps,
                meta={
                    "span_ids": span_ids,
                    # 可加更多统计，比如 steps 数、开始/结束时间等
                    "num_steps": len(steps),
                    "started_at": sp_list_sorted[0].ts.isoformat(),
                    "finished_at": sp_list_sorted[-1].ts.isoformat(),
                },
            )
            episodes.append(ep)

        # 3) 只保留最新若干个 episode（按 finished_at 排序）
        episodes_sorted = sorted(
            episodes,
            key=lambda e: e.meta.get("finished_at", ""),
        )
        if limit_episodes is not None and limit_episodes > 0:
            episodes_sorted = episodes_sorted[-limit_episodes:]

        return episodes_sorted
约定：后面你在写 tracer.span() 的时候，只要在 meta 或 input 里加上 task_id，
EpisodeBuilder 就能把同一个任务过程中的 spans 聚合成一条 episode。

三、KVTraceEpisodeProvider：从 KV 中读 spans → 生成 episodes
接下来实现 SleepLoop 用到的 episodes_provider 真身。

从 KV 中读 span 列表

转成 TraceSpan

丢给 TraceEpisodeBuilder 聚合成 episodes

返回给 SkillMiningJob 使用（记得 SkillMiner 里用的是 dict，咱们 .model_dump() 一下即可）

新建文件：quentin/traces/episode_provider.py

# quentin/traces/episode_provider.py

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Optional

from quentin.infra.storage.kv_store import KVStore
from quentin.traces.schemas import TraceSpan, TraceEpisode
from quentin.traces.episode_builder import TraceEpisodeBuilder


@dataclass
class KVTraceEpisodeProvider:
    """
    一个基于 KVStore 的 episodes_provider 实现。

    假设：
    - 所有 TraceSpan 以 JSON 形式存储在：
        key = "trace:span:{span_id}"
    - 有一个 index key 记录所有 span_id 顺序：
        key = "trace:index:spans"  ->  [span_id1, span_id2, ...]

    如果你现有的 Tracer 存储格式不一样：
    - 要么调整 Tracer 的写入逻辑
    - 要么自己实现一个自定义 Provider 满足 get_recent_episodes 接口
    """

    kv: KVStore
    span_prefix: str = "trace:span:"
    span_index_key: str = "trace:index:spans"

    builder: TraceEpisodeBuilder = TraceEpisodeBuilder()

    async def _load_recent_spans(self, span_limit: int) -> List[TraceSpan]:
        """
        从 KV 中读取最近 span_limit 条 spans（倒序）。
        """
        ids = await self.kv.get(self.span_index_key) or []
        if not ids:
            return []

        # 去重并保留最新 span_limit 个
        # index 里假定是 append 顺序（旧→新），所以取末尾
        ids = list(dict.fromkeys(ids))[-span_limit:]

        spans: List[TraceSpan] = []
        for sid in ids:
            raw = await self.kv.get(f"{self.span_prefix}{sid}")
            if not raw:
                continue
            try:
                span = TraceSpan(**raw)
                spans.append(span)
            except Exception:
                # 某些旧格式 / 脏数据可以忽略
                continue

        return spans

    async def get_recent_episodes(self, limit: int = 100) -> List[Dict[str, Any]]:
        """
        提供给 SleepLoop.SkillMiningJob 使用的接口。

        返回值是「字典列表」，兼容 SkillMiner 预期：
        [
          {
            "episode_id": "...",
            "domain": "...",
            "steps": [...],
            "meta": {...}
          },
          ...
        ]
        """
        # 为了凑够 limit 个 episode，大致抓 limit * 10 个 spans（可以以后调参）
        spans = await self._load_recent_spans(span_limit=limit * 10)

        if not spans:
            return []

        episodes: List[TraceEpisode] = self.builder.build_episodes_from_spans(
            spans,
            limit_episodes=limit,
        )
        # SkillMiner 期望的是 dict
        return [ep.model_dump() for ep in episodes]
这一层是 SleepLoop / SkillMiner 和 Tracer 之间的“胶水层”。
只要你的 Tracer 把 spans 写进 KV 对应 key，这个 provider 就能工作。

四、Tracer 存储约定说明（轻量说明，不强行改你现有实现）
为了让上面的 provider 能用，我们需要一个约定：

Tracer 在写 span 时：

把 span 转成 TraceSpan 的结构（字段尽量对齐）

存到 key = "trace:span:{span_id}"

同时在 key = "trace:index:spans" 这个 list 里 append 这个 span_id

你现有的 Tracer.span(...) 可能已经在做类似事情，如果不是，可以朝这个方向微调。比如：

# 伪代码（仅示意）
span_id = "span_" + uuid.uuid4().hex
span = TraceSpan(
    span_id=span_id,
    layer=layer,
    event=event,
    module=module,
    input=input or {},
    output=output or {},
    meta=meta or {},
)

await kv.save(f"trace:span:{span_id}", span.model_dump())
await kv.append_list("trace:index:spans", span_id)
只要保证这一点，EpisodesProvider 就通了。

六、CLI：快速查看 episode 抽取情况（可选但很有用）
再给你一个小命令，方便你检查 EpisodeBuilder / Provider 是否工作正常。

修改：quentin/cli/qctl.py

@cli.command("trace-episodes")
@click.option("--limit", default=10, show_default=True,
              help="Number of recent episodes to show.")
def trace_episodes_cmd(limit: int):
    """
    Dump recent trace episodes (for debugging / research):

    - 使用 KVTraceEpisodeProvider 从 KV 中读取 spans
    - 聚合成 episodes
    - 打印基本信息
    """

    async def _run():
        import json
        settings = Settings()
        runtime = AgentRuntime.create_from_settings(settings)

        provider = runtime.sleep_loop.episodes_provider  # 或直接创建 KVTraceEpisodeProvider(kv=runtime.kv_store)
        episodes = await provider.get_recent_episodes(limit=limit)

        print(json.dumps(episodes, ensure_ascii=False, indent=2))

    asyncio.run(_run())
