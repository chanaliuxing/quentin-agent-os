from __future__ import annotations

from dataclasses import dataclass, field, asdict
from datetime import datetime
from typing import Dict, Any, Literal, Optional

MemoryTier = Literal["STS", "MTS", "LTS"]
MemoryKind = Literal["episodic", "semantic", "skill", "other"]


@dataclass
class UpdatePolicy:
    """
    对应蓝皮书里的 update_policy 字段：
      - promotion_rule: 什么时候提升 tier
      - decay_rule    : 什么时候衰减 / 删除
      - consolidation_batch: "nightly" / "weekly" / "manual"
    v1 先当作元数据存着，具体规则由 SleepLoop 解释。
    """
    promotion_rule: str = "auto"
    decay_rule: str = "auto"
    consolidation_batch: str = "nightly"

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "UpdatePolicy":
        if d is None:
            return cls()
        return cls(
            promotion_rule=d.get("promotion_rule", "auto"),
            decay_rule=d.get("decay_rule", "auto"),
            consolidation_batch=d.get("consolidation_batch", "nightly"),
        )


@dataclass
class MemoryRecord:
    """
    Memory Fabric 的基础单位。

    - memory_id      : 唯一 ID
    - tier           : STS / MTS / LTS
    - kind           : episodic / semantic / skill / other
    - content        : 原始内容（文本为主，后续可扩展为结构化）
    - vector_ref     : 向量存储引用（例如外部向量库的 key），v1 先留字段
    - meta           : 额外元数据（domain、task_id、trace_id、来源模型等）

    - created_at     : 创建时间
    - updated_at     : 最近更新
    - last_accessed  : 最近访问
    - access_count   : 总访问次数

    - update_policy  : 升级/遗忘策略（蓝皮书字段）
    """
    memory_id: str
    tier: MemoryTier
    kind: MemoryKind
    content: str
    vector_ref: Optional[str] = None
    meta: Dict[str, Any] = field(default_factory=dict)

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    last_accessed_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    access_count: int = 0

    update_policy: UpdatePolicy = field(default_factory=UpdatePolicy)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        d["update_policy"] = self.update_policy.to_dict()
        return d

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "MemoryRecord":
        return cls(
            memory_id=d["memory_id"],
            tier=d.get("tier", "STS"),
            kind=d.get("kind", "other"),
            content=d.get("content", ""),
            vector_ref=d.get("vector_ref"),
            meta=d.get("meta", {}) or {},
            created_at=d.get("created_at", ""),
            updated_at=d.get("updated_at", ""),
            last_accessed_at=d.get("last_accessed_at", ""),
            access_count=int(d.get("access_count", 0)),
            update_policy=UpdatePolicy.from_dict(d.get("update_policy") or {}),
        )
二、MemoryFabric：三层记忆 + 简单 JSON 存储
新文件：quentin/memory/fabric.py

# quentin/memory/fabric.py
from __future__ import annotations

import json
import logging
import uuid
from datetime import datetime
from pathlib import Path
from typing import Dict, Any, List, Optional, Iterable

from quentin.memory.models import MemoryRecord, MemoryTier, MemoryKind

logger = logging.getLogger(__name__)


class FileBackedMemoryFabric:
    """
    v1：使用单个 JSON 文件存所有记忆：
      {
        "records": {
          "<memory_id>": { ... MemoryRecord dict ... },
          ...
        }
      }

    未来如果你接入向量库 / 数据库，只要保持外部接口不变即可。
    """

    def __init__(self, path: str | Path = "data/memory_fabric.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._loaded = False
        self._records: Dict[str, MemoryRecord] = {}

    # ---------- 内部加载/保存 ----------

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                raw = json.loads(self.path.read_text(encoding="utf-8") or "{}")
                recs_raw = raw.get("records", {}) or {}
                self._records = {
                    mid: MemoryRecord.from_dict(rd) for mid, rd in recs_raw.items()
                }
            except Exception:
                logger.exception("Failed to load MemoryFabric from %s", self.path)
                self._records = {}
        else:
            self._records = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        raw = {
            "records": {mid: r.to_dict() for mid, r in self._records.items()},
        }
        self.path.write_text(
            json.dumps(raw, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    # ---------- 基础 CRUD ----------

    def create(
        self,
        *,
        tier: MemoryTier,
        kind: MemoryKind,
        content: str,
        meta: Optional[Dict[str, Any]] = None,
        vector_ref: Optional[str] = None,
    ) -> MemoryRecord:
        self._load()
        mid = str(uuid.uuid4())
        rec = MemoryRecord(
            memory_id=mid,
            tier=tier,
            kind=kind,
            content=content,
            meta=meta or {},
            vector_ref=vector_ref,
        )
        self._records[mid] = rec
        self._save()
        return rec

    def upsert(self, rec: MemoryRecord) -> None:
        self._load()
        rec.updated_at = datetime.utcnow().isoformat()
        self._records[rec.memory_id] = rec
        self._save()

    def get(self, memory_id: str) -> Optional[MemoryRecord]:
        self._load()
        return self._records.get(memory_id)

    def delete(self, memory_id: str) -> None:
        self._load()
        if memory_id in self._records:
            del self._records[memory_id]
            self._save()

    # ---------- 查询 & 迭代 ----------

    def list_all(self) -> List[MemoryRecord]:
        self._load()
        return list(self._records.values())

    def iter_by_tier(
        self,
        tier: MemoryTier,
        *,
        limit: Optional[int] = None,
    ) -> Iterable[MemoryRecord]:
        self._load()
        count = 0
        for rec in self._records.values():
            if rec.tier != tier:
                continue
            yield rec
            count += 1
            if limit is not None and count >= limit:
                break

    def search_by_meta(
        self,
        *,
        tier: Optional[MemoryTier] = None,
        meta_contains: Optional[Dict[str, Any]] = None,
        limit: Optional[int] = None,
    ) -> List[MemoryRecord]:
        """
        非向量检索，只是简单按 meta 里的 key/value 过滤。
        v1 主要用于 SleepLoop、调试。
        """
        self._load()
        res: List[MemoryRecord] = []
        for rec in self._records.values():
            if tier and rec.tier != tier:
                continue
            if meta_contains:
                ok = True
                for k, v in meta_contains.items():
                    if rec.meta.get(k) != v:
                        ok = False
                        break
                if not ok:
                    continue
            res.append(rec)
            if limit is not None and len(res) >= limit:
                break
        return res

    # ---------- 访问记录 & 层级移动 ----------

    def record_access(self, memory_id: str) -> None:
        rec = self.get(memory_id)
        if not rec:
            return
        rec.access_count += 1
        rec.last_accessed_at = datetime.utcnow().isoformat()
        self.upsert(rec)

    def move_to_tier(self, memory_id: str, new_tier: MemoryTier) -> None:
        rec = self.get(memory_id)
        if not rec:
            return
        if rec.tier == new_tier:
            return
        rec.tier = new_tier
        rec.updated_at = datetime.utcnow().isoformat()
        self.upsert(rec)

# quentin/memory/sleep_loop.py
from __future__ import annotations

import logging
from dataclasses import dataclass, asdict
from datetime import datetime, timedelta
from typing import Dict, Any, List, Optional

from quentin.memory.fabric import FileBackedMemoryFabric
from quentin.memory.models import MemoryRecord, MemoryTier

logger = logging.getLogger(__name__)


@dataclass
class SleepLoopConfig:
    """
    对应蓝皮书 "Sleep / Consolidation Loop" 的关键参数。

    v1 规则比较简单：
      - STS → MTS:
          - access_count >= sts_to_mts_min_access
          - age_hours >= sts_to_mts_min_age_hours
      - MTS → LTS:
          - access_count >= mts_to_lts_min_access
          - age_days   >= mts_to_lts_min_age_days

      - 遗忘：
          - STS 若 age_hours > sts_max_age_hours 且 access_count == 0 → 删除
          - MTS 若 age_days   > mts_decay_days   且 access_count 很低 → 删除
    """
    sts_to_mts_min_access: int = 3
    sts_to_mts_min_age_hours: float = 1.0

    mts_to_lts_min_access: int = 5
    mts_to_lts_min_age_days: float = 3.0

    sts_max_age_hours: float = 72.0
    mts_decay_days: float = 30.0

    max_records_per_run: int = 1000

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class SleepLoopReport:
    promoted_sts_to_mts: int = 0
    promoted_mts_to_lts: int = 0
    deleted_sts: int = 0
    deleted_mts: int = 0
    notes: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class SleepLoop:
    """
    每次 run_once 做三件事：
      1) 从 STS/MTS 中挑出需要提升的记录
      2) 从 STS/MTS 中找出需要遗忘/删除的记录
      3) 预留“语义总结 / 技能抽取”的 Hook（未来接 MultiAICouncil）

    注意：
      - 这里只负责 Memory Fabric 层级的变化，不负责写入新技能 / semantic memory。
        但会在日志里说明哪些记录“候选可总结”。
    """

    def __init__(
        self,
        *,
        memory_fabric: Optional[FileBackedMemoryFabric] = None,
        config: Optional[SleepLoopConfig] = None,
    ) -> None:
        self.memory_fabric = memory_fabric or FileBackedMemoryFabric()
        self.config = config or SleepLoopConfig()

    # ---------- 主入口 ----------

    def run_once(self) -> SleepLoopReport:
        logger.info("SleepLoop: start run_once()")
        report = SleepLoopReport()

        # 1) 处理 STS
        sts_records = list(
            self.memory_fabric.iter_by_tier(
                "STS",
                limit=self.config.max_records_per_run,
            )
        )
        self._process_sts(sts_records, report)

        # 2) 处理 MTS
        mts_records = list(
            self.memory_fabric.iter_by_tier(
                "MTS",
                limit=self.config.max_records_per_run,
            )
        )
        self._process_mts(mts_records, report)

        logger.info("SleepLoop: finished run_once() with %s", report.to_dict())
        return report

    # ---------- 具体规则 ----------

    def _process_sts(self, records: List[MemoryRecord], report: SleepLoopReport) -> None:
        now = datetime.utcnow()

        for rec in records:
            created = self._parse_ts(rec.created_at) or now
            age_hours = (now - created).total_seconds() / 3600.0

            # 1) promotion: STS -> MTS
            if (
                rec.access_count >= self.config.sts_to_mts_min_access
                and age_hours >= self.config.sts_to_mts_min_age_hours
            ):
                self.memory_fabric.move_to_tier(rec.memory_id, "MTS")
                report.promoted_sts_to_mts += 1
                continue

            # 2) decay/forget: STS 过期且从未访问
            if (
                rec.access_count == 0
                and age_hours >= self.config.sts_max_age_hours
            ):
                self.memory_fabric.delete(rec.memory_id)
                report.deleted_sts += 1

    def _process_mts(self, records: List[MemoryRecord], report: SleepLoopReport) -> None:
        now = datetime.utcnow()

        for rec in records:
            created = self._parse_ts(rec.created_at) or now
            age_days = (now - created).days

            # 1) promotion: MTS -> LTS
            if (
                rec.access_count >= self.config.mts_to_lts_min_access
                and age_days >= self.config.mts_to_lts_min_age_days
            ):
                self.memory_fabric.move_to_tier(rec.memory_id, "LTS")
                report.promoted_mts_to_lts += 1
                continue

            # 2) decay/forget: MTS 太老且几乎没人用
            if (
                age_days >= self.config.mts_decay_days
                and rec.access_count <= 1
            ):
                self.memory_fabric.delete(rec.memory_id)
                report.deleted_mts += 1

    @staticmethod
    def _parse_ts(s: str) -> Optional[datetime]:
        if not s:
            return None
        try:
            return datetime.fromisoformat(s)
        except Exception:
            return None

把“很老且几乎不用的 MTS”删除
