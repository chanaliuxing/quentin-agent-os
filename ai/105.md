# quentin/eval/harness.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, List, Optional
import logging

from quentin.runtime.system import TaskResult

logger = logging.getLogger(__name__)


@dataclass
class EvalCase:
    """
    一个离线评测样本（用来跑 A/B 测试）。

    对于通用智能体来说，一个 case 基本就是：
      - goal: 任务目标（自然语言）
      - context: 附加上下文（domain, user_profile, extra flags）
      - expected: 可选的“期望输出”或标签（例如正确答案、期望结构）
    """
    case_id: str
    goal: str
    context: Dict[str, Any]
    expected: Optional[Any] = None
    meta: Dict[str, Any] = None

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        if d["meta"] is None:
            d["meta"] = {}
        return d


@dataclass
class EvalCaseResult:
    """
    单个 case 的评测结果。
    """
    case_id: str
    variant_id: str
    success: float        # 0.0 ~ 1.0
    status: str
    latency_sec: Optional[float]
    raw_metrics: Dict[str, Any]
    notes: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class VariantEvalSummary:
    """
    一个 variant 在整个实验里的整体表现。
    """
    experiment_id: str
    variant_id: str
    num_cases: int
    success_rate: float
    avg_latency_sec: Optional[float]
    extra_metrics: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class EvalHarness:
    """
    v1 版 Eval Harness（评测框架）。

    目标：
      - 给 ExperimentOrchestrator 提供统一的 evaluate 接口
      - 默认用最简单的规则：status == "success" 视为 success=1.0
      - 把延迟等指标一起打包，方便后续扩展为更复杂的打分逻辑

    以后你可以：
      - 根据 domain（Trading / Legal / Desktop）注册自定义打分器
      - 引入自动对比 expected vs output 的语义评测（LLM-as-judge）
    """

    def __init__(self, *, name: str = "default-eval-harness") -> None:
        self.name = name

    # --------------------------------------------------
    # 单个任务结果的评测
    # --------------------------------------------------
    def evaluate_task_result(
        self,
        *,
        experiment_id: str,
        variant_id: str,
        case: EvalCase,
        task_result: TaskResult,
        config: Optional[Dict[str, Any]] = None,
    ) -> EvalCaseResult:
        """
        对单个 (variant, case) 的 TaskResult 做评测。

        v1 策略：
          - status == "success" -> success = 1.0
          - status == "failed" 或 "error" -> success = 0.0
          - latency 从 trace.top_level_metrics["latency_sec"] 读取（如果有）
        """
        trace = task_result.trace
        status = trace.status or "unknown"
        metrics = trace.top_level_metrics or {}

        if status == "success":
            success = 1.0
        elif status in ("failed", "error"):
            success = 0.0
        else:
            success = 0.0

        latency = metrics.get("latency_sec")

        notes = ""
        if status != "success":
            notes = f"status={status}; may inspect trace for details."

        raw_metrics = {
            "status": status,
            "latency_sec": latency,
            "llm_calls": metrics.get("llm_calls"),
            "experiment_id": experiment_id,
            "variant_id": variant_id,
        }

        # 以后可以在这里注入 domain-specific 的评分逻辑
        # 比如 Trading domain: 根据收益/回撤计算 score
        # 比如 Legal domain: 根据正确性/合规性评估 score

        return EvalCaseResult(
            case_id=case.case_id,
            variant_id=variant_id,
            success=success,
            status=status,
            latency_sec=latency,
            raw_metrics=raw_metrics,
            notes=notes,
        )

    # --------------------------------------------------
    # 聚合一个 variant 的所有 case 结果
    # --------------------------------------------------
    def aggregate_variant_metrics(
        self,
        *,
        experiment_id: str,
        variant_id: str,
        case_results: List[EvalCaseResult],
    ) -> VariantEvalSummary:
        """
        对一个 variant 的所有 case_result 做聚合。

        默认指标：
          - success_rate: mean(success)
          - avg_latency_sec: mean(latency_sec)（忽略 None）
        """
        num_cases = len(case_results)
        if num_cases == 0:
            return VariantEvalSummary(
                experiment_id=experiment_id,
                variant_id=variant_id,
                num_cases=0,
                success_rate=0.0,
                avg_latency_sec=None,
                extra_metrics={},
            )

        success_sum = 0.0
        latency_sum = 0.0
        latency_count = 0

        for r in case_results:
            success_sum += r.success
            if r.latency_sec is not None:
                latency_sum += r.latency_sec
                latency_count += 1

        success_rate = success_sum / num_cases
        avg_latency = latency_sum / latency_count if latency_count > 0 else None

        extra = {
            "num_success_cases": sum(1 for r in case_results if r.success >= 1.0),
            "num_failed_cases": sum(1 for r in case_results if r.success < 1.0),
        }

        return VariantEvalSummary(
            experiment_id=experiment_id,
            variant_id=variant_id,
            num_cases=num_cases,
            success_rate=success_rate,
            avg_latency_sec=avg_latency,
            extra_metrics=extra,
        )

# quentin/research/experiments.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging
import time
import uuid

from quentin.eval.harness import (
    EvalHarness,
    EvalCase,
    EvalCaseResult,
    VariantEvalSummary,
)
from quentin.runtime.system import MetaAgentRuntime, TaskResult

logger = logging.getLogger(__name__)


@dataclass
class ExperimentTask:
    """
    一个实验里的单个任务定义（近似于 EvalCase 的核心部分）。

    实际跑的时候，会转成 EvalCase。
    """
    task_id: str
    goal: str
    context: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class ExperimentConfig:
    """
    实验配置（机器可读）：

    - experiment_id: 唯一 ID
    - name / description: 便于人类识别
    - variants: 变体列表（ID 字符串），每个 ID 对应一个 MetaAgentRuntime
                 实现上：ExperimentOrchestrator 构造时注入 variant_id -> runtime 的 mapping
    - tasks: 任务列表
    - meta: 额外信息（domain / notes / tags）
    """
    experiment_id: str
    name: str
    description: str
    variants: List[str]
    tasks: List[ExperimentTask]
    meta: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "experiment_id": self.experiment_id,
            "name": self.name,
            "description": self.description,
            "variants": self.variants,
            "tasks": [t.to_dict() for t in self.tasks],
            "meta": self.meta,
        }

    @staticmethod
    def from_dict(data: Dict[str, Any]) -> ExperimentConfig:
        tasks = [
            ExperimentTask(
                task_id=t.get("task_id"),
                goal=t.get("goal"),
                context=t.get("context") or {},
            )
            for t in data.get("tasks", [])
        ]
        return ExperimentConfig(
            experiment_id=data.get("experiment_id"),
            name=data.get("name"),
            description=data.get("description", ""),
            variants=list(data.get("variants") or []),
            tasks=tasks,
            meta=dict(data.get("meta") or {}),
        )


@dataclass
class VariantRunReport:
    """
    一个 variant 在某次 run_experiment 中的结果：
      - case_results: 每个 case 的评测结果
      - summary: 聚合指标
    """
    variant_id: str
    case_results: List[EvalCaseResult]
    summary: VariantEvalSummary

    def to_dict(self) -> Dict[str, Any]:
        return {
            "variant_id": self.variant_id,
            "case_results": [c.to_dict() for c in self.case_results],
            "summary": self.summary.to_dict(),
        }


@dataclass
class ExperimentRunResult:
    """
    一次完整实验运行的结果。

    - run_id: 本次运行 ID
    - experiment_id: 对应哪个实验
    - started_at / finished_at: 时间戳
    - variant_reports: 每个 variant 的报告
    """
    run_id: str
    experiment_id: str
    started_at: float
    finished_at: float
    variant_reports: Dict[str, VariantRunReport]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "run_id": self.run_id,
            "experiment_id": self.experiment_id,
            "started_at": self.started_at,
            "finished_at": self.finished_at,
            "variant_reports": {
                vid: report.to_dict() for vid, report in self.variant_reports.items()
            },
        }


@dataclass
class ExperimentRecord:
    """
    完整的实验记录：
      - config: 实验配置
      - runs: 历史运行记录列表
    """
    config: ExperimentConfig
    runs: List[ExperimentRunResult]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "config": self.config.to_dict(),
            "runs": [r.to_dict() for r in self.runs],
        }

    @staticmethod
    def from_dict(data: Dict[str, Any]) -> ExperimentRecord:
        cfg = ExperimentConfig.from_dict(data.get("config") or {})
        runs_data = data.get("runs") or []
        runs: List[ExperimentRunResult] = []
        for r in runs_data:
            vr_dict = r.get("variant_reports") or {}
            variant_reports: Dict[str, VariantRunReport] = {}
            for vid, vrd in vr_dict.items():
                case_results = [
                    EvalCaseResult(
                        case_id=cr.get("case_id"),
                        variant_id=cr.get("variant_id"),
                        success=float(cr.get("success", 0.0)),
                        status=cr.get("status", "unknown"),
                        latency_sec=cr.get("latency_sec"),
                        raw_metrics=cr.get("raw_metrics") or {},
                        notes=cr.get("notes") or "",
                    )
                    for cr in vrd.get("case_results", [])
                ]
                summary_data = vrd.get("summary") or {}
                summary = VariantEvalSummary(
                    experiment_id=summary_data.get("experiment_id", cfg.experiment_id),
                    variant_id=summary_data.get("variant_id", vid),
                    num_cases=int(summary_data.get("num_cases", len(case_results))),
                    success_rate=float(summary_data.get("success_rate", 0.0)),
                    avg_latency_sec=summary_data.get("avg_latency_sec"),
                    extra_metrics=summary_data.get("extra_metrics") or {},
                )
                variant_reports[vid] = VariantRunReport(
                    variant_id=vid,
                    case_results=case_results,
                    summary=summary,
                )
            runs.append(
                ExperimentRunResult(
                    run_id=r.get("run_id"),
                    experiment_id=r.get("experiment_id", cfg.experiment_id),
                    started_at=float(r.get("started_at", 0.0)),
                    finished_at=float(r.get("finished_at", 0.0)),
                    variant_reports=variant_reports,
                )
            )
        return ExperimentRecord(config=cfg, runs=runs)


class ExperimentOrchestrator:
    """
    L3 Research Layer 的核心组件之一：实验编排器。

    职责：
      - 管理 ExperimentConfig / ExperimentRecord
      - 调用多个 MetaAgentRuntime，对同一批任务进行 A/B 测试
      - 使用 EvalHarness 计算指标
      - 提供：
          * create_experiment(config)
          * run_experiment(experiment_id)
          * compare_variants(experiment_id, variant_a, variant_b)
          * export_results(experiment_id)

    设计思路：
      - variants_runtimes: variant_id -> MetaAgentRuntime
        （即不同版本/配置的智能体 runtime）
      - 持久化：每个 experiment 一个 JSON 文件，位于 data/research/experiments/<id>.json
    """

    def __init__(
        self,
        *,
        repo_root: str | Path,
        variants_runtimes: Dict[str, MetaAgentRuntime],
        eval_harness: Optional[EvalHarness] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.variants_runtimes = variants_runtimes
        self.eval_harness = eval_harness or EvalHarness()

        self.storage_dir = self.repo_root / "data" / "research" / "experiments"
        self.storage_dir.mkdir(parents=True, exist_ok=True)

        self._experiments: Dict[str, ExperimentRecord] = {}
        self._load_existing_experiments()

    # --------------------------------------------------
    # 公共 API：创建 / 运行 / 比较 / 导出
    # --------------------------------------------------
    def create_experiment(
        self,
        *,
        name: str,
        description: str,
        variants: Optional[List[str]] = None,
        tasks: Optional[List[ExperimentTask]] = None,
        meta: Optional[Dict[str, Any]] = None,
        experiment_id: Optional[str] = None,
    ) -> ExperimentRecord:
        """
        蓝皮书 API：create_experiment(config)

        这里用参数显式列出，内部构造 ExperimentConfig。
        """
        exp_id = experiment_id or f"exp::{uuid.uuid4().hex}"
        variants = variants or list(self.variants_runtimes.keys())
        tasks = tasks or []
        meta = meta or {}

        cfg = ExperimentConfig(
            experiment_id=exp_id,
            name=name,
            description=description,
            variants=variants,
            tasks=tasks,
            meta=meta,
        )

        record = ExperimentRecord(config=cfg, runs=[])
        self._experiments[exp_id] = record
        self._save_experiment(record)
        return record

    def run_experiment(
        self,
        experiment_id: str,
        *,
        max_tasks: Optional[int] = None,
    ) -> ExperimentRunResult:
        """
        蓝皮书 API：run_experiment(experiment_id)

        对指定实验的所有 variants 和 tasks 执行一轮完整评测。
        """
        record = self._get_experiment_or_raise(experiment_id)
        cfg = record.config

        # 1) 构造 EvalCase 列表
        eval_cases: List[EvalCase] = []
        for t in cfg.tasks:
            eval_cases.append(
                EvalCase(
                    case_id=t.task_id,
                    goal=t.goal,
                    context=t.context,
                    expected=None,
                    meta={"experiment_id": experiment_id},
                )
            )
        if max_tasks is not None:
            eval_cases = eval_cases[:max_tasks]

        start_t = time.time()
        variant_reports: Dict[str, VariantRunReport] = {}

        # 2) 对每个 variant 跑这批 case
        for vid in cfg.variants:
            runtime = self.variants_runtimes.get(vid)
            if runtime is None:
                logger.warning("Experiment %s: variant %s has no runtime, skip", experiment_id, vid)
                continue

            case_results: List[EvalCaseResult] = []

            for case in eval_cases:
                # 调用 runtime.run_task
                ctx = {
                    **case.context,
                    "domain": cfg.meta.get("domain", "general"),
                    "experiment_id": experiment_id,
                    "variant_id": vid,
                }
                task_result: TaskResult = runtime.run_task(case.goal, ctx)

                # 交给 EvalHarness 打分
                eval_res = self.eval_harness.evaluate_task_result(
                    experiment_id=experiment_id,
                    variant_id=vid,
                    case=case,
                    task_result=task_result,
                    config={"source": "experiment_orchestrator"},
                )
                case_results.append(eval_res)

            # 聚合 variant 指标
            summary = self.eval_harness.aggregate_variant_metrics(
                experiment_id=experiment_id,
                variant_id=vid,
                case_results=case_results,
            )

            variant_reports[vid] = VariantRunReport(
                variant_id=vid,
                case_results=case_results,
                summary=summary,
            )

        finished_t = time.time()
        run_id = f"run::{uuid.uuid4().hex}"

        run_result = ExperimentRunResult(
            run_id=run_id,
            experiment_id=experiment_id,
            started_at=start_t,
            finished_at=finished_t,
            variant_reports=variant_reports,
        )

        record.runs.append(run_result)
        self._save_experiment(record)
        return run_result

    def compare_variants(
        self,
        experiment_id: str,
        variant_a: Optional[str] = None,
        variant_b: Optional[str] = None,
    ) -> Dict[str, Any]:
        """
        蓝皮书 API：compare_variants(variant_a, variant_b)

        v1 实现：
          - 使用最近一次 run_experiment 的结果
          - 如果传入 variant_a / variant_b，则只比较这两个
          - 否则返回所有 variants 的 success_rate 排序列表
        """
        record = self._get_experiment_or_raise(experiment_id)
        if not record.runs:
            raise ValueError(f"Experiment {experiment_id} has no runs yet.")

        last_run = record.runs[-1]
        reports = last_run.variant_reports

        def _summary_for(vid: str) -> VariantEvalSummary:
            rep = reports.get(vid)
            if rep is None:
                raise ValueError(f"Variant {vid} not found in last run of {experiment_id}")
            return rep.summary

        if variant_a and variant_b:
            sa = _summary_for(variant_a)
            sb = _summary_for(variant_b)
            return {
                "experiment_id": experiment_id,
                "last_run_id": last_run.run_id,
                "variant_a": sa.to_dict(),
                "variant_b": sb.to_dict(),
                "better": (
                    variant_a if sa.success_rate > sb.success_rate
                    else variant_b if sb.success_rate > sa.success_rate
                    else "tie"
                ),
            }

        # 没有指定具体 variant，就返回排序列表
        summaries = [rep.summary for rep in reports.values()]
        summaries_sorted = sorted(
            summaries,
            key=lambda s: s.success_rate,
            reverse=True,
        )
        return {
            "experiment_id": experiment_id,
            "last_run_id": last_run.run_id,
            "ranked_variants": [s.to_dict() for s in summaries_sorted],
        }

    def export_results(self, experiment_id: str) -> Dict[str, Any]:
        """
        蓝皮书 API：export_results(experiment_id)

        v1 实现：
          - 返回 ExperimentRecord 的完整 dict
          - 同时确保磁盘上已经有最新 JSON
        """
        record = self._get_experiment_or_raise(experiment_id)
        self._save_experiment(record)  # 确保持久化
        return record.to_dict()

    # --------------------------------------------------
    # 内部：加载 / 保存
    # --------------------------------------------------
    def _load_existing_experiments(self) -> None:
        """
        启动时从 data/research/experiments 目录加载所有 experiment JSON。
        """
        for path in self.storage_dir.glob("*.json"):
            try:
                with path.open("r", encoding="utf-8") as f:
                    data = json.load(f)
                record = ExperimentRecord.from_dict(data)
                self._experiments[record.config.experiment_id] = record
            except Exception:
                logger.exception("Failed to load experiment file: %s", path)

    def _save_experiment(self, record: ExperimentRecord) -> None:
        exp_id = record.config.experiment_id
        path = self.storage_dir / f"{exp_id.replace('::', '_')}.json"
        data = record.to_dict()
        tmp_path = path.with_suffix(".json.tmp")
        with tmp_path.open("w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        tmp_path.replace(path)

    def _get_experiment_or_raise(self, experiment_id: str) -> ExperimentRecord:
        record = self._experiments.get(experiment_id)
        if record is None:
            raise KeyError(f"Experiment not found: {experiment_id}")
        return record
