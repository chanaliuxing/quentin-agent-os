# quentin/metrics/aggregator.py
from __future__ import annotations

from dataclasses import asdict
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import datetime as dt

from quentin.metrics.schema import RunMetrics


def load_run_metrics_from_dir(runs_dir: str | Path) -> List[RunMetrics]:
    """
    从 data/runs/blueprint_self_iterate 目录中读取所有 metrics_*.json。

    返回按 started_at 升序排序的 RunMetrics 列表。
    """
    runs_path = Path(runs_dir).resolve()
    metrics_files = sorted(runs_path.glob("metrics_*.json"))
    results: List[RunMetrics] = []

    for path in metrics_files:
        try:
            with path.open("r", encoding="utf-8") as f:
                data = json.load(f)
        except Exception:
            continue

        try:
            rm = RunMetrics(**data)
        except TypeError:
            # 字段不匹配时忽略这条，避免整个汇总崩掉
            continue

        results.append(rm)

    # 按 started_at 排序（如果格式不标准，就保持原顺序）
    def _parse_ts(ts: str) -> dt.datetime:
        try:
            return dt.datetime.fromisoformat(ts)
        except Exception:
            return dt.datetime.min

    results.sort(key=lambda r: _parse_ts(r.started_at))
    return results


def summarize_progress(metrics_list: List[RunMetrics]) -> Dict[str, Any]:
    """
    对一个 RunMetrics 序列做整体概览。

    返回一个 dict，适合直接打印或转成 JSON：
      - total_runs
      - latest_run (dict)
      - first_run (dict)
      - improvement (done_increase, missing_decrease, done_ratio_change, 等)
      - avg_experiment_success_rate
    """
    if not metrics_list:
        return {
            "total_runs": 0,
            "latest_run": None,
            "first_run": None,
            "improvement": None,
            "avg_experiment_success_rate": None,
        }

    first = metrics_list[0]
    latest = metrics_list[-1]

    # 蓝皮书能力覆盖度变化
    first_total = max(first.total_capabilities, 1)
    latest_total = max(latest.total_capabilities, 1)

    first_done_ratio = first.num_done / first_total
    latest_done_ratio = latest.num_done / latest_total

    first_missing_ratio = first.num_missing / first_total
    latest_missing_ratio = latest.num_missing / latest_total

    improvement = {
        "done_increase": latest.num_done - first.num_done,
        "missing_decrease": first.num_missing - latest.num_missing,
        "done_ratio_change": latest_done_ratio - first_done_ratio,
        "missing_ratio_change": first_missing_ratio - latest_missing_ratio,
    }

    # 实验成功率（简单平均）
    success_rates: List[float] = []
    for m in metrics_list:
        if m.experiments_executed > 0:
            success_rates.append(m.experiments_succeeded / m.experiments_executed)
    avg_success_rate = sum(success_rates) / len(success_rates) if success_rates else None

    return {
        "total_runs": len(metrics_list),
        "latest_run": asdict(latest),
        "first_run": asdict(first),
        "improvement": improvement,
        "avg_experiment_success_rate": avg_success_rate,
    }


def to_timeseries(metrics_list: List[RunMetrics]) -> Dict[str, List[Any]]:
    """
    把一串 RunMetrics 转成简单的时间序列视图，方便喂给绘图 / dashboard。

    返回字段：
      - timestamps
      - num_done_series
      - num_missing_series
      - experiments_succeeded_series
      - experiments_failed_series
    """
    ts: List[str] = []
    done_series: List[int] = []
    missing_series: List[int] = []
    exp_succ: List[int] = []
    exp_fail: List[int] = []

    for m in metrics_list:
        ts.append(m.started_at)
        done_series.append(m.num_done)
        missing_series.append(m.num_missing)
        exp_succ.append(m.experiments_succeeded)
        exp_fail.append(m.experiments_failed)

    return {
        "timestamps": ts,
        "num_done_series": done_series,
        "num_missing_series": missing_series,
        "experiments_succeeded_series": exp_succ,
        "experiments_failed_series": exp_fail,
    }

# scripts/metrics_summary.py
"""
命令行查看蓝皮书自迭代的整体进度。

用法（在仓库根目录）：
    python -m scripts.metrics_summary

可选参数：
    --runs-dir PATH   指定 runs 目录（默认 data/runs/blueprint_self_iterate）
"""

from __future__ import annotations

import argparse
import json
from pathlib import Path

from quentin.metrics.aggregator import (
    load_run_metrics_from_dir,
    summarize_progress,
    to_timeseries,
)


def main() -> None:
    parser = argparse.ArgumentParser(description="Show summary of blueprint self-iteration progress.")
    parser.add_argument(
        "--runs-dir",
        type=str,
        default="data/runs/blueprint_self_iterate",
        help="Directory containing metrics_*.json files.",
    )
    parser.add_argument(
        "--json",
        action="store_true",
        help="Output full summary as JSON (otherwise print human-readable text).",
    )
    args = parser.parse_args()

    runs_dir = Path(args.runs_dir).resolve()
    metrics_list = load_run_metrics_from_dir(runs_dir)

    if not metrics_list:
        print(f"No metrics_*.json found in {runs_dir}")
        return

    summary = summarize_progress(metrics_list)

    if args.json:
        print(json.dumps(summary, ensure_ascii=False, indent=2))
        return

    # 人类可读模式
    print("==== Blueprint Self-Iteration Summary ====")
    print(f"Runs directory : {runs_dir}")
    print(f"Total runs     : {summary['total_runs']}")

    first = summary["first_run"]
    latest = summary["latest_run"]
    improvement = summary["improvement"]

    if first and latest:
        print("\n-- First run --")
        print(f"  run_id       : {first['run_id']}")
        print(f"  started_at   : {first['started_at']}")
        print(f"  total caps   : {first['total_capabilities']}")
        print(f"  done         : {first['num_done']}")
        print(f"  missing      : {first['num_missing']}")
        print(f"  partial      : {first['num_partial']}")
        print(f"  unknown      : {first['num_unknown']}")

        print("\n-- Latest run --")
        print(f"  run_id       : {latest['run_id']}")
        print(f"  started_at   : {latest['started_at']}")
        print(f"  total caps   : {latest['total_capabilities']}")
        print(f"  done         : {latest['num_done']}")
        print(f"  missing      : {latest['num_missing']}")
        print(f"  partial      : {latest['num_partial']}")
        print(f"  unknown      : {latest['num_unknown']}")

        if improvement:
            print("\n-- Improvement (first -> latest) --")
            print(f"  done_increase      : {improvement['done_increase']}")
            print(f"  missing_decrease   : {improvement['missing_decrease']}")
            print(f"  done_ratio_change  : {improvement['done_ratio_change']:.4f}")
            print(f"  missing_ratio_change: {improvement['missing_ratio_change']:.4f}")

    avg_success = summary["avg_experiment_success_rate"]
    if avg_success is not None:
        print(f"\nAvg experiment success rate: {avg_success:.3f}")

    print("\nTip: run with --json to get a machine-readable summary.")


if __name__ == "__main__":
    main()
