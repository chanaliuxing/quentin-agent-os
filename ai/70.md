# quentin/blueprint/auto_patcher.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, List, Optional
import logging

from quentin.blueprint.gap_tracker import BlueprintGapTracker, CapabilityStatus, BlueprintGapReport
from quentin.selfprog.engine import SelfProgrammingEngine
from quentin.selfprog.models import Patch
from quentin.research.experiments import ResearchManager, Experiment

logger = logging.getLogger(__name__)


@dataclass
class GapPatchResult:
    capability_id: str
    status_before: str          # missing/partial/...
    patch_id: Optional[str]
    action: str                 # "proposed" / "skipped"
    reason: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class GapAutoPatchReport:
    blueprint_path: str
    total_gaps: int
    attempted: int
    proposed: int
    skipped: int
    details: List[GapPatchResult]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "blueprint_path": self.blueprint_path,
            "total_gaps": self.total_gaps,
            "attempted": self.attempted,
            "proposed": self.proposed,
            "skipped": self.skipped,
            "details": [d.to_dict() for d in self.details],
        }


class BlueprintGapAutoPatcher:
    """
    BlueprintGapAutoPatcher v1：

    - 调用 BlueprintGapTracker.scan() 获取各 capability 的 implemented/partial/missing
    - 挑出 missing / partial 的能力
    - 自动为每个 capability 生成一个自编程 Patch（SelfProgrammingEngine）
    - （可选）之后由别的流程执行 patch-eval / patch-apply

    目标：让“蓝皮书缺口”变成一串可追踪的补丁候选，而不是只停留在文档。
    """

    def __init__(
        self,
        *,
        blueprint_path: str | Path = "blueprints/meta_agent_v3.json",
        selfprog_engine: Optional[SelfProgrammingEngine] = None,
        research_manager: Optional[ResearchManager] = None,
    ) -> None:
        self.blueprint_path = str(blueprint_path)
        self.gap_tracker = BlueprintGapTracker(blueprint_path=self.blueprint_path)
        self.selfprog_engine = selfprog_engine or SelfProgrammingEngine()
        self.research_manager = research_manager or ResearchManager()

    # ---------- 主入口 ----------

    def propose_patches_for_gaps(
        self,
        *,
        status_filter: List[str] = ("missing", "partial"),
        auto_eval: bool = False,
    ) -> GapAutoPatchReport:
        """
        核心流程：

        1. 先调用 gap_tracker.scan()，拿到 BlueprintGapReport
        2. 遍历 capabilities，挑出 status in status_filter 的项
        3. 每个 gap 根据 module → target_path 推导文件路径
        4. 调 selfprog_engine.propose_patch(...) 生成代码补丁
        5. 如 auto_eval=True，再对每个 patch 调一次 evaluate_patch(...)
        6. 最后记录一个 Research 实验 run
        """
        bp_report: BlueprintGapReport = self.gap_tracker.scan()
        caps = bp_report.capabilities

        gap_caps: List[CapabilityStatus] = [
            c for c in caps if c.status in status_filter
        ]
        results: List[GapPatchResult] = []

        for cap in gap_caps:
            res = self._handle_single_gap(cap, auto_eval=auto_eval)
            results.append(res)

        total_gaps = len(gap_caps)
        proposed = sum(1 for r in results if r.action == "proposed")
        skipped = sum(1 for r in results if r.action == "skipped")

        report = GapAutoPatchReport(
            blueprint_path=self.blueprint_path,
            total_gaps=total_gaps,
            attempted=total_gaps,
            proposed=proposed,
            skipped=skipped,
            details=results,
        )

        self._record_research(report)
        return report

    # ---------- 内部：处理单个 capability gap ----------

    def _handle_single_gap(
        self,
        cap: CapabilityStatus,
        *,
        auto_eval: bool,
    ) -> GapPatchResult:
        """
        对一个 capability 的缺口发起自编程 patch 建议。
        """
        # 若蓝皮书根本没写 module，没法做，直接跳过
        if not cap.module:
            return GapPatchResult(
                capability_id=cap.id,
                status_before=cap.status,
                patch_id=None,
                action="skipped",
                reason="Blueprint capability has no module field; cannot map to a file.",
            )

        target_path = self._module_to_path(cap.module)

        if not target_path:
            return GapPatchResult(
                capability_id=cap.id,
                status_before=cap.status,
                patch_id=None,
                action="skipped",
                reason=f"Cannot infer file path from module '{cap.module}'.",
            )

        # 生成问题描述：解释 gap，并提醒 LLM 参考蓝皮书
        problem = self._build_problem_description(cap, target_path)

        change_type = "new_module" if cap.status == "missing" else "refactor"

        try:
            patch: Patch = self.selfprog_engine.propose_patch(
                target_path=target_path,
                problem_description=problem,
                change_type=change_type,
                target_symbol=cap.symbol,
            )
        except Exception as e:
            logger.exception(
                "BlueprintGapAutoPatcher: propose_patch failed for capability %s",
                cap.id,
            )
            return GapPatchResult(
                capability_id=cap.id,
                status_before=cap.status,
                patch_id=None,
                action="skipped",
                reason=f"SelfProgrammingEngine.propose_patch failed: {e}",
            )

        # 如需要，自动做静态评估
        if auto_eval and patch.patch_id:
            try:
                self.selfprog_engine.evaluate_patch(patch.patch_id)
            except Exception:
                logger.exception(
                    "BlueprintGapAutoPatcher: evaluate_patch failed for patch_id=%s",
                    patch.patch_id,
                )

        return GapPatchResult(
            capability_id=cap.id,
            status_before=cap.status,
            patch_id=patch.patch_id,
            action="proposed",
            reason=f"Patch proposed for capability {cap.id} (status={cap.status}).",
        )

    # ---------- 内部：module → path ----------

    def _module_to_path(self, module_name: str) -> Optional[str]:
        """
        把 python module 名字转换成 repo 相对路径。

        例如：
          "quentin.selfprog.engine" -> "quentin/selfprog/engine.py"
        """
        if not module_name:
            return None
        parts = module_name.split(".")
        # 简单处理：最后一段当作文件名
        path = Path(*parts[:-1]) / f"{parts[-1]}.py"
        return str(path).replace("\\", "/")

    # ---------- 内部：构造问题描述 ----------

    def _build_problem_description(
        self,
        cap: CapabilityStatus,
        target_path: str,
    ) -> str:
        lines = [
            f"Blueprint capability ID: {cap.id}",
            f"Requirement ID      : {cap.requirement_id}",
            f"Current status      : {cap.status}",
            f"Target module       : {cap.module}",
            f"Target symbol       : {cap.symbol or '<unspecified>'}",
            f"Target file path    : {target_path}",
            "",
            "The blueprint says this capability should exist as part of a self-evolving agent.",
            "Your job as SELF-PROGRAMMING ENGINE is to create or complete the minimal implementation",
            "needed to make this capability real, in a safe and modular way.",
            "",
            "Constraints:",
            "- The implementation should be consistent with a layered, research-friendly meta-agent design.",
            "- Do NOT modify kernel/ or safety core; focus on the specified module/symbol only.",
            "- Provide clear, commented code and at least a basic test plan or test function.",
            "",
            "Existing scan detail (from BlueprintGapTracker):",
            cap.detail or "<no extra detail>",
        ]
        return "\n".join(lines)

    # ---------- 内部：写入 ResearchLayer ----------

    def _record_research(self, report: GapAutoPatchReport) -> None:
        rm = self.research_manager

        capability = "blueprint:gap_autopatch"
        requirement_id = "blueprint_gap_closure_rate"

        exps = rm.list_experiments()
        exp: Optional[Experiment] = None
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "blueprint_gap_autopatch"
            ):
                exp = e
                break

        if exp is None:
            exp = rm.create_experiment(
                name="BlueprintGapAutoPatch",
                purpose="blueprint_gap_autopatch",
                capability=capability,
                requirement_id=requirement_id,
                tags=["blueprint", "gap", "selfprog"],
            )

        if report.total_gaps > 0:
            proposal_rate = report.proposed / report.total_gaps
        else:
            proposal_rate = 0.0

        metrics = {
            "total_gaps": report.total_gaps,
            "attempted": report.attempted,
            "proposed": report.proposed,
            "skipped": report.skipped,
            "proposal_rate": proposal_rate,
        }
        context = {
            "blueprint_path": report.blueprint_path,
        }

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics=metrics,
            context=context,
        )

def cmd_blueprint_autopatch(args: argparse.Namespace) -> None:
    statuses = [s.strip() for s in args.status.split(",") if s.strip()]
    autopatcher = BlueprintGapAutoPatcher(
        blueprint_path=args.blueprint_path,
    )
    report = autopatcher.propose_patches_for_gaps(
        status_filter=statuses,
        auto_eval=args.eval,
    )
    data = report.to_dict()

    if args.json:
        import json as _json
        print(_json.dumps(data, ensure_ascii=False, indent=2))
        return

    print(f"Blueprint: {data['blueprint_path']}")
    print(
        f"Gaps targeted: {data['total_gaps']} | "
        f"proposed={data['proposed']} | skipped={data['skipped']}"
    )
    print("\nDetails:")
    for d in data["details"]:
        print(
            f"- cap={d['capability_id']} | status_before={d['status_before']} | "
            f"action={d['action']} | patch_id={d['patch_id']} | reason={d['reason']}"
        )
