二、BlueprintSpec：加载 YAML 的封装
文件：quentin/blueprint/spec.py

# quentin/blueprint/spec.py
from __future__ import annotations

import json
import logging
from dataclasses import dataclass, field
from pathlib import Path
from typing import Any, Dict, List, Optional

import yaml

logger = logging.getLogger(__name__)


@dataclass
class CapabilityRequirement:
    id: str
    type: str
    description: str


@dataclass
class CapabilitySpec:
    name: str
    layer: str
    category: str
    description: str
    components: List[str] = field(default_factory=list)
    requirements: List[CapabilityRequirement] = field(default_factory=list)
    metrics: List[str] = field(default_factory=list)
    definition_of_done: List[str] = field(default_factory=list)


@dataclass
class BlueprintSpec:
    id: str
    version: str
    codename: str
    raw: Dict[str, Any]
    capabilities: Dict[str, CapabilitySpec]

    @classmethod
    def load_from_file(cls, path: str | Path) -> BlueprintSpec:
        """
        从 meta_agent_v3.yaml 加载蓝皮书 spec。
        """
        path = Path(path)
        if not path.exists():
            raise FileNotFoundError(f"Blueprint spec file not found: {path}")

        with path.open("r", encoding="utf-8") as f:
            data: Dict[str, Any] = yaml.safe_load(f)

        bp = data.get("blueprint", {})
        capabilities_data = data.get("capabilities", {})

        capabilities: Dict[str, CapabilitySpec] = {}
        for cap_name, cap_info in capabilities_data.items():
            reqs: List[CapabilityRequirement] = []
            for r in cap_info.get("requirements", []):
                reqs.append(
                    CapabilityRequirement(
                        id=str(r.get("id")),
                        type=str(r.get("type", "functional")),
                        description=str(r.get("description", "")),
                    )
                )

            capabilities[cap_name] = CapabilitySpec(
                name=cap_name,
                layer=str(cap_info.get("layer", "")),
                category=str(cap_info.get("category", "")),
                description=str(cap_info.get("description", "")),
                components=list(cap_info.get("components", [])),
                requirements=reqs,
                metrics=list(cap_info.get("metrics", [])),
                definition_of_done=list(cap_info.get("definition_of_done", [])),
            )

        return cls(
            id=str(bp.get("id", "meta_agent_v3")),
            version=str(bp.get("version", "0.0")),
            codename=str(bp.get("codename", "")),
            raw=data,
            capabilities=capabilities,
        )

    def to_json(self, indent: int = 2) -> str:
        return json.dumps(self.raw, ensure_ascii=False, indent=indent)
说明：

这段代码会把你刚才那整份 meta_agent_v3.yaml 全读进去，并且把 capabilities.* 转成结构化对象。

后面 BlueprintGapTracker 会直接用 BlueprintSpec.capabilities 来做检查。

三、BlueprintGapTracker：扫描实现 / 占位 / 缺失
文件：quentin/blueprint/tracker.py

# quentin/blueprint/tracker.py
from __future__ import annotations

import importlib
import inspect
import json
import logging
from dataclasses import dataclass
from types import ModuleType
from typing import Any, Dict, List, Optional, Tuple

from quentin.blueprint.spec import BlueprintSpec, CapabilitySpec

logger = logging.getLogger(__name__)


@dataclass
class ComponentStatus:
    component: str              # e.g. "quentin.memory.fabric.MemoryFabric"
    exists: bool
    is_placeholder: bool
    details: Dict[str, Any]


@dataclass
class CapabilityStatus:
    name: str                   # capability name, e.g. "memory_fabric"
    overall_status: str         # "implemented" | "placeholder" | "missing"
    components: List[ComponentStatus]
    missing_requirements: List[str]


class BlueprintGapTracker:
    """
    对照 meta_agent_v3 蓝皮书，检查当前代码库实现情况。
    输出每个 Capability 的状态：
    - implemented   : 所有组件存在，且至少有一部分非占位
    - placeholder   : 组件存在，但大部分是占位/空逻辑
    - missing       : 至少有一个关键组件不存在
    """

    def __init__(
        self,
        blueprint: BlueprintSpec,
        placeholder_detection_enabled: bool = True,
    ) -> None:
        self.blueprint = blueprint
        self.placeholder_detection_enabled = placeholder_detection_enabled

    @classmethod
    def from_file(cls, path: str) -> "BlueprintGapTracker":
        bp = BlueprintSpec.load_from_file(path)
        return cls(blueprint=bp)

    # ---------- 外部主入口 ----------

    def scan(self) -> Dict[str, CapabilityStatus]:
        """
        扫描所有 capabilities，返回 {capability_name: CapabilityStatus}
        """
        results: Dict[str, CapabilityStatus] = {}

        for cap_name, cap_spec in self.blueprint.capabilities.items():
            try:
                cap_status = self._scan_single_capability(cap_spec)
            except Exception as e:
                logger.exception("Error scanning capability %s", cap_name)
                # 出错时，为了不让整个 scan 崩溃，标记为 missing，附上错误信息
                cap_status = CapabilityStatus(
                    name=cap_name,
                    overall_status="missing",
                    components=[],
                    missing_requirements=[f"scan_error: {e}"],
                )
            results[cap_name] = cap_status

        return results

    # ---------- 核心逻辑 ----------

    def _scan_single_capability(self, cap: CapabilitySpec) -> CapabilityStatus:
        """
        针对一个 capability：
        - 检查所有 components 是否存在
        - 粗略判断是否“占位”
        - 推导 overall_status
        """
        comp_statuses: List[ComponentStatus] = []

        for comp in cap.components:
            cs = self._check_component(comp)
            comp_statuses.append(cs)

        exists_flags = [c.exists for c in comp_statuses]
        placeholder_flags = [c.is_placeholder for c in comp_statuses if c.exists]

        if not exists_flags or not any(exists_flags):
            overall = "missing"
        else:
            # 至少有一个存在
            if placeholder_flags and all(placeholder_flags):
                overall = "placeholder"
            elif placeholder_flags and any(placeholder_flags):
                # 混合情况：有些占位，有些不是
                overall = "partial"
            else:
                overall = "implemented"

        # 简单策略：不是 implemented 的，就把所有 requirement id 当作 missing_requirements
        missing_requirements: List[str] = []
        if overall != "implemented":
            missing_requirements = [r.id for r in cap.requirements]

        return CapabilityStatus(
            name=cap.name,
            overall_status=overall,
            components=comp_statuses,
            missing_requirements=missing_requirements,
        )

    def _check_component(self, dotted_path: str) -> ComponentStatus:
        """
        检查某个组件路径是否存在，并尝试判断是否占位实现。

        dotted_path 示例：
        - "quentin.memory.fabric.MemoryFabric"
        - "quentin.traces.episode_builder.TraceEpisodeBuilder"
        """
        details: Dict[str, Any] = {}

        module_name, attr_name = self._split_module_attr(dotted_path)
        if not module_name:
            return ComponentStatus(
                component=dotted_path,
                exists=False,
                is_placeholder=False,
                details={"error": "invalid_path"},
            )

        try:
            module = importlib.import_module(module_name)
        except Exception as e:
            details["import_error"] = str(e)
            return ComponentStatus(
                component=dotted_path,
                exists=False,
                is_placeholder=False,
                details=details,
            )

        attr = getattr(module, attr_name, None)
        if attr is None:
            details["missing_attr"] = True
            return ComponentStatus(
                component=dotted_path,
                exists=False,
                is_placeholder=False,
                details=details,
            )

        details["attr_type"] = str(type(attr))

        if not self.placeholder_detection_enabled:
            # 只检查存在性，不做占位识别
            return ComponentStatus(
                component=dotted_path,
                exists=True,
                is_placeholder=False,
                details=details,
            )

        is_ph, ph_reason = self._is_placeholder(attr)
        if ph_reason:
            details["placeholder_reason"] = ph_reason

        return ComponentStatus(
            component=dotted_path,
            exists=True,
            is_placeholder=is_ph,
            details=details,
        )

    # ---------- 辅助函数 ----------

    @staticmethod
    def _split_module_attr(dotted_path: str) -> Tuple[Optional[str], str]:
        """
        从 "package.module.ClassName" 里拆成：
        - module_name = "package.module"
        - attr_name = "ClassName"
        """
        if "." not in dotted_path:
            return None, dotted_path
        parts = dotted_path.split(".")
        module_name = ".".join(parts[:-1])
        attr_name = parts[-1]
        return module_name, attr_name

    def _is_placeholder(self, obj: Any) -> Tuple[bool, Optional[str]]:
        """
        粗略判断一个对象是不是“占位实现”。

        主要启发式：
        - 源码基本只有 pass
        - 或 raise NotImplementedError
        - 或只有注释/TODO，没有具体逻辑
        """
        try:
            source = inspect.getsource(obj)
        except (OSError, TypeError):  # builtin / C 扩展等
            return False, None

        src = source.strip()

        # 统一小写做简单匹配
        lower = src.lower()

        # 非常简单的启发式，你后续可以根据实际代码风格继续加规则
        if "notimplementederror" in lower:
            return True, "raises NotImplementedError"

        # 如果去掉空行和注释后几乎没有实质代码
        lines = [line.strip() for line in src.splitlines()]
        code_lines = [
            ln for ln in lines
            if ln
            and not ln.startswith("#")
            and not ln.startswith('"""')
            and not ln.startswith("'''")
        ]

        # 如果主体只有函数/类定义 + pass
        # 例如：
        # def foo(...):
        #     pass
        if len(code_lines) <= 3 and any("pass" == ln or ln.endswith("pass") for ln in code_lines):
            return True, "only pass / empty body"

        # 可以继续加规则，如 TODO/XXX 之类
        if "todo" in lower and "pass" in lower and len(code_lines) <= 6:
            return True, "mostly TODO with pass"

        return False, None

    # ---------- 输出工具 ----------

    @staticmethod
    def format_results_as_dict(
        results: Dict[str, CapabilityStatus]
    ) -> Dict[str, Any]:
        """
        把 CapabilityStatus 映射成可以直接 JSON dump 的字典。
        """
        out: Dict[str, Any] = {}
        for name, cs in results.items():
            out[name] = {
                "overall_status": cs.overall_status,
                "missing_requirements": cs.missing_requirements,
                "components": [
                    {
                        "component": c.component,
                        "exists": c.exists,
                        "is_placeholder": c.is_placeholder,
                        "details": c.details,
                    }
                    for c in cs.components
                ],
            }
        return out

    @staticmethod
    def format_results_as_json(
        results: Dict[str, CapabilityStatus],
        indent: int = 2,
    ) -> str:
        return json.dumps(
            BlueprintGapTracker.format_results_as_dict(results),
            ensure_ascii=False,
            indent=indent,
        )
四、简单 CLI：qctl blueprint-scan
如果你已经有自己的 CLI，可以把这段挂进去；如果暂时没有，可以先建个最小可跑的脚本。

文件：quentin/cli/qctl.py（示例）

# quentin/cli/qctl.py
from __future__ import annotations

import argparse
import logging
from pathlib import Path

from quentin.blueprint.tracker import BlueprintGapTracker

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


def cmd_blueprint_scan(args: argparse.Namespace) -> None:
    bp_path = args.blueprint or "quentin/blueprint/meta_agent_v3.yaml"
    tracker = BlueprintGapTracker.from_file(bp_path)
    results = tracker.scan()
    print(BlueprintGapTracker.format_results_as_json(results))


def main() -> None:
    parser = argparse.ArgumentParser(prog="qctl", description="Project Quentin CLI")
    subparsers = parser.add_subparsers(dest="command", required=True)

    # blueprint-scan
    p_scan = subparsers.add_parser(
        "blueprint-scan", help="Scan current repo against meta_agent_v3 blueprint"
    )
    p_scan.add_argument(
        "--blueprint",
        type=str,
        default="quentin/blueprint/meta_agent_v3.yaml",
        help="Path to blueprint YAML file",
    )
    p_scan.set_defaults(func=cmd_blueprint_scan)

    args = parser.parse_args()
    args.func(args)


if __name__ == "__main__":
    main()
之后在项目根目录加一个入口：
