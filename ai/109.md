from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, List, Optional
import importlib
import inspect
import logging

from quentin.blueprint.spec import BlueprintSpec, BlueprintCapability, load_default_spec_v3
from quentin.blueprint.gap_tracker import CapabilitiesRegistry, CapabilityStatus

logger = logging.getLogger(__name__)


@dataclass
class CapabilityProbeResult:
    """
    单条能力的探测结果（来自静态规则 / import 检查 / 将来可能来自 LLM 分析）。

    status_guess:
      - "done" / "partial" / "missing" / "unknown"
    confidence:
      - 0.0 ~ 1.0（简单启发式置信度）
    evidence:
      - 简短文字说明，例如 "found class ExperimentOrchestrator in module quentin.research.experiments"
    """
    capability_id: str
    status_guess: str
    confidence: float
    evidence: str
    source: str  # 哪个 probe 给出的结果（static/import/llm/...）

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class CapabilityScanReport:
    """
    一次扫描的整体报告。
    """
    spec_version: str
    repo_root: str
    probe_results: List[CapabilityProbeResult]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "spec_version": self.spec_version,
            "repo_root": self.repo_root,
            "probe_results": [r.to_dict() for r in self.probe_results],
        }


class CapabilityScanner:
    """
    蓝皮书能力自动扫描器 v1：

      - 读取 BlueprintSpec（蓝皮书能力定义）
      - 使用简单的静态规则检查 repo 是否具备某些关键模块/类
      - 更新 CapabilitiesRegistry（missing/partial/done）
      - 返回 CapabilityScanReport

    设计目标：
      - 现在先靠静态规则把“我们已经写好的模块”标记为 done
      - 未来可以新增：
          * 文件级模式匹配（grep / AST 分析）
          * LLM / 多模型 council 对代码片段做语义判断
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        spec: Optional[BlueprintSpec] = None,
        registry: Optional[CapabilitiesRegistry] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.spec: BlueprintSpec = spec or load_default_spec_v3()
        self.registry: CapabilitiesRegistry = registry or CapabilitiesRegistry(repo_root=self.repo_root)

    # --------------------------------------------------
    # 对外主入口
    # --------------------------------------------------
    def scan_and_update_registry(self) -> CapabilityScanReport:
        """
        扫描所有能力项，对每个能力尝试给出一个 status_guess，
        并写回 CapabilitiesRegistry。

        当前策略：
          - 只对“静态规则覆盖到的能力”更新为 done/partial
          - 其它保持原状（unknown 或之前的人工标记）
        """
        probe_results: List[CapabilityProbeResult] = []

        # 1) 静态 heuristic 检查（import + 文件路径）
        probe_results.extend(self._run_static_import_probes())

        # TODO: 以后在这里挂 LLM / AST / 多模型 council probes

        # 2) 根据 probe_results 更新 registry
        for pr in probe_results:
            # 简单合并策略：
            #   - 如果静态 probe 判为 done，直接写 done
            #   - 如果判为 partial 且 registry 还没标 done，就写 partial
            #   - missing / unknown 暂时不覆盖已有的 done/partial（防止误伤）
            current = self.registry.get(pr.capability_id)
            current_status = current.status if current is not None else "unknown"

            new_status = current_status
            if pr.status_guess == "done":
                new_status = "done"
            elif pr.status_guess == "partial":
                if current_status not in ("done", "partial"):
                    new_status = "partial"
            elif pr.status_guess in ("missing", "unknown"):
                # 不主动覆盖 done/partial
                if current_status == "unknown":
                    new_status = pr.status_guess

            if new_status != current_status:
                self.registry.set_status(
                    pr.capability_id,
                    new_status,
                    evidence=f"[{pr.source}] {pr.evidence}",
                )

        return CapabilityScanReport(
            spec_version=self.spec.version,
            repo_root=str(self.repo_root),
            probe_results=probe_results,
        )

                return importlib.import_module(module_name)
            except Exception:
                logger.debug("CapabilityScanner: import failed: %s", module_name)
                return None

            results.append(CapabilityProbeResult(
                capability_id=cap_id,
                status_guess="done",
                confidence=0.9,
                evidence="Found quentin.runtime.system and quentin.research.layer modules.",
                source="static-import",
            ))

然后在类里加方法：

    def auto_update_capabilities_from_repo(
        self,
        *,
        repo_root: Optional[str | Path] = None,
    ) -> Dict[str, Any]:
        """
        使用 CapabilityScanner 对当前 repo 做一次扫描，
        自动更新 CapabilitiesRegistry（哪些蓝皮书能力已实现/部分实现）。

        返回一个 dict：
          - scan_report: 本次扫描的详细结果（probe_results）
          - registry_snapshot: 更新后的能力状态快照
        """
        root = Path(repo_root or self.repo_root).resolve()

        scanner = CapabilityScanner(
            repo_root=root,
            spec=self.blueprint_spec,
            registry=self.capabilities_registry,
        )
        scan_report: CapabilityScanReport = scanner.scan_and_update_registry()

        return {
            "scan_report": scan_report.to_dict(),
            "registry_snapshot": self.capabilities_registry.export(),
        }
这样你就有了两个关键闭环入口：

更新“我们做到哪儿了”：

runtime.auto_update_capabilities_from_repo()
围绕蓝皮书 gap 做自迭代实验：

runtime.run_blueprint_improvement_cycle(max_gaps=5)
这两个加起来，就是一个完整的 v1 自迭代 loop：
扫描能力 → 标记进度 → 找蓝皮书 gap → 生成实验计划 → ResearchLayer 跑实验。
