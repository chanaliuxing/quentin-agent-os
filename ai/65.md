一、Plan 模型 + 简单解析器
先约定一个最简单、LLM 很容易遵守的技能调用语法（你后面可以写进 Planner 的 prompt）：

STEP 1: THINK do X Y Z...

STEP 2: CALL_SKILL id=<skill_id> input={"text": "....", "foo": 123}

STEP 3: THINK summarize the result...
只要行里出现 CALL_SKILL，我们就尝试解析 id= 和 input= 后面的 JSON。

新文件：quentin/core/plan.py
# quentin/core/plan.py
from __future__ import annotations

import json
import logging
import re
from dataclasses import dataclass
from typing import Any, Dict, List, Optional

logger = logging.getLogger(__name__)


@dataclass
class PlanStep:
    index: int                     # 步骤序号（从 1 开始）
    kind: str                      # "thought" | "skill_call" | "action" | "unknown"
    raw_text: str                  # 原始文本
    skill_id: Optional[str] = None # 若为 skill_call，则填充
    input_json: Optional[Dict[str, Any]] = None  # skill input


class SimplePlanParser:
    """
    v1 计划解析器：基于文本的“行解析”，识别 CALL_SKILL 语句。

    支持格式示例：
      STEP 2: CALL_SKILL id=abcd-123 input={"text": "hello", "n": 2}
    """

    # 匹配 CALL_SKILL id=xxx input=xxx
    _skill_pattern = re.compile(
        r"CALL_SKILL\s+id=(?P<id>[^\s]+)\s+input=(?P<input>.+)$",
        re.IGNORECASE,
    )

    def parse(self, plan_text: str) -> List[PlanStep]:
        lines = plan_text.splitlines()
        steps: List[PlanStep] = []
        idx = 1

        for line in lines:
            stripped = line.strip()
            if not stripped:
                continue

            m = self._skill_pattern.search(stripped)
            if m:
                skill_id = m.group("id")
                input_str = m.group("input").strip()
                input_json = None
                try:
                    # 尝试从 input= 后面解析 JSON
                    # 允许前后有多余文字，尝试从第一个 { 找到最后一个 }
                    first = input_str.find("{")
                    last = input_str.rfind("}")
                    if first != -1 and last != -1 and last > first:
                        json_sub = input_str[first : last + 1]
                    else:
                        json_sub = input_str
                    input_json = json.loads(json_sub)
                except Exception as e:
                    logger.warning(
                        "SimplePlanParser: failed to parse input JSON in line=%r: %s",
                        line,
                        e,
                    )

                steps.append(
                    PlanStep(
                        index=idx,
                        kind="skill_call",
                        raw_text=line,
                        skill_id=skill_id,
                        input_json=input_json,
                    )
                )
            else:
                # 其它都先当作 thought，后续可以扩展 "tool_call" / "subtask" 等
                steps.append(
                    PlanStep(
                        index=idx,
                        kind="thought",
                        raw_text=line,
                    )
                )
            idx += 1

        return steps
二、SkillAwareExecutor：识别并执行技能步骤
我们做一个独立的执行器，不动你原来的 TaskResult 结构，只做 “在计划里找到 CALL_SKILL 并执行”，并且：

写 Trace：记录每个 skill 调用和结果

可选写记忆：把成功结果写进 STS，作为 episodic memory（接口先留个 “best effort”）

新文件：quentin/core/executor.py
# quentin/core/executor.py
from __future__ import annotations

import logging
from dataclasses import dataclass, asdict
from datetime import datetime
from typing import Any, Dict, List, Optional

from quentin.core.plan import SimplePlanParser, PlanStep
from quentin.core.task import TaskSpec
from quentin.core.trace import FileBackedTraceStore
from quentin.memory.fabric import FileBackedMemoryFabric  # 假定已存在
from quentin.skills.manager import SkillManager, SkillRunResult
from quentin.skills.registry import FileBackedSkillRegistry

logger = logging.getLogger(__name__)


@dataclass
class ExecutedSkillCall:
    step_index: int
    skill_id: str
    success: bool
    input_json: Optional[Dict[str, Any]]
    output: Any
    error: Optional[str]
    used_model: Optional[str]
    started_at: str
    finished_at: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class SkillAwareExecutor:
    """
    计划执行器 v1：只负责执行计划中的 CALL_SKILL 步骤。

    - 不替代原有的“LLM 思考 + 直接生成答案”的路径；
    - 只做：
        * 解析 plan_text -> PlanStep 列表
        * 对每个 skill_call 调用 SkillManager.run_skill(...)
        * 记录 Trace 事件
        * 可选写入 MemoryFabric（episodic skill_result）
    """

    def __init__(
        self,
        *,
        trace_store: Optional[FileBackedTraceStore] = None,
        memory_fabric: Optional[FileBackedMemoryFabric] = None,
        skill_manager: Optional[SkillManager] = None,
    ) -> None:
        self.parser = SimplePlanParser()
        self.trace_store = trace_store or FileBackedTraceStore()
        self.memory_fabric = memory_fabric or FileBackedMemoryFabric()
        self.skill_manager = skill_manager or SkillManager(
            registry=FileBackedSkillRegistry()
        )

    # ---------- 主入口 ----------

    def execute_skill_calls(
        self,
        task: TaskSpec,
        plan_text: str,
    ) -> List[ExecutedSkillCall]:
        """
        扫描 plan_text，执行所有 skill_call 步骤，返回执行结果列表。
        """
        steps: List[PlanStep] = self.parser.parse(plan_text)
        executed: List[ExecutedSkillCall] = []

        for step in steps:
            if step.kind != "skill_call":
                continue
            if not step.skill_id:
                logger.warning(
                    "SkillAwareExecutor: step %s missing skill_id, skip. line=%r",
                    step.index,
                    step.raw_text,
                )
                continue

            exec_result = self._run_single_skill_step(task, step)
            if exec_result:
                executed.append(exec_result)

        return executed

    # ---------- 内部 ----------

    def _run_single_skill_step(
        self,
        task: TaskSpec,
        step: PlanStep,
    ) -> Optional[ExecutedSkillCall]:
        skill = self.skill_manager.registry.get_skill(step.skill_id)
        if not skill:
            logger.warning(
                "SkillAwareExecutor: skill_id=%s not found for step=%s",
                step.skill_id,
                step.index,
            )
            self._record_trace(
                task_id=task.task_id,
                step_index=step.index,
                kind="skill_call_missing_skill",
                payload={
                    "skill_id": step.skill_id,
                    "raw_text": step.raw_text,
                },
            )
            return None

        input_payload = step.input_json or {}

        self._record_trace(
            task_id=task.task_id,
            step_index=step.index,
            kind="skill_call_started",
            payload={
                "skill_id": skill.skill_id,
                "skill_name": skill.name,
                "input": input_payload,
            },
        )

        res: SkillRunResult = self.skill_manager.run_skill(
            skill,
            input_payload=input_payload,
            expect_json=True,
        )

        exec_result = ExecutedSkillCall(
            step_index=step.index,
            skill_id=skill.skill_id,
            success=res.success,
            input_json=input_payload,
            output=res.output,
            error=res.error,
            used_model=res.used_model,
            started_at=res.started_at,
            finished_at=res.finished_at or datetime.utcnow().isoformat(),
        )

        # Trace 记录
        self._record_trace(
            task_id=task.task_id,
            step_index=step.index,
            kind="skill_call_finished",
            payload=exec_result.to_dict(),
        )

        # 可选：写入 MemoryFabric，作为 STS 的 episodic skill_result
        self._write_skill_result_memory(task, exec_result)

        return exec_result

    def _record_trace(
        self,
        *,
        task_id: str,
        step_index: int,
        kind: str,
        payload: Dict[str, Any],
    ) -> None:
        try:
            self.trace_store.record_event(
                task_id=task_id,
                step=f"exec_step_{step_index}",
                role="system",
                kind=kind,
                payload=payload,
            )
        except Exception:
            logger.exception("SkillAwareExecutor: failed to record trace event.")

    def _write_skill_result_memory(
        self,
        task: TaskSpec,
        exec_result: ExecutedSkillCall,
    ) -> None:
        """
        v1：best-effort 写入一个简单记忆条目（若 MemoryFabric 支持 add/append 类方法）。

        - tier: STS
        - type: "skill_result"
        """
        mf = self.memory_fabric
        if mf is None:
            return

        # 为了不和你现有接口冲突，这里做一个“鸭子类型”调用，
        # 你可以按自己的 MemoryFabric API 调整。
        try:
            if hasattr(mf, "add"):
                mf.add(
                    {
                        "task_id": task.task_id,
                        "kind": "skill_result",
                        "tier": "STS",
                        "skill_id": exec_result.skill_id,
                        "success": exec_result.success,
                        "output": exec_result.output,
                        "created_at": datetime.utcnow().isoformat(),
                    }
                )
        except Exception:
            logger.exception("SkillAwareExecutor: failed to write skill_result memory.")
这里 MemoryFabric 的具体方法我只能“鸭子调用”，你可以按自己实现调整为 add_memory(...) / save(...) 等。

四、现在这一层在蓝皮书里的位置
这一步完成后，你的系统多了一条完整链路：

L1 Base Layer – Planner

在看到 Memory Fabric & Skills 后，写出一个包含 CALL_SKILL 的计划文本。

L1 Base Layer – Executor（SkillAwareExecutor）

解析 plan → 执行技能 → 写 Trace & Memory（STS skill_result）。

LLM Execution 阶段

在上下文中拿到 “有哪些技能结果可用”，可以基于这些结果给出更好的最终答案。

这基本上就是蓝皮书里 Skill & Code Evolution Loop 在运行时这一块的落地：

“已有技能被 Planner 发现 → Executor 真正调用 → 结果进入 Trace & Memory → 将来 Sleep Loop 与 Research Layer 再分析这些 Skill 的表现。”
