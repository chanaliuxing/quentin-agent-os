@dataclass
class ExperimentVariant:
    variant_id: str
    kind: str = "patch"  # "patch" / "strategy" / ...
    ref: str = ""        # 如 patch_id 或策略配置 ID
    description: str = ""
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "ExperimentVariant":
        return cls(
            variant_id=d["variant_id"],
            kind=d.get("kind", "patch"),
            ref=d.get("ref", ""),
            description=d.get("description", ""),
            metadata=d.get("metadata", {}) or {},
        )


@dataclass
class ExperimentRun:
    run_id: str
    experiment_id: str
    variant_id: str
    metrics: Dict[str, Any]
    context: Dict[str, Any] = field(default_factory=dict)
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "ExperimentRun":
        return cls(
            run_id=d["run_id"],
            experiment_id=d["experiment_id"],
            variant_id=d["variant_id"],
            metrics=d.get("metrics", {}) or {},
            context=d.get("context", {}) or {},
            created_at=d.get("created_at", ""),
        )


@dataclass
class Experiment:
    experiment_id: str
    name: str
    purpose: str
    capability: str
    requirement_id: str

    tags: List[str] = field(default_factory=list)
    variants: List[ExperimentVariant] = field(default_factory=list)
    aggregated_stats: Dict[str, Any] = field(default_factory=dict)

    status: str = "active"  # "active" / "completed" / "archived"
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return {
            "experiment_id": self.experiment_id,
            "name": self.name,
            "purpose": self.purpose,
            "capability": self.capability,
            "requirement_id": self.requirement_id,
            "tags": list(self.tags),
            "variants": [v.to_dict() for v in self.variants],
            "aggregated_stats": self.aggregated_stats,
            "status": self.status,
            "created_at": self.created_at,
            "updated_at": self.updated_at,
        }

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "Experiment":
        variants = [
            ExperimentVariant.from_dict(v) for v in d.get("variants", []) or []
        ]
        return cls(
            experiment_id=d["experiment_id"],
            name=d.get("name", d["experiment_id"]),
            purpose=d.get("purpose", ""),
            capability=d.get("capability", ""),
            requirement_id=d.get("requirement_id", ""),
            tags=list(d.get("tags", []) or []),
            variants=variants,
            aggregated_stats=d.get("aggregated_stats", {}) or {},
            status=d.get("status", "active"),
            created_at=d.get("created_at", ""),
            updated_at=d.get("updated_at", ""),
        )


class ResearchManager:
    """
    Research & Experiment 管理：
    - 维护 experiments + runs 的 JSON 存储
    - 为 EvolutionStrategy 提供聚合统计
    """

    def __init__(self, path: str | Path = "data/experiments.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._loaded = False
        self._experiments: Dict[str, Experiment] = {}
        self._runs: Dict[str, ExperimentRun] = {}

    # ---------- 基础存取 ----------

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                raw = json.loads(self.path.read_text(encoding="utf-8") or "{}")
                exps_raw = raw.get("experiments", {}) or {}
                runs_raw = raw.get("runs", {}) or {}
                self._experiments = {
                    eid: Experiment.from_dict(ed) for eid, ed in exps_raw.items()
                }
                self._runs = {
                    rid: ExperimentRun.from_dict(rd) for rid, rd in runs_raw.items()
                }
            except Exception:
                logger.exception("Failed to load experiments from %s", self.path)
                self._experiments = {}
                self._runs = {}
        else:
            self._experiments = {}
            self._runs = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        raw = {
            "experiments": {
                eid: e.to_dict() for eid, e in self._experiments.items()
            },
            "runs": {
                rid: r.to_dict() for rid, r in self._runs.items()
            },
        }
        self.path.write_text(
            json.dumps(raw, ensure_ascii=False, indent=2), encoding="utf-8"
        )

    # ---------- 实验 CRUD ----------

    def list_experiments(self) -> List[Experiment]:
        self._load()
        return list(self._experiments.values())

    def get_experiment(self, experiment_id: str) -> Optional[Experiment]:
        self._load()
        return self._experiments.get(experiment_id)

    def create_experiment(
        self,
        name: str,
        purpose: str,
        capability: str,
        requirement_id: str,
        tags: Optional[List[str]] = None,
    ) -> Experiment:
        self._load()
        eid = str(uuid.uuid4())
        exp = Experiment(
            experiment_id=eid,
            name=name,
            purpose=purpose,
            capability=capability,
            requirement_id=requirement_id,
            tags=list(tags or []),
        )
        self._experiments[eid] = exp
        self._save()
        return exp

    # ---------- 记录 Run & 聚合 ----------

    def record_run(
        self,
        experiment_id: str,
        variant_id: str,
        metrics: Dict[str, Any],
        context: Optional[Dict[str, Any]] = None,
    ) -> ExperimentRun:
        self._load()
        if experiment_id not in self._experiments:
            raise ValueError(f"Experiment not found: {experiment_id}")

        run_id = str(uuid.uuid4())
        run = ExperimentRun(
            run_id=run_id,
            experiment_id=experiment_id,
            variant_id=variant_id,
            metrics=metrics,
            context=context or {},
        )
        self._runs[run_id] = run

        # 更新 aggregated_stats
        self._recompute_aggregated_stats(experiment_id)

        self._experiments[experiment_id].updated_at = datetime.utcnow().isoformat()
        self._save()
        return run

    def _recompute_aggregated_stats(self, experiment_id: str) -> None:
        exp = self._experiments.get(experiment_id)
        if not exp:
            return

        # variant_id -> [runs...]
        per_variant: Dict[str, List[ExperimentRun]] = {}
        for r in self._runs.values():
            if r.experiment_id != experiment_id:
                continue
            per_variant.setdefault(r.variant_id, []).append(r)

        aggregated: Dict[str, Any] = {
            "total_runs": 0,
            "variants": {},
        }

        for vid, runs in per_variant.items():
            n_runs = len(runs)
            aggregated["total_runs"] += n_runs

            success_count = 0
            success_weighted = 0.0
            reward_sum = 0.0
            reward_count = 0
            last_run_at = ""

            for r in runs:
                m = r.metrics or {}
                success = m.get("success")
                if isinstance(success, bool):
                    success_count += 1 if success else 0
                    success_weighted += 1.0 if success else 0.0

                reward = m.get("reward_delta")
                if isinstance(reward, (int, float)):
                    reward_sum += float(reward)
                    reward_count += 1

                if not last_run_at or r.created_at > last_run_at:
                    last_run_at = r.created_at

            success_rate = (
                success_weighted / n_runs if n_runs > 0 and success_count > 0 else 0.0
            )
            mean_reward = (
                reward_sum / reward_count if reward_count > 0 else None
            )

            aggregated["variants"][vid] = {
                "n_runs": n_runs,
                "success_rate": success_rate,
                "mean_reward_delta": mean_reward,
                "last_run_at": last_run_at,
            }

        exp.aggregated_stats = aggregated

    # ---------- 给 EvolutionStrategy 用的 task 粗粒度统计 ----------

    def get_task_level_stats(
        self,
        capability: str,
        requirement_id: str,
    ) -> Dict[str, Any]:
        """
        汇总某个 (capability, requirement_id) 下所有 experiment 的统计。
        """
        self._load()
        exps = [
            e
            for e in self._experiments.values()
            if e.capability == capability and e.requirement_id == requirement_id
        ]
        if not exps:
            return {}

        total_runs = 0
        success_weighted_sum = 0.0
        reward_weighted_sum = 0.0
        reward_weighted_count = 0

        for e in exps:
            stats = e.aggregated_stats or {}
            total_runs += int(stats.get("total_runs", 0))

            variants = stats.get("variants", {}) or {}
            for v in variants.values():
                n = int(v.get("n_runs", 0))
                sr = float(v.get("success_rate", 0.0))
                total_runs += 0  # 已计入上面，这里只用来加权
                success_weighted_sum += sr * n

                rv = v.get("mean_reward_delta")
                if isinstance(rv, (int, float)):
                    reward_weighted_sum += float(rv) * n
                    reward_weighted_count += n

        mean_success = (
            success_weighted_sum / total_runs if total_runs > 0 else None
        )
        mean_reward = (
            reward_weighted_sum / reward_weighted_count
            if reward_weighted_count > 0
            else None
        )

        return {
            "total_runs": total_runs,
            "mean_success": mean_success,
            "mean_reward_delta": mean_reward,
        }

    # ---------- 和 CodePatch 的桥接：记录 patch 评估 ----------

    def record_patch_evaluation(self, patch: CodePatch) -> None:
        """
        把某个 patch 的 eval_summary 记为实验数据。

        策略：
        - 每个 (capability, requirement_id) 有一个自动实验：
          purpose = "patch_auto"
        - 每个 patch_id 是一个 variant
        - metrics:
            success      = patch.eval_summary.success
            reward_delta = +1.0 (成功) / 0.0 (未知) / -1.0 (失败)
        """
        self._load()

        capability = patch.capability
        requirement_id = patch.requirement_id

        if not capability or not requirement_id:
            # 信息不足就跳过，不抛异常
            logger.warning(
                "record_patch_evaluation: patch %s missing capability/requirement_id",
                patch.patch_id,
            )
            return

        # 找或建 experiment
        exp = None
        for e in self._experiments.values():
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "patch_auto"
            ):
                exp = e
                break

        if exp is None:
            exp = self.create_experiment(
                name=f"PatchAuto-{capability}-{requirement_id}",
                purpose="patch_auto",
                capability=capability,
                requirement_id=requirement_id,
                tags=["patch", "auto"],
            )

        # 找或建 variant
        variant_id = patch.patch_id
        found = False
        for v in exp.variants:
            if v.variant_id == variant_id:
                found = True
                break
        if not found:
            exp.variants.append(
                ExperimentVariant(
                    variant_id=variant_id,
                    kind="patch",
                    ref=patch.patch_id,
                    description=f"Patch for {capability}/{requirement_id}",
                )
            )

        # 构造 metrics
        ev = patch.eval_summary or {}
        success = bool(ev.get("success"))
        if "success" not in ev:
            ev["success"] = success

        if "reward_delta" not in ev:
            # 简单规则：成功 +1，失败 -1，未知 0
            if success:
                ev["reward_delta"] = 1.0
            else:
                ev["reward_delta"] = -1.0 if ev.get("exit_code", 0) != 0 else 0.0

        context = {
            "patch_id": patch.patch_id,
            "test_command": ev.get("test_command"),
            "notes": ev.get("notes"),
        }

        self._experiments[exp.experiment_id] = exp
        self._save()  # 确保 experiment 持久化，再写 run

        self.record_run(
            experiment_id=exp.experiment_id,
            variant_id=variant_id,
            metrics=ev,
            context=context,
        )

@dataclass
class BudgetRule:
    capability: str  # 也可以是 "*" 表示默认
    max_tokens_per_day: float = 1e9
    max_patches_per_day: int = 1000

    enabled: bool = True

        return asdict(self)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "BudgetRule":

            capability=d["capability"],
            max_tokens_per_day=float(d.get("max_tokens_per_day", 1e9)),
            max_patches_per_day=int(d.get("max_patches_per_day", 1000)),
            max_experiments_per_day=int(d.get("max_experiments_per_day", 100)),
            enabled=bool(d.get("enabled", True)),
        )


@dataclass
class BudgetUsage:
    capability: str
    date: str
    tokens_used: float = 0.0
    patches_used: int = 0
    experiments_used: int = 0

    def to_dict(self) -> Dict[str, Any]:

            date=d["date"],
            tokens_used=float(d.get("tokens_used", 0.0)),
            patches_used=int(d.get("patches_used", 0)),

class BudgetController:
    """
    简单的 per-capability 配额控制：
    - 有一条默认规则 "*"，可以在 budget.json 手工编辑
    - 可以按 capability 自定义规则
    """

    def __init__(self, path: str | Path = "data/budget.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._loaded = False
        self._rules: Dict[str, BudgetRule] = {}
        self._usage: Dict[str, BudgetUsage] = {}

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                raw = json.loads(self.path.read_text(encoding="utf-8") or "{}")
                rules_raw = raw.get("rules", {}) or {}
                usage_raw = raw.get("usage", {}) or {}
                self._rules = {
                    cap: BudgetRule.from_dict(rd) for cap, rd in rules_raw.items()
                }
                self._usage = {
                    cap: BudgetUsage.from_dict(ud) for cap, ud in usage_raw.items()
                }
            except Exception:
                logger.exception("Failed to load budget from %s", self.path)
                self._rules = {}
                self._usage = {}
        else:
            self._rules = {}
            self._usage = {}

        # 确保有一条默认规则
        if "*" not in self._rules:
            self._rules["*"] = BudgetRule(capability="*")
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        raw = {
            "rules": {cap: r.to_dict() for cap, r in self._rules.items()},
            "usage": {cap: u.to_dict() for cap, u in self._usage.items()},
        }
        self.path.write_text(
            json.dumps(raw, ensure_ascii=False, indent=2), encoding="utf-8"
        )

    # ---------- 规则 & 使用 ----------

    def get_rule(self, capability: str) -> BudgetRule:
        self._load()
        if capability in self._rules:
            return self._rules[capability]
        return self._rules["*"]

    def _ensure_usage_today(self, capability: str) -> BudgetUsage:
        self._load()
        today = date.today().isoformat()
        key = capability or "*"
        usage = self._usage.get(key)
        if not usage or usage.date != today:
            usage = BudgetUsage(
                capability=key,
                date=today,
                tokens_used=0.0,
                patches_used=0,
                experiments_used=0,
            )
            self._usage[key] = usage
        return usage

    def can_spend(
        self,
        *,
        capability: str,
        cost_type: str,  # "tokens" / "patches" / "experiments"
        amount: float | int,
    ) -> bool:
        """
        判断是否还有配额。不会自动记录，只是 check。
        """
        self._load()
        rule = self.get_rule(capability)
        if not rule.enabled:
            return False

        usage = self._ensure_usage_today(capability)

        if cost_type == "tokens":
            return (usage.tokens_used + float(amount)) <= rule.max_tokens_per_day
        elif cost_type == "patches":
            return (usage.patches_used + int(amount)) <= rule.max_patches_per_day
        elif cost_type == "experiments":
            return (usage.experiments_used + int(amount)) <= rule.max_experiments_per_day
        else:
            logger.warning("Unknown cost_type=%s", cost_type)
            return True

    def record_spend(
        self,
        *,
        capability: str,
        cost_type: str,
        amount: float | int,
    ) -> None:
        """
        记录支出（不会再次检查 can_spend）。
        """
        self._load()
        usage = self._ensure_usage_today(capability)

        if cost_type == "tokens":
            usage.tokens_used += float(amount)
        elif cost_type == "patches":
            usage.patches_used += int(amount)
        elif cost_type == "experiments":
            usage.experiments_used += int(amount)
        else:
            logger.warning("Unknown cost_type=%s", cost_type)

        self._usage[usage.capability] = usage
        self._save()

# quentin/research/evolution.py
from __future__ import annotations

import logging
from dataclasses import dataclass, asdict
from typing import Dict, Any, List, Optional, Tuple

from quentin.blueprint.gap_tracker import BlueprintGapTracker, BlueprintGap
from quentin.research.experiments import ResearchManager
from quentin.research.budget import BudgetController
from quentin.self_improvement.manager import SelfImprovementManager
from quentin.self_improvement.schemas import ImprovementTask

logger = logging.getLogger(__name__)


@dataclass
class TaskScore:
    task_id: str
    capability: str
    requirement_id: str
    score: float
    factors: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class EvolutionStrategyManager:
    """
    根据蓝皮书 / 实验数据 / 预算，给出“下一步要优化哪些任务”。

    评分大致结构：
      score = blueprint_score * experiment_factor * budget_factor
    """

    def __init__(
        self,
        *,
        blueprint_tracker: Optional[BlueprintGapTracker] = None,
        research_manager: Optional[ResearchManager] = None,
        budget_controller: Optional[BudgetController] = None,
        si_manager: Optional[SelfImprovementManager] = None,
    ) -> None:
        self.blueprint_tracker = blueprint_tracker or BlueprintGapTracker()
        self.research_manager = research_manager or ResearchManager()
        self.budget_controller = budget_controller or BudgetController()
        self.si_manager = si_manager or SelfImprovementManager()

    # ---------- 主入口 ----------

    def suggest_next_tasks(
        self,
        *,
        limit: int = 10,
        allowed_statuses: Optional[List[str]] = None,
    ) -> List[Tuple[ImprovementTask, TaskScore]]:
        """
        返回按优先级排序的 ImprovementTask 列表及其评分。
        """
        tasks = self.si_manager.list_tasks()
        gaps = self.blueprint_tracker.scan()

        gap_index: Dict[Tuple[str, str], BlueprintGap] = {}
        for g in gaps:
            key = (g.capability_id, g.requirement_id)
            gap_index[key] = g

        scores: List[Tuple[ImprovementTask, TaskScore]] = []

        allowed_statuses_lower = (
            {s.lower() for s in allowed_statuses} if allowed_statuses else None
        )

        for t in tasks:
            status = getattr(t, "overall_status", "") or ""
            if allowed_statuses_lower and status.lower() not in allowed_statuses_lower:
                continue

            cap = getattr(t, "capability", "") or ""
            req = getattr(t, "requirement_id", "") or ""

            gap = gap_index.get((cap, req))
            stats = self.research_manager.get_task_level_stats(cap, req)
            ts = self._score_task(t, gap, stats)
            scores.append((t, ts))

        scores.sort(key=lambda x: x[1].score, reverse=True)
        return scores[:limit]

    # ---------- 打分逻辑 ----------

    def _score_task(
        self,
        task: ImprovementTask,
        gap: Optional[BlueprintGap],
        stats: Dict[str, Any],
    ) -> TaskScore:
        cap = getattr(task, "capability", "") or ""
        req = getattr(task, "requirement_id", "") or ""

        # 1) blueprint_score：来自蓝皮书优先级 + status
        if gap:
            base_priority = float(gap.priority)
            status = (gap.spec_status or "").lower()
            status_weight_map = {
                "missing": 1.0,
                "placeholder": 0.8,
                "partial": 0.5,
                "ready": 0.2,
            }
            status_weight = status_weight_map.get(status, 0.3)
            blueprint_score = base_priority * 10.0 * status_weight
        else:
            blueprint_score = 1.0

        # 2) experiment_factor：看历史实验表现
        total_runs = int(stats.get("total_runs", 0) or 0)
        mean_success = stats.get("mean_success", None)

        if total_runs == 0 or mean_success is None:
            experiment_factor = 1.0  # 还没探索过，保持中性
        else:
            # 成功率越高，说明方向“有产出”，略微放大优先级
            if mean_success < 0.3:
                experiment_factor = 0.7
            elif mean_success < 0.7:
                experiment_factor = 1.0
            else:
                experiment_factor = 1.2

        # 3) budget_factor：看预算是否允许
        if self.budget_controller.can_spend(
            capability=cap,
            cost_type="patches",
            amount=1,
        ):
            budget_factor = 1.0
        else:
            budget_factor = 0.1  # 预算不够，强行降权

        score = blueprint_score * experiment_factor * budget_factor

        factors = {
            "blueprint_score": blueprint_score,
            "experiment_factor": experiment_factor,
            "budget_factor": budget_factor,
            "total_runs": total_runs,
            "mean_success": mean_success,
        }

        return TaskScore(
            task_id=task.task_id,
            capability=cap,
            requirement_id=req,
            score=score,
            factors=factors,
        )

    def evaluate_patch(self, patch_id: str) -> CodePatch:
        """
        v1：假设 patch 已经由人工或其他工具应用到了 repo_root。
        """
        patch = self.patch_store.get(patch_id)
        if not patch:
            raise ValueError(f"Patch not found: {patch_id}")

        metrics = self.eval_harness.evaluate_patch(
            patch,
            apply_notes="Assumes patch was applied manually or by another tool.",
        )
        patch.eval_summary = metrics.to_dict()
        patch.status = "evaluated" if metrics.success else "rejected"
        self.patch_store.upsert(patch)

        # 把评估结果写入 ResearchManager，方便 EvolutionStrategy 用
        try:
            self.research_manager.record_patch_evaluation(patch)
        except Exception:
            logger.exception(
                "Failed to record patch evaluation to ResearchManager for patch_id=%s",
                patch.patch_id,
            )

        return patch
