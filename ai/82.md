# quentin/memory/fabric.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional, Iterable, Tuple
from datetime import datetime, timedelta
from pathlib import Path
import json
import logging
import uuid
import math

logger = logging.getLogger(__name__)


# ---------- 常量 ----------

TIER_STS = "STS"  # Short-Term Store
TIER_MTS = "MTS"  # Mid-Term Store
TIER_LTS = "LTS"  # Long-Term Store

DEFAULT_STS_MAX_ITEMS = 2000
DEFAULT_MTS_MAX_ITEMS = 5000
DEFAULT_LTS_MAX_ITEMS = 20000


# ---------- 数据结构 ----------

@dataclass
class MemoryConfig:
    """
    MemoryFabric v1 配置：

    - max_items_per_tier : 每一层最大条目数（超过会触发衰减）
    - sts_decay_minutes   : STS 默认存活时间（超过可能被清理）
    - mts_decay_days      : MTS 默认存活时间（超过可能下沉或压缩）
    - lts_decay_days      : LTS 几乎不删，这里只是留接口
    """

    max_items_per_tier: Dict[str, int] = field(
        default_factory=lambda: {
            TIER_STS: DEFAULT_STS_MAX_ITEMS,
            TIER_MTS: DEFAULT_MTS_MAX_ITEMS,
            TIER_LTS: DEFAULT_LTS_MAX_ITEMS,
        }
    )
    sts_decay_minutes: int = 120  # 2 小时
    mts_decay_days: int = 30
    lts_decay_days: int = 365 * 10  # 这里基本不动

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class MemoryItem:
    """
    核心记忆单元：

    - mem_id        : 唯一 ID
    - tier          : 所属记忆层（STS/MTS/LTS）
    - kind          : 类型（episode/semantic_summary/skill_ref/...）
    - text          : 主文本内容，用于检索 & 提示
    - metadata      : 附加信息，例如 domain / task_id / goal_id / tags 等
    - importance    : 0~1，主观重要性（用于排序 & 升级）
    - created_at    : 创建时间
    - last_access_at: 最近访问时间
    - access_count  : 被命中次数
    - frozen        : 是否冻结（不再被衰减/删除）
    """

    mem_id: str
    tier: str
    kind: str
    text: str
    metadata: Dict[str, Any] = field(default_factory=dict)
    importance: float = 0.5
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    last_access_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    access_count: int = 0
    frozen: bool = False

    def touch(self) -> None:
        self.last_access_at = datetime.utcnow().isoformat()
        self.access_count += 1

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "MemoryItem":
        return cls(**data)


@dataclass
class MemoryQueryResult:
    """
    查询结果项：

    - item : 命中的 MemoryItem
    - score: 相关度得分（0~1）
    """

    item: MemoryItem
    score: float

    def to_dict(self) -> Dict[str, Any]:
        d = self.item.to_dict()
        d["score"] = self.score
        return d


# ---------- 工具函数：简单相似度（v1 先用词重叠） ----------

def _simple_text_score(query: str, text: str) -> float:
    """
    非向量化版本的极简相似度：
    - 分词：按空格/简单符号
    - 计算 词交集 / 词并集
    只是为了让 MemoryFabric 在没有向量库的情况下也能跑起来。
    未来可以被 SelfProgrammingEngine 自动升级为真正的向量检索。
    """
    if not query or not text:
        return 0.0

    def _tokenize(s: str) -> List[str]:
        # 粗糙分词：按空格和标点切
        for ch in [",", ".", "!", "?", ":", ";", "\n", "\t"]:
            s = s.replace(ch, " ")
        return [w.lower() for w in s.split() if w.strip()]

    q_tokens = set(_tokenize(query))
    t_tokens = set(_tokenize(text))

    if not q_tokens or not t_tokens:
        return 0.0

    inter = len(q_tokens & t_tokens)
    union = len(q_tokens | t_tokens)
    if union == 0:
        return 0.0
    return inter / union


# ---------- MemoryFabric 主体 ----------

class MemoryFabric:
    """
    MemoryFabric v1：三层记忆 + 简单查询 + 升级/衰减策略（骨架可进化）。

    设计目标：
    - **三层**：STS / MTS / LTS
    - **最小可运行**：
        - 支持写入短期/中期/长期记忆
        - 支持简单的基于文本重叠的检索
        - 支持 run_maintenance() 做非常粗的 promotion/decay
    - **可自我进化**：
        - 具体 promotion_rule / decay_rule / 向量检索可以被 SelfProgrammingEngine 替换 & 改进
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        config: Optional[MemoryConfig] = None,
        sts_path: str | Path = "data/memory_sts.jsonl",
        mts_path: str | Path = "data/memory_mts.jsonl",
        lts_path: str | Path = "data/memory_lts.jsonl",
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.config = config or MemoryConfig()

        self.sts_path = self.repo_root / sts_path
        self.mts_path = self.repo_root / mts_path
        self.lts_path = self.repo_root / lts_path

        for p in [self.sts_path, self.mts_path, self.lts_path]:
            p.parent.mkdir(parents=True, exist_ok=True)

        # 内存缓存（v1 简单实现：运行时加载部分，写入时直接 append）
        self._cache: Dict[str, Dict[str, MemoryItem]] = {
            TIER_STS: {},
            TIER_MTS: {},
            TIER_LTS: {},
        }
        self._loaded_flags: Dict[str, bool] = {
            TIER_STS: False,
            TIER_MTS: False,
            TIER_LTS: False,
        }

    # ----- 公共 API：存储 -----

    def store_short_term(self, record: Dict[str, Any]) -> MemoryItem:
        """
        供 CognitiveLoop 调用：
        - tier=STS
        - kind/metadata/importance 由调用方传入或使用默认值
        """
        return self._store(record, tier=TIER_STS)

    def store_mid_term(self, record: Dict[str, Any]) -> MemoryItem:
        """
        供 SleepLoop 等调用：
        - tier=MTS
        """
        return self._store(record, tier=TIER_MTS)

    def store_long_term(self, record: Dict[str, Any]) -> MemoryItem:
        """
        将非常稳定的知识写入 LTS。
        """
        return self._store(record, tier=TIER_LTS)

    # 与之前代码兼容的通用入口
    def add_episode(self, tier: str, content: Dict[str, Any]) -> MemoryItem:
        if tier not in (TIER_STS, TIER_MTS, TIER_LTS):
            tier = TIER_STS
        return self._store(content, tier=tier)

    # ----- 公共 API：查询 -----

    def query(
        self,
        *,
        query_text: str,
        top_k: int = 5,
        tiers: Optional[List[str]] = None,
        kind: Optional[str] = None,
        domain: Optional[str] = None,
    ) -> List[MemoryQueryResult]:
        """
        简单查询：
        - 遍历目标层的所有 MemoryItem
        - 用 _simple_text_score 计算相关度
        - 按 score+importance 排序
        """
        if not query_text:
            return []

        tiers = tiers or [TIER_STS, TIER_MTS, TIER_LTS]
        items = list(self._iter_items(tiers=tiers))

        scored: List[Tuple[float, MemoryItem]] = []
        for item in items:
            if kind and item.kind != kind:
                continue
            if domain:
                # domain 存在于 metadata["domain"]
                if item.metadata.get("domain") != domain:
                    continue

            score = _simple_text_score(query_text, item.text)
            if score <= 0.0:
                continue
            # importance 加权
            final_score = 0.7 * score + 0.3 * float(item.importance)
            scored.append((final_score, item))

        # 按分数排序
        scored.sort(key=lambda x: x[0], reverse=True)
        results: List[MemoryQueryResult] = []
        for sc, item in scored[:top_k]:
            item.touch()
            results.append(MemoryQueryResult(item=item, score=sc))
        return results

    # ----- 公共 API：维护策略 -----

    def run_maintenance(self) -> None:
        """
        维护例程（可由 SleepLoop 或 Cron 调用）：

        v1 策略（非常简单，但可被未来自编程升级）：
        - STS:
            - 删除超出 sts_decay_minutes 且非 frozen 的旧条目
            - 若数量超过 max_items 时，按重要度+最近访问时间删尾部
            - 对 access_count >= 3 且 importance >= 0.6 的条目，考虑升级到 MTS
        - MTS:
            - 若数量超过阈值，删掉非常久未访问、importance 较低的条目
            - 对 access_count 高且 importance 高的条目，可以升级到 LTS
        - LTS:
            - 暂不主动删除，只更新 last_access_at
        """
        now = datetime.utcnow()

        # STS: decay + promotion
        self._load_tier(TIER_STS)
        sts_items = self._cache[TIER_STS]
        decay_delta = timedelta(minutes=self.config.sts_decay_minutes)

        to_delete_sts: List[str] = []
        to_promote_sts: List[MemoryItem] = []

        for mem_id, item in list(sts_items.items()):
            if item.frozen:
                continue
            try:
                created = datetime.fromisoformat(item.created_at)
            except Exception:
                created = now
            if now - created > decay_delta:
                to_delete_sts.append(mem_id)
                continue

            if item.access_count >= 3 and item.importance >= 0.6:
                to_promote_sts.append(item)

        for mem_id in to_delete_sts:
            self._delete_item(TIER_STS, mem_id)

        for item in to_promote_sts:
            self._promote_item(item, target_tier=TIER_MTS)

        # 控制 STS 数量
        self._apply_capacity_limit(TIER_STS)

        # MTS: 只做简单容量控制和少量升级
        self._apply_capacity_limit(TIER_MTS)
        # LTS: 目前不动

    # ----- 内部：存储 & 加载 & 迭代 -----

    def _store(self, record: Dict[str, Any], *, tier: str) -> MemoryItem:
        kind = str(record.get("kind") or "episode")
        text = str(record.get("text") or record.get("content") or "")

        metadata = dict(record.get("metadata") or {})
        importance = float(record.get("importance") or 0.5)

        mem = MemoryItem(
            mem_id=str(uuid.uuid4()),
            tier=tier,
            kind=kind,
            text=text,
            metadata=metadata,
            importance=max(0.0, min(1.0, importance)),
        )

        # 写入文件
        path = self._tier_path(tier)
        try:
            with path.open("a", encoding="utf-8") as f:
                f.write(json.dumps(mem.to_dict(), ensure_ascii=False) + "\n")
        except Exception:
            logger.exception("MemoryFabric._store: failed to append to %s", path)

        # 写入缓存
        self._load_tier(tier)  # 确保 cache 存在
        self._cache[tier][mem.mem_id] = mem
        return mem

    def _tier_path(self, tier: str) -> Path:
        if tier == TIER_STS:
            return self.sts_path
        if tier == TIER_MTS:
            return self.mts_path
        if tier == TIER_LTS:
            return self.lts_path
        # 默认回落 STS
        return self.sts_path

    def _load_tier(self, tier: str) -> None:
        if self._loaded_flags.get(tier):
            return
        path = self._tier_path(tier)
        cache = self._cache[tier]
        if not path.exists():
            self._loaded_flags[tier] = True
            return
        try:
            with path.open("r", encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if not line:
                        continue
                    try:
                        obj = json.loads(line)
                        mem = MemoryItem.from_dict(obj)
                        cache[mem.mem_id] = mem
                    except Exception:
                        logger.exception("MemoryFabric._load_tier: parse error in %s", path)
        except Exception:
            logger.exception("MemoryFabric._load_tier: failed to read %s", path)
        self._loaded_flags[tier] = True

    def _iter_items(self, *, tiers: Iterable[str]) -> Iterable[MemoryItem]:
        for tier in tiers:
            if tier not in self._cache:
                continue
            self._load_tier(tier)
            for mem in self._cache[tier].values():
                yield mem

    # ----- 内部：删除 / 提升 / 容量控制 -----

    def _delete_item(self, tier: str, mem_id: str) -> None:
        if tier not in self._cache:
            return
        cache = self._cache[tier]
        if mem_id not in cache:
            return
        # 从缓存删
        del cache[mem_id]
        # 重新写入文件（简单实现：重写整个 tier 文件）
        self._rewrite_tier_file(tier)

    def _rewrite_tier_file(self, tier: str) -> None:
        path = self._tier_path(tier)
        cache = self._cache[tier]
        try:
            with path.open("w", encoding="utf-8") as f:
                for mem in cache.values():
                    f.write(json.dumps(mem.to_dict(), ensure_ascii=False) + "\n")
        except Exception:
            logger.exception("MemoryFabric._rewrite_tier_file: failed to write %s", path)

    def _promote_item(self, item: MemoryItem, target_tier: str) -> None:
        """
        将 MemoryItem 从当前 tier 提升到目标 tier：
        - 在原 tier 删除
        - 在目标 tier 追加
        - tier 字段更新为 target_tier
        """
        if item.tier == target_tier:
            return

        # 从源层删除
        src_tier = item.tier
        if src_tier in self._cache and item.mem_id in self._cache[src_tier]:
            del self._cache[src_tier][item.mem_id]
            self._rewrite_tier_file(src_tier)

        # 更新 tier
        item.tier = target_tier
        # 重置部分计数（可调整策略）
        item.access_count = 0
        item.importance = min(1.0, max(0.5, item.importance))

        # 写入目标层
        path = self._tier_path(target_tier)
        try:
            with path.open("a", encoding="utf-8") as f:
                f.write(json.dumps(item.to_dict(), ensure_ascii=False) + "\n")
        except Exception:
            logger.exception("MemoryFabric._promote_item: failed to append to %s", path)

        self._load_tier(target_tier)
        self._cache[target_tier][item.mem_id] = item

    def _apply_capacity_limit(self, tier: str) -> None:
        """
        若某层条目数超过 max_items_per_tier[tier]，则按
        (importance, last_access_at) 排序，删除尾部。
        """
        self._load_tier(tier)
        cache = self._cache[tier]
        max_items = self.config.max_items_per_tier.get(tier)
        if max_items is None or len(cache) <= max_items:
            return

        # 冻结的条目不删
        items = [m for m in cache.values() if not m.frozen]
        if len(items) <= max_items:
            return

        def _score_for_eviction(item: MemoryItem) -> float:
            try:
                last = datetime.fromisoformat(item.last_access_at)
            except Exception:
                last = datetime.utcnow()
            # 越旧越容易删，importance 越高越不删
            age_days = (datetime.utcnow() - last).total_seconds() / 86400.0
            # 排序分数越大越先删
            return age_days * (1.0 + (1.0 - item.importance))

        # 从要删的里选出需要删除的数量
        num_to_delete = len(cache) - max_items
        # 排序，得分高的先删
        items.sort(key=_score_for_eviction, reverse=True)
        to_delete_ids = [m.mem_id for m in items[:num_to_delete]]

        for mem_id in to_delete_ids:
            if mem_id in cache:
                del cache[mem_id]

        self._rewrite_tier_file(tier)

三、现在蓝皮书 Memory 部分的落地情况

三层记忆：STS / MTS / LTS ✅
