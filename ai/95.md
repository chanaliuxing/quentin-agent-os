# quentin/research/runners/patch_test_runner.py
from __future__ import annotations

from typing import Any, Dict, Optional
import logging

from quentin.research.orchestrator import ExperimentRunner
from quentin.research.experiments import (
    ExperimentConfig,
    VariantConfig,
    VariantResult,
)
from quentin.self_programming.engine import SelfProgrammingEngine
from quentin.self_programming.store import PatchStore
from quentin.self_programming.schema import PatchCandidate, PatchReport

logger = logging.getLogger(__name__)


class PatchTestRunner(ExperimentRunner):
    """
    针对 PatchCandidate 的实验 Runner。

    设计思路：
    - 每个 variant 对应一个 patch_id（或 "baseline"）
    - 调用 SelfProgrammingEngine.run_tests_for_patch()
    - 把 PatchReport 映射到 VariantResult.metrics

    用法：
    - ExperimentConfig.target_component = "patch_test"
    - VariantConfig.params:
        - "patch_id": str   # PatchCandidate ID
    - context 可选字段：
        - "test_suites": ["unit", "static_analysis", ...]
        - "extra_context": dict
    """

    def __init__(
        self,
        engine: SelfProgrammingEngine,
        *,
        patch_store: Optional[PatchStore] = None,
    ) -> None:
        self.engine = engine
        self.patch_store = patch_store or engine.patch_store

    def run_variant(
        self,
        experiment: ExperimentConfig,
        variant: VariantConfig,
        context: Optional[Dict[str, Any]] = None,
    ) -> VariantResult:
        ctx = dict(context or {})
        test_suites = ctx.get("test_suites") or ["unit", "static_analysis"]
        extra_ctx = ctx.get("extra_context") or {}

        patch_id = variant.params.get("patch_id")
        if not patch_id:
            # baseline 或无 patch 情况
            logger.info(
                "PatchTestRunner: variant=%s has no patch_id; treated as baseline.",
                variant.variant_id,
            )
            metrics = {
                "score": 0.0,
                "tests_passed_ratio": 0.0,
                "risk_score": 0.0,
            }
            extra = {
                "patch_report": None,
                "is_baseline": True,
            }
            return VariantResult(
                variant_id=variant.variant_id,
                metrics=metrics,
                extra=extra,
            )

        patch = self.patch_store.load_candidate(patch_id)
        if patch is None:
            logger.warning(
                "PatchTestRunner: patch_id=%s not found for variant=%s",
                patch_id,
                variant.variant_id,
            )
            metrics = {
                "score": 0.0,
                "tests_passed_ratio": 0.0,
                "risk_score": 1.0,
            }
            extra = {
                "patch_report": None,
                "error": f"patch_id={patch_id} not found",
            }
            return VariantResult(
                variant_id=variant.variant_id,
                metrics=metrics,
                extra=extra,
            )

        report: PatchReport = self.engine.run_tests_for_patch(
            patch=patch,
            test_suites=test_suites,
            extra_context=extra_ctx,
        )

        # 根据 PatchReport 计算简单指标
        total_tests = len(report.tests) or 1
        passed_tests = sum(1 for t in report.tests if t.passed)
        passed_ratio = passed_tests / total_tests

        # v1：score 先简单用 (passed_ratio * (1 - risk_score))
        risk_score = float(report.risk_score or 1.0)
        score = passed_ratio * max(0.0, 1.0 - risk_score)

        metrics = {
            "score": float(score),
            "tests_passed_ratio": float(passed_ratio),
            "risk_score": float(risk_score),
            "coverage": float(report.coverage or 0.0),
        }

        extra = {
            "patch_report": report.to_dict(),
            "patch_id": patch_id,
        }

        return VariantResult(
            variant_id=variant.variant_id,
            metrics=metrics,
            extra=extra,
        )

# quentin/research/runners/skill_candidate_runner.py
from __future__ import annotations

from typing import Any, Dict, Optional
import logging

from quentin.research.orchestrator import ExperimentRunner
from quentin.research.experiments import (
    ExperimentConfig,
    VariantConfig,
    VariantResult,
)
from quentin.self_programming.engine import SelfProgrammingEngine
from quentin.skills.store import SkillStore
from quentin.skills.schema import SkillCandidate

logger = logging.getLogger(__name__)


class SkillCandidateRunner(ExperimentRunner):
    """
    针对 SkillCandidate 的离线评测 Runner（stub 版）。

    用法：
    - ExperimentConfig.target_component = "skill_candidate"
    - VariantConfig.params:
        - "skill_candidate_id": str
    - context:
        - 预留 "eval_dataset" / "task_factory" 等，未来可扩展为真正的行为测试
    """

    def __init__(
        self,
        engine: SelfProgrammingEngine,
        *,
        skill_store: Optional[SkillStore] = None,
    ) -> None:
        self.engine = engine
        self.skill_store = skill_store or engine.skill_store

    def run_variant(
        self,
        experiment: ExperimentConfig,
        variant: VariantConfig,
        context: Optional[Dict[str, Any]] = None,
    ) -> VariantResult:
        ctx = dict(context or {})
        candidate_id = variant.params.get("skill_candidate_id")

        if not candidate_id:
            logger.warning(
                "SkillCandidateRunner: variant=%s has no skill_candidate_id",
                variant.variant_id,
            )
            metrics = {
                "score": 0.0,
                "has_test_cases": 0.0,
                "estimated_risk": 1.0,
            }
            extra = {
                "candidate": None,
                "error": "missing skill_candidate_id",
            }
            return VariantResult(
                variant_id=variant.variant_id,
                metrics=metrics,
                extra=extra,
            )

        cand = self.skill_store.load_candidate(candidate_id)
        if cand is None:
            logger.warning(
                "SkillCandidateRunner: skill_candidate_id=%s not found",
                candidate_id,
            )
            metrics = {
                "score": 0.0,
                "has_test_cases": 0.0,
                "estimated_risk": 1.0,
            }
            extra = {
                "candidate": None,
                "error": "candidate not found",
            }
            return VariantResult(
                variant_id=variant.variant_id,
                metrics=metrics,
                extra=extra,
            )

        # v1 stub：简单按照是否有 test_cases 来打分
        has_tests = len(cand.proposed_skill.test_cases) > 0
        has_tests_float = 1.0 if has_tests else 0.0

        # 估算风险：有测试则略低
        estimated_risk = 0.8 if has_tests else 1.0
        score = has_tests_float * (1.0 - estimated_risk * 0.5)

        metrics = {
            "score": float(score),
            "has_test_cases": has_tests_float,
            "estimated_risk": float(estimated_risk),
        }

        extra = {
            "candidate": cand.to_dict(),
        }

        return VariantResult(
            variant_id=variant.variant_id,
            metrics=metrics,
            extra=extra,
        )

    # ---------- 与 ExperimentOrchestrator 集成的 Helper ---------- #

    def create_patch_experiment(
        self,
        *,
        patch_ids: List[str],
        name: str = "patch_eval",
        description: str = "Evaluate patch candidates via PatchTestRunner.",
    ) -> ExperimentConfig:
        """
        便捷方法：为一组 patch_ids 构造 ExperimentConfig。
        每个 patch_id 对应一个 variant。
        """
        variants: List[VariantConfig] = []
        for pid in patch_ids:
            variants.append(
                VariantConfig(
                    variant_id=f"patch_{pid[:8]}",
                    description=f"Patch {pid}",
                    params={"patch_id": pid},
                )
            )

        exp = ExperimentConfig(
            experiment_id=None,  # 由 ExperimentStore 分配
            name=name,
            description=description,
            target_component="patch_test",
            variants=variants,
        )
        exp = self.experiment_store.create_experiment(exp)
        return exp

    def run_patch_experiment(
        self,
        experiment_id: str,
        *,
        test_suites: Optional[List[str]] = None,
        extra_context: Optional[Dict[str, Any]] = None,
    ):
        """
        使用 PatchTestRunner 对某个 patch 实验进行评测。
        """
        experiment = self.experiment_store.get_experiment(experiment_id)
        if experiment is None:
            raise ValueError(f"Experiment {experiment_id} not found")

        runner = PatchTestRunner(engine=self)
        self.orchestrator.register_runner("patch_test", runner)

        ctx: Dict[str, Any] = {
            "test_suites": test_suites or ["unit", "static_analysis"],
            "extra_context": extra_context or {},
        }
        return self.orchestrator.run_experiment(experiment_id, context=ctx)

    def create_skill_experiment(
        self,
        *,
        candidate_ids: List[str],
        name: str = "skill_candidate_eval",
        description: str = "Evaluate skill candidates via SkillCandidateRunner.",
    ) -> ExperimentConfig:
        """
        为一组 skill_candidate_id 创建 ExperimentConfig。
        """
        variants: List[VariantConfig] = []
        for cid in candidate_ids:
            variants.append(
                VariantConfig(
                    variant_id=f"skill_{cid[:8]}",
                    description=f"SkillCandidate {cid}",
                    params={"skill_candidate_id": cid},
                )
            )

        exp = ExperimentConfig(
            experiment_id=None,
            name=name,
            description=description,
            target_component="skill_candidate",
            variants=variants,
        )
        exp = self.experiment_store.create_experiment(exp)
        return exp

    def run_skill_experiment(
        self,
        experiment_id: str,
        *,
        context: Optional[Dict[str, Any]] = None,
    ):
        """
        使用 SkillCandidateRunner 对 skill candidate 实验进行评测。
        """
        experiment = self.experiment_store.get_experiment(experiment_id)
        if experiment is None:
            raise ValueError(f"Experiment {experiment_id} not found")

        runner = SkillCandidateRunner(engine=self)
        self.orchestrator.register_runner("skill_candidate", runner)

        ctx = context or {}
        return self.orchestrator.run_experiment(experiment_id, context=ctx)

五、现在「自编程 + 实验」这一块的完整状态（和蓝皮书对齐）
蓝皮书第 4 章要的能力：

可验证的自编程能力

✅ PatchCandidate / SkillCandidate 有统一 schema；

✅ SelfProgrammingEngine 能生成候选（即便现在是 stub）；

✅ PatchTestRunner / SkillCandidateRunner 能把候选转成 Experiment + VariantResult；

✅ 结果中有 metrics（score / risk / tests_passed_ratio 等）和详细报告（PatchReport / candidate 内容）。

Research-Friendly 实验接口

✅ Patch / Skill 都可以通过 ExperimentOrchestrator 运行；

✅ ResearchInterface 有统一入口可触发这些评测；

✅ 所有数据（patch candidate, report, skill candidate）都落在 data/patches/, data/skills/ 下，方便你后面做 offline 分析 / 可视化。

清晰边界（安全内核不可改）

✅ SelfProgrammingEngine.is_target_allowed() 限制 PatchTargetRef 不能碰 kernel / schema / risk；

✅ 默认只对 mutable 目标或 Skill 层做自动修改。
