# quentin/core/task.py
from __future__ import annotations

from dataclasses import dataclass, field, asdict
from datetime import datetime
from typing import Dict, Any, Optional


@dataclass
class TaskSpec:
    """
    一个通用任务描述（domain-agnostic）：

    - task_id : 唯一 ID（可以由调用方指定，也可以在 CognitiveLoop 里自动生成）
    - goal    : 自然语言目标描述
    - domain  : 所属领域（"general" / "research" / "desktop" / ...）
    - input   : 结构化输入（后续可以挂 JSON、上下文等）
    - metadata: 额外信息（调用者、优先级、tags、request_id 等）
    """
    task_id: str
    goal: str
    domain: str = "general"
    input: Dict[str, Any] = field(default_factory=dict)
    metadata: Dict[str, Any] = field(default_factory=dict)
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class TaskResult:
    """
    一次任务执行的结果摘要：

    - task_id       : 对应的 TaskSpec
    - success       : 是否成功（CognitiveLoop 级别判断）
    - final_answer  : 最终输出（文本）
    - artifacts     : 其它产物（结构化数据、代码片段等）
    - error_message : 失败时的错误说明
    - started_at    : 开始时间
    - finished_at   : 结束时间
    - metrics       : 时长、token 估计、重试次数等
    - lessons       : Critic/反省产生的 lessons_learned（会写入 Memory）
    """
    task_id: str
    success: bool
    final_answer: str
    artifacts: Dict[str, Any] = field(default_factory=dict)
    error_message: Optional[str] = None
    started_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    finished_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    metrics: Dict[str, Any] = field(default_factory=dict)
    lessons: Optional[str] = None

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

# quentin/core/trace.py
from __future__ import annotations

import json
import logging
import uuid
from dataclasses import dataclass, asdict, field
from datetime import datetime
from pathlib import Path
from typing import Dict, Any, List, Optional, Iterable

logger = logging.getLogger(__name__)


@dataclass
class TraceEvent:
    """
    执行过程中的一个事件：

    - event_id   : 唯一 ID
    - task_id    : 所属任务
    - step       : 步骤名，例如 "plan", "execute", "critic"
    - role       : "planner" / "executor" / "critic" / "memory"
    - kind       : "llm_call" / "tool_call" / "memory_write" / "info"
    - payload    : 任意 JSON，可记录 prompt/response 摘要、tool 调用参数等
    - created_at : 时间戳
    """
    event_id: str
    task_id: str
    step: str
    role: str
    kind: str
    payload: Dict[str, Any] = field(default_factory=dict)
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "TraceEvent":
        return cls(
            event_id=d["event_id"],
            task_id=d["task_id"],
            step=d.get("step", ""),
            role=d.get("role", ""),
            kind=d.get("kind", ""),
            payload=d.get("payload", {}) or {},
            created_at=d.get("created_at", ""),
        )


class FileBackedTraceStore:
    """
    非常简单的 Trace 存储：一个 JSON 文件记录所有事件。

    结构：
      {
        "events": [
          { ...TraceEvent... },
          ...
        ]
      }
    """

    def __init__(self, path: str | Path = "data/traces.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._loaded = False
        self._events: List[TraceEvent] = []

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                raw = json.loads(self.path.read_text(encoding="utf-8") or "{}")
                evs_raw = raw.get("events", []) or []
                self._events = [TraceEvent.from_dict(e) for e in evs_raw]
            except Exception:
                logger.exception("Failed to load traces from %s", self.path)
                self._events = []
        else:
            self._events = []
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        raw = {
            "events": [e.to_dict() for e in self._events],
        }
        self.path.write_text(
            json.dumps(raw, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    # ---------- 核心接口 ----------

    def record_event(
        self,
        *,
        task_id: str,
        step: str,
        role: str,
        kind: str,
        payload: Dict[str, Any],
    ) -> TraceEvent:
        self._load()
        ev = TraceEvent(
            event_id=str(uuid.uuid4()),
            task_id=task_id,
            step=step,
            role=role,
            kind=kind,
            payload=payload,
        )
        self._events.append(ev)
        self._save()
        return ev

    def list_by_task(self, task_id: str) -> List[TraceEvent]:
        self._load()
        return [e for e in self._events if e.task_id == task_id]

    def iter_all(self) -> Iterable[TraceEvent]:
        self._load()
        return iter(self._events)

# quentin/core/critic.py
from __future__ import annotations

import logging
from dataclasses import dataclass, asdict
from typing import Dict, Any, List, Optional

from quentin.core.task import TaskSpec, TaskResult
from quentin.core.trace import TraceEvent
from quentin.llm.multi_council import MultiAICouncil, CouncilDecision

logger = logging.getLogger(__name__)


@dataclass
class CriticFeedback:
    """
    对一次 Task 的整体评价：

    - ok              : 是否整体看起来合理
    - confidence      : 模糊的信心评分（0.0~1.0）
    - issues          : 发现的问题列表（字符串）
    - suggestions     : 改进建议
    - lessons_learned : 适合写入 Memory 的总结
    """
    ok: bool
    confidence: float
    issues: List[str]
    suggestions: str
    lessons_learned: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class Critic:
    """
    利用 MultiAICouncil 实现反思 / 评价。

    未来可以：
      - 针对特定 domain 使用不同的模板
      - 根据历史失败次数调整反思强度（更多轮会议）
    """

    def __init__(self, council: MultiAICouncil) -> None:
        self.council = council

    def review(
        self,
        task: TaskSpec,
        result: TaskResult,
        trace_events: List[TraceEvent],
    ) -> CriticFeedback:
        trace_text = self._format_trace(trace_events)

        question = (
            "You are an AI critic reviewing the performance of another AI system.\n"
            "Assess whether the task's goal was achieved, identify issues, and "
            "produce concise lessons learned.\n"
        )

        context = (
            f"Task goal:\n{task.goal}\n\n"
            f"Domain: {task.domain}\n\n"
            f"Final answer:\n{result.final_answer}\n\n"
            f"Error message (if any):\n{result.error_message}\n\n"
            f"Trace events (high level):\n{trace_text}\n\n"
            "Please be concrete and avoid vague praise."
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="reflection",
        )

        # v1：简单 parse，一律认为 ok=True、confidence=0.7，后续可结构化输出
        feedback = CriticFeedback(
            ok=True,
            confidence=0.7,
            issues=[],
            suggestions=decision.final_answer,
            lessons_learned=decision.final_answer,
        )
        return feedback

    def _format_trace(self, events: List[TraceEvent]) -> str:
        """
        把结构化 Trace 简单压成文本，以便给 LLM 做上下文。
        """
        lines: List[str] = []
        for e in events[-20:]:  # 只取最后 20 条，防止上下文爆炸
            lines.append(
                f"[{e.step}/{e.role}/{e.kind}] payload_keys={list((e.payload or {}).keys())}"
            )
        return "\n".join(lines)
v1 先让 Critic 输出一段“文字总结”，同时作为 lessons_learned 写入 Memory。
后面你可以升级成结构化 JSON 输出，把 issues / 封装成更多维度的指标。

# quentin/core/cognitive_loop.py
from __future__ import annotations

import logging
import uuid
from datetime import datetime
from typing import Dict, Any, List, Optional

from quentin.core.task import TaskSpec, TaskResult
from quentin.core.trace import FileBackedTraceStore, TraceEvent
from quentin.core.critic import Critic, CriticFeedback
from quentin.llm.multi_council import MultiAICouncil
from quentin.llm.types import LLMConfig
from quentin.memory.fabric import FileBackedMemoryFabric
from quentin.memory.models import MemoryRecord

logger = logging.getLogger(__name__)


class CognitiveLoop:
    """
    对应蓝皮书里的“认知循环”最小可运行版本：
      1) 从外部接到一个 TaskSpec（目标）
      2) 规划：MultiAICouncil 开会，生成 plan
      3) 执行：MultiAICouncil 或单模型执行（当前版本 purely LLM）
      4) 反省：Critic（MultiAICouncil）评估
      5) 写入 Trace + Memory（STS）

    未来可以挂：
      - 真正的工具/动作执行（桌面自动化、LegalBot、Trading 等）
      - 复杂的重试策略 / 多轮对话
    """

    def __init__(
        self,
        *,
        council: Optional[MultiAICouncil] = None,
        trace_store: Optional[FileBackedTraceStore] = None,
        memory_fabric: Optional[FileBackedMemoryFabric] = None,
    ) -> None:
        if council is None:
            default_members = [
                LLMConfig(provider="ollama", model="qwen3-vl:latest", role="planner"),
                LLMConfig(provider="ollama", model="deepseek-r1:1.5b", role="critic"),
            ]
            council = MultiAICouncil(members=default_members)

        self.council = council
        self.trace_store = trace_store or FileBackedTraceStore()
        self.memory_fabric = memory_fabric or FileBackedMemoryFabric()
        self.critic = Critic(self.council)

    # ---------- 一次性任务入口（给 CLI 用） ----------

    def run_oneoff(
        self,
        *,
        goal: str,
        domain: str = "general",
        input_data: Optional[Dict[str, Any]] = None,
        metadata: Optional[Dict[str, Any]] = None,
    ) -> TaskResult:
        task_id = str(uuid.uuid4())
        spec = TaskSpec(
            task_id=task_id,
            goal=goal,
            domain=domain,
            input=input_data or {},
            metadata=metadata or {},
        )
        return self.run_task(spec)

    # ---------- 核心认知循环 ----------

    def run_task(self, spec: TaskSpec) -> TaskResult:
        logger.info("CognitiveLoop: start task_id=%s goal=%s", spec.task_id, spec.goal)
        started_at = datetime.utcnow().isoformat()

        # 0) 收集记忆上下文
        memory_context = self._collect_memory_context(spec)

        # 1) 规划阶段（多 AI 开会）
        plan_text = self._plan(spec, memory_context)

        # 2) 执行阶段（当前版本仅 LLM，返回 final_answer）
        final_answer = self._execute(spec, plan_text, memory_context)

        # 3) 生成 TaskResult（先不考虑工具级错误）
        result = TaskResult(
            task_id=spec.task_id,
            success=True,
            final_answer=final_answer,
            artifacts={
                "plan": plan_text,
                "memory_context_used": memory_context,
            },
            error_message=None,
            started_at=started_at,
            finished_at=datetime.utcnow().isoformat(),
            metrics={},
        )

        # 4) 反省（Critic）
        trace_events = self.trace_store.list_by_task(spec.task_id)
        feedback = self.critic.review(spec, result, trace_events)
        result.lessons = feedback.lessons_learned
        result.metrics["critic_confidence"] = feedback.confidence

        # 5) 写入 Memory（STS：episodic）
        self._write_memory(spec, result, feedback)

        logger.info(
            "CognitiveLoop: finished task_id=%s success=%s",
            spec.task_id,
            result.success,
        )
        return result

    # ---------- 内部步骤 ----------

    def _collect_memory_context(self, spec: TaskSpec) -> str:
        """
        v1：简单从 MemoryFabric 中拉取与 domain 匹配的 LTS/MTS 记忆，
        拼成一段上下文文本，并记录 access_count。
        """
        pieces: List[str] = []

        # 先 LTS，再 MTS
        for tier in ("LTS", "MTS"):
            recs = self.memory_fabric.search_by_meta(
                tier=tier,
                meta_contains={"domain": spec.domain},
                limit=5,
            )
            for r in recs:
                self.memory_fabric.record_access(r.memory_id)
                pieces.append(
                    f"[{tier}] {r.meta.get('kind', r.kind)} | "
                    f"from_task={r.meta.get('task_id', '')} | "
                    f"content={r.content}"
                )

        return "\n".join(pieces)

    def _plan(self, spec: TaskSpec, memory_context: str) -> str:
        question = (
            "You are a planning agent in a self-improving AI system.\n"
            "Given the task goal and some prior memories, propose a concise plan "
            "with concrete steps. Prefer 3-7 steps.\n"
        )
        context = (
            f"Task goal:\n{spec.goal}\n\n"
            f"Domain: {spec.domain}\n\n"
            f"Input:\n{spec.input}\n\n"
            f"Relevant memories:\n{memory_context}\n"
        )

        decision = self.council.deliberate(
            question=question,
            context=context,
            mode="plan",
        )

        self.trace_store.record_event(
            task_id=spec.task_id,
            step="plan",
            role="planner",
            kind="llm_council",
            payload={
                "question": question,
                "context_summary": context[:1000],
                "decision": decision.to_dict(),
            },
        )
        return decision.final_answer

    def _execute(self, spec: TaskSpec, plan_text: str, memory_context: str) -> str:
        """
        v1：执行阶段仍然用 MultiAICouncil，后续可以拆成：
          - Planner 子 Agent 负责拆步骤
          - Executor 子 Agent 负责调工具
        """
        question = (
            "You are an executor agent in a self-improving AI system.\n"
            "Follow the given plan to complete the task and produce a final answer.\n"
        )
        context = (
            f"Task goal:\n{spec.goal}\n\n"
            f"Plan:\n{plan_text}\n\n"
            f"Domain: {spec.domain}\n\n"
            f"Input:\n{spec.input}\n\n"
            f"Relevant memories:\n{memory_context}\n"
        )

        decision = self.council.deliberate(
            question=question,
            context=context,
            mode="execute",
        )

        self.trace_store.record_event(
            task_id=spec.task_id,
            step="execute",
            role="executor",
            kind="llm_council",
            payload={
                "question": question,
                "context_summary": context[:1000],
                "decision": decision.to_dict(),
            },
        )
        return decision.final_answer

    def _write_memory(
        self,
        spec: TaskSpec,
        result: TaskResult,
        feedback: CriticFeedback,
    ) -> None:
        """
        把本次任务作为一条 episodic STS 记忆写入 MemoryFabric。
        """
        summary = (
            f"Task goal: {spec.goal}\n\n"
            f"Final answer (truncated):\n{result.final_answer[:1200]}\n\n"
            f"Lessons learned:\n{feedback.lessons_learned}"
        )

        meta: Dict[str, Any] = {
            "domain": spec.domain,
            "task_id": spec.task_id,
            "kind": "task_episode",
            "critic_confidence": feedback.confidence,
        }

        rec: MemoryRecord = self.memory_fabric.create(
            tier="STS",
            kind="episodic",
            content=summary,
            meta=meta,
        )

        self.trace_store.record_event(
            task_id=spec.task_id,
            step="memory",
            role="system",
            kind="memory_write",
            payload={
                "memory_id": rec.memory_id,
                "tier": rec.tier,
                "kind": rec.kind,
            },
        )
