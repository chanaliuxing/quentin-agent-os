@dataclass
class MemoryItem:
    """
    Memory Fabric 的基本单元。

    - memory_id : 唯一 ID
    - tier      : STS / MTS / LTS
    - kind      : 记忆类型，如 "skill_result" / "semantic_summary" / "task_lesson"
    - content   : 主要文本内容（方便 LLM 检索和 summarization）
    - payload   : 结构化数据（如 skill_result 的 JSON）

    - task_id   : 来源任务（可选）
    - domain    : 任务域（general/desktop/legal/trading/…）

    - stats     : 简单访问统计（以后 SleepLoop 也可以用它做 promotion 策略）
    - update_policy : 从蓝皮书中来的元信息，先以字符串描述的规则占位
    """

    memory_id: str
    tier: MemoryTier
    kind: str
    content: str
    payload: Dict[str, Any] = field(default_factory=dict)

    task_id: Optional[str] = None
    domain: str = "general"

    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    access_count: int = 0
    last_accessed_at: Optional[str] = None

    update_policy: Dict[str, Any] = field(
        default_factory=lambda: {
            "promotion_rule": "heuristic_v1",
            "decay_rule": "never_for_now",
            "consolidation_batch": "manual_or_sleep_loop",
        }
    )

    metadata: Dict[str, Any] = field(default_factory=dict)

    def touch(self) -> None:
        self.access_count += 1
        now = datetime.utcnow().isoformat()
        self.last_accessed_at = now
        self.updated_at = now

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "MemoryItem":
        return cls(
            memory_id=d["memory_id"],
            tier=d.get("tier", "STS"),
            kind=d.get("kind", "generic"),
            content=d.get("content", ""),
            payload=d.get("payload", {}) or {},
            task_id=d.get("task_id"),
            domain=d.get("domain", "general"),
            created_at=d.get("created_at", datetime.utcnow().isoformat()),
            updated_at=d.get("updated_at", datetime.utcnow().isoformat()),
            access_count=int(d.get("access_count", 0)),
            last_accessed_at=d.get("last_accessed_at"),
            update_policy=d.get("update_policy", {}) or {},
            metadata=d.get("metadata", {}) or {},

class FileBackedMemoryFabric:
    """
    v1 MemoryFabric 实现：所有记忆存到一个 JSON 文件中。

    结构：
      {
        "items": {
          "<memory_id>": { ...MemoryItem... },
          ...
        }
      }
    """

    def __init__(self, path: Union[str, Path] = "data/memory.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._loaded = False
        self._items: Dict[str, MemoryItem] = {}

    # ---------- 内部 load/save ----------

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                raw = json.loads(self.path.read_text(encoding="utf-8") or "{}")
                items_raw = raw.get("items", {}) or {}
                self._items = {
                    mid: MemoryItem.from_dict(md) for mid, md in items_raw.items()
                }
            except Exception:
                logger.exception("MemoryFabric: failed to load, using empty store.")
                self._items = {}
        else:
            self._items = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        raw = {
            "items": {mid: m.to_dict() for mid, m in self._items.items()},
        }
        self.path.write_text(
            json.dumps(raw, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    # ---------- CRUD API ----------

    def add(self, item: Union[MemoryItem, Dict[str, Any]]) -> MemoryItem:
        """
        兼容之前的“鸭子调用”：
          - 若传 dict，则自动生成 MemoryItem
          - 若传 MemoryItem，直接存
        """
        self._load()

        if isinstance(item, dict):
            # 若没有 memory_id，由调用者保证唯一性——这里做一个简单 fallback
            from uuid import uuid4

            d = dict(item)
            if "memory_id" not in d:
                d["memory_id"] = str(uuid4())
            if "tier" not in d:
                d["tier"] = "STS"
            if "kind" not in d:
                d["kind"] = "generic"
            if "content" not in d:
                # 如果没有 content，就从 payload 中提个摘要
                payload = d.get("payload") or {}
                content = str(payload)[:500]
                d["content"] = content
            mem = MemoryItem.from_dict(d)
        else:
            mem = item

        self._items[mem.memory_id] = mem
        self._save()
        return mem

    def get(self, memory_id: str) -> Optional[MemoryItem]:
        self._load()
        m = self._items.get(memory_id)
        if m:
            m.touch()
            self._save()
        return m

    def list_items(
        self,
        *,
        tier: Optional[str] = None,
        kind: Optional[str] = None,
        limit: Optional[int] = None,
    ) -> List[MemoryItem]:
        self._load()
        items = list(self._items.values())
        if tier:
            items = [m for m in items if m.tier == tier]
        if kind:
            items = [m for m in items if m.kind == kind]

        # 按 created_at 排序（新到旧）
        items.sort(key=lambda m: m.created_at, reverse=True)
        if limit is not None:
            items = items[:limit]
        return items

    def promote(self, memory_id: str, new_tier: str) -> None:
        self._load()
        m = self._items.get(memory_id)
        if not m:
            return
        m.tier = new_tier
        m.updated_at = datetime.utcnow().isoformat()
        self._items[memory_id] = m
        self._save()

    def delete(self, memory_id: str) -> None:
        self._load()
        if memory_id in self._items:
            del self._items[memory_id]
            self._save()

    def count_by_tier(self) -> Dict[str, int]:
        self._load()
        counts: Dict[str, int] = {"STS": 0, "MTS": 0, "LTS": 0}
        for m in self._items.values():
            if m.tier in counts:
                counts[m.tier] += 1
        return counts

class ResearchManager:
    """
    v1：用 JSON 文件记录实验定义和运行结果。

    结构：
      {
        "experiments": {
          "<experiment_id>": { ...Experiment... },
          ...
        },
        "runs": {
          "<run_id>": { ...ExperimentRun... },
          ...
        }
      }
    """

    def __init__(self, path: Union[str, Path] = "data/research.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._loaded = False
        self._experiments: Dict[str, Experiment] = {}
        self._runs: Dict[str, ExperimentRun] = {}

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                raw = json.loads(self.path.read_text(encoding="utf-8") or "{}")
                exps_raw = raw.get("experiments", {}) or {}
                runs_raw = raw.get("runs", {}) or {}
                self._experiments = {
                    eid: Experiment.from_dict(ed) for eid, ed in exps_raw.items()
                }
                self._runs = {
                    rid: ExperimentRun.from_dict(rd) for rid, rd in runs_raw.items()
                }
            except Exception:
                logger.exception("ResearchManager: failed to load, using empty store.")
                self._experiments = {}
                self._runs = {}
        else:
            self._experiments = {}
            self._runs = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        raw = {
            "experiments": {
                e_id: e.to_dict() for e_id, e in self._experiments.items()
            },
            "runs": {r_id: r.to_dict() for r_id, r in self._runs.items()},
        }
        self.path.write_text(
            json.dumps(raw, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    # ---------- Experiment CRUD ----------

    def list_experiments(self) -> List[Experiment]:
        self._load()
        return list(self._experiments.values())

    def get_experiment(self, experiment_id: str) -> Optional[Experiment]:
        self._load()
        return self._experiments.get(experiment_id)

    def create_experiment(
        self,
        *,
        name: str,
        purpose: str,
        capability: str,
        requirement_id: str,
        tags: Optional[List[str]] = None,
    ) -> Experiment:
        self._load()

            experiment_id=experiment_id,
            name=name,
            purpose=purpose,
            capability=capability,
            requirement_id=requirement_id,
            tags=tags or [],
        )
        self._experiments[experiment_id] = exp
        self._save()
        return exp

    # ---------- Run ----------

    def record_run(
        self,
        *,
        experiment_id: str,
        variant_id: str,
        metrics: Dict[str, Any],
        context: Optional[Dict[str, Any]] = None,
    ) -> ExperimentRun:
        self._load()
        if experiment_id not in self._experiments:
            raise ValueError(f"Experiment not found: {experiment_id}")

        run_id = str(uuid.uuid4())

            variant_id=variant_id,
            metrics=metrics,
            context=context or {},
        )
        self._runs[run_id] = run
        self._save()
        return run

        self._load()
        runs = [r for r in self._runs.values() if r.experiment_id == experiment_id]

class SleepLoop:
    """
    睡眠 / 巩固循环 v1：

    - 目标对齐蓝皮书 3.3：
        * 从 STS 中提炼 semantic memory（MTS/LTS）
        * 决定哪些 STS 记忆可以被删除
        * 为 Research Layer 提供指标

    v1 策略（刻意简单）：
      1. 取最近 N 条 STS 记忆（默认 50）
      2. 用 MultiAICouncil 生成“本轮总结”
      3. 把总结写入 MTS 作为 'semantic_summary'
      4. 将已参与总结的 STS 记忆标记为删除（模拟“遗忘”）
      5. 将本轮过程写入 ResearchManager

    将来可以扩展：
      - 细分不同 kind（skill_result / task_lesson ...）
      - 给部分记忆提升到 LTS（长期知识）
      - 输出训练数据对（failure → success）
    """

    def __init__(
        self,
        *,
        memory_fabric: Optional[FileBackedMemoryFabric] = None,
        council: Optional[MultiAICouncil] = None,
        research_manager: Optional[ResearchManager] = None,
        state_path: Optional[Path] = None,
        max_sts_per_run: int = 50,
    ) -> None:
        self.memory_fabric = memory_fabric or FileBackedMemoryFabric()
        if council is None:
            default_members = [
                LLMConfig(
                    provider="ollama",
                    model="qwen3-vl:latest",
                    role="consolidator",
                ),
                LLMConfig(
                    provider="ollama",
                    model="deepseek-r1:1.5b",
                    role="critic",
                ),
            ]
            council = MultiAICouncil(members=default_members)
        self.council = council

        self.research_manager = research_manager or ResearchManager()
        self.state_path = state_path or Path("data/sleep_state.json")
        self.max_sts_per_run = max_sts_per_run

    # ---------- 状态管理 ----------

    def _load_state(self) -> Dict[str, Any]:
        if self.state_path.exists():
            try:
                return json.loads(self.state_path.read_text(encoding="utf-8") or "{}")
            except Exception:
                logger.exception("SleepLoop: failed to load state; using empty.")
        return {}

    def _save_state(self, state: Dict[str, Any]) -> None:
        self.state_path.parent.mkdir(parents=True, exist_ok=True)
        self.state_path.write_text(
            json.dumps(state, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    # ---------- 主入口 ----------

    def run_once(self) -> SleepRunReport:
        state = self._load_state()
        started_at = datetime.utcnow().isoformat()

        # 1) 取 STS 记忆（最近 self.max_sts_per_run 条）
        sts_items: List[MemoryItem] = self.memory_fabric.list_items(
            tier="STS",
            limit=self.max_sts_per_run,
        )
        if not sts_items:
            finished_at = datetime.utcnow().isoformat()
            report = SleepRunReport(
                run_id=started_at,
                started_at=started_at,
                finished_at=finished_at,
                sts_processed=0,
                sts_deleted=0,
                mts_created=0,
                lts_created=0,
                notes="no STS items to consolidate",
            )
            self._record_research(report)
            self._save_state({"last_run_at": finished_at})
            return report

        # 2) 拼接上下文给 MultiAICouncil 做“夜间总结”
        ctx_lines: List[str] = []
        for m in sts_items:
            line = f"[{m.kind}/{m.domain}] {m.content[:300]}"
            ctx_lines.append(line)
        context = "\n".join(ctx_lines)

        question = (
            "You are the SLEEP/CONSOLIDATION loop of a self-improving agent.\n"
            "Given a list of episodic STS memories (skill results, lessons, etc.), "
            "you must summarize the key reusable knowledge and patterns.\n\n"
            "Please output in the following format:\n"
            "SEMANTIC_SUMMARY:\n"
            "<multi-paragraph natural language summary>\n\n"
            "LESSONS:\n"
            "- <bullet point 1>\n"
            "- <bullet point 2>\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="sleep_consolidation",
        )
        answer = decision.final_answer or ""

        semantic_summary, lessons_block = self._split_answer(answer)

        # 3) 把总结写入 MTS 作为 semantic_summary
        mts_created = 0
        if semantic_summary.strip():
            mem = MemoryItem(
                memory_id=f"sleep_mts_{started_at}",
                tier="MTS",
                kind="semantic_summary",
                content=semantic_summary.strip(),
                payload={
                    "lessons": lessons_block.strip(),
                    "source": "sleep_loop",
                    "num_sts_merged": len(sts_items),
                    "started_at": started_at,
                },
                domain="cross",
            )
            self.memory_fabric.add(mem)
            mts_created = 1

        # 4) 删除已处理的 STS（模拟遗忘）
        sts_deleted = 0
        for m in sts_items:
            try:
                self.memory_fabric.delete(m.memory_id)
                sts_deleted += 1
            except Exception:
                logger.exception(
                    "SleepLoop: failed to delete STS memory_id=%s", m.memory_id
                )

        finished_at = datetime.utcnow().isoformat()
        report = SleepRunReport(
            run_id=started_at,
            started_at=started_at,
            finished_at=finished_at,
            sts_processed=len(sts_items),
            sts_deleted=sts_deleted,
            mts_created=mts_created,
            lts_created=0,  # v1 先不做 LTS 提升逻辑
            notes=f"consolidated {len(sts_items)} STS items into {mts_created} MTS summary",
        )

        # 5) 写入 Research Layer
        self._record_research(report)
        self._save_state({"last_run_at": finished_at})
        return report

    # ---------- 辅助 ----------

    def _split_answer(self, text: str) -> tuple[str, str]:
        """
        从回答中粗略分离 SEMANTIC_SUMMARY 和 LESSONS 段落。
        """
        marker = "LESSONS:"
        idx = text.find(marker)
        if idx == -1:
            return text, ""
        summary = text[:idx].replace("SEMANTIC_SUMMARY:", "").strip()
        lessons = text[idx + len(marker) :].strip()
        return summary, lessons

    def _record_research(self, report: SleepRunReport) -> None:
        """
        将本次 SleepLoop 运行记录为一个 experiment run。
        """
        rm = self.research_manager

        # 找或建一个 experiment
        capability = "memory:sleep_loop"
        requirement_id = "consolidation_quality"

        exps = rm.list_experiments()
        exp: Optional[Experiment] = None
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "sleep_consolidation"
            ):
                exp = e
                break
        if exp is None:
            exp = rm.create_experiment(
                name="SleepLoop-Consolidation",
                purpose="sleep_consolidation",
                capability=capability,
                requirement_id=requirement_id,
                tags=["memory", "sleep"],
            )

        metrics = {
            "sts_processed": report.sts_processed,
            "sts_deleted": report.sts_deleted,
            "mts_created": report.mts_created,
            "lts_created": report.lts_created,
        }
        context = {
            "run_id": report.run_id,
            "notes": report.notes,
            "started_at": report.started_at,
            "finished_at": report.finished_at,
        }

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics=metrics,
            context=context,
        )
