一、研究预算层：BudgetController
新文件：quentin/research/budget.py

这个模块先做一个**“轻量预算系统”**：

默认有一份内建的 per-capability 预算

可以用 data/research_budget.json 覆盖

每个 capability 有：

priority：静态优先级（越大越重要）

max_experiments_per_day：每天最多自改实验数（占位）

max_tokens_per_day：理论 token 上限（占位）

暂时不做精确消耗，只做“预算因子”用来调权重

# quentin/research/budget.py
from __future__ import annotations

import json
import logging
from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Dict, Any, Optional

logger = logging.getLogger(__name__)


@dataclass
class CapabilityBudget:
    """
    针对一个 capability 的预算配置（v1 版）。
    目前只是轻量级的“软约束”，用于 EvolutionStrategyManager 计算优先级。
    """
    capability: str

    # 静态优先级：0~1，越大表示越重要（默认 0.5）
    priority: float = 0.5

    # 理论预算（占位字段，未来可以逐步接入真实消耗统计）
    max_experiments_per_day: int = 10
    max_tokens_per_day: int = 200_000

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "CapabilityBudget":
        return cls(
            capability=data["capability"],
            priority=float(data.get("priority", 0.5)),
            max_experiments_per_day=int(data.get("max_experiments_per_day", 10)),
            max_tokens_per_day=int(data.get("max_tokens_per_day", 200_000)),
        )


class BudgetController:
    """
    Research & Evolution 层的预算管理器 v1：

    - 从 data/research_budget.json 读取配置（如果不存在就用内建默认）
    - 提供 get_capability_budget(capability) 接口
    - 未来可以加：记录每日实际消耗、剩余预算等
    """

    def __init__(self, path: str | Path = "data/research_budget.json") -> None:
        self.path = Path(path)
        self._cache: Dict[str, CapabilityBudget] = {}
        self._loaded = False

    # ---------- 内部：加载 & 默认 ----------

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                with self.path.open("r", encoding="utf-8") as f:
                    raw = json.load(f)
                for cap, cfg in raw.items():
                    cfg["capability"] = cap
                    self._cache[cap] = CapabilityBudget.from_dict(cfg)
                logger.info("Loaded research budget from %s", self.path)
            except Exception as e:
                logger.exception("Failed to load research budget from %s", self.path)
                self._cache = {}
        else:
            logger.info("Budget file %s not found, using defaults.", self.path)
            self._cache = {}
        self._loaded = True

    def _default_budget(self, capability: str) -> CapabilityBudget:
        # v1 默认：所有 capability 统一 0.5 权重，
        # 可以在 data/research_budget.json 里按需覆盖。
        return CapabilityBudget(
            capability=capability,
            priority=0.5,
            max_experiments_per_day=10,
            max_tokens_per_day=200_000,
        )

    # ---------- 对外接口 ----------

    def get_capability_budget(self, capability: str) -> CapabilityBudget:
        self._load()
        if capability in self._cache:
            return self._cache[capability]
        # 未配置则用默认，并不立即写回文件，方便之后手工维护。
        return self._default_budget(capability)

    def dump_current_to_file(self) -> None:
        """
        把当前内存中的预算配置写回到 JSON 文件。
        通常用于你手动编辑/脚本调整之后做持久化。
        """
        self._load()
        data: Dict[str, Any] = {}
        for cap, cfg in self._cache.items():
            d = cfg.to_dict()
            d.pop("capability", None)
            data[cap] = d

        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self.path.open("w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        logger.info("Research budget dumped to %s", self.path)

2. EvolutionStrategyManager 实现
class EvolutionStrategyManager:
    """
    L3：Research & Evolution Layer 的策略管理 v1。

    负责回答：
    - 下一轮应该优先优化哪些 capability / requirement / task？
    - 依据是什么？（gap 大？实验效果不好？预算允许？）

    目前完全是“数据驱动 heuristics”，不会再调用 LLM，
    方便你后面把这套逻辑替换成 bandit / RL / 更复杂的优化器。
    """

    def __init__(
        self,
        research_manager: Optional[ResearchManager] = None,
        budget_controller: Optional[BudgetController] = None,
        si_manager: Optional[SelfImprovementManager] = None,
    ) -> None:
        self.research_manager = research_manager or ResearchManager()
        self.budget_controller = budget_controller or BudgetController()
        self.si_manager = si_manager or SelfImprovementManager()

    # ---------- 对外：选出 topN 自我迭代候选 ----------

    def suggest_next_tasks(
        self,
        limit: int = 10,
    ) -> List[OptimizationCandidate]:
        """
        综合考虑：
        - ImprovementTask 的 gap 程度（overall_status）
        - Experiment 的当前效果（avg_score / success_rate / num_variants）
        - Capability 的预算优先级（priority）
        计算一个 priority_score，返回 top N 候选任务。
        """
        tasks = self.si_manager.list_tasks()
        # 实验聚合（按 capability + requirement_id）
        cap_rows = self.research_manager.top_capabilities(limit=10_000)
        # 重组成键值映射，方便快速查询
        exp_index: Dict[tuple, Dict[str, Any]] = {}
        for r in cap_rows:
            key = (r["capability"], r["requirement_id"])
            exp_index[key] = r

        candidates: List[OptimizationCandidate] = []
        for t in tasks:
            score, signals, reason = self._compute_priority_for_task(t, exp_index)
            if score <= 0:
                # 得分为 0 的任务就不推了（例如已经 ready 且没有明显改进空间）
                continue
            candidate = OptimizationCandidate(
                task_id=t.task_id,
                capability=t.capability,
                requirement_id=t.requirement_id,
                priority_score=score,
                reason=reason,
                signals=signals,
            )
            candidates.append(candidate)

        candidates.sort(key=lambda c: c.priority_score, reverse=True)
        return candidates[:limit]

    # ---------- 核心逻辑：给单 task 打分 ----------

    def _compute_priority_for_task(
        self,
        task: ImprovementTask,
        exp_index: Dict[tuple, Dict[str, Any]],
    ) -> tuple[float, Dict[str, Any], str]:
        """
        计算某个 ImprovementTask 的优先级：

        priority_score ~= gap_factor * improvement_potential * budget_factor

        gap_factor:
          - 来自 task.overall_status（missing / placeholder / partial / ready）
        improvement_potential:
          - 来自 experiment 的 avg_score / success_rate / num_variants
        budget_factor:
          - 来自 BudgetController 的 capability.priority
        """
        cap = task.capability
        req = task.requirement_id
        key = (cap, req)
        # 如果 req 为 None，则也试一下 "ALL"
        exp = exp_index.get(key) or exp_index.get((cap, "ALL"))

        # 1) gap_factor：看任务当前状态
        status = (task.overall_status or "").lower()
        if status in ("missing", "not_implemented"):
            gap_factor = 1.0
        elif status in ("placeholder", "stub"):
            gap_factor = 0.8
        elif status in ("partial", "wip", "in_progress"):
            gap_factor = 0.6
        elif status in ("ready", "done", "completed"):
            # 已经 ready，仍然允许小幅迭代，但权重较低
            gap_factor = 0.2
        else:
            # 未知状态，给个中等权重
            gap_factor = 0.5

        # 2) improvement_potential：看实验效果
        if exp is None:
            # 没有任何实验记录 => “完全没尝试过”，优先试一试
            improvement_potential = 0.9
            avg_score = None
            success_rate = None
            num_variants = 0
        else:
            avg_score = float(exp.get("avg_score", 0.0))
            success_rate = float(exp.get("success_rate", 0.0))
            num_variants = int(exp.get("num_variants", 0))

            # 越低的 avg_score / success_rate，改进空间越大
            space = 1.0 - avg_score
            # 有一些实验记录说明可学信号较多，给一点加成（对数防炸）
            volume_factor = 1.0 + min(num_variants, 20) / 20.0
            improvement_potential = max(0.1, space * volume_factor)

        # 3) budget_factor：来自 capability 的优先级
        budget = self.budget_controller.get_capability_budget(cap)
        budget_factor = max(0.1, min(budget.priority, 1.0))

        # 综合
        priority_score = gap_factor * improvement_potential * budget_factor

        # 辅助信息 & 人类可读 reason
        signals: Dict[str, Any] = {
            "gap_factor": gap_factor,
            "improvement_potential": improvement_potential,
            "budget_factor": budget_factor,
            "avg_score": avg_score,
            "success_rate": success_rate,
            "num_variants": num_variants,
            "task_status": status,
        }

        reason_parts = [
            f"gap_factor={gap_factor:.2f} (status={status or 'unknown'})",
            f"improvement_potential={improvement_potential:.2f} (avg_score={avg_score}, success_rate={success_rate}, num_variants={num_variants})",
            f"budget_factor={budget_factor:.2f} (cap={cap})",
        ]
        reason = " | ".join(reason_parts)

        return priority_score, signals, reason
这里所有权重都是可解释的 heuristic，完全数据驱动，不再调 LLM：

gap_factor：越“没做完”的模块优先级越高；

improvement_potential：实验越“糟糕 / 未尝试”，越有改进空间；

budget_factor：人为给某些能力更高权重。

def cmd_strategy_next_tasks(args: argparse.Namespace) -> None:
    mgr = EvolutionStrategyManager()
    candidates = mgr.suggest_next_tasks(limit=args.limit)
    if not candidates:
        print("No optimization candidates found.")
        return

    print("Suggested next ImprovementTasks for self-evolution:")
    for c in candidates:
        cap = c.capability
        req = c.requirement_id or "ALL"
        print(
            f"- score={c.priority_score:.3f} :: task={c.task_id} :: cap={cap}/{req}"
        )
        print(f"  reason: {c.reason}")
