@dataclass

      - id: "memory.fabric.tiers"
      - section: "memory"

    name: str

    notes: Optional[str] = None

    def to_dict(self) -> Dict[str, Any]:

@dataclass
class BlueprintSpec:
    """
    蓝皮书整体规格（机器可读版）：

    - version: "v3"
    - codename: "Project Quentin / Research-Integrated Edition"
    - capabilities: 所有能力项列表
    """
    version: str
    codename: str
    capabilities: List[BlueprintCapability]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "version": self.version,
            "codename": self.codename,
            "capabilities": [c.to_dict() for c in self.capabilities],
        }

    def get_capability(self, cap_id: str) -> Optional[BlueprintCapability]:
        for c in self.capabilities:
            if c.id == cap_id:
                return c
        return None

    def list_by_section(self, section: str) -> List[BlueprintCapability]:
        return [c for c in self.capabilities if c.section == section]

from __future__ import annotations

from dataclasses import dataclass, asdict
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List, Optional
import json

from quentin.blueprint.spec import BlueprintSpec, BlueprintCapability


@dataclass
class CapabilityStatus:
    """
    当前系统对某个蓝皮书能力的实现状态。

    status:
      - "unknown": 未评估
      - "missing": 明确没有实现
      - "partial": 部分实现
      - "done": 已实现（满足 v1 要求）
    """
    capability_id: str
    status: str = "unknown"
    evidence: Optional[str] = None
    last_update_ts: str = ""
    notes: Optional[str] = None

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @staticmethod
    def from_dict(data: Dict[str, Any]) -> CapabilityStatus:
        return CapabilityStatus(
            capability_id=data["capability_id"],
            status=data.get("status", "unknown"),
            evidence=data.get("evidence"),
            last_update_ts=data.get("last_update_ts", ""),
            notes=data.get("notes"),
        )


class CapabilitiesRegistry:
    """
    能力状态注册表：cap_id -> CapabilityStatus。

    持久化在 data/blueprint/capabilities.json 里，
    方便人工或 LLM 工具链更新“我们做到哪儿了”。
    """

    def __init__(self, *, repo_root: str | Path = ".") -> None:
        self.repo_root = Path(repo_root).resolve()
        self.storage_path = self.repo_root / "data" / "blueprint" / "capabilities.json"
        self.storage_path.parent.mkdir(parents=True, exist_ok=True)

        self._items: Dict[str, CapabilityStatus] = {}
        self._load()

    def _load(self) -> None:
        if not self.storage_path.exists():
            return
        try:
            with self.storage_path.open("r", encoding="utf-8") as f:
                data = json.load(f)
        except Exception:
            return
        for item in data.get("capabilities", []):
            try:
                status = CapabilityStatus.from_dict(item)
                self._items[status.capability_id] = status
            except Exception:
                continue

    def _save(self) -> None:
        data = {
            "capabilities": [s.to_dict() for s in self._items.values()],
        }
        tmp = self.storage_path.with_suffix(".tmp")
        with tmp.open("w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        tmp.replace(self.storage_path)

    # --------------------------------------------------
    # 对外 API
    # --------------------------------------------------
    def get(self, capability_id: str) -> Optional[CapabilityStatus]:
        return self._items.get(capability_id)

    def set_status(
        self,
        capability_id: str,
        status: str,
        *,
        evidence: Optional[str] = None,
        notes: Optional[str] = None,
    ) -> CapabilityStatus:
        """
        手动/自动更新某个能力的状态。
        """
        now = datetime.utcnow().isoformat()
        cs = self._items.get(capability_id) or CapabilityStatus(
            capability_id=capability_id,
        )
        cs.status = status
        cs.evidence = evidence
        cs.notes = notes
        cs.last_update_ts = now
        self._items[capability_id] = cs
        self._save()
        return cs

    def export(self) -> Dict[str, Any]:
        return {
            "capabilities": [s.to_dict() for s in self._items.values()],
        }


@dataclass
class CapabilityGap:
    """
    单条能力的“差距视图”：蓝皮书定义 + 当前实现状态。
    """
    capability: BlueprintCapability
    status: CapabilityStatus

    @property
    def gap_status(self) -> str:
        """
        用一个统一的 gap 标签：
          - "missing": 未实现但 required=True
          - "partial": 部分实现
          - "ok": 已实现
          - "unknown": 没有状态记录
        """
        s = self.status.status
        if s == "done":
            return "ok"
        if s == "partial":
            return "partial"
        if s == "missing":
            return "missing"
        return "unknown"

    def to_dict(self) -> Dict[str, Any]:
        return {
            "capability": self.capability.to_dict(),
            "status": self.status.to_dict(),
            "gap_status": self.gap_status,
        }


@dataclass
class BlueprintGapReport:
    """
    蓝皮书整体 diff 报告。

    按 gap_status 分类列出所有能力项，并统计数量。
    """
    spec_version: str
    missing: List[CapabilityGap]
    partial: List[CapabilityGap]
    ok: List[CapabilityGap]
    unknown: List[CapabilityGap]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "spec_version": self.spec_version,
            "missing": [g.to_dict() for g in self.missing],
            "partial": [g.to_dict() for g in self.partial],
            "ok": [g.to_dict() for g in self.ok],
            "unknown": [g.to_dict() for g in self.unknown],
            "stats": {
                "missing": len(self.missing),
                "partial": len(self.partial),
                "ok": len(self.ok),
                "unknown": len(self.unknown),
            },
        }


class BlueprintGapTracker:
    """
    蓝皮书差距追踪器：

      - 输入：BlueprintSpec + CapabilitiesRegistry
      - 输出：BlueprintGapReport（哪些已实现、缺失、部分实现）

    未来你可以：
      - 用一个 LLM agent 去扫描代码/文件，自动更新 CapabilitiesRegistry，
        再用这个 tracker 做“总结”。
    """

    def __init__(
        self,
        *,
        spec: BlueprintSpec,
        registry: CapabilitiesRegistry,
    ) -> None:
        self.spec = spec
        self.registry = registry

    def scan(self) -> BlueprintGapReport:
        missing: List[CapabilityGap] = []
        partial: List[CapabilityGap] = []
        ok: List[CapabilityGap] = []
        unknown: List[CapabilityGap] = []

        for cap in self.spec.capabilities:
            cs = self.registry.get(cap.id) or CapabilityStatus(
                capability_id=cap.id,
                status="unknown",
                evidence=None,
                last_update_ts="",
                notes=None,
            )
            gap = CapabilityGap(capability=cap, status=cs)
            gs = gap.gap_status

            if gs == "ok":
                ok.append(gap)
            elif gs == "partial":
                partial.append(gap)
            elif gs == "missing":
                # 只统计 required=True 的 missing，非 required 可以放到 partial/unknown
                if cap.required:
                    missing.append(gap)
                else:
                    partial.append(gap)
            else:
                unknown.append(gap)

        return BlueprintGapReport(
            spec_version=self.spec.version,
            missing=missing,
            partial=partial,
            ok=ok,
            unknown=unknown,
        )

class BlueprintImprover:
    """
    蓝皮书自迭代的“策划器”：

      - 可以只用一个 LLM（llm_client）
      - 也可以用 MultiModelCouncil 做多模型“开会”，然后由会议结果产生计划

    用法：
      improver = BlueprintImprover(llm=..., council=council_or_None)
      plans = improver.propose_improvement_plans(gap_report)
      （然后交给 ExperimentOrchestrator.create_experiment(...)）
    """

    def __init__(
        self,
        *,
        llm: Optional[Any] = None,
        council: Optional[MultiModelCouncil] = None,
        default_domain: str = "blueprint-improvement",
    ) -> None:
        if llm is None and council is None:
            raise ValueError("Either llm or council must be provided.")
        self.llm = llm
        self.council = council
        self.default_domain = default_domain

    # --------------------------------------------------
    # 核心接口：根据 gap 报告生成改进计划
    # --------------------------------------------------
    def propose_improvement_plans(
        self,
        gap_report: BlueprintGapReport,
        *,
        max_gaps: int = 10,
    ) -> List[GapImprovementPlan]:
        """
        针对 gap_report 中的 missing/partial 能力，生成改进实验计划。

        v1 策略：
          - 优先处理 missing，然后是 partial
          - 最多处理 max_gaps 条能力
          - 每个能力调用一次 LLM / MultiModelCouncil，要求输出 JSON
        """
        candidates: List[CapabilityGap] = []
        candidates.extend(gap_report.missing)
        candidates.extend(gap_report.partial)
        candidates = candidates[:max_gaps]

        plans: List[GapImprovementPlan] = []

        for gap in candidates:
            try:
                plan = self._plan_for_single_gap(gap)
            except Exception:
                logger.exception("BlueprintImprover: failed to plan for cap=%s", gap.capability.id)
                continue
            if plan is not None:
                plans.append(plan)

        return plans

    # --------------------------------------------------
    # 内部：单个能力的计划生成
    # --------------------------------------------------
    def _plan_for_single_gap(self, gap: CapabilityGap) -> Optional[GapImprovementPlan]:
        cap = gap.capability
        gs = gap.gap_status

        prompt = self._build_prompt_for_gap(cap, gs)

        if self.council is not None:
            # 用多模型会议结果作为“最终答复”，final_answer 里要求是 JSON
            deliberation = self.council.deliberate(
                question=prompt,
                context={
                    "capability_id": cap.id,
                    "section": cap.section,
                    "gap_status": gs,
                    "blueprint_version": "v3",
                },
            )
            raw = deliberation.final_answer
        else:
            raw = _call_llm(self.llm, prompt)

        try:
            data = json.loads(raw)
        except Exception:
            logger.warning(
                "BlueprintImprover: LLM output for cap=%s is not valid JSON, raw=%r",
                cap.id,
                raw[:300],
            )
            return None

        # 期待的 JSON 结构：
        # {
        #   "priority": 1,
        #   "experiment_name": "...",
        #   "experiment_description": "...",
        #   "tasks": [
        #     {
        #       "goal": "...",
        #       "context": {...}
        #     },
        #     ...
        #   ]
        # }
        priority = int(data.get("priority", 5))
        exp_name = data.get("experiment_name") or f"improve::{cap.id}"
        exp_desc = data.get("experiment_description") or cap.description

        tasks_json = data.get("tasks") or []
        tasks: List[ExperimentTask] = []

        for i, t in enumerate(tasks_json):
            goal = t.get("goal") or f"Improve capability {cap.id} (task {i})"
            context = t.get("context") or {}
            # 确保带上 domain 信息
            if "domain" not in context:
                context["domain"] = self.default_domain
            task_id = t.get("task_id") or f"{cap.id}::t{i}"
            tasks.append(
                ExperimentTask(
                    task_id=task_id,
                    goal=goal,
                    context=context,
                )
            )

        return GapImprovementPlan(
            capability_id=cap.id,
            gap_status=gs,
            priority=priority,
            experiment_name=exp_name,
            experiment_description=exp_desc,
            tasks=tasks,
        )

    # --------------------------------------------------
    # Prompt 设计
    # --------------------------------------------------
    def _build_prompt_for_gap(self, cap: BlueprintCapability, gap_status: str) -> str:
        """
        把单个 capability gap 转成 LLM 可理解的描述，
        要求输出一个 JSON 改进方案。
        """
        cap_json = json.dumps(cap.to_dict(), ensure_ascii=False, indent=2)
        return (
            "你是一名帮助构建通用自进化智能体的系统架构师。\n"
            "我们有一份蓝皮书（Meta-Agent v3），其中定义了很多能力项。\n"
            "现在有一个能力还没有完全实现，请你为它设计一个“研究实验计划”，"
            "帮助系统在 sandbox/staging 环境中逐步实现该能力。\n\n"
            "【能力定义（蓝皮书项）】\n"
            f"{cap_json}\n\n"
            f"【当前 gap 状态】{gap_status}\n\n"
            "请你输出一个 JSON 对象，字段必须包含：\n"
            "  - \"priority\": 数字，1=最高优先级，数字越小越重要。\n"
            "  - \"experiment_name\": 此能力的实验名称（简短英文或中英文混合）。\n"
            "  - \"experiment_description\": 实验要验证或实现的内容简介。\n"
            "  - \"tasks\": 一个数组，每个元素是一个任务对象：\n"
            "      * \"goal\": 自然语言描述该任务要做什么（例如：实现某个模块骨架、补充测试等）。\n"
            "      * \"context\": 一个 JSON 对象（可为空），用于提供额外信息（如建议的文件路径、需要使用的循环/模块名称等）。\n\n"
            "要求：\n"
            "  - 不要写任何解释性文字，只输出一个 JSON 对象。\n"
            "  - 任务要尽量具体、可执行，适合交给一个通用智能体 runtime 去跑。\n"
        )


def _call_llm(llm: Any, prompt: str) -> str:
    """
    统一 LLM 调用接口：

      - llm.complete(prompt=...)
      - llm.generate(prompt=...)
      - 直接可调用 llm(prompt)
    """
    if hasattr(llm, "complete"):
        return llm.complete(prompt)
    if hasattr(llm, "generate"):
        return llm.generate(prompt)
    if callable(llm):
        return llm(prompt)
    raise RuntimeError("LLM client has no complete/generate/callable interface")
现在你就有了一个 “蓝皮书 → gap → LLM 生成实验计划” 的完整链条。
