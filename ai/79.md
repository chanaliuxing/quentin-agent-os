# ---------- Data Models ----------

@dataclass
class GoalSpec:
    """
    Goal & Definition-of-Done 结构化描述：

    - goal_id          : 全局唯一 ID
    - title            : 目标标题（简短）
    - description      : 详细目标描述（自然语言）
    - domain           : 领域（'core' / 'desktop' / 'legal' / 'trading' / ...）
    - environment      : 环境（'sandbox' / 'staging' / 'prod'）
    - success_criteria : 必须满足的成功条件（自然语言列表）
    - hard_constraints : 严格约束（不能违反，比如：不得执行真实交易）
    - soft_preferences : 软偏好（优先考虑但不是硬性条件）
    - created_by       : 'human' / 'agent' / 'system'
    - parent_task_id   : 对应的 task / conversation / run ID
    - metadata         : 额外信息（比如 UI 上下文、用户 ID 等）
    """

    goal_id: str
    title: str
    description: str
    domain: str
    environment: str
    success_criteria: List[str] = field(default_factory=list)
    hard_constraints: List[str] = field(default_factory=list)
    soft_preferences: List[str] = field(default_factory=list)
    created_by: str = "agent"
    parent_task_id: Optional[str] = None
    metadata: Dict[str, Any] = field(default_factory=dict)
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def new(
        cls,
        *,
        title: str,
        description: str,
        domain: str = "core",
        environment: str = "sandbox",
        success_criteria: Optional[List[str]] = None,
        hard_constraints: Optional[List[str]] = None,
        soft_preferences: Optional[List[str]] = None,
        created_by: str = "agent",
        parent_task_id: Optional[str] = None,
        metadata: Optional[Dict[str, Any]] = None,
    ) -> "GoalSpec":
        return cls(
            goal_id=str(uuid.uuid4()),
            title=title,
            description=description,
            domain=domain,
            environment=environment,
            success_criteria=success_criteria or [],
            hard_constraints=hard_constraints or [],
            soft_preferences=soft_preferences or [],
            created_by=created_by,
            parent_task_id=parent_task_id,
            metadata=metadata or {},
        )


@dataclass
class DoneCheckContext:
    """
    用于 Done 检查的上下文：

    - task_id         : 本次执行的 task / run ID
    - plan_summary    : Planner 的计划摘要（自然语言）
    - actions_summary : 执行过程摘要（动作列表/Tool 调用概要）
    - final_output    : 给用户/上层的最终输出（自然语言）
    - metrics         : 数值指标，例如：success flag、error_count、latency 等
    - logs_excerpt    : 关键日志片段（可选）
    """

    task_id: str
    plan_summary: str
    actions_summary: str
    final_output: str
    metrics: Dict[str, Any] = field(default_factory=dict)
    logs_excerpt: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class DoneCheckResult:
    """
    Done Council 的决策结果：

    - is_done                : 是否认为任务已经完成
    - confidence             : 0~1 信心
    - reasons                : 自然语言理由列表
    - violated_constraints   : 被认为违反/未满足的约束
    - improvement_suggestions: 若未完成或可改进，给出的建议
    - raw_answer             : AI 原始 JSON 文本（方便 debug）
    """

    is_done: bool
    confidence: float
    reasons: List[str] = field(default_factory=list)
    violated_constraints: List[str] = field(default_factory=list)
    improvement_suggestions: List[str] = field(default_factory=list)
    raw_answer: str = ""

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


# ---------- GoalStore：持久化目标 ----------

class GoalStore:
    """
    v1 的简单持久化：把 GoalSpec 以 JSONL 存在 data/goals.jsonl 中。

    未来可以替换成 SQLite / Postgres，而不影响模块对外接口。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        goals_path: str | Path = "data/goals.jsonl",
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.goals_path = self.repo_root / goals_path
        self.goals_path.parent.mkdir(parents=True, exist_ok=True)

    def save(self, goal: GoalSpec) -> None:
        rec = goal.to_dict()
        try:
            with self.goals_path.open("a", encoding="utf-8") as f:
                f.write(json.dumps(rec, ensure_ascii=False) + "\n")
        except Exception:
            logger.exception("GoalStore.save: failed to append goal to %s", self.goals_path)

    def load_all(self, limit: Optional[int] = None) -> List[GoalSpec]:
        if not self.goals_path.exists():
            return []
        res: List[GoalSpec] = []
        try:
            with self.goals_path.open("r", encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if not line:
                        continue
                    try:
                        obj = json.loads(line)
                        res.append(GoalSpec(**obj))
                        if limit is not None and len(res) >= limit:
                            break
                    except Exception:
                        logger.exception("GoalStore.load_all: parse error in %s", self.goals_path)
        except Exception:
            logger.exception("GoalStore.load_all: failed to read %s", self.goals_path)
        return res


# ---------- DoneCouncil：多 AI 会议判定 Done ----------

class DoneCouncil:
    """
    多 AI 会议模式的 Done 判定器：

    - 使用 MultiAICouncil（Speed/Safety/Cost/DomainExpert）做 deliberation
    - 统一 JSON 输出协议，便于上层自动处理和做研究

    JSON 协议：

    {
      "is_done": true/false,
      "confidence": 0.0~1.0,
      "reasons": ["..."],
      "violated_constraints": ["..."],
      "improvement_suggestions": ["..."]
    }
    """

    def __init__(
        self,
        *,
        router: Optional[ModelRouter] = None,
        council: Optional[MultiAICouncil] = None,
        domain: str = "core",
    ) -> None:
        self.router = router or ModelRouter()
        if council is None:
            council = make_decision_council(router=self.router, domain=domain, mode="done_check")
        self.council = council
        self.domain = domain

    def check_done(
        self,
        *,
        goal: GoalSpec,
        context: DoneCheckContext,
    ) -> DoneCheckResult:
        goal_json = json.dumps(goal.to_dict(), ensure_ascii=False, indent=2)
        ctx_json = json.dumps(context.to_dict(), ensure_ascii=False, indent=2)

        question = (
            "You are a MULTI-AGENT COUNCIL that decides whether a task has reached its goal.\n"
            "You must read the GOAL (with success criteria and constraints) and the EXECUTION CONTEXT\n"
            "(plan, actions, final output, metrics) and decide if the task is DONE.\n\n"
            "Return ONLY ONE JSON object with the following structure:\n"
            "{\n"
            "  \"is_done\": true/false,\n"
            "  \"confidence\": 0.0-1.0,\n"
            "  \"reasons\": [\"...\"],\n"
            "  \"violated_constraints\": [\"...\"],\n"
            "  \"improvement_suggestions\": [\"...\"]\n"
            "}\n"
            "Be strict about constraints: if any HARD constraint is clearly violated, is_done MUST be false.\n"
        )

        context_block = (
            "=== GOAL SPEC ===\n"
            f"{goal_json}\n\n"
            "=== EXECUTION CONTEXT ===\n"
            f"{ctx_json}\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context_block,
            mode="done_check",
        )
        ans = decision.final_answer or "{}"

        try:
            s = ans.strip()
            first = s.find("{")
            last = s.rfind("}")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)
            is_done = bool(data.get("is_done", False))
            confidence = float(data.get("confidence", 0.0))
            reasons = data.get("reasons") or []
            violated = data.get("violated_constraints") or []
            suggestions = data.get("improvement_suggestions") or []

            # 兼容意外类型
            if not isinstance(reasons, list):
                reasons = [str(reasons)]
            if not isinstance(violated, list):
                violated = [str(violated)]
            if not isinstance(suggestions, list):
                suggestions = [str(suggestions)]
        except Exception:
            logger.exception("DoneCouncil.check_done: failed to parse council JSON: %r", ans)
            is_done = False
            confidence = 0.0
            reasons = ["Failed to parse council JSON; default to NOT done."]
            violated = []
            suggestions = []

        return DoneCheckResult(
            is_done=is_done,
            confidence=max(0.0, min(1.0, confidence)),
            reasons=[str(r) for r in reasons],
            violated_constraints=[str(v) for v in violated],
            improvement_suggestions=[str(s) for s in suggestions],
            raw_answer=ans,
        )


# ---------- GoalDoneManager：对外统一入口 ----------

class GoalDoneManager:
    """
    统一的 Goal & Done 管理：

    - 创建/保存 GoalSpec
    - 调用 DoneCouncil 评估是否完成
    - 将结果写入：
        - ResearchManager（capability: goal_done_definition）
        - ExecutionTraceStore（capability_id: goal_done_definition）

    让 “任务是否完成” 也变成一种可观测、可评估、可实验的能力。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        goal_store: Optional[GoalStore] = None,
        router: Optional[ModelRouter] = None,
        traces_store: Optional[ExecutionTraceStore] = None,
        research_manager: Optional[ResearchManager] = None,
        domain: str = "core",
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.goal_store = goal_store or GoalStore(repo_root=self.repo_root)
        self.router = router or ModelRouter()
        self.traces_store = traces_store or ExecutionTraceStore()
        self.research_manager = research_manager or ResearchManager()
        self.domain = domain
        self.council = DoneCouncil(router=self.router, domain=domain)

    # ---- Goal 管理 ----

    def create_and_save_goal(
        self,
        *,
        title: str,
        description: str,
        domain: str = "core",
        environment: str = "sandbox",
        success_criteria: Optional[List[str]] = None,
        hard_constraints: Optional[List[str]] = None,
        soft_preferences: Optional[List[str]] = None,
        created_by: str = "agent",
        parent_task_id: Optional[str] = None,
        metadata: Optional[Dict[str, Any]] = None,
    ) -> GoalSpec:
        goal = GoalSpec.new(
            title=title,
            description=description,
            domain=domain,
            environment=environment,
            success_criteria=success_criteria,
            hard_constraints=hard_constraints,
            soft_preferences=soft_preferences,
            created_by=created_by,
            parent_task_id=parent_task_id,
            metadata=metadata,
        )
        self.goal_store.save(goal)
        return goal

    # ---- Done 检查 + 记录 ----

    def check_done_and_record(
        self,
        *,
        goal: GoalSpec,
        context: DoneCheckContext,
    ) -> DoneCheckResult:
        result = self.council.check_done(goal=goal, context=context)

        # 写 Research experiment
        capability = "goal_done_definition"
        requirement_id = "done_accuracy"

        exps = self.research_manager.list_experiments()
        exp = None
        for e in exps:
            if e.capability == capability and e.requirement_id == requirement_id:
                exp = e
                break
        if exp is None:
            exp = self.research_manager.create_experiment(
                name="GoalDoneDefinition",
                purpose="check_done",
                capability=capability,
                requirement_id=requirement_id,
                tags=["goal", "done", "council"],
            )

        metrics = {
            # 目前只能记录 AI 自己的 confidence，
            # 以后可接入“人类是否同意”的标注作为 label。
            "confidence": result.confidence,
            "is_done_flag": 1.0 if result.is_done else 0.0,
        }

        self.research_manager.record_run(
            experiment_id=exp.experiment_id,
            variant_id="council_v1",
            metrics=metrics,
            context={
                "goal": goal.to_dict(),
                "done_context": context.to_dict(),
                "result": result.to_dict(),
            },
        )

        # 写 ExecutionTrace（供 BlueprintGapTracker 使用）
        trace = ExecutionTrace(
            trace_id=str(uuid.uuid4()),
            task_id=context.task_id,
            component_id="goals.done_definition.v1",
            blueprint_capability_id="goal_done_definition",
            target_module="quentin/goals/done_definition.py",
            target_symbol="GoalDoneManager.check_done_and_record",
            success=True,
            error_type=None,
            error_message=None,
            start_time=datetime.utcnow().isoformat(),
            end_time=datetime.utcnow().isoformat(),
            metrics=metrics,
            tags={
                "domain": goal.domain,
                "environment": goal.environment,
                "goal_id": goal.goal_id,
            },
        )
        self.traces_store.append(trace)

        return result
