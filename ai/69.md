# quentin/kernel/safety.py
from __future__ import annotations

from dataclasses import dataclass
from pathlib import Path
from typing import Tuple, Literal


SelfProgDecision = Literal["allow", "warn", "deny"]


@dataclass
class SelfProgPolicyResult:
    decision: SelfProgDecision  # "allow" | "warn" | "deny"
    reason: str


class SafetyKernel:
    """
    v1 SafetyKernel：集中管理几类关键决策：

    1. 某个文件能不能被自编程引擎修改（self-programming）
    2. 某种实验/patch 是否可以在当前环境自动执行（后续扩展）

    政策要求对齐蓝皮书：
      - kernel/ 及安全关键模块：只读，不允许自编程
      - 建议把可自动修改模块放在 mutable/ 下
    """

    def __init__(self, repo_root: str | Path = ".") -> None:
        self.repo_root = Path(repo_root).resolve()

    # ---------- 自编程相关 ----------

    def check_self_programming_target(self, target_path: str) -> SelfProgPolicyResult:
        """
        对目标路径做安全判断：

        - kernel/ 或 quentin/kernel 下：deny
        - mutable/ 下：allow
        - 其他：warn（允许但提醒最好迁到 mutable/）
        """
        rel = target_path.replace("\\", "/")

        # 绝对路径转相对
        if rel.startswith(str(self.repo_root)):
            rel = str(Path(rel).relative_to(self.repo_root)).replace("\\", "/")

        # 禁止修改 kernel
        if rel.startswith("quentin/kernel") or rel.startswith("kernel/"):
            return SelfProgPolicyResult(
                decision="deny",
                reason="target_path is under kernel/, which is immutable for self-programming.",
            )

        # 推荐 mutable 目录用于自动可变代码
        if rel.startswith("mutable/"):
            return SelfProgPolicyResult(
                decision="allow",
                reason="target_path is under mutable/, allowed for self-programming.",
            )

        # 其他：允许，但给 warning
        return SelfProgPolicyResult(
            decision="warn",
            reason=(
                "target_path is outside kernel, self-programming is allowed "
                "but it's recommended to keep auto-updatable code under mutable/."
            ),
        )
后续你想加更多策略（比如 Trading / Legal 目录限制）就在这里扩展，其他模块都只调用 SafetyKernel，不各自乱写规则。

# quentin/blueprint/gap_tracker.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from importlib import import_module
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging

from quentin.research.experiments import ResearchManager, Experiment

logger = logging.getLogger(__name__)


@dataclass
class CapabilityStatus:
    """
    蓝皮书中的一个 capability 在当前 repo 的落实情况。

    - id            : 蓝皮书 capability id
    - requirement_id: 蓝皮书里的 requirement_id（如 memory_fabric_v1）
    - module        : 期望的 python module 路径
    - symbol        : 期望存在的类/函数名
    - status        : "implemented" / "partial" / "missing" / "unknown"
    - detail        : 文本说明
    """
    id: str
    requirement_id: str
    module: Optional[str]
    symbol: Optional[str]
    status: str
    detail: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class BlueprintGapReport:
    """
    一次 scan 的总报告。
    """
    blueprint_path: str
    total_capabilities: int
    implemented: int
    partial: int
    missing: int
    unknown: int
    capabilities: List[CapabilityStatus]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "blueprint_path": self.blueprint_path,
            "total_capabilities": self.total_capabilities,
            "implemented": self.implemented,
            "partial": self.partial,
            "missing": self.missing,
            "unknown": self.unknown,
            "capabilities": [c.to_dict() for c in self.capabilities],
        }


class BlueprintGapTracker:
    """
    BlueprintGapTracker v1：

    - 从蓝皮书 spec JSON 中读取 capabilities 列表
    - 对每一项尝试 import module + getattr(symbol)
    - 得出 implemented/partial/missing/unknown
    - 把覆盖率写入 ResearchManager（方便你后面看“蓝皮书落地率”）

    只做“存在性检测”，不评估质量；质量由 EvalHarness / LLM 负责。
    """

    def __init__(
        self,
        *,
        blueprint_path: str | Path = "blueprints/meta_agent_v3.json",
        research_manager: Optional[ResearchManager] = None,
    ) -> None:
        self.blueprint_path = str(blueprint_path)
        self.research_manager = research_manager or ResearchManager()

    # ---------- 主入口 ----------

    def scan(self) -> BlueprintGapReport:
        spec = self._load_blueprint()
        caps_raw = spec.get("capabilities", []) or []
        statuses: List[CapabilityStatus] = []

        for item in caps_raw:
            cid = str(item.get("id", "<unknown>"))
            requirement_id = str(item.get("requirement_id", "unknown"))
            module = item.get("module")
            symbol = item.get("symbol")

            status, detail = self._check_capability(module, symbol)

            statuses.append(
                CapabilityStatus(
                    id=cid,
                    requirement_id=requirement_id,
                    module=module,
                    symbol=symbol,
                    status=status,
                    detail=detail,
                )
            )

        implemented = sum(1 for c in statuses if c.status == "implemented")
        partial = sum(1 for c in statuses if c.status == "partial")
        missing = sum(1 for c in statuses if c.status == "missing")
        unknown = sum(1 for c in statuses if c.status == "unknown")

        report = BlueprintGapReport(
            blueprint_path=self.blueprint_path,
            total_capabilities=len(statuses),
            implemented=implemented,
            partial=partial,
            missing=missing,
            unknown=unknown,
            capabilities=statuses,
        )

        self._record_research(report)
        return report

    # ---------- 内部：加载蓝皮书 JSON ----------

    def _load_blueprint(self) -> Dict[str, Any]:
        p = Path(self.blueprint_path)
        if not p.exists():
            logger.warning("BlueprintGapTracker: blueprint file not found: %s", p)
            return {"capabilities": []}
        try:
            return json.loads(p.read_text(encoding="utf-8") or "{}")
        except Exception:
            logger.exception("BlueprintGapTracker: failed to load blueprint JSON.")
            return {"capabilities": []}

    # ---------- 内部：检测单个 capability ----------

    def _check_capability(
        self,
        module_name: Optional[str],
        symbol_name: Optional[str],
    ) -> tuple[str, str]:
        """
        返回 (status, detail)
        """
        if not module_name:
            return "unknown", "No module specified in blueprint; cannot check."

        # 先看 module 能否 import
        try:
            mod = import_module(module_name)
        except Exception:
            return "missing", f"Module '{module_name}' cannot be imported."

        # 如果没有定义 symbol，就认为 module 存在但无法精确判断
        if not symbol_name:
            return "partial", f"Module '{module_name}' imported, but no symbol specified in blueprint."

        if not hasattr(mod, symbol_name):
            return "partial", f"Module '{module_name}' imported, but symbol '{symbol_name}' not found."

        return "implemented", f"Module '{module_name}' and symbol '{symbol_name}' both exist."

    # ---------- 内部：写入 ResearchLayer ----------

    def _record_research(self, report: BlueprintGapReport) -> None:
        rm = self.research_manager

        capability = "blueprint:coverage"
        requirement_id = "blueprint_implementation_rate"

        exps = rm.list_experiments()
        exp: Optional[Experiment] = None
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "blueprint_gap_scan"
            ):
                exp = e
                break

        if exp is None:
            exp = rm.create_experiment(
                name="BlueprintGapScan",
                purpose="blueprint_gap_scan",
                capability=capability,
                requirement_id=requirement_id,
                tags=["blueprint", "coverage"],
            )

        if report.total_capabilities > 0:
            coverage = report.implemented / report.total_capabilities
        else:
            coverage = 0.0

        metrics = {
            "total_capabilities": report.total_capabilities,
            "implemented": report.implemented,
            "partial": report.partial,
            "missing": report.missing,
            "unknown": report.unknown,
            "coverage": coverage,
        }
        context = {
            "blueprint_path": report.blueprint_path,
        }

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics=metrics,
            context=context,
        )

def cmd_blueprint_scan(args: argparse.Namespace) -> None:
    tracker = BlueprintGapTracker(blueprint_path=args.blueprint_path)
    report = tracker.scan()
    data = report.to_dict()

    if args.json:
        import json as _json
        print(_json.dumps(data, ensure_ascii=False, indent=2))
        return

    print(f"Blueprint: {data['blueprint_path']}")
    print(
        f"Capabilities: total={data['total_capabilities']} | "
        f"implemented={data['implemented']} | "
        f"partial={data['partial']} | "
        f"missing={data['missing']} | "
        f"unknown={data['unknown']}"
    )
    print("\nDetails:")
    for c in data["capabilities"]:
        print(
            f"- [{c['status']}] {c['id']} "
            f"(module={c['module']}, symbol={c['symbol']}) "
            f"=> {c['detail']}"
        )
