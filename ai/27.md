# ============================================

from __future__ import annotations

from typing import Dict, Any, List, Optional
from datetime import datetime
from pydantic import BaseModel, Field


class ExecutionStep(BaseModel):
    """
    单个执行步骤：
    - kind: "domain_task" / "tool_call" / "model_call" / ...
    - name: 例如 "generic.handle_task" / "generic.echo"
    """
    step_id: str
    kind: str
    name: str

    input: Dict[str, Any] = Field(default_factory=dict)
    output: Optional[Dict[str, Any]] = None
    error: Optional[str] = None

    started_at: datetime = Field(default_factory=datetime.utcnow)
    finished_at: Optional[datetime] = None

    success: bool = True

    meta: Dict[str, Any] = Field(default_factory=dict)


class ExecutionTrace(BaseModel):
    """
    一次任务执行的完整 Trace（多步）。
    """
    trace_id: str
    task_id: str
    domain: str
    goal: str
    task_input: Dict[str, Any] = Field(default_factory=dict)

    started_at: datetime = Field(default_factory=datetime.utcnow)
    finished_at: Optional[datetime] = None

    steps: List[ExecutionStep] = Field(default_factory=list)

    success: bool = True
    error: Optional[str] = None

    metrics: Dict[str, Any] = Field(default_factory=dict)
    meta: Dict[str, Any] = Field(default_factory=dict)


class ExecutionSummary(BaseModel):
    """
    Trace 的压缩版本，给记忆 / 评测 / 运维看。
    """
    summary_id: str
    trace_id: str
    task_id: str
    domain: str
    goal: str

    success: bool
    error: Optional[str] = None

    steps_count: int
    started_at: datetime
    finished_at: datetime

    # 粗粒度指标（后续可以补 token / cost / reward 等）
    metrics: Dict[str, Any] = Field(default_factory=dict)

    # 与 lessons_learned 在 MemoryFabric 中的链接
    lesson_memory_ids: List[str] = Field(default_factory=list)

    meta: Dict[str, Any] = Field(default_factory=dict)


class CriticReview(BaseModel):
    """
    Critic 对一次 Trace 的总结。
    - summary: 人类可读的一句话总结
    - lessons: 若干条可写入记忆的经验教训
    """
    summary: str
    lessons: List[str] = Field(default_factory=list)
    meta: Dict[str, Any] = Field(default_factory=dict)
二、Critic：review Trace → 给出 lessons（占位版）
新建文件：quentin/cognitive/critic.py

当前版不依赖 LLM，先用简单规则生成内容；以后你在这里接 ModelRouter 即可。

# ============================================
# quentin/cognitive/critic.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict

from quentin.infra.logging import Tracer
from quentin.runtime.model_router import ModelRouter
from quentin.cognitive.schemas import ExecutionTrace, CriticReview


@dataclass
class Critic:
    """
    Critic v1（占位版）：
    - 当前实现：不用 LLM，基于 Trace 成功/失败简单生成几条 lesson
    - 后续可以在 review_with_llm 中接入 ModelRouter，生成更复杂的反思
    """
    tracer: Tracer
    model_router: ModelRouter

    async def review(self, trace: ExecutionTrace) -> CriticReview:
        """
        核心接口：给一次 ExecutionTrace 做整体评价。
        """
        if trace.success:
            summary = "Task completed successfully."
            lessons = [
                "This approach worked. Consider extracting it as a reusable skill if it repeats.",
            ]
        else:
            summary = "Task failed or encountered errors."
            lessons = [
                "Investigate the failing step and log more context for debugging.",
                "Consider adding fallback strategies or retries for the critical step.",
            ]

        # 这里可以以后扩展：
        # review_llm = await self.review_with_llm(trace)
        # 合并两者结果

        self.tracer.span(
            "L1",
            "critic_review",
            module="critic",
            input={"trace_id": trace.trace_id},
            output={"success": trace.success},
        )

        return CriticReview(summary=summary, lessons=lessons)

    async def review_with_llm(self, trace: ExecutionTrace) -> CriticReview:
        """
        预留接口：真正用 LLM 做复杂反思的地方。
        当前先返回空结果，避免强依赖。
        """
        # 将来在这里调用 self.model_router 生成更细致的分析。
        return CriticReview(summary="(LLM critic not enabled yet.)", lessons=[])
这样 CognitiveLoop 可以先用一个 完全可离线运行 的 Critic，未来再换成 LLM 驱动的版本，不动上层结构。

三、CognitiveLoop：执行 DomainTask → 写 Trace & Summary → 写记忆
新建文件：quentin/cognitive/loop.py

# ============================================
# quentin/cognitive/loop.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass
from typing import Dict, Any, Optional
from datetime import datetime
import uuid

from quentin.infra.logging import Tracer
from quentin.infra.storage.kv_store import KVStore
from quentin.domains.base import DomainTask, DomainRegistry
from quentin.tools.manager import ToolManager
from quentin.memory.fabric import MemoryFabric
from quentin.memory.schemas import MemoryItem, MemoryTier
from quentin.cognitive.schemas import (
    ExecutionStep,
    ExecutionTrace,
    ExecutionSummary,
)
from quentin.cognitive.critic import Critic


@dataclass
class CognitiveLoop:
    """
    Cognitive Loop v1：

    - 接收一个 DomainTask
    - 调用 DomainRegistry 执行
    - 记录 ExecutionTrace
    - 调用 Critic 得到 lessons_learned
    - 把 lessons 写入 MemoryFabric（STS）
    - 持久化 Trace & Summary 到 KV，给 Sleep / Immune / Research 用
    """
    tracer: Tracer
    kv: KVStore
    memory: MemoryFabric
    tool_manager: ToolManager
    domain_registry: DomainRegistry
    critic: Critic

    trace_prefix: str = "exec:trace:"
    summary_prefix: str = "exec:summary:"
    summary_index_key: str = "exec:index:summary"
    error_index_key: str = "exec:index:error"

    # ---------- KV 工具 ----------

    async def _save_trace(self, trace: ExecutionTrace) -> None:
        key = f"{self.trace_prefix}{trace.trace_id}"
        await self.kv.save(key, trace.model_dump())

    async def _save_summary(self, summary: ExecutionSummary) -> None:
        key = f"{self.summary_prefix}{summary.trace_id}"
        await self.kv.save(key, summary.model_dump())
        await self.kv.append_list(self.summary_index_key, summary.trace_id)
        if not summary.success:
            await self.kv.append_list(self.error_index_key, summary.trace_id)

    # ---------- 公开主入口：运行任务 ----------

    async def run_task(self, task: DomainTask) -> ExecutionSummary:
        """
        执行一次任务，生成 Trace + Summary + lessons_learned。
        """
        now = datetime.utcnow()
        trace_id = f"trace_{uuid.uuid4().hex}"
        task_id = f"task_{uuid.uuid4().hex}"

        trace = ExecutionTrace(
            trace_id=trace_id,
            task_id=task_id,
            domain=task.domain,
            goal=task.goal,
            task_input=task.input,
            started_at=now,
            success=True,
        )

        # Step 1: 调 domain adapter 执行任务
        step = ExecutionStep(
            step_id=f"step_{uuid.uuid4().hex}",
            kind="domain_task",
            name=f"{task.domain}.handle_task",
            input={"goal": task.goal, "input": task.input},
            started_at=datetime.utcnow(),
        )

        try:
            result = await self.domain_registry.route_task(task)
            step.output = result
            step.success = True
        except Exception as e:
            step.error = repr(e)
            step.success = False
            trace.success = False
            trace.error = repr(e)

        step.finished_at = datetime.utcnow()
        trace.steps.append(step)
        trace.finished_at = datetime.utcnow()

        # 简单 metrics
        total_latency = (trace.finished_at - trace.started_at).total_seconds()
        trace.metrics["latency_sec"] = total_latency
        trace.metrics["steps"] = len(trace.steps)

        # Step 2: 调 Critic 得到 lessons_learned
        critic_review = await self.critic.review(trace)

        # Step 3: 把 lessons 写入 MemoryFabric（STS）
        lesson_memory_ids = []
        for lesson_text in critic_review.lessons:
            mem = MemoryItem(
                memory_id=f"lesson_{uuid.uuid4().hex}",
                kind="lesson",
                content=lesson_text,
                metadata={
                    "source": "cognitive_loop",
                    "trace_id": trace.trace_id,
                    "task_id": trace.task_id,
                    "domain": trace.domain,
                    "goal": trace.goal,
                },
                memory_tier=MemoryTier.STS,
            )
            await self.memory.add_memory(mem)
            lesson_memory_ids.append(mem.memory_id)

        # Step 4: 生成 ExecutionSummary，保存到 KV
        summary = ExecutionSummary(
            summary_id=f"summary_{uuid.uuid4().hex}",
            trace_id=trace.trace_id,
            task_id=trace.task_id,
            domain=trace.domain,
            goal=trace.goal,
            success=trace.success,
            error=trace.error,
            steps_count=len(trace.steps),
            started_at=trace.started_at,
            finished_at=trace.finished_at or trace.started_at,
            metrics=trace.metrics,
            lesson_memory_ids=lesson_memory_ids,
            meta={"critic_summary": critic_review.summary},
        )

        await self._save_trace(trace)
        await self._save_summary(summary)

        # Trace 一定要有
        self.tracer.span(
            "L1",
            "cognitive_task_run",
            module="cognitive_loop",
            input={
                "trace_id": trace.trace_id,
                "task_id": trace.task_id,
                "domain": trace.domain,
                "goal": trace.goal,
            },
            output={
                "success": trace.success,
                "latency_sec": total_latency,
                "lessons": len(lesson_memory_ids),
            },
        )

        return summary

@cli.command("demo-cognitive-task")
@click.option("--domain", "domain_name", default="generic", show_default=True)
@click.option("--goal", required=True, help="Task goal.")
@click.option("--input", "input_json", default="{}", help="JSON string as task.input payload.")
def demo_cognitive_task_cmd(domain_name: str, goal: str, input_json: str):
    """
    演示：用 CognitiveLoop 跑一条任务（默认 generic 域），
    自动生成 Trace / Summary / lessons_learned。
    """
    async def _run():
        import json
        settings = Settings()
        runtime = await _init_runtime_with_domains(settings)

        try:
            payload = json.loads(input_json)
        except Exception:
            payload = {}

        task = DomainTask(domain=domain_name, goal=goal, input=payload)
        summary = await runtime.cognitive_loop.run_task(task)

        print("ExecutionSummary:")
        print(json.dumps(summary.model_dump(), ensure_ascii=False, indent=2))

    asyncio.run(_run())
