# quentin/evolution/skill_loop.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, List, Optional
import logging
import uuid
import textwrap

from quentin.self_programming.engine import (
    SelfProgrammingEngine,
    PatchCandidate,
    PatchTestReport,
)
from quentin.orchestrator.core import LanguageModel

logger = logging.getLogger(__name__)


@dataclass
class SkillCandidate:
    """
    从成功 Trace 中抽取出来的“技能候选”。

    v1 先只是结构占位，后面可以真的把长流程提炼成独立函数/模块。
    """
    skill_id: str
    name: str
    description: str
    source_trace_ids: List[str]
    code_suggestion: Optional[str] = None
    meta: Dict[str, Any] = None

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        return d


class SkillEvolutionLoop:
    """
    技能与代码进化循环（Skill & Code Evolution Loop）v1 实现。

    职责：
    - 从成功 Trace 中挖潜在技能（目前只是占位）
    - 从 issues 列表（包括 ImmuneCase / BlueprintGap 等）生成 PatchCandidate
    - 调用 SelfProgrammingEngine 跑测试，返回 PatchTestReport
    """

    def __init__(
        self,
        engine: SelfProgrammingEngine,
        *,
        llm: Optional[LanguageModel] = None,
        repo_root: str | Path = ".",
    ) -> None:
        self.engine = engine
        self.llm = llm
        self.repo_root = Path(repo_root).resolve()

    # --------------------------------------------------------------
    # 对 SleepLoop / Runtime 暴露的总入口
    # --------------------------------------------------------------
    def run_once(
        self,
        *,
        successful_traces: List[Dict[str, Any]],
        issues: List[Dict[str, Any]],
        test_suites: Optional[List[str]] = None,
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        一次完整的进化轮次。

        输入：
        - successful_traces: SleepLoop 准备好的“成功 Trace 摘要”
        - issues: ImmuneLoop / Blueprint 等准备好的 issue 列表
        - test_suites: 要跑哪些测试（unit/static/…）
        - extra_context: 额外上下文（比如来源：sleep_cycle / research_cycle）

        输出：
        - skill_candidates
        - patch_candidates
        - patch_reports
        - promotable_patches（由 SelfProgrammingEngine.select_patches_to_promote 决定）
        """
        ctx = extra_context or {}

        # 1) 从成功 trace 中挖技能候选（v1 只是结构占位）
        skill_candidates = self.generate_skills_from_traces(successful_traces)

        # 2) 从 issues 生成 PatchCandidate
        patch_candidates = self.generate_patches_from_issues(issues)

        # 3) 跑测试
        suites = test_suites or ["unit", "static_analysis"]
        patch_reports = self.evaluate_patches(
            patch_candidates,
            test_suites=suites,
            extra_context=ctx,
        )

        # 4) 选可推广 patch
        promotable = self.engine.select_patches_to_promote(
            patch_reports,
            max_promote=1,
        )

        return {
            "skill_candidates": [s.to_dict() for s in skill_candidates],
            "patch_candidates": [p.to_dict() for p in patch_candidates],
            "patch_reports": [r.to_dict() for r in patch_reports],
            "promotable_patches": [r.to_dict() for r in promotable],
        }

    # --------------------------------------------------------------
    # 1. 从成功 Trace 中挖技能（目前只是 stub）
    # --------------------------------------------------------------
    def generate_skills_from_traces(
        self,
        successful_traces: List[Dict[str, Any]],
    ) -> List[SkillCandidate]:
        """
        v1：只做占位，后面再用 LLM 对 Trace 做 pattern mining。

        现在先简单：每个成功 trace 生成一个很粗糙的 SkillCandidate。
        """
        candidates: List[SkillCandidate] = []
        for t in successful_traces:
            trace_id = t.get("trace_id")
            task_id = t.get("task_id", "unknown_task")
            domain = t.get("domain", "general")

            skill = SkillCandidate(
                skill_id=f"skill_from_trace::{trace_id}",
                name=f"auto_skill_for_{task_id}",
                description=f"Auto-extracted skill placeholder from trace {trace_id} in domain {domain}.",
                source_trace_ids=[trace_id],
                code_suggestion=None,
                meta={"status": "placeholder"},
            )
            candidates.append(skill)

        return candidates

    # --------------------------------------------------------------
    # 2. 从 issues 生成 PatchCandidate —— 重点：接入 LLM
    # --------------------------------------------------------------
    def generate_patches_from_issues(
        self,
        issues: List[Dict[str, Any]],
    ) -> List[PatchCandidate]:
        """
        核心：从 issue 列表生成 PatchCandidate。

        逻辑（v1）：
        - 对每个 issue：
            - 找到 target.file_path
            - 读当前文件内容（不存在则视为新文件）
            - 如果有 llm：
                - 用 llm 生成该文件的完整新版本（不是 diff，而是“重写后的完整文件内容”）
            - 否则：
                - 生成一个带 TODO 的占位 patch_text
        """
        candidates: List[PatchCandidate] = []
        for issue in issues:
            try:
                cand = self._build_patch_candidate_for_issue(issue)
                if cand is not None:
                    candidates.append(cand)
            except Exception:
                logger.exception("SkillEvolutionLoop: failed to build patch for issue=%r", issue)

        logger.info("SkillEvolutionLoop.generate_patches_from_issues: %d candidates", len(candidates))
        return candidates

    def _build_patch_candidate_for_issue(
        self,
        issue: Dict[str, Any],
    ) -> Optional[PatchCandidate]:
        target = issue.get("target") or {}
        file_path = target.get("file_path")
        if not file_path:
            logger.warning("SkillEvolutionLoop: issue has no file_path: %r", issue)
            return None

        abs_path = self.repo_root / file_path
        try:
            current_code = abs_path.read_text(encoding="utf-8") if abs_path.exists() else ""
        except Exception:
            logger.exception("SkillEvolutionLoop: failed to read %s", abs_path)
            current_code = ""

        reason = issue.get("reason", "")
        issue_id = issue.get("issue_id", f"issue::{uuid.uuid4().hex}")
        severity = (issue.get("metrics") or {}).get("severity", "medium")

        if self.llm is None:
            # 没有 LLM 时，生成一个占位 PatchCandidate，方便后续人工或别的脚本补齐
            patch_text = self._build_stub_patch_text(file_path, current_code, reason)
        else:
            patch_text = self._build_patch_with_llm(
                file_path=file_path,
                current_code=current_code,
                issue=issue,
            )

        patch_id = f"patch::{issue_id}::{uuid.uuid4().hex}"

        candidate = PatchCandidate(
            patch_id=patch_id,
            target=target,
            description=f"Auto-generated patch for issue: {reason}",
            patch_text=patch_text,
            tests=[],
            meta={
                "source_issue": issue_id,
                "severity": severity,
                "file_path": file_path,
            },
        )
        return candidate

    def _build_stub_patch_text(self, file_path: str, current_code: str, reason: str) -> str:
        """
        在没有 LLM 的情况下，用一个带 TODO 的 patch 占位。
        """
        banner = textwrap.dedent(
            f"""
            # NOTE: This file is auto-generated as a placeholder by SkillEvolutionLoop.
            # Reason: {reason}
            # File: {file_path}
            #
            # TODO: Implement according to the Meta-Agent v3 blueprint.
            """
        ).strip("\n")

        if not current_code.strip():
            return banner + "\n\n" + "pass\n"
        else:
            return banner + "\n\n" + current_code

    def _build_patch_with_llm(
        self,
        *,
        file_path: str,
        current_code: str,
        issue: Dict[str, Any],
    ) -> str:
        """
        用 LLM 生成新的文件内容。

        设计选择：
        - 不生成 diff，而是生成“该文件的完整新版本”。
        - SelfProgrammingEngine 在应用 patch 时，可以把 patch_text 直接写回文件
          （或者写到临时目录跑测试）。
        """
        reason = issue.get("reason", "")
        issue_id = issue.get("issue_id", "unknown_issue")

        system_prompt = (
            "You are a senior Python engineer working inside a self-evolving AI agent repository.\n"
            "Your job is to modify ONE file to address a specific issue from a blueprint or immune case.\n"
            "You must output ONLY the full new content of that file, as valid Python code.\n"
            "Do NOT include explanations, comments about what you changed, or diff markers.\n"
            "Just return the final code for the file."
        )

        user_prompt = textwrap.dedent(
            f"""
            Repository context:
              - This repo implements a Meta-Agent v3 blueprint (research-integrated, self-programming system).
              - You are editing the file: {file_path}

            Issue to fix (from blueprint/immune system):
            - Issue ID: {issue_id}
            - Reason: {reason}
            - Target meta: {issue.get("target")}

            Current file content (if empty, the file does not exist yet):
            ---------------- CURRENT FILE START ----------------
            {current_code}
            ---------------- CURRENT FILE END ----------------

            Requirements:
            1. Keep the overall architecture consistent with a research-friendly, modular design.
            2. Use clear, production-quality Python.
            3. If the file is missing, create a minimal but functional skeleton that satisfies the issue.
            4. Do NOT rely on external frameworks beyond standard Python and the existing quentin.* modules.
            5. Output ONLY the full new file content.
            """
        ).strip("\n")

        messages = [
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_prompt},
        ]

        try:
            code = self.llm.generate(messages, temperature=0.2)
        except Exception:
            logger.exception("SkillEvolutionLoop: LLM patch generation failed for %s", file_path)
            return self._build_stub_patch_text(file_path, current_code, reason)

        return code

    # --------------------------------------------------------------
    # 3. 调用 SelfProgrammingEngine 跑测试
    # --------------------------------------------------------------
    def evaluate_patches(
        self,
        patch_candidates: List[PatchCandidate],
        *,
        test_suites: List[str],
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> List[PatchTestReport]:
        """
        对每个 PatchCandidate 运行测试，返回 PatchTestReport 列表。

        self.engine.run_tests_for_patch 目前可能还是 stub，
        但整体调用链已经打通。
        """
        ctx = extra_context or {}
        reports: List[PatchTestReport] = []

        for cand in patch_candidates:
            try:
                report = self.engine.run_tests_for_patch(
                    candidate=cand,
                    test_suites=test_suites,
                    extra_context=ctx,
                )
                reports.append(report)
            except Exception:
                logger.exception(
                    "SkillEvolutionLoop.evaluate_patches: tests failed for patch_id=%s",
                    getattr(cand, "patch_id", "<unknown>"),
                )

        return reports

# quentin/blueprint/improver.py
from __future__ import annotations

from typing import Any, Dict, List, Optional
from pathlib import Path
import logging

from quentin.blueprint.gap_tracker import BlueprintGapTracker, BlueprintGapReport, BlueprintGap
from quentin.evolution.skill_loop import SkillEvolutionLoop
from quentin.self_programming.engine import SelfProgrammingEngine
from quentin.orchestrator.core import LanguageModel

logger = logging.getLogger(__name__)


class BlueprintImprover:
    """
    根据 BlueprintGapTracker 的报告：
    - 把 gap 转成 issues 列表
    - 让 SkillEvolutionLoop / SelfProgrammingEngine 尝试生成 patch
    - 跑测试 + 选出 promotable patches

    v1：流程打通，patch 生成基于可选的 LLM。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        tracker: Optional[BlueprintGapTracker] = None,
        engine: Optional[SelfProgrammingEngine] = None,
        skill_evolution: Optional[SkillEvolutionLoop] = None,
        llm: Optional[LanguageModel] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.tracker = tracker or BlueprintGapTracker(repo_root=self.repo_root)
        self.engine = engine or SelfProgrammingEngine(repo_root=self.repo_root)

        # 如果调用方已经传了一个带 llm 的 SkillEvolutionLoop，就用它；
        # 否则在这里新建一个（可以把 llm 传进来）。
        self.skill_evolution = skill_evolution or SkillEvolutionLoop(
            self.engine,
            llm=llm,
            repo_root=self.repo_root,
        )

    # ---- 对外主入口 ---- #
    def run_once(
        self,
        *,
        max_gaps: int = 20,
        test_suites: Optional[List[str]] = None,
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        一次 Blueprint 改进循环：
        （略，同之前）
        """
        report = self.tracker.scan()
        gaps = report.gaps[:max_gaps]

        logger.info("BlueprintImprover.run_once: gaps=%d (total=%d)", len(gaps), len(report.gaps))

        issues = self._gaps_to_issues(gaps)

        patch_candidates = self.skill_evolution.generate_patches_from_issues(issues)

        suites = test_suites or ["unit", "static_analysis"]
        patch_reports = self.skill_evolution.evaluate_patches(
            patch_candidates,
            test_suites=suites,
            extra_context=extra_context or {"source": "blueprint_improver"},
        )

        promotable = self.engine.select_patches_to_promote(
            patch_reports,
            max_promote=1,
        )

        return {
            "blueprint_report": report.to_dict(),
            "issues": issues,
            "patch_candidates": [p.to_dict() for p in patch_candidates],
            "patch_reports": [r.to_dict() for r in patch_reports],
            "promotable_patches": [r.to_dict() for r in promotable],
        }

    # 下面 _gaps_to_issues 保持不变……
    def _gaps_to_issues(self, gaps: List[BlueprintGap]) -> List[Dict[str, Any]]:
        issues: List[Dict[str, Any]] = []
        for gap in gaps:
            severity = "medium"
            if gap.issue_type == "missing_file":
                severity = "high"
            elif gap.issue_type == "stub_implementation":
                severity = "medium"

            target = {
                "target_type": "python_module",
                "identifier": gap.path,
                "file_path": gap.path,
                "symbol_name": None,
                "meta": {
                    "component_id": gap.component_id,
                    "issue_type": gap.issue_type,
                },
            }

            issue = {
                "issue_id": f"blueprint_gap::{gap.component_id}",
                "source": "blueprint_gap",
                "target": target,
                "reason": gap.details,
                "metrics": {
                    "severity": severity,
                },
            }
            issues.append(issue)

        return issues
三、一键脚本：tools/run_self_evolution_cycle.py
这个脚本做一件事：在本地跑一轮自迭代循环：

用 Qwen + MultiModelCouncil 搭一个最简 runtime + orchestrator；

跑几条测试任务（让系统产生 trace & memory）；

跑 SleepLoop（记忆 + skill/patch 演化入口）；

跑 ImmuneLoop（异常聚合）；

跑 BlueprintImprover（蓝皮书 gap → patch candidates → 测试 → promotable patches）。

注意：SelfProgrammingEngine.run_tests_for_patch 现在大概率还是 stub，所以“测试”不会真正跑，但链路已经通了。

文件：tools/run_self_evolution_cycle.py

# tools/run_self_evolution_cycle.py
from __future__ import annotations

import argparse
import logging
from pathlib import Path

from quentin.llm.ollama_client import OllamaLLM
from quentin.orchestrator.core import SimpleOrchestrator
from quentin.orchestrator.council import MultiModelCouncil
from quentin.runtime.system import MetaAgentRuntime
from quentin.blueprint.improver import BlueprintImprover


def setup_logging(level: str = "INFO") -> None:
    logging.basicConfig(
        level=getattr(logging, level.upper(), logging.INFO),
        format="%(asctime)s [%(levelname)s] %(name)s: %(message)s",
    )


def build_runtime(repo_root: Path) -> MetaAgentRuntime:
    """
    使用本地 Ollama 模型构建一个最简 MetaAgentRuntime。
    - 用 qwen3-vl:latest 作为主模型
    - 其他模型作为 council 成员
    """
    # 主模型（规划 / 反思 / aggregator）
    qwen_main = OllamaLLM(model="qwen3-vl:latest")

    # 其他成员模型（可以按需精简/扩展）
    deepseek_small = OllamaLLM(model="deepseek-r1:1.5b")
    qwen_vision = OllamaLLM(model="qwen2.5vl:latest")
    uncensored = OllamaLLM(model="my-deepseek-uncensored:latest")

    members = [
        ("qwen_main", qwen_main),
        ("deepseek_small", deepseek_small),
        ("qwen_vision", qwen_vision),
        ("uncensored", uncensored),
    ]

    council = MultiModelCouncil(
        members=members,
        aggregator=qwen_main,
    )

    orchestrator = SimpleOrchestrator(
        llm=qwen_main,
        council=council,
    )

    runtime = MetaAgentRuntime(
        repo_root=str(repo_root),
        orchestrator=orchestrator,
        council=council,
    )
    return runtime


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Run one self-evolution cycle for the Meta-Agent v3 blueprint repo.",
    )
    parser.add_argument(
        "--repo-root",
        type=str,
        default=str(Path(__file__).resolve().parents[1]),
        help="Path to the repository root (default: project root).",
    )
    parser.add_argument(
        "--log-level",
        type=str,
        default="INFO",
        help="Logging level (DEBUG, INFO, WARNING, ERROR).",
    )
    parser.add_argument(
        "--max-gaps",
        type=int,
        default=10,
        help="Max number of blueprint gaps to process in this run.",
    )
    parser.add_argument(
        "--no-demo-tasks",
        action="store_true",
        help="If set, skip running demo tasks (only run evolution on existing traces).",
    )

    args = parser.parse_args()
    setup_logging(args.log_level)

    repo_root = Path(args.repo_root).resolve()
    logging.getLogger(__name__).info("Repo root: %s", repo_root)

    # 1) 构建 Runtime（含 orchestrator / council / loops / engine 等）
    runtime = build_runtime(repo_root)

    # 2) 可选：先跑几条 demo 任务，产生一些 trace & memory
    if not args.no_demo_tasks:
        demo_goals = [
            "设计一份针对 Meta-Agent v3 蓝皮书的单元测试清单。",
            "为自编程引擎构思一个更严格的测试策略。",
            "总结当前系统的三大自我改进方向。",
        ]
        for goal in demo_goals:
            result = runtime.run_task(goal, {"domain": "general"})
            logging.getLogger(__name__).info(
                "Demo task done: %s | summary keys=%s",
                result.task_id,
                list(result.trace.execution_summary.keys()),
            )

    # 3) 跑 SleepLoop（记忆巩固 + skill/patch 演化入口）
    sleep_result = runtime.run_sleep_cycle()
    logging.getLogger(__name__).info("Sleep cycle result: %s", sleep_result)

    # 4) 跑 ImmuneLoop（异常聚合）
    immune_result = runtime.run_immune_cycle()
    logging.getLogger(__name__).info("Immune cycle result: %s", immune_result)

    # 5) 跑 BlueprintImprover（蓝皮书 gap -> patch candidates -> 测试）
    #    这里我们尽量复用 runtime 的 engine / skill_evolution_loop，
    #    并把同一个 LLM 传进去。
    from quentin.llm.ollama_client import OllamaLLM  # 复用 qwen_main 作 code LLM

    code_llm = OllamaLLM(model="qwen3-vl:latest")

    improver = BlueprintImprover(
        repo_root=repo_root,
        engine=runtime.self_programming_engine,
        skill_evolution=runtime.skill_evolution_loop,
        llm=code_llm,
    )

    bp_result = improver.run_once(max_gaps=args.max_gaps)
    logging.getLogger(__name__).info(
        "Blueprint improver summary: %s",
        bp_result["blueprint_report"]["summary"],
    )
    logging.getLogger(__name__).info(
        "Promotable patches: %d",
        len(bp_result["promotable_patches"]),
    )

    print("\n=== Self-Evolution Cycle Finished ===")
    print("Blueprint gaps:", bp_result["blueprint_report"]["summary"])
    print("Promotable patches:", len(bp_result["promotable_patches"]))


if __name__ == "__main__":
    main()
