一、极简 Orchestrator 骨架
文件：quentin/orchestrator/core.py

这个 Orchestrator 做三件事：

用 单 LLM 或 multi-LLM council 做「任务规划」

执行一个 stub 的“执行步骤”（以后你可以换成真实 executor）

可选再来一次「反思/总结」，也可以走 council

最终返回 (output, ExecutionTrace)，满足 MetaAgentRuntime 的调用约定。

# quentin/orchestrator/core.py
from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Optional, Tuple
from datetime import datetime
import logging
import uuid

from quentin.trace.spec import ExecutionTrace, StepTrace

logger = logging.getLogger(__name__)


# ---- 简单的 LLM 接口定义（你之后可以接 Ollama / OpenAI / DeepSeek 等） ---- #

class LanguageModel:
    """
    通用 LLM 接口抽象。

    你可以实现一个 OllamaLLM(LanguageModel) 通过 http://localhost:11434/api 调用，
    也可以做一个 OpenAICompatibleLLM 等。
    """

    def generate(self, messages: List[Dict[str, str]], **kwargs: Any) -> str:
        raise NotImplementedError


# ---- Orchestrator ---- #

@dataclass
class OrchestratorConfig:
    """
    控制 orchestrator 内部行为的简单配置。
    """
    enable_reflection: bool = True


class SimpleOrchestrator:
    """
    极简 orchestrator 骨架。

    目标：
    - 不依赖任何具体领域（完全通用）
    - 能走完一条“plan → execute → reflect”的 Cognitive Loop
    - 能在 plan / reflect 阶段选择使用单 LLM 或多 AI council

    约定：
    - run_task(request, context) 必须返回 (output, ExecutionTrace)
    """

    def __init__(
        self,
        *,
        llm: LanguageModel,
        config: Optional[OrchestratorConfig] = None,
        council: Optional[Any] = None,  # 你的 Multi-LLM Council 实现，如果有的话
    ) -> None:
        self.llm = llm
        self.config = config or OrchestratorConfig()
        self.council = council

    # ==================================================================
    # 公共入口
    # ==================================================================
    def run_task(
        self,
        *,
        request: Any,
        context: Optional[Dict[str, Any]] = None,
    ) -> Tuple[Any, ExecutionTrace]:
        """
        request: 预期是 MetaAgentRuntime.TaskRequest，但不强依赖类型，
                 只要求有 task_id / goal / domain / priority / context 属性。

        context:
            - runtime_config: 来自 MetaAgentRuntime 的整体配置（含 council_policy）
            - council: 如有的话，Multi-Model Council 实例
        """
        ctx = dict(context or {})
        runtime_cfg = ctx.get("runtime_config") or {}
        council_policy = (runtime_cfg.get("council_policy") or {})
        council = ctx.get("council") or self.council

        trace = ExecutionTrace(
            trace_id=uuid.uuid4().hex,
            task_id=request.task_id,
            domain=request.domain,
            status="unknown",
            steps=[],
            execution_summary={},
            lessons_learned={},
            top_level_metrics={},
            meta={
                "priority": request.priority,
                "created_at": datetime.utcnow().isoformat(),
            },
        )

        # 1) 规划阶段
        plan_text, plan_step = self._plan_with_optional_council(
            request=request,
            council=council,
            council_policy=council_policy,
        )
        trace.steps.append(plan_step)

        # 2) 执行阶段（v1 stub：只回显 plan）
        exec_result, exec_step = self._execute_stub(request, plan_text)
        trace.steps.append(exec_step)

        # 3) 反思阶段（可选，也可以用 council）
        reflection_text = None
        if self.config.enable_reflection:
            reflection_text, reflect_step = self._reflect_with_optional_council(
                request=request,
                plan_text=plan_text,
                exec_result=exec_result,
                council=council,
                council_policy=council_policy,
            )
            trace.steps.append(reflect_step)

        # 4) 汇总
        trace.status = "success"
        trace.execution_summary = {
            "goal": request.goal,
            "plan": plan_text,
            "exec_result": exec_result,
            "reflection": reflection_text,
        }
        trace.lessons_learned = {
            "notes": reflection_text or "",
        }
        trace.top_level_metrics = {
            "steps": len(trace.steps),
        }

        output = {
            "goal": request.goal,
            "plan": plan_text,
            "result": exec_result,
            "reflection": reflection_text,
        }

        return output, trace

    # ==================================================================
    # 规划阶段：可选多 AI 开会
    # ==================================================================
    def _plan_with_optional_council(
        self,
        *,
        request: Any,
        council: Optional[Any],
        council_policy: Dict[str, Any],
    ) -> Tuple[str, StepTrace]:
        """
        根据 council_policy 决定：
        - 是否由 多 AI Council 做规划
        - 还是走单 LLM

        约定 council.deliberate(...) 返回 (final_answer, debug_info)。
        """
        started_at = datetime.utcnow().isoformat()
        use_council = bool(council and council_policy.get("use_for_planning", True))

        if use_council:
            try:
                final_answer, debug = council.deliberate(
                    topic="task_planning",
                    question=request.goal,
                    context=request.context or {},
                )
                plan_text = final_answer
                details = {
                    "mode": "council",
                    "debug": debug,
                }
            except Exception:
                logger.exception("SimpleOrchestrator: council planning failed, fallback to single LLM")
                plan_text = self._single_llm_plan(request)
                details = {"mode": "single_llm_fallback"}
        else:
            plan_text = self._single_llm_plan(request)
            details = {"mode": "single_llm"}

        finished_at = datetime.utcnow().isoformat()
        step = StepTrace(
            step_id=uuid.uuid4().hex,
            role="planner",
            action="plan_task",
            input={
                "goal": request.goal,
                "context": request.context or {},
            },
            output={
                "plan": plan_text,
                "details": details,
            },
            metrics={},
            error=None,
            started_at=started_at,
            finished_at=finished_at,
        )
        return plan_text, step

    def _single_llm_plan(self, request: Any) -> str:
        """
        使用一个 LLM 做简单规划。
        """
        messages = [
            {
                "role": "system",
                "content": (
                    "You are a helpful planner inside a meta-agent system. "
                    "Given a high-level user goal, produce a concise step-by-step plan "
                    "that an automation agent can follow. Always be explicit and numbered."
                ),
            },
            {
                "role": "user",
                "content": f"Goal: {request.goal}\nContext: {request.context or {}}",
            },
        ]
        try:
            return self.llm.generate(messages)
        except Exception:
            logger.exception("SimpleOrchestrator: single LLM planning failed")
            return "1. Understand the goal.\n2. Think about possible steps.\n3. Execute them one by one."

    # ==================================================================
    # 执行阶段（目前是 stub）
    # ==================================================================
    def _execute_stub(self, request: Any, plan_text: str) -> Tuple[Dict[str, Any], StepTrace]:
        """
        v1 stub：不真正调用外部工具，只把 plan 回显为“计划结果”。

        将来你可以在这里：
        - 解析 plan 为子任务
        - 调用具体工具 / desktop automation / legalbot / trading adapter 等。
        """
        started_at = datetime.utcnow().isoformat()

        result = {
            "status": "stub_execution",
            "note": "Execution is not implemented yet. This is a placeholder.",
            "echo_plan": plan_text,
        }

        finished_at = datetime.utcnow().isoformat()
        step = StepTrace(
            step_id=uuid.uuid4().hex,
            role="executor",
            action="execute_plan_stub",
            input={
                "goal": request.goal,
                "plan": plan_text,
            },
            output=result,
            metrics={},
            error=None,
            started_at=started_at,
            finished_at=finished_at,
        )
        return result, step

    # ==================================================================
    # 反思阶段：可选多 AI 合议
    # ==================================================================
    def _reflect_with_optional_council(
        self,
        *,
        request: Any,
        plan_text: str,
        exec_result: Dict[str, Any],
        council: Optional[Any],
        council_policy: Dict[str, Any],
    ) -> Tuple[str, StepTrace]:
        """
        对本次任务做“自我反思 / 总结”。

        同样支持：
        - 多 AI Council
        - 单 LLM
        """
        started_at = datetime.utcnow().isoformat()
        use_council = bool(council and council_policy.get("use_for_reflection", True))

        if use_council:
            try:
                final_answer, debug = council.deliberate(
                    topic="task_reflection",
                    question="Please critique and improve the plan and execution.",
                    context={
                        "goal": request.goal,
                        "plan": plan_text,
                        "result": exec_result,
                    },
                )
                reflection = final_answer
                details = {
                    "mode": "council",
                    "debug": debug,
                }
            except Exception:
                logger.exception("SimpleOrchestrator: council reflection failed, fallback to single LLM")
                reflection = self._single_llm_reflection(request, plan_text, exec_result)
                details = {"mode": "single_llm_fallback"}
        else:
            reflection = self._single_llm_reflection(request, plan_text, exec_result)
            details = {"mode": "single_llm"}

        finished_at = datetime.utcnow().isoformat()
        step = StepTrace(
            step_id=uuid.uuid4().hex,
            role="critic",
            action="reflect_on_task",
            input={
                "goal": request.goal,
                "plan": plan_text,
                "result": exec_result,
            },
            output={
                "reflection": reflection,
                "details": details,
            },
            metrics={},
            error=None,
            started_at=started_at,
            finished_at=finished_at,
        )
        return reflection, step

    def _single_llm_reflection(
        self,
        request: Any,
        plan_text: str,
        exec_result: Dict[str, Any],
    ) -> str:
        messages = [
            {
                "role": "system",
                "content": (
                    "You are a critic and reflection helper inside a meta-agent system. "
                    "Given the goal, plan, and (possibly stub) execution result, "
                    "produce a short reflection: what was good, what can be improved, "
                    "and what we should remember as a lesson."
                ),
            },
            {
                "role": "user",
                "content": (
                    f"Goal: {request.goal}\n\n"
                    f"Plan:\n{plan_text}\n\n"
                    f"Result:\n{exec_result}"
                ),
            },
        ]
        try:
            return self.llm.generate(messages)
        except Exception:
            logger.exception("SimpleOrchestrator: single LLM reflection failed")

# quentin/blueprint/gap_tracker.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging

logger = logging.getLogger(__name__)

try:
    import yaml  # type: ignore
except ImportError:
    yaml = None


# ---------- 数据结构 ---------- #

@dataclass
class ComponentSpec:
    """
    蓝皮书中对单个组件的描述（精简版）。

    真实蓝皮书可以更复杂，这里先满足：
    - id: 逻辑组件 ID
    - path: 相对 repo_root 的路径
    - kind: "python_module" / "config" / "data" / ...
    - required: 是否必须存在
    - min_status: "skeleton" | "full"（期望的最小完成度）
    """
    id: str
    path: str
    kind: str = "python_module"
    required: bool = True
    min_status: str = "skeleton"
    meta: Dict[str, Any] = None

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        return d


@dataclass
class BlueprintSpec:
    version: str
    components: List[ComponentSpec]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "version": self.version,
            "components": [c.to_dict() for c in self.components],
        }


@dataclass
class BlueprintGap:
    """
    蓝皮书与实际 repo 之间发现的一个 gap。
    """
    component_id: str
    path: str
    issue_type: str        # "missing_file" / "stub_implementation" / "unknown"
    details: str

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class BlueprintGapReport:
    """
    一次 scan() 的完整报告。
    """
    spec_version: str
    gaps: List[BlueprintGap]

    def to_dict(self) -> Dict[str, Any]:
        return {
            "spec_version": self.spec_version,
            "gaps": [g.to_dict() for g in self.gaps],
            "summary": {
                "gap_count": len(self.gaps),
                "missing_files": len([g for g in self.gaps if g.issue_type == "missing_file"]),
                "stub_implementations": len([g for g in self.gaps if g.issue_type == "stub_implementation"]),
            },
        }


# ---------- GapTracker 实现 ---------- #

class BlueprintGapTracker:
    """
    根据机器可读蓝皮书 spec 扫描整个 repo 的落地情况。

    v1 关注三件事：
    1. spec 中 required=true 的组件是否有对应文件存在；
    2. python_module 是否只是“占位”（大量 TODO / pass，函数/类极少）；
    3. 输出一个结构化报告（方便后续喂给 ResearchLayer / 自编程引擎）。

    将来可以扩展：
    - 检查测试是否存在
    - 检查文档是否存在
    - 检查代码中是否有关键类/函数名
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        spec_path: Optional[str | Path] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.spec_path = self._resolve_spec_path(spec_path)

    # ---- Public API ---- #
    def scan(self) -> BlueprintGapReport:
        """
        扫描并返回 gap 报告。

        如果 spec 不存在，返回空报告，但不会抛异常。
        """
        spec = self._load_spec()
        if spec is None:
            logger.warning("BlueprintGapTracker: spec not found; returning empty report.")
            return BlueprintGapReport(spec_version="unknown", gaps=[])

        gaps: List[BlueprintGap] = []

        for comp in spec.components:
            comp_gaps = self._scan_component(comp)
            gaps.extend(comp_gaps)

        report = BlueprintGapReport(spec_version=spec.version, gaps=gaps)
        return report

    # ---- 内部：加载 spec ---- #
    def _resolve_spec_path(self, spec_path: Optional[str | Path]) -> Optional[Path]:
        """
        优先顺序：
        1. 调用者显式指定的路径
        2. repo_root/spec/blueprint_v3.json
        3. repo_root/spec/blueprint_v3.yaml 或 yml
        """
        if spec_path is not None:
            return Path(spec_path).resolve()

        candidates = [
            self.repo_root / "spec" / "blueprint_v3.json",
            self.repo_root / "spec" / "blueprint_v3.yaml",
            self.repo_root / "spec" / "blueprint_v3.yml",
        ]
        for p in candidates:
            if p.exists():
                return p

        return None

    def _load_spec(self) -> Optional[BlueprintSpec]:
        if not self.spec_path or not self.spec_path.exists():
            return None

        try:
            text = self.spec_path.read_text(encoding="utf-8")
        except Exception:
            logger.exception("BlueprintGapTracker: failed to read spec file %s", self.spec_path)
            return None

        raw: Dict[str, Any]
        try:
            if self.spec_path.suffix.lower() == ".json":
                raw = json.loads(text)
            else:
                if yaml is None:
                    logger.error("BlueprintGapTracker: yaml not available but spec is YAML")
                    return None
                raw = yaml.safe_load(text)  # type: ignore
        except Exception:
            logger.exception("BlueprintGapTracker: failed to parse spec file %s", self.spec_path)
            return None

        version = str(raw.get("version", "unknown"))
        components_raw = raw.get("components") or []

        components: List[ComponentSpec] = []
        for c in components_raw:
            try:
                comp = ComponentSpec(
                    id=c["id"],
                    path=c["path"],
                    kind=c.get("kind", "python_module"),
                    required=bool(c.get("required", True)),
                    min_status=c.get("min_status", "skeleton"),
                    meta=c.get("meta") or {},
                )
                components.append(comp)
            except Exception:
                logger.exception("BlueprintGapTracker: invalid component spec: %r", c)

        return BlueprintSpec(version=version, components=components)

    # ---- 内部：按组件扫描 ---- #
    def _scan_component(self, comp: ComponentSpec) -> List[BlueprintGap]:
        gaps: List[BlueprintGap] = []
        full_path = self.repo_root / comp.path

        if comp.required and not full_path.exists():
            gaps.append(
                BlueprintGap(
                    component_id=comp.id,
                    path=comp.path,
                    issue_type="missing_file",
                    details="File does not exist but component is marked as required.",
                )
            )
            return gaps

        if not full_path.exists():
            # 非 required 且不存在就直接略过
            return gaps

        # 目前只对 python_module 做简单占位检测
        if comp.kind == "python_module" and full_path.suffix == ".py":
            if self._looks_like_stub_python(full_path, comp):
                gaps.append(
                    BlueprintGap(
                        component_id=comp.id,
                        path=comp.path,
                        issue_type="stub_implementation",
                        details="Python module looks like a stub (mostly TODO/pass, very few defs).",
                    )
                )

        # 将来这里可以增加更多 kind 的检查，如 config/data/schema/test 等
        return gaps

    def _looks_like_stub_python(self, path: Path, comp: ComponentSpec) -> bool:
        """
        粗略判断一个 python 文件是否“还停留在占位阶段”。

        简单启发式：
        - 行数很少，且包含 pass 或 TODO
        - 或者：
          - def/class 数量 <= 1
          - 且包含大量 TODO/pseudo 注释
        """
        try:
            text = path.read_text(encoding="utf-8")
        except Exception:
            logger.exception("BlueprintGapTracker: failed to read %s", path)
            return False

        lines = [ln.rstrip() for ln in text.splitlines()]
        non_empty = [ln for ln in lines if ln.strip()]
        line_count = len(non_empty)

        # 很短而且有明显占位标记
        if line_count <= 10:
            if any("TODO" in ln or "pass" == ln.strip() for ln in non_empty):
                return True

        # 统计 def / class 数量
        def_count = sum(1 for ln in non_empty if ln.lstrip().startswith("def "))
        class_count = sum(1 for ln in non_empty if ln.lstrip().startswith("class "))

        todo_like = sum(
            1
            for ln in non_empty
            if "TODO" in ln or "FIXME" in ln or "pass" == ln.strip()
        )

        # def/class 很少，同时 TODO 很多
        if (def_count + class_count) <= 1 and todo_like >= 2:
            return True

        # 如果 spec 要求 min_status=full，可以在这里更严格
        if comp.min_status == "full":
            # 这里先简单一点：如果 def+class<2 就判定为未完全实现
            if (def_count + class_count) < 2:
                return True

        return False
