## 一、Batch 批量治理核心：`quentin/governance/batch.py`

```python
# quentin/governance/batch.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, List, Optional

import json
from datetime import datetime

from quentin.governance.patch_council import PatchGovernanceCouncil, PatchDecision


@dataclass
class BatchPatchGovernResult:
    """
    一次批量治理的总结果。
    """
    total_reports: int
    processed: int
    skipped: int
    decisions: List[PatchDecision]
    started_at: str
    finished_at: str

    def to_dict(self) -> Dict[str, Any]:
        return {
            "total_reports": self.total_reports,
            "processed": self.processed,
            "skipped": self.skipped,
            "decisions": [d.to_dict() for d in self.decisions],
            "started_at": self.started_at,
            "finished_at": self.finished_at,
        }


class BatchPatchGovernor:
    """
    扫描 data/patches/reports 下所有 PatchReport，
    对尚未决策的 proposal 执行治理决策（可选 auto_apply）。
    """

    def __init__(
        self,
        *,
        repo_root: str | Path,
        patch_council: PatchGovernanceCouncil,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.patch_council = patch_council

        self._reports_dir = self.repo_root / "data" / "patches" / "reports"
        self._decisions_dir = self.repo_root / "data" / "governance" / "patch_decisions"
        self._batch_runs_dir = self.repo_root / "data" / "governance" / "batch_runs"
        self._batch_runs_dir.mkdir(parents=True, exist_ok=True)

    # --------------------------------------------------
    # 对外：批量治理
    # --------------------------------------------------
    def auto_govern_all(
        self,
        *,
        metrics: Dict[str, Any],
        immune_snapshot: Optional[Dict[str, Any]] = None,
        auto_apply: bool = False,
        only_recommended: Optional[List[str]] = None,
        max_patches: Optional[int] = None,
    ) -> BatchPatchGovernResult:
        """
        参数：
          - metrics / immune_snapshot: 来自 MetricsAggregator + ImmuneLoop
          - auto_apply: 是否允许对通过检查的 patch 自动在 sandbox apply
          - only_recommended: 只治理哪些 PatchReport 的 recommended_action
                例: ["accept"] 则只治理建议接受的 patch
          - max_patches: 最多处理多少个 patch

        返回：
          - BatchPatchGovernResult
        """
        start_ts = datetime.utcnow().isoformat()

        reports = self._list_patch_reports()
        total_reports = len(reports)

        decisions: List[PatchDecision] = []
        processed = 0
        skipped = 0

        for report_path in reports:
            proposal_id = report_path.stem  # 默认报告名是 <proposal_id>.json

            # 1) 已有决策则跳过
            if self._has_existing_decision(proposal_id):
                skipped += 1
                continue

            # 2) 如果指定 only_recommended，则检查 PatchReport 里的推荐动作，不匹配就跳过
            if only_recommended is not None:
                try:
                    with report_path.open("r", encoding="utf-8") as f:
                        rp = json.load(f)
                    rec = rp.get("recommended_action")
                    if rec not in only_recommended:
                        skipped += 1
                        continue
                except Exception:
                    # 报告坏了/解析失败，安全起见也跳过
                    skipped += 1
                    continue

            # 3) 调用理事会治理 + 可选自动 apply
            decision = self.patch_council.govern_patch(
                proposal_id=proposal_id,
                metrics=metrics,
                immune_snapshot=immune_snapshot,
                auto_apply=auto_apply,
            )
            decisions.append(decision)
            processed += 1

            if max_patches is not None and processed >= max_patches:
                break

        finished_ts = datetime.utcnow().isoformat()
        result = BatchPatchGovernResult(
            total_reports=total_reports,
            processed=processed,
            skipped=skipped,
            decisions=decisions,
            started_at=start_ts,
            finished_at=finished_ts,
        )

        # 落盘一次 batch run 记录
        self._persist_batch_result(result)
        return result

    # --------------------------------------------------
    # 内部：列出所有 PatchReport
    # --------------------------------------------------
    def _list_patch_reports(self) -> List[Path]:
        if not self._reports_dir.exists():
            return []
        return sorted(self._reports_dir.glob("*.json"))

    # --------------------------------------------------
    # 内部：判断某 proposal 是否已有治理决策
    # --------------------------------------------------
    def _has_existing_decision(self, proposal_id: str) -> bool:
        if not self._decisions_dir.exists():
            return False
        # 规则：任何以 <proposal_id>_ 开头的决策文件都视为已有决策
        matches = list(self._decisions_dir.glob(f"{proposal_id}_*.json"))
        return len(matches) > 0

    # --------------------------------------------------
    # 内部：保存本次 batch run 结果
    # --------------------------------------------------
    def _persist_batch_result(self, result: BatchPatchGovernResult) -> None:
        ts = datetime.utcnow().strftime("%Y%m%dT%H%M%S")
        path = self._batch_runs_dir / f"batch_govern_{ts}.json"
        tmp = path.with_suffix(".tmp")

        with tmp.open("w", encoding="utf-8") as f:
            json.dump(result.to_dict(), f, ensure_ascii=False, indent=2)

        tmp.replace(path)
```

## 二、命令行脚本：`scripts/auto_govern_all_patches.py`

> 一条命令：扫描所有 patch reports → 用 AI 理事会 + 安全内核批量决策 → 可选自动在 sandbox apply。

```python
# scripts/auto_govern_all_patches.py
"""
批量治理所有已有 PatchReport，并可选自动在 sandbox 环境应用补丁。

用法（在 repo 根目录）：

  # 只对推荐 action = "accept" 的 patch 做治理，不自动应用
  python -m scripts.auto_govern_all_patches --only-recommended accept

  # 对所有 patch 做治理（不筛选 recommended_action），也不自动应用
  python -m scripts.auto_govern_all_patches

  # 对推荐 accept 的 patch 做治理，并在安全内核允许的前提下自动应用
  python -m scripts.auto_govern_all_patches --only-recommended accept --auto-apply
"""

from __future__ import annotations

import argparse
from pathlib import Path
from typing import List, Optional

from quentin.runtime.system import MetaAgentRuntime
from scripts.blueprint_self_iterate import create_runtime_for_blueprint_self_iteration
from quentin.traces.core import JsonlTraceRecorder
from quentin.metrics.aggregator import MetricsAggregator
from quentin.governance.patch_council import PatchGovernanceCouncil
from quentin.governance.batch import BatchPatchGovernor


def _parse_only_recommended(raw: Optional[str]) -> Optional[List[str]]:
    if not raw:
        return None
    # 支持逗号分隔： "accept,reject"
    items = [x.strip() for x in raw.split(",") if x.strip()]
    return items or None


def main() -> None:
    parser = argparse.ArgumentParser(description="Batch govern all patch reports with AI council and safety kernel.")
    parser.add_argument("--repo-root", type=str, default=".", help="Repository root (default: current dir).")
    parser.add_argument(
        "--only-recommended",
        type=str,
        default=None,
        help="Only process patch reports whose 'recommended_action' in this list "
             "(comma separated, e.g. 'accept,needs_review'). Default: no filter.",
    )
    parser.add_argument(
        "--auto-apply",
        action="store_true",
        help="If set, allow governance council + safety kernel to automatically apply patches in sandbox.",
    )
    parser.add_argument(
        "--max-patches",
        type=int,
        default=None,
        help="Max number of patches to process in this batch.",
    )
    parser.add_argument(
        "--max-events",
        type=int,
        default=2000,
        help="Max events to scan from trace when computing recent metrics.",
    )
    args = parser.parse_args()

    repo_root = Path(args.repo_root).resolve()

    # 1) 初始化 runtime（带 council、loops、research_layer 等）
    runtime: MetaAgentRuntime = create_runtime_for_blueprint_self_iteration(repo_root=repo_root)

    # 2) 收集最近 metrics + ImmuneLoop snapshot
    metrics_agg = MetricsAggregator(repo_root=repo_root)
    recent_metrics = metrics_agg.compute_recent_metrics(
        max_events=args.max_events,
        window_seconds=None,
    )

    try:
        immune_action = runtime.run_immune_check()
        immune_snapshot = {
            "action_type": getattr(immune_action, "action_type", None),
            "reason": getattr(immune_action, "reason", None),
            "metadata": getattr(immune_action, "metadata", {}),
        }
    except Exception as e:
        immune_snapshot = {
            "action_type": "error",
            "reason": f"run_immune_check_failed: {e!r}",
            "metadata": {},
        }

    # 3) 构造 PatchGovernanceCouncil + BatchPatchGovernor
    recorder = JsonlTraceRecorder(repo_root=repo_root)
    council_callable = getattr(runtime, "council", None) or runtime.model_router

    patch_council = PatchGovernanceCouncil(
        repo_root=repo_root,
        council_callable=council_callable,
        trace_recorder=recorder,
    )
    batch_governor = BatchPatchGovernor(
        repo_root=repo_root,
        patch_council=patch_council,
    )

    only_rec = _parse_only_recommended(args.only_recommended)

    # 4) 执行批量治理
    result = batch_governor.auto_govern_all(
        metrics=recent_metrics,
        immune_snapshot=immune_snapshot,
        auto_apply=args.auto_apply,
        only_recommended=only_rec,
        max_patches=args.max_patches,
    )

    # 5) 控制台输出 summary
    print("=== Batch Patch Governance Result ===")
    print("total_reports :", result.total_reports)
    print("processed     :", result.processed)
    print("skipped       :", result.skipped)
    print("started_at    :", result.started_at)
    print("finished_at   :", result.finished_at)
    print("decisions     :", len(result.decisions))

    for d in result.decisions:
        print("- proposal_id       :", d.proposal_id)
        print("  action           :", d.recommended_action)
        print("  env              :", d.environment)
        print("  risk_level       :", d.risk_level)
        print("  auto_apply       :", d.auto_apply)
        print("  rationale        :", d.rationale[:200].replace("\n", " "))
        if "apply_log" in d.constraints:
            log = d.constraints["apply_log"]
            print("  apply_status     :", log.get("status"))
            print("  apply_target     :", log.get("target_path"))
            print("  apply_backup     :", log.get("backup_path"))
        print()
    

if __name__ == "__main__":
    main()
```

## 三、现在这条“自我迭代通路”已经闭环成这样：

从蓝皮书 v3 的视角，你的 repo 已经有了一个完整 **「发现缺口 → 生成补丁 → 评测 → 审批 → 应用」** 的循环：

1. **蓝皮书 → Gap 扫描 → PatchProposal 生成**

   * `BlueprintGapTracker.scan()`
   * `ResearchLayer.run_blueprint_improvement_cycle(...)`
   * `SelfProgrammingEngine.run_cycle(...)` → `data/patches/proposals/*.json` + patch 实验

2. **Patch 离线 A/B 评测**

   * `scripts.eval_patch`
   * `PatchEvaluator + EvalHarness + ExperimentOrchestrator`
   * 输出 `data/patches/reports/<proposal_id>.json`（PatchReport）

3. **单个 Patch 的 AI 理事会 + 安全内核治理**

   * `scripts.govern_patch`
   * `PatchGovernanceCouncil.govern_patch(...)`
   * 输出 `data/governance/patch_decisions/<proposal_id>_*.json`
   * 如允许：在 sandbox 上自动调用 `PatchApplier` 改 mutable/ 下代码

4. **批量治理循环（这一部刚做完）**

   * `scripts.auto_govern_all_patches`
   * `BatchPatchGovernor.auto_govern_all(...)`
   * 可以设定：

     * `--only-recommended accept`：只治理蓝皮书认为“值得尝试”的 patch
     * `--auto-apply`：在安全内核允许前提下自动在 sandbox 应用

5. **所有行为（patch proposal / eval / governance / apply）都写入 Trace + JSON**

   * 这些 Trace + JSON 又会成为下一轮蓝皮书自我改进循环的输入。
