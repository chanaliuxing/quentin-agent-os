# quentin/skills/schemas.py

from __future__ import annotations

from datetime import datetime
from enum import Enum
from typing import Any, Dict, List, Optional

from pydantic import BaseModel, Field


class SkillKind(str, Enum):
    """
    Skill 类型：
    - procedural: 由一串步骤组成的流程（宏命令）
    - tool_wrapper: 对某个工具的包装
    - code: 纯代码实现的函数技能
    """
    procedural = "procedural"
    tool_wrapper = "tool_wrapper"
    code = "code"


class SkillDefinition(BaseModel):
    """
    一个 Skill 的结构化定义。

    v1 设计：
    - 对于 code 型技能：只需要 code_ref（module + attr）
    - 对于 procedural 型：使用 plan_steps 存储步骤列表
    """

    skill_id: str
    name: str
    version: str = "1.0.0"

    kind: SkillKind = SkillKind.procedural

    description: str = ""
    domain: Optional[str] = None        # "desktop" / "legal" / "trading" / "demo" / ...

    tags: List[str] = Field(default_factory=list)

    # 对于 code 型技能：指定模块路径和函数名
    code_ref: Optional[Dict[str, str]] = None
    # 对于 procedural 型技能：一个步骤列表（具体格式由业务决定）
    plan_steps: Optional[List[Dict[str, Any]]] = None

    created_at: datetime = Field(default_factory=datetime.utcnow)
    updated_at: datetime = Field(default_factory=datetime.utcnow)

    active: bool = True
    supersedes: Optional[str] = None     # 被替代的 skill_id

    meta: Dict[str, Any] = Field(default_factory=dict)


class SkillStats(BaseModel):
    """
    每个 Skill 的性能统计信息。
    """
    skill_id: str

    calls: int = 0
    success: int = 0
    failure: int = 0

    avg_latency_sec: float = 0.0
    last_called_at: Optional[datetime] = None

    meta: Dict[str, Any] = Field(default_factory=dict)

# quentin/skills/library.py

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Dict, Any, List, Optional

import uuid

from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.skills.schemas import SkillDefinition, SkillStats


@dataclass
class SkillLibrary:
    """
    SkillLibrary v1：

    - 提供 Skill 的注册、查询、列出、统计更新
    - 所有数据存 KV，方便以后切换 Redis / DB / 文件
    """

    tracer: Tracer
    kv: KVStore

    skill_prefix: str = "skill:def:"
    stats_prefix: str = "skill:stats:"
    index_key: str = "skill:index:all"
    name_index_key: str = "skill:index:name"

    # ---------- 工具方法 ----------

    async def _save_skill(self, skill: SkillDefinition) -> None:
        key = f"{self.skill_prefix}{skill.skill_id}"
        await self.kv.save(key, skill.model_dump())
        await self.kv.append_list(self.index_key, skill.skill_id)
        # name 索引：允许按 name 查最新版本
        await self.kv.hset(self.name_index_key, skill.name, skill.skill_id)

    async def _save_stats(self, stats: SkillStats) -> None:
        key = f"{self.stats_prefix}{stats.skill_id}"
        await self.kv.save(key, stats.model_dump())

    # ---------- 注册与获取 ----------

    async def register_skill(
        self,
        name: str,
        kind: str,
        description: str = "",
        domain: Optional[str] = None,
        tags: Optional[List[str]] = None,
        code_ref: Optional[Dict[str, str]] = None,
        plan_steps: Optional[List[Dict[str, Any]]] = None,
        supersedes: Optional[str] = None,
        meta: Optional[Dict[str, Any]] = None,
    ) -> SkillDefinition:
        """
        注册一个新的 SkillDefinition。
        v1 不做去重策略，由调用方控制命名。
        """
        skill_id = f"skill_{uuid.uuid4().hex}"
        skill = SkillDefinition(
            skill_id=skill_id,
            name=name,
            kind=kind,
            description=description,
            domain=domain,
            tags=tags or [],
            code_ref=code_ref,
            plan_steps=plan_steps,
            supersedes=supersedes,
            meta=meta or {},
        )

        await self._save_skill(skill)
        await self._save_stats(SkillStats(skill_id=skill_id))

        self.tracer.span(
            "L1",
            "skill_registered",
            module="skills",
            input={
                "skill_id": skill_id,
                "name": name,
                "kind": kind,
                "domain": domain,
            },
        )

        return skill

    async def get_skill(self, skill_id: str) -> Optional[SkillDefinition]:
        data = await self.kv.get(f"{self.skill_prefix}{skill_id}")
        if not data:
            return None
        return SkillDefinition(**data)

    async def get_skill_by_name(self, name: str) -> Optional[SkillDefinition]:
        skill_id = await self.kv.hget(self.name_index_key, name)
        if not skill_id:
            return None
        return await self.get_skill(skill_id)

    async def list_skills(self, limit: int = 100) -> List[SkillDefinition]:
        ids = await self.kv.get(self.index_key) or []
        ids = list(dict.fromkeys(ids))[-limit:]
        res: List[SkillDefinition] = []
        for sid in ids:
            data = await self.kv.get(f"{self.skill_prefix}{sid}")
            if data:
                res.append(SkillDefinition(**data))
        return res

    # ---------- 统计更新 ----------

    async def get_stats(self, skill_id: str) -> SkillStats:
        data = await self.kv.get(f"{self.stats_prefix}{skill_id}")
        if not data:
            stats = SkillStats(skill_id=skill_id)
            await self._save_stats(stats)
            return stats
        return SkillStats(**data)

    async def update_stats(
        self,
        skill_id: str,
        success: bool,
        latency_sec: float,
    ) -> None:
        from datetime import datetime

        stats = await self.get_stats(skill_id)
        stats.calls += 1
        if success:
            stats.success += 1
        else:
            stats.failure += 1

        # 简单移动平均
        if stats.calls == 1:
            stats.avg_latency_sec = latency_sec
        else:
            alpha = 0.2
            stats.avg_latency_sec = (
                alpha * latency_sec + (1 - alpha) * stats.avg_latency_sec
            )

        stats.last_called_at = datetime.utcnow()

        await self._save_stats(stats)

        self.tracer.span(
            "L1",
            "skill_stats_updated",
            module="skills",
            input={
                "skill_id": skill_id,
                "success": success,
                "latency_sec": latency_sec,
            },
        )

# quentin/skills/miner.py

from __future__ import annotations

from dataclasses import dataclass
from typing import Any, Dict, List, Optional

import uuid

from quentin.infra.logging import Tracer
from quentin.skills.library import SkillLibrary
from quentin.skills.schemas import SkillDefinition, SkillKind


# ---- Trace Episode 的简单类型约定 ----
# 实际上你可以定义自己的结构，这里只约定最小字段。


TraceStep = Dict[str, Any]
TraceEpisode = Dict[str, Any]
# episode 预期结构示例：
# {
#   "episode_id": "ep_xxx",
#   "domain": "demo",
#   "steps": [
#       {"tool": "add", "args": {"a": 1, "b": 2}, "result": 3},
#       {"tool": "square", "args": {"x": 3}, "result": 9},
#   ],
#   "meta": {...}
# }


@dataclass
class SkillPatternCandidate:
    """
    从 Trace 中挖到的“技能候选”模式。
    """

    name: str
    domain: Optional[str]
    occurrences: int
    plan_steps: List[TraceStep]
    example_episode_ids: List[str]
    description: str = ""
    tags: List[str] = None
    meta: Dict[str, Any] = None


@dataclass
class SkillMiner:
    """
    SkillMiner v1：

    - 接收若干 Trace episodes
    - 以非常简单的规则生成“宏技能候选”（proc 类型）
    - 把它们注册进 SkillLibrary

    这里的挖掘算法刻意保持简单：
    - 每个 episode 的 steps 长度 >= 2，就视为一个潜在的 procedural skill
    - 用 steps 的 "tool" 序列作为 pattern key
    - 对出现次数 >= min_occurrences 的 pattern，生成一个 skill
    """

    tracer: Tracer
    skill_library: SkillLibrary
    min_steps: int = 2
    min_occurrences: int = 2

    async def mine_from_episodes(
        self,
        episodes: List[TraceEpisode],
        domain: Optional[str] = None,
    ) -> List[SkillDefinition]:
        """
        从 episodes 中挖掘技能，并写入 SkillLibrary。
        返回 newly created skills。
        """
        if not episodes:
            return []

        # 1) 聚合 pattern -> 统计出现次数
        pattern_map: Dict[str, Dict[str, Any]] = {}

        for ep in episodes:
            ep_domain = domain or ep.get("domain")
            if ep_domain is None:
                ep_domain = "unknown"

            steps: List[TraceStep] = ep.get("steps") or []
            if len(steps) < self.min_steps:
                continue

            # 使用 tool 序列作为 pattern key（极简版本）
            tool_seq = [step.get("tool", "?") for step in steps]
            pattern_key = "|".join(tool_seq)

            entry = pattern_map.setdefault(
                pattern_key,
                {
                    "count": 0,
                    "steps": steps,
                    "episodes": [],
                    "domain": ep_domain,
                },
            )
            entry["count"] += 1
            entry["episodes"].append(ep.get("episode_id"))

        # 2) 对出现次数达到阈值的 pattern 创建 SkillDefinition
        created_skills: List[SkillDefinition] = []

        for pattern_key, info in pattern_map.items():
            count = info["count"]
            if count < self.min_occurrences:
                continue

            ep_ids = info["episodes"]
            steps = info["steps"]
            skill_domain = info["domain"]

            # 生成 skill 名称
            name = f"macro_{pattern_key}_{uuid.uuid4().hex[:8]}"
            description = (
                f"Auto-mined macro skill from {count} episodes, tools: {pattern_key}"
            )

            plan_steps = steps  # 直接把 Trace 步骤作为 plan

            skill = await self.skill_library.register_skill(
                name=name,
                kind=SkillKind.procedural.value,
                description=description,
                domain=skill_domain,
                tags=["auto-mined", "macro"],
                plan_steps=plan_steps,
                meta={
                    "pattern_key": pattern_key,
                    "episodes": ep_ids[:10],  # 只记一部分样例
                },
            )

            created_skills.append(skill)

            self.tracer.span(
                "L3",
                "skill_mined",
                module="skill_miner",
                input={
                    "pattern_key": pattern_key,
                    "domain": skill_domain,
                    "occurrences": count,
                },
                output={"skill_id": skill.skill_id, "name": skill.name},
            )

        return created_skills
这个 SkillMiner 就是蓝皮书里 3.4 “从成功 Trace 中挖掘可抽象的技能模式（pattern mining）” 的骨架。
以后你可以把 mine_from_episodes 换成任何复杂算法，只要接口不变。

3.1 SleepJob 抽象、SleepLoop 主体

# quentin/sleep/loop.py

from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any, Dict, List

from datetime import datetime

from quentin.infra.logging import Tracer
from quentin.memory.fabric import MemoryFabric   # 假定已存在
from quentin.skills.miner import SkillMiner


@dataclass
class SleepBudget:
    """
    本次 SleepLoop 允许消耗的资源预算。
    v1 只是放在接口上，具体限制逻辑由各个 Job 自己处理。
    """
    max_traces: int = 1000
    max_tokens: int = 50_000
    max_seconds: int = 300


@dataclass
class SleepJobResult:
    name: str
    metrics: Dict[str, Any]
    new_memories: int = 0
    new_skills: int = 0


class SleepJob:
    """
    所有 SleepJob 的抽象基类。
    """

    name: str = "base"

    async def run(self, budget: SleepBudget) -> SleepJobResult:
        raise NotImplementedError


@dataclass
class MemoryConsolidationJob(SleepJob):
    """
    执行记忆巩固与晋级的 Job。

    这里假定 MemoryFabric 提供一个 consolidate_once() 接口，
    返回 {"promoted": n1, "decayed": n2, ...} 之类的统计信息。
    """

    memory: MemoryFabric
    tracer: Tracer
    name: str = "memory_consolidation"

    async def run(self, budget: SleepBudget) -> SleepJobResult:
        self.tracer.span(
            "L2",
            "sleep_memory_consolidation_start",
            module="sleep",
            input={"budget": budget.__dict__},
        )

        try:
            stats = await self.memory.consolidate_once(budget=budget.__dict__)
        except AttributeError:
            # 如果你目前的 MemoryFabric 还没实现 consolidate_once，可以先返回空结果
            stats = {"promoted": 0, "decayed": 0}

        self.tracer.span(
            "L2",
            "sleep_memory_consolidation_end",
            module="sleep",
            output=stats,
        )

        return SleepJobResult(
            name=self.name,
            metrics=stats,
            new_memories=stats.get("promoted", 0),
        )


@dataclass
class SkillMiningJob(SleepJob):
    """
    使用 SkillMiner 从 Trace Episodes 中挖掘技能。

    注意：
    - SleepLoop 并不直接关心 Trace 是从哪里来的
    - 你只需要实现一个 episodes_provider，返回 episodes 列表即可
    """

    tracer: Tracer
    skill_miner: SkillMiner
    episodes_provider: Any  # 一个带有 async get_recent_episodes(limit) 的对象
    name: str = "skill_mining"

    async def run(self, budget: SleepBudget) -> SleepJobResult:
        max_eps = budget.max_traces

        # episodes_provider 接口约定：
        #   async def get_recent_episodes(self, limit: int) -> List[TraceEpisode]
        episodes = await self.episodes_provider.get_recent_episodes(limit=max_eps)

        if not episodes:
            self.tracer.span(
                "L2",
                "sleep_skill_mining_skip",
                module="sleep",
                output={"reason": "no_episodes"},
            )
            return SleepJobResult(name=self.name, metrics={"episodes": 0})

        domain = None  # 交给 SkillMiner 自己从 episodes 里决定
        skills = await self.skill_miner.mine_from_episodes(episodes, domain=domain)

        metrics = {
            "episodes": len(episodes),
            "new_skills": len(skills),
        }

        self.tracer.span(
            "L2",
            "sleep_skill_mining_end",
            module="sleep",
            output=metrics,
        )

        return SleepJobResult(
            name=self.name,
            metrics=metrics,
            new_skills=len(skills),
        )


@dataclass
class SleepLoop:
    """
    SleepLoop v1：

    - 管理多个 SleepJob
    - 在 run_once() 中串行执行它们
    - 汇总运行结果（方便 Observability 展示）
    """

    tracer: Tracer
    memory: MemoryFabric
    skill_miner: SkillMiner
    episodes_provider: Any

    jobs: List[SleepJob] = field(default_factory=list)

    def __post_init__(self) -> None:
        # 默认注册两个 Job：记忆巩固 + 技能挖掘
        if not self.jobs:
            self.jobs = [
                MemoryConsolidationJob(memory=self.memory, tracer=self.tracer),
                SkillMiningJob(
                    tracer=self.tracer,
                    skill_miner=self.skill_miner,
                    episodes_provider=self.episodes_provider,
                ),
            ]

    async def run_once(self, budget: SleepBudget) -> Dict[str, Any]:
        """
        顺序执行所有 SleepJob，返回整体统计情况。
        """
        started = datetime.utcnow()
        results: List[SleepJobResult] = []

        for job in self.jobs:
            res = await job.run(budget)
            results.append(res)

        finished = datetime.utcnow()

        summary = {
            "started_at": started.isoformat(),
            "finished_at": finished.isoformat(),
            "jobs": [
                {
                    "name": r.name,
                    "metrics": r.metrics,
                    "new_memories": r.new_memories,
                    "new_skills": r.new_skills,
                }
                for r in results
            ],
        }

        self.tracer.span(
            "L2",
            "sleep_loop_run_once",
            module="sleep",
            output=summary,
        )

        return summary

@cli.command("sleep-run-once")
@click.option("--max-traces", default=1000, show_default=True)
@click.option("--max-tokens", default=50000, show_default=True)
@click.option("--max-seconds", default=300, show_default=True)
def sleep_run_once_cmd(max_traces: int, max_tokens: int, max_seconds: int):
    """
    Run SleepLoop once with given budget.

    当前实现：
    - MemoryConsolidationJob 可能是空实现（取决于你是否实现了 MemoryFabric.consolidate_once）
    - SkillMiningJob 默认用 DummyEpisodesProvider（不会挖出 anything）
    你可以后续替换 episodes_provider，让它真的从 Trace 里挖 skill。
    """

    async def _run():
        import json
        from quentin.sleep.loop import SleepBudget
        settings = Settings()
        runtime = AgentRuntime.create_from_settings(settings)

        budget = SleepBudget(
            max_traces=max_traces,
            max_tokens=max_tokens,
            max_seconds=max_seconds,
        )

        summary = await runtime.sleep_loop.run_once(budget)
        print(json.dumps(summary, ensure_ascii=False, indent=2))

        # 顺带展示一下当前 SkillLibrary 里有哪些技能
        skills = await runtime.skill_library.list_skills(limit=20)
        print("\n=== Skills (up to 20) ===")
        for s in skills:
            print(f"- {s.skill_id} | {s.name} | {s.kind.value} | domain={s.domain}")

    asyncio.run(_run())
