from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional, Iterable, Tuple
from pathlib import Path
from datetime import datetime
import json
import logging
import uuid

logger = logging.getLogger(__name__)


# ---- 常量 & 简单工具 ----

SKILL_STATUS_CANDIDATE = "candidate"    # SleepLoop 刚抽出来，还没实现
SKILL_STATUS_ACTIVE = "active"          # 已实现，可用于 Planner/Executor
SKILL_STATUS_DEPRECATED = "deprecated"  # 废弃，不再推荐

DEFAULT_REGISTRY_PATH = "data/skills_registry.jsonl"
DEFAULT_CANDIDATES_PATH = "data/skills.jsonl"


def _simple_text_score(query: str, text: str) -> float:
    """
    和 MemoryFabric 一致的极简文本相似度。
    以后可以被 SelfProgrammingEngine 自动升级为向量检索。
    """
    if not query or not text:
        return 0.0

    def _tokenize(s: str):
        for ch in [",", ".", "!", "?", ":", ";", "\n", "\t", "，", "。", "？", "！"]:
            s = s.replace(ch, " ")
        return [w.lower() for w in s.split() if w.strip()]

    q = set(_tokenize(query))
    t = set(_tokenize(text))
    if not q or not t:
        return 0.0

    inter = len(q & t)
    union = len(q | t)
    if union == 0:
        return 0.0
    return inter / union


# ---- 数据结构：SkillSpec ----

@dataclass
class SkillSpec:
    """
    SkillSpec：蓝皮书里的“技能定义”工程版。

    - skill_id              : 唯一 ID
    - title                 : 技能名
    - description           : 技能说明（自然语言）
    - domain                : 适用领域（core/desktop/legal/trading/...）
    - blueprint_capability  : 对应蓝皮书里的 capability_id（可选）
    - status                : candidate/active/deprecated
    - implementation_module : 实现所在模块（如 "quentin/tools/desktop.py"）
    - implementation_symbol : 实现函数名（如 "click_button_with_image"）
    - created_at            : 创建时间
    - updated_at            : 最近更新时间
    - metadata              : 附加信息（来源 trace、LLM 版本等）
    """

    skill_id: str
    title: str
    description: str
    domain: str = "core"
    blueprint_capability: Optional[str] = None
    status: str = SKILL_STATUS_CANDIDATE
    implementation_module: Optional[str] = None
    implementation_symbol: Optional[str] = None
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    metadata: Dict[str, Any] = field(default_factory=dict)

    # 派生属性
    @property
    def is_implemented(self) -> bool:
        return bool(self.implementation_module and self.implementation_symbol)

    @property
    def is_active(self) -> bool:
        return self.status == SKILL_STATUS_ACTIVE and self.is_implemented

    def touch(self) -> None:
        self.updated_at = datetime.utcnow().isoformat()

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

    @classmethod
    def from_dict(cls, data: Dict[str, Any]) -> "SkillSpec":
        return cls(**data)


# ---- SkillRegistry：底层存储（JSONL） ----

class SkillRegistry:
    """
    SkillRegistry v1：用 JSONL 存 skill 规范。

    设计：
    - 内存里用 dict[skill_id] 管理
    - 每次写操作后，重写整个 registry 文件（简单但稳定）
    - 提供 CRUD & 列表 & 查询基础
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        registry_path: str | Path = DEFAULT_REGISTRY_PATH,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.registry_path = self.repo_root / registry_path
        self.registry_path.parent.mkdir(parents=True, exist_ok=True)

        self._skills: Dict[str, SkillSpec] = {}
        self._loaded = False

    # --- 加载 & 保存 ---

    def _ensure_loaded(self) -> None:
        if self._loaded:
            return
        if not self.registry_path.exists():
            self._loaded = True
            return
        try:
            with self.registry_path.open("r", encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if not line:
                        continue
                    try:
                        obj = json.loads(line)
                        sk = SkillSpec.from_dict(obj)
                        self._skills[sk.skill_id] = sk
                    except Exception:
                        logger.exception("SkillRegistry: failed to parse line in %s", self.registry_path)
        except Exception:
            logger.exception("SkillRegistry: failed to read %s", self.registry_path)
        self._loaded = True

    def _flush(self) -> None:
        try:
            with self.registry_path.open("w", encoding="utf-8") as f:
                for sk in self._skills.values():
                    f.write(json.dumps(sk.to_dict(), ensure_ascii=False) + "\n")
        except Exception:
            logger.exception("SkillRegistry: failed to write %s", self.registry_path)

    # --- CRUD ---

    def upsert(self, spec: SkillSpec) -> None:
        self._ensure_loaded()
        self._skills[spec.skill_id] = spec
        self._flush()

    def get(self, skill_id: str) -> Optional[SkillSpec]:
        self._ensure_loaded()
        return self._skills.get(skill_id)

    def delete(self, skill_id: str) -> None:
        self._ensure_loaded()
        if skill_id in self._skills:
            del self._skills[skill_id]
            self._flush()

    def all(self) -> List[SkillSpec]:
        self._ensure_loaded()
        return list(self._skills.values())

    def find_by_title_domain(self, title: str, domain: str) -> Optional[SkillSpec]:
        self._ensure_loaded()
        norm_title = title.strip().lower()
        for sk in self._skills.values():
            if sk.title.strip().lower() == norm_title and sk.domain == domain:
                return sk
        return None


# ---- SkillManager：对外接口 ----

class SkillManager:
    """
    SkillManager v1：

    - 从 SleepLoop 产出的 candidates 文件（skills.jsonl）同步技能
    - 维护 SkillRegistry（技能生命周期）
    - 给 Planner / Executor / Research 提供：
        - list_skills(...)
        - suggest_for_task(...)
        - mark_implemented(...)
        - mark_active/mark_deprecated(...)
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        registry_path: str | Path = DEFAULT_REGISTRY_PATH,
        candidates_path: str | Path = DEFAULT_CANDIDATES_PATH,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.registry = SkillRegistry(
            repo_root=self.repo_root,
            registry_path=registry_path,
        )
        self.candidates_path = self.repo_root / candidates_path
        self.candidates_path.parent.mkdir(parents=True, exist_ok=True)

    # ---- 与 SleepLoop 的技能候选同步 ----

    def sync_from_candidates(self, *, max_new: Optional[int] = None) -> int:
        """
        从 data/skills.jsonl 读取 SkillCandidate，合并到 SkillRegistry。

        逻辑：
        - 按 (title, domain) 去重；如果已有同名同域技能，则：
            - 若旧技能是 candidate，则更新 description/metadata；
            - 若旧技能已经 active，则不覆盖，仅追加 metadata.source。
        - 新建技能默认 status=candidate，implementation 为空。
        - 返回新增/更新的技能数量。
        """
        if not self.candidates_path.exists():
            logger.info("SkillManager.sync_from_candidates: no candidates file, skip.")
            return 0

        changed = 0
        count_new = 0

        try:
            with self.candidates_path.open("r", encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if not line:
                        continue
                    try:
                        obj = json.loads(line)
                    except Exception:
                        logger.exception("SkillManager.sync_from_candidates: invalid JSON in %s", self.candidates_path)
                        continue

                    title = str(obj.get("title") or "Untitled Skill").strip()
                    description = str(obj.get("description") or "").strip()
                    domain = str(obj.get("domain") or "core").strip()
                    blueprint_cap = obj.get("blueprint_cap")

                    # 统一 metadata：保留原 candidate 的字段
                    metadata = {
                        "source": "SleepLoop",
                        "blueprint_cap": blueprint_cap,
                        "raw_candidate": obj,
                        "synced_at": datetime.utcnow().isoformat(),
                    }

                    existing = self.registry.find_by_title_domain(title, domain)
                    if existing is None:
                        # 创建新技能
                        sk = SkillSpec(
                            skill_id=str(uuid.uuid4()),
                            title=title,
                            description=description,
                            domain=domain,
                            blueprint_capability=blueprint_cap,
                            status=SKILL_STATUS_CANDIDATE,
                            metadata=metadata,
                        )
                        self.registry.upsert(sk)
                        changed += 1
                        count_new += 1
                        if max_new is not None and count_new >= max_new:
                            break
                    else:
                        # 更新已有技能的描述和 metadata（仅适用于 candidate）
                        if existing.status == SKILL_STATUS_CANDIDATE:
                            existing.description = description or existing.description
                            existing.metadata.setdefault("candidate_updates", []).append(metadata)
                            existing.touch()
                            self.registry.upsert(existing)
                            changed += 1
                        else:
                            # active/dep 的技能只追加 metadata，不改描述
                            existing.metadata.setdefault("candidate_updates", []).append(metadata)
                            existing.touch()
                            self.registry.upsert(existing)
                            changed += 1

        except Exception:
            logger.exception("SkillManager.sync_from_candidates: failed to read %s", self.candidates_path)

        return changed

    # ---- 列表 / 查询 ----

    def list_skills(
        self,
        *,
        status: Optional[str] = None,
        domain: Optional[str] = None,
        implemented: Optional[bool] = None,
    ) -> List[SkillSpec]:
        """
        列出技能：
        - status: candidate/active/deprecated
        - domain: core/desktop/legal/trading/...
        - implemented: True/False 过滤是否已经绑定代码实现
        """
        skills = self.registry.all()
        result: List[SkillSpec] = []
        for sk in skills:
            if status and sk.status != status:
                continue
            if domain and sk.domain != domain:
                continue
            if implemented is not None:
                if implemented != sk.is_implemented:
                    continue
            result.append(sk)
        return result

    def suggest_for_task(
        self,
        *,
        task_text: str,
        domain: Optional[str] = None,
        top_k: int = 5,
        include_candidates: bool = True,
    ) -> List[Dict[str, Any]]:
        """
        给 Planner/Executor 用的：根据任务文本推荐相关技能。

        - 默认优先使用 active+implemented 的技能；
        - 如果 include_candidates=True，则候选技能也可以被推荐（但 is_implemented=False）。
        - 返回简化字典列表，方便直接写入 Prompt 或日志。
        """
        if not task_text:
            return []

        skills = self.registry.all()
        scored: List[Tuple[float, SkillSpec]] = []

        for sk in skills:
            if domain and sk.domain not in (domain, "cross"):
                continue
            if sk.status == SKILL_STATUS_DEPRECATED:
                continue
            if not include_candidates and sk.status == SKILL_STATUS_CANDIDATE:
                continue

            score = _simple_text_score(task_text, sk.title + " " + sk.description)
            if score <= 0.0:
                continue

            # 实现好的 active 技能加权
            weight = 1.0
            if sk.is_active:
                weight = 1.3
            elif sk.status == SKILL_STATUS_CANDIDATE:
                weight = 0.9

            scored.append((score * weight, sk))

        scored.sort(key=lambda x: x[0], reverse=True)
        results: List[Dict[str, Any]] = []
        for sc, sk in scored[:top_k]:
            results.append(
                {
                    "skill_id": sk.skill_id,
                    "title": sk.title,
                    "description": sk.description,
                    "domain": sk.domain,
                    "blueprint_capability": sk.blueprint_capability,
                    "status": sk.status,
                    "implemented": sk.is_implemented,
                    "implementation_module": sk.implementation_module,
                    "implementation_symbol": sk.implementation_symbol,
                    "score": sc,
                }
            )
        return results

    # ---- 状态管理：激活 / 废弃 / 绑定实现 ----

    def mark_implemented(
        self,
        *,
        skill_id: str,
        module: str,
        symbol: str,
        auto_activate: bool = True,
    ) -> Optional[SkillSpec]:
        """
        当某个技能已经被 SelfProgrammingEngine 或人工实现为代码时：
        - 绑定 implementation_module / implementation_symbol
        - 可选：自动将 status=active
        """
        sk = self.registry.get(skill_id)
        if sk is None:
            logger.warning("SkillManager.mark_implemented: skill %s not found", skill_id)
            return None

        sk.implementation_module = module
        sk.implementation_symbol = symbol
        if auto_activate:
            sk.status = SKILL_STATUS_ACTIVE
        sk.touch()
        self.registry.upsert(sk)
        return sk

    def mark_active(self, skill_id: str) -> Optional[SkillSpec]:
        sk = self.registry.get(skill_id)
        if sk is None:
            return None
        sk.status = SKILL_STATUS_ACTIVE
        sk.touch()
        self.registry.upsert(sk)
        return sk

    def mark_deprecated(self, skill_id: str) -> Optional[SkillSpec]:
        sk = self.registry.get(skill_id)
        if sk is None:
            return None
        sk.status = SKILL_STATUS_DEPRECATED
        sk.touch()
        self.registry.upsert(sk)
        return sk

def cmd_skills_sync(args: argparse.Namespace) -> None:
    mgr = SkillManager()
    changed = mgr.sync_from_candidates()

def cmd_skills_list(args: argparse.Namespace) -> None:
    implemented: Optional[bool] = None
    if args.implemented is not None:
        v = args.implemented.strip().lower()
        if v in ("true", "1", "yes"):
            implemented = True
        elif v in ("false", "0", "no"):
            implemented = False

    mgr = SkillManager()
    skills = mgr.list_skills(
        status=args.status,
        domain=args.domain,
        implemented=implemented,
    )
    print(f"Total skills: {len(skills)}")
    for sk in skills:
        impl_flag = "Y" if sk.is_implemented else "N"
        print(
            f"- {sk.skill_id[:8]} | {sk.status:10s} | impl={impl_flag} | "
            f"{sk.domain:8s} | {sk.title}"
        )


def cmd_skills_suggest(args: argparse.Namespace) -> None:
    mgr = SkillManager()
    suggestions = mgr.suggest_for_task(
        task_text=args.text,
        domain=args.domain,
        top_k=args.top_k,
        include_candidates=True,
    )
    print(f"Suggested skills for domain={args.domain or '*'} text='{args.text[:40]}...':")
    for s in suggestions:
        print(
            f"- {s['skill_id'][:8]} | score={s['score']:.2f} | "
            f"{s['status']:10s} | impl={s['implemented']} | {s['title']}"
        )

SleepLoop

读取 skills.jsonl，合并到 SkillRegistry ✅

支持 list_skills / suggest_for_task / 标记实现状态 ✅

SelfProgrammingEngine（前面已经有）：

可针对某个模块文件生成 patch ✅（下一步可把“实现新技能”也挂上来）

剩余的 gap（下一步可以做）：

当 SelfProgrammingEngine 生成了一个新工具函数时，
自动在 SkillManager 里调用 mark_implemented()，绑定 implementation_module + symbol；

在 Planner / Executor 里，自动调用 SkillManager.suggest_for_task()，
把技能描述塞进 LLM 的 prompt 里，引导它优先使用这些技能；
