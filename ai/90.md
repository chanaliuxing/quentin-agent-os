# quentin/immune/anomaly.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, List, Optional
from datetime import datetime
import logging

logger = logging.getLogger(__name__)


@dataclass
class MetricSnapshot:
    """
    用于描述某一时刻的一些关键指标。
    这里的字段你可以随时扩展，当前只做示例。
    """
    attempt_index: int
    total_attempts: int
    success_attempts: int
    error_attempts: int
    cost_tokens: float = 0.0
    latency_ms: float = 0.0
    reward_score: float = 0.0  # 可以是任务收益、评分等
    timestamp: str = datetime.utcnow().isoformat()

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class AnomalySignal:
    """
    异常信号：Immune Loop 的输入。
    """
    anomaly_type: str          # 如 "high_failure_rate" / "high_cost" / "timeout" 等
    severity: str              # "low" | "medium" | "high" | "critical"
    reason: str                # 人类可读的异常描述
    suggested_actions: List[str]  # 如 ["degrade_mode", "disable_self_programming"]

    metrics: Dict[str, Any]
    created_at: str = datetime.utcnow().isoformat()

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class AnomalyDetector:
    """
    v1: 简单规则 + 日志，后续可以接统计/ML 模型。

    使用方式：
    - detector = AnomalyDetector()
    - signal = detector.analyze(goal, metrics, snapshot, recent_snapshots)
    - 若 signal 为 None = 正常；否则表示检测到异常。
    """

    def __init__(self, config: Optional[Dict[str, Any]] = None) -> None:
        self.config = config or {}
        # 一些默认阈值
        self.failure_rate_threshold = float(self.config.get("failure_rate_threshold", 0.8))
        self.min_attempts_for_failure = int(self.config.get("min_attempts_for_failure", 3))
        self.max_tokens_per_goal = float(self.config.get("max_tokens_per_goal", 50_000))
        self.max_latency_ms = float(self.config.get("max_latency_ms", 60_000))  # 60 秒

    def analyze(
        self,
        *,
        goal: Any,
        metrics: Dict[str, Any],
        snapshot: MetricSnapshot,
        recent_snapshots: List[MetricSnapshot],
    ) -> Optional[AnomalySignal]:
        """
        根据 metrics + 当前 snapshot + 最近的历史，判断是否出现异常。
        """
        total = snapshot.total_attempts
        errors = snapshot.error_attempts
        failure_rate = (errors / total) if total > 0 else 0.0
        cost_tokens = snapshot.cost_tokens
        latency_ms = snapshot.latency_ms

        # 1) 高失败率
        if total >= self.min_attempts_for_failure and failure_rate >= self.failure_rate_threshold:
            reason = (
                f"High failure rate detected: {failure_rate:.2f} over {total} attempts "
                f"(threshold={self.failure_rate_threshold})."
            )
            logger.warning("[AnomalyDetector] %s", reason)
            return AnomalySignal(
                anomaly_type="high_failure_rate",
                severity="high",
                reason=reason,
                suggested_actions=["degrade_mode", "disable_self_programming", "trigger_reflection"],
                metrics={
                    "failure_rate": failure_rate,
                    "total_attempts": total,
                    "error_attempts": errors,
                },
            )

        # 2) 成本过高
        if cost_tokens >= self.max_tokens_per_goal > 0:
            reason = (
                f"High token cost detected: {cost_tokens} tokens used for this goal "
                f"(threshold={self.max_tokens_per_goal})."
            )
            logger.warning("[AnomalyDetector] %s", reason)
            return AnomalySignal(
                anomaly_type="high_cost",
                severity="medium",
                reason=reason,
                suggested_actions=["degrade_mode", "limit_planning_depth"],
                metrics={
                    "cost_tokens": cost_tokens,
                },
            )

        # 3) 延迟过高
        if latency_ms >= self.max_latency_ms > 0:
            reason = (
                f"High latency detected: {latency_ms:.1f} ms "
                f"(threshold={self.max_latency_ms})."
            )
            logger.warning("[AnomalyDetector] %s", reason)
            return AnomalySignal(
                anomaly_type="high_latency",
                severity="medium",
                reason=reason,
                suggested_actions=["degrade_mode"],
                metrics={
                    "latency_ms": latency_ms,
                },
            )

        # TODO: 未来可以在这里加入更复杂的统计/ML 检测，如：
        # - reward_score 持续很低
        # - 行为异常模式（某个技能反复导致崩溃）
        # - 多目标/跨域的异常相关性

        return None

from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, List, Optional
from pathlib import Path
from datetime import datetime
import json
import uuid
import logging

from quentin.immune.anomaly import AnomalySignal

logger = logging.getLogger(__name__)


@dataclass
class ImmuneCase:
    """
    一个完整的“免疫案例”：一次异常周期的上下文。
    """
    case_id: str
    goal_summary: Dict[str, Any]
    anomaly: Dict[str, Any]
    metrics: Dict[str, Any]
    recent_traces: List[Dict[str, Any]]
    created_at: str = datetime.utcnow().isoformat()
    tags: List[str] = None

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        if d.get("tags") is None:
            d["tags"] = []
        return d


class ImmuneCaseStore:
    """
    v1: 简单的 JSONL / JSON 文件存储。
    后续可以换成 DB / 对象存储。

    - append_case(case): 追加一个 ImmuneCase
    - list_cases():      简单列举（示例，不是高性能设计）
    """

    def __init__(self, *, repo_root: str | Path = ".") -> None:
        self.repo_root = Path(repo_root).resolve()
        self.base_dir = self.repo_root / "data" / "immune_cases"
        self.base_dir.mkdir(parents=True, exist_ok=True)
        self.jsonl_path = self.base_dir / "immune_cases.jsonl"

    def append_case(
        self,
        *,
        goal: Any,
        anomaly: AnomalySignal,
        metrics: Dict[str, Any],
        recent_traces: List[Dict[str, Any]],
    ) -> ImmuneCase:
        case_id = uuid.uuid4().hex
        goal_summary = self._summarize_goal(goal)
        case = ImmuneCase(
            case_id=case_id,
            goal_summary=goal_summary,
            anomaly=anomaly.to_dict(),
            metrics=metrics,
            recent_traces=recent_traces,
            tags=[anomaly.anomaly_type, anomaly.severity],
        )
        line = json.dumps(case.to_dict(), ensure_ascii=False)
        try:
            with self.jsonl_path.open("a", encoding="utf-8") as f:
                f.write(line + "\n")
        except Exception:
            logger.exception("ImmuneCaseStore: failed to append case to %s", self.jsonl_path)

        # 也可以每 case 单独存一份，便于调试
        try:
            per_file = self.base_dir / f"{case_id}.json"
            per_file.write_text(
                json.dumps(case.to_dict(), ensure_ascii=False, indent=2),
                encoding="utf-8",
            )
        except Exception:
            logger.exception("ImmuneCaseStore: failed to write per-case json file for %s", case_id)

        return case

    def list_cases(self, limit: int = 100) -> List[ImmuneCase]:
        cases: List[ImmuneCase] = []
        if not self.jsonl_path.exists():
            return cases
        try:
            with self.jsonl_path.open("r", encoding="utf-8") as f:
                for _ in range(limit):
                    line = f.readline()
                    if not line:
                        break
                    obj = json.loads(line)
                    cases.append(
                        ImmuneCase(
                            case_id=obj.get("case_id", ""),
                            goal_summary=obj.get("goal_summary", {}),
                            anomaly=obj.get("anomaly", {}),
                            metrics=obj.get("metrics", {}),
                            recent_traces=obj.get("recent_traces", []),
                            created_at=obj.get("created_at", ""),
                            tags=obj.get("tags", []),
                        )
                    )
        except Exception:
            logger.exception("ImmuneCaseStore: failed to read cases from %s", self.jsonl_path)
        return cases

    def _summarize_goal(self, goal: Any) -> Dict[str, Any]:
        if isinstance(goal, dict):
            g = goal
        else:
            g = {}
            for attr in ("id", "title", "description", "domain", "metadata"):
                if hasattr(goal, attr):
                    g[attr] = getattr(goal, attr)
        # 可以做进一步裁剪
        return g

# quentin/immune/loop.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from typing import Any, Dict, List, Optional
import logging

from quentin.immune.anomaly import AnomalyDetector, AnomalySignal, MetricSnapshot
from quentin.immune.store import ImmuneCaseStore, ImmuneCase
from quentin.research.interface import ResearchInterface

logger = logging.getLogger(__name__)


@dataclass
class ImmuneAction:
    """
    Immune Loop 给系统的“处置建议”。

    - degrade_mode             : 是否建议进入降级模式（比如只用安全模型 / 禁用复杂实验）
    - disable_self_programming : 是否暂时关闭自编程 / patch 生成
    - revert_version           : 是否建议回滚到上一稳定版本（当前只是标记，具体回滚由外部执行）
    - notes                    : 解释/理由
    - anomaly                  : 对应的异常信号
    - immune_case_id           : 关联的 ImmuneCase，方便后续追踪
    """
    degrade_mode: bool
    disable_self_programming: bool
    revert_version: bool
    notes: str
    anomaly: Dict[str, Any]
    immune_case_id: Optional[str]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class ImmuneLoop:
    """
    Immune Loop v1：异常检测 + 异常止损 + ImmuneCase 归档。

    使用方式：
    - 在 CognitiveLoop 每次尝试后调用：
      action = immune_loop.after_attempt(...)
    - 如果 action 非 None，则根据建议调整：降级、自编程开关、版本回滚等。
    """

    def __init__(
        self,
        *,
        repo_root: str | None = None,
        anomaly_detector: Optional[AnomalyDetector] = None,
        case_store: Optional[ImmuneCaseStore] = None,
        research_interface: Optional[ResearchInterface] = None,
    ) -> None:
        from pathlib import Path

        self.repo_root = Path(repo_root or ".").resolve()
        self.anomaly_detector = anomaly_detector or AnomalyDetector()
        self.case_store = case_store or ImmuneCaseStore(repo_root=self.repo_root)
        self.research_interface = research_interface or ResearchInterface()

    def after_attempt(
        self,
        *,
        goal: Any,
        metrics: Dict[str, Any],
        snapshot: MetricSnapshot,
        recent_snapshots: List[MetricSnapshot],
        recent_traces: List[Dict[str, Any]],
    ) -> Optional[ImmuneAction]:
        """
        每次尝试（一次 plan+execute 循环）结束后调用。

        - goal             : 当前目标
        - metrics          : 累计指标（total_attempts, error_attempts 等）
        - snapshot         : 当前这一轮的 MetricSnapshot
        - recent_snapshots : 最近几轮 snapshot（用于趋势分析）
        - recent_traces    : 最近几轮的 Trace 摘要，用 ImmuneCase 记录

        若返回 None：无异常；
        若返回 ImmuneAction：建议采取降级/回滚/关闭自编程等措施。
        """
        signal = self.anomaly_detector.analyze(
            goal=goal,
            metrics=metrics,
            snapshot=snapshot,
            recent_snapshots=recent_snapshots,
        )
        if signal is None:
            return None

        # 生成 ImmuneCase
        immune_case = self.case_store.append_case(
            goal=goal,
            anomaly=signal,
            metrics=metrics,
            recent_traces=recent_traces,
        )

        # 通知 Research Layer
        try:
            self.research_interface.submit_immune_case(immune_case)
        except Exception:
            logger.exception("ImmuneLoop: failed to submit immune case to ResearchInterface")

        # 从异常类型和严重级别中推导 ImmuneAction（简单 rule-based，后续可以学习优化）
        action = self._derive_action_from_signal(signal, immune_case)
        return action

    def _derive_action_from_signal(self, signal: AnomalySignal, case: ImmuneCase) -> ImmuneAction:
        degrade = False
        disable_self_prog = False
        revert = False
        notes_parts: List[str] = [signal.reason]

        if signal.anomaly_type == "high_failure_rate":
            degrade = True
            disable_self_prog = True
            notes_parts.append("High failure rate: suggest degrade mode and disable self-programming.")
        elif signal.anomaly_type == "high_cost":
            degrade = True
            notes_parts.append("High cost: suggest degrade mode and limit planning depth.")
        elif signal.anomaly_type == "high_latency":
            degrade = True
            notes_parts.append("High latency: suggest degrade mode.")

        # 根据严重级别进一步加码
        if signal.severity in ("high", "critical"):
            # v1 先简单设为需要考虑回滚；具体怎么回滚由外部控制
            revert = True
            notes_parts.append(f"Severity={signal.severity}: consider reverting to last stable version.")

        notes = " ".join(notes_parts)

        return ImmuneAction(
            degrade_mode=degrade,
            disable_self_programming=disable_self_prog,
            revert_version=revert,
            notes=notes,
            anomaly=signal.to_dict(),
            immune_case_id=case.case_id,
        )

    def run_goal(self, goal: "Goal", context: Dict[str, Any]) -> None:
        failure_count = 0
        metrics = {
            "total_attempts": 0,
            "success_attempts": 0,
            "error_attempts": 0,
            "cost_tokens": 0.0,
            "latency_ms": 0.0,
        }

        last_plan_dict: Dict[str, Any] = {}
        last_exec_result: Dict[str, Any] = {}
        recent_traces: List[Dict[str, Any]] = []
        recent_snapshots: List[MetricSnapshot] = []

        while True:
            metrics["total_attempts"] += 1

            # TODO: 这里可以记录每一轮开始时间，结束后计算 latency
            plan = self._run_planner(goal, context)
            last_plan_dict = {
                "steps": getattr(plan, "steps", []),
                "raw_output": getattr(plan, "raw_output", None),
            }

            exec_result = self._run_executor(goal, plan, context)
            last_exec_result = exec_result

            # 从 exec_result 提取简单指标
            if isinstance(exec_result, dict):
                success = bool(exec_result.get("success", False))
                step_records = exec_result.get("steps") or []
                # 这里先简单不算 tokens 和 latency，你后面可以接真实数据
                tokens_used = float(exec_result.get("meta", {}).get("tokens_used", 0.0)) if exec_result.get("meta") else 0.0
                latency_ms = float(exec_result.get("meta", {}).get("latency_ms", 0.0)) if exec_result.get("meta") else 0.0
            else:
                success = False
                step_records = []
                tokens_used = 0.0
                latency_ms = 0.0

            if success:
                metrics["success_attempts"] += 1
            else:
                metrics["error_attempts"] += 1
                failure_count += 1

            metrics["cost_tokens"] += tokens_used
            metrics["latency_ms"] = latency_ms  # 可以改成平均或最大值

            recent_traces.append(
                {
                    "attempt_index": metrics["total_attempts"],
                    "steps": step_records,
                }
            )
            if len(recent_traces) > 5:
                recent_traces = recent_traces[-5:]

            snapshot = MetricSnapshot(
                attempt_index=metrics["total_attempts"],
                total_attempts=metrics["total_attempts"],
                success_attempts=metrics["success_attempts"],
                error_attempts=metrics["error_attempts"],
                cost_tokens=metrics["cost_tokens"],
                latency_ms=metrics["latency_ms"],
                reward_score=0.0,  # TODO: 如果有 reward，可以填上
            )
            recent_snapshots.append(snapshot)
            if len(recent_snapshots) > 10:
                recent_snapshots = recent_snapshots[-10:]

            execution_summary = {
                "plan_steps": last_plan_dict.get("steps"),
                "executor_result": last_exec_result,
                "attempt_index": metrics["total_attempts"],
                "failure_count": failure_count,
            }

            # ✅ 先做 done 判定
            done_decision = self.done_council.assess(
                goal=goal,
                execution_summary=execution_summary,
                metrics=metrics,
                importance="normal",
            )
            if done_decision.is_done:
                logger.info(
                    "CognitiveLoop: goal %s considered DONE by done_council (%s)",
                    getattr(goal, "id", None),
                    done_decision.verdict,
                )
                break

            # ✅ ImmuneLoop：检测异常，可能改变 degrade_mode / self_programming_enabled
            immune_action = self.immune_loop.after_attempt(
                goal=goal,
                metrics=metrics,
                snapshot=snapshot,
                recent_snapshots=recent_snapshots,
                recent_traces=recent_traces,
            )
            if immune_action is not None:
                self._apply_immune_action(immune_action, goal)

            # ✅ 再看是否需要触发 reflection（多 AI 反省）
            if self._should_trigger_reflection(goal, metrics, failure_count):
                advice = self.reflection_council.reflect(
                    goal=goal,
                    last_plan=last_plan_dict,

六、这一阶段和蓝皮书的对齐度
到这一步，我们已经把蓝皮书中关于 Immune Loop + Anomaly Detection + Research 接口 的核心骨架落地：

✅ Immune Loop
