from __future__ import annotations

from typing import Dict, Any, List, Optional
from datetime import datetime
from enum import Enum
from pydantic import BaseModel, Field


class ExperimentStatus(str, Enum):
    draft = "draft"
    running = "running"
    completed = "completed"
    failed = "failed"
    cancelled = "cancelled"


class ExperimentRunStatus(str, Enum):
    pending = "pending"
    running = "running"
    succeeded = "succeeded"
    failed = "failed"


class VariantConfig(BaseModel):
    """
    一个“变体”的定义：
    - kind: 如 "model_router" / "planner" / "skill" / "generic"
    - params: 变体的超参数（交给具体模块去解释）
    """
    variant_id: str
    label: Optional[str] = None
    description: Optional[str] = None

    kind: str = "generic"
    params: Dict[str, Any] = Field(default_factory=dict)
    meta: Dict[str, Any] = Field(default_factory=dict)


class ExperimentConfig(BaseModel):
    """
    用来“申请创建实验”的配置：
    - 不含状态，只描述：评测方式 + 变体们 + 数据集等
    """
    name: str
    description: Optional[str] = None

    eval_kind: str = "offline"          # offline / online / custom
    dataset: Optional[str] = None       # 比如 "desktop_small", "legal_dev_set"

    variants: List[VariantConfig]

    meta: Dict[str, Any] = Field(default_factory=dict)

    # 可选指定 experiment_id，否则由系统生成
    experiment_id: Optional[str] = None


class ExperimentRunResult(BaseModel):
    """
    某一次 (experiment, variant) 组合的评测结果。
    """
    run_id: str
    experiment_id: str
    variant_id: str

    status: ExperimentRunStatus = ExperimentRunStatus.succeeded

    metrics: Dict[str, Any] = Field(default_factory=dict)
    error: Optional[str] = None

    started_at: datetime = Field(default_factory=datetime.utcnow)
    finished_at: Optional[datetime] = None

    meta: Dict[str, Any] = Field(default_factory=dict)


class ExperimentRecord(BaseModel):
    """
    持久化在 KV 里的实验记录：
    - = ExperimentConfig + 状态 + runs
    """
    experiment_id: str
    name: str
    description: Optional[str] = None

    eval_kind: str
    dataset: Optional[str] = None
    variants: List[VariantConfig]

    status: ExperimentStatus = ExperimentStatus.draft

    created_at: datetime = Field(default_factory=datetime.utcnow)
    updated_at: datetime = Field(default_factory=datetime.utcnow)

    runs: List[ExperimentRunResult] = Field(default_factory=list)

    meta: Dict[str, Any] = Field(default_factory=dict)

# ============================================
# quentin/experiments/registry.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass
from typing import List, Optional
from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.experiments.schemas import VariantConfig


@dataclass
class VariantRegistry:
    """
    变体注册表：
    - 注册不同的策略 / 模型路由 / Planner 版本
    - 实验时只用 variant_id 拿配置，不直接依赖具体模块
    """
    kv: KVStore
    tracer: Tracer

    prefix: str = "variant:config:"
    index_key: str = "variant:index:all"

    async def register(self, cfg: VariantConfig) -> VariantConfig:
        key = f"{self.prefix}{cfg.variant_id}"
        await self.kv.save(key, cfg.model_dump())
        await self.kv.append_list(self.index_key, cfg.variant_id)

        self.tracer.span(
            "L3",
            "variant_registered",
            module="variant_registry",
            input={"variant_id": cfg.variant_id, "kind": cfg.kind},
        )
        return cfg

    async def get(self, variant_id: str) -> Optional[VariantConfig]:
        data = await self.kv.get(f"{self.prefix}{variant_id}")
        if not data:
            return None
        return VariantConfig(**data)

    async def list(self, limit: int = 100) -> List[VariantConfig]:
        ids = await self.kv.get(self.index_key) or []
        ids = list(dict.fromkeys(ids))[-limit:]
        res: List[VariantConfig] = []
        for vid in ids:
            data = await self.kv.get(f"{self.prefix}{vid}")
            if data:
                res.append(VariantConfig(**data))
        return res

@dataclass
class EvalHarness:

    当前实现：
    - 不真正跑任务，只返回一个占位 metrics，保证实验框架能跑通。
    - 你可以在 run_variant 中接入任何评测逻辑。
    """
    tracer: Tracer
    kv: KVStore

    async def run_variant(
        self,
        exp: ExperimentRecord,
        variant: VariantConfig,
    ) -> ExperimentRunResult:
        """
        评测单个 variant。
        当前版本只是返回一个占位结果。
        """
        started = datetime.utcnow()

        # TODO：未来你可以在这里：
        # - 根据 exp.eval_kind 选择 offline / online 方案
        # - 对 exp.dataset 做离线评测
        # - 或者调用 runtime.cognitive_loop 跑一批任务

        metrics: Dict[str, Any] = {
            "score": 0.0,
            "notes": "EvalHarness v1 placeholder; please implement domain-specific eval logic.",
            "variant_label": variant.label or variant.variant_id,
            "variant_kind": variant.kind,
        }

        finished = datetime.utcnow()

        run = ExperimentRunResult(
            run_id=f"run_{uuid.uuid4().hex}",
            experiment_id=exp.experiment_id,
            variant_id=variant.variant_id,
            status=ExperimentRunStatus.succeeded,
            metrics=metrics,
            started_at=started,
            finished_at=finished,
        )

        self.tracer.span(
            "L3",
            "eval_variant",
            module="eval_harness",
            input={
                "experiment_id": exp.experiment_id,
                "variant_id": variant.variant_id,
            },
            output={"score": metrics["score"]},
        )

        return run

# ============================================
# quentin/experiments/orchestrator.py
# ============================================

from __future__ import annotations

from dataclasses import dataclass
from typing import List, Dict, Any, Optional
from datetime import datetime
import uuid

from quentin.infra.storage.kv_store import KVStore
from quentin.infra.logging import Tracer
from quentin.eval.harness import EvalHarness
from quentin.experiments.schemas import (
    ExperimentConfig,
    ExperimentRecord,
    ExperimentStatus,
    ExperimentRunResult,
    ExperimentRunStatus,
    VariantConfig,
)


@dataclass
class ExperimentOrchestrator:
    """
    Research & Evolution Layer v1 的核心：

    - create_experiment(config)
    - run_experiment(experiment_id)
    - compare_variants(experiment_id, metric="score")
    - export_results(experiment_id)

    当前只实现 offline、顺序跑所有 variants。
    """
    tracer: Tracer
    kv: KVStore
    eval_harness: EvalHarness

    exp_prefix: str = "exp:record:"
    exp_index_key: str = "exp:index:all"

    # ---------- KV 工具 ----------

    async def _save_record(self, record: ExperimentRecord) -> None:
        key = f"{self.exp_prefix}{record.experiment_id}"
        await self.kv.save(key, record.model_dump())
        await self.kv.append_list(self.exp_index_key, record.experiment_id)

    async def _load_record(self, experiment_id: str) -> Optional[ExperimentRecord]:
        data = await self.kv.get(f"{self.exp_prefix}{experiment_id}")
        if not data:
            return None
        return ExperimentRecord(**data)

    async def list_experiments(self, limit: int = 50) -> List[ExperimentRecord]:
        ids = await self.kv.get(self.exp_index_key) or []
        ids = list(dict.fromkeys(ids))[-limit:]
        res: List[ExperimentRecord] = []
        for eid in ids:
            data = await self.kv.get(f"{self.exp_prefix}{eid}")
            if data:
                res.append(ExperimentRecord(**data))
        return res

    async def get_experiment(self, experiment_id: str) -> Optional[ExperimentRecord]:
        return await self._load_record(experiment_id)

    # ---------- API：创建实验 ----------

    async def create_experiment(self, cfg: ExperimentConfig) -> ExperimentRecord:
        exp_id = cfg.experiment_id or f"exp_{uuid.uuid4().hex}"
        now = datetime.utcnow()
        record = ExperimentRecord(
            experiment_id=exp_id,
            name=cfg.name,
            description=cfg.description,
            eval_kind=cfg.eval_kind,
            dataset=cfg.dataset,
            variants=cfg.variants,
            status=ExperimentStatus.draft,
            created_at=now,
            updated_at=now,
            runs=[],
            meta=cfg.meta,
        )
        await self._save_record(record)

        self.tracer.span(
            "L3",
            "experiment_created",
            module="experiment_orchestrator",
            input={"experiment_id": exp_id, "name": cfg.name},
        )
        return record

    # ---------- API：运行实验 ----------

    async def run_experiment(self, experiment_id: str) -> Optional[ExperimentRecord]:
        record = await self._load_record(experiment_id)
        if not record:
            return None

        record.status = ExperimentStatus.running
        record.updated_at = datetime.utcnow()
        await self._save_record(record)

        runs: List[ExperimentRunResult] = []

        for variant in record.variants:
            try:
                run = await self.eval_harness.run_variant(record, variant)
            except Exception as e:
                run = ExperimentRunResult(
                    run_id=f"run_{uuid.uuid4().hex}",
                    experiment_id=record.experiment_id,
                    variant_id=variant.variant_id,
                    status=ExperimentRunStatus.failed,
                    error=repr(e),
                    metrics={},
                )
            runs.append(run)

        record.runs.extend(runs)
        record.updated_at = datetime.utcnow()

        # 决定整体状态
        if any(r.status == ExperimentRunStatus.failed for r in record.runs):
            record.status = ExperimentStatus.failed
        else:
            record.status = ExperimentStatus.completed

        await self._save_record(record)

        self.tracer.span(
            "L3",
            "experiment_run_completed",
            module="experiment_orchestrator",
            input={"experiment_id": experiment_id},
            output={
                "status": record.status.value,
                "runs": len(record.runs),
            },
        )

        return record

    # ---------- API：比较变体 ----------

    async def compare_variants(
        self,
        experiment_id: str,
        metric: str = "score",
        higher_is_better: bool = True,
    ) -> Optional[Dict[str, Any]]:
        record = await self._load_record(experiment_id)
        if not record or not record.runs:
            return None

        # 聚合每个 variant 的 score（当前以最后一次 run 为准）
        by_variant: Dict[str, ExperimentRunResult] = {}
        for r in record.runs:
            by_variant[r.variant_id] = r

        def get_score(r: ExperimentRunResult) -> float:
            val = r.metrics.get(metric, None)
            if isinstance(val, (int, float)):
                return float(val)
            return 0.0

        best_run: Optional[ExperimentRunResult] = None
        for r in by_variant.values():
            if not best_run:
                best_run = r
                continue
            if higher_is_better:
                if get_score(r) > get_score(best_run):
                    best_run = r
            else:
                if get_score(r) < get_score(best_run):
                    best_run = r

        if not best_run:
            return None

        return {
            "experiment_id": record.experiment_id,
            "metric": metric,
            "higher_is_better": higher_is_better,
            "best_variant_id": best_run.variant_id,
            "best_score": get_score(best_run),
            "best_run": best_run.model_dump(),
        }

    # ---------- API：导出完整结果 ----------

    async def export_results(self, experiment_id: str) -> Optional[Dict[str, Any]]:
        record = await self._load_record(experiment_id)
        if not record:
            return None

        compare = await self.compare_variants(experiment_id)

        return {
            "experiment": record.model_dump(),
            "compare": compare,
        }

    async def get_experiment_stats(self) -> Dict[str, Any]:
        orch = self.runtime.experiment_orchestrator

        return {
            "experiments": total,
            "completed": completed,
            "failed": failed,
            "last_experiment": {
                "experiment_id": last.experiment_id,
                "name": last.name,
                "status": last.status.value,
                "created_at": last.created_at.isoformat(),
