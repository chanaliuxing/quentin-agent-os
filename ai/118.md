from __future__ import annotations

from dataclasses import dataclass, asdict
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, Optional

import json

from quentin.self_programming.schema import (
    PatchProposal,
    PatchTarget,
    PatchTestPlan,
)
from quentin.traces.core import TraceRecorder, TraceEventType, new_trace_event


@dataclass
class PatchApplyLog:
    """
    记录一次 Patch 应用的元信息，用于审计和回滚。
    """
    proposal_id: str
    target_path: str             # 相对于 repo_root
    backup_path: str             # 备份文件路径（相对于 repo_root）
    applied_at: str
    dry_run: bool
    status: str                  # "applied" | "dry_run" | "failed"
    message: str
    meta: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


def _now_iso() -> str:
    return datetime.utcnow().isoformat()
2. PatchApplier：从 Proposal → 应用 / 回滚
class PatchApplier:
    """
    负责 Patch 的安全应用和回滚。

    关键点：
      - 只允许修改 repo_root/mutable/** 下的文件
      - 应用前必备份原文件到 data/patches/backups/
      - 默认 dry_run = True，只在 --apply 显式指定时才真的写文件
    """

    def __init__(self, repo_root: str | Path, trace_recorder: TraceRecorder) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.trace_recorder = trace_recorder

        self._patches_dir = self.repo_root / "data" / "patches"
        self._proposals_dir = self._patches_dir / "proposals"
        self._backups_dir = self._patches_dir / "backups"
        self._applied_dir = self._patches_dir / "applied"

        self._backups_dir.mkdir(parents=True, exist_ok=True)
        self._applied_dir.mkdir(parents=True, exist_ok=True)

    # ----------------------------
    # 对外：按 proposal_id 应用 patch
    # ----------------------------
    def apply_by_id(self, proposal_id: str, *, dry_run: bool = True) -> PatchApplyLog:
        proposal = self._load_proposal_by_id(proposal_id)
        if proposal is None:
            log = PatchApplyLog(
                proposal_id=proposal_id,
                target_path="",
                backup_path="",
                applied_at=_now_iso(),
                dry_run=dry_run,
                status="failed",
                message=f"Proposal {proposal_id} not found.",
                meta={},
            )
            self._record_apply_log(log)
            return log

        return self.apply_proposal(proposal, dry_run=dry_run)

    # ----------------------------
    # 对外：直接用 PatchProposal 应用
    # ----------------------------
    def apply_proposal(self, proposal: PatchProposal, *, dry_run: bool = True) -> PatchApplyLog:
        target_rel = Path(proposal.target.path)
        target_abs = self.repo_root / target_rel

        # 安全检查：必须在 mutable/ 下
        if not str(target_rel).startswith("mutable/"):
            msg = f"Refuse to patch non-mutable path: {target_rel}"
            log = PatchApplyLog(
                proposal_id=proposal.proposal_id,
                target_path=str(target_rel),
                backup_path="",
                applied_at=_now_iso(),
                dry_run=dry_run,
                status="failed",
                message=msg,
                meta={"reason": "outside_mutable_dir"},
            )
            self._record_apply_log(log)
            return log

        # 读取原内容（不存在时视为空文件）
        try:
            original_content = target_abs.read_text(encoding="utf-8") if target_abs.exists() else ""
        except Exception as e:
            msg = f"Failed to read target file: {e!r}"
            log = PatchApplyLog(
                proposal_id=proposal.proposal_id,
                target_path=str(target_rel),
                backup_path="",
                applied_at=_now_iso(),
                dry_run=dry_run,
                status="failed",
                message=msg,
                meta={},
            )
            self._record_apply_log(log)
            return log

        # 生成新内容（当前版本：简单策略）
        if proposal.patch_type == "full_file":
            new_content = proposal.patch_content
        elif proposal.patch_type == "snippet":
            # 非破坏性：先不替换，直接在文件末尾追加“候选实现”，由人/后续逻辑来整合
            new_content = (
                original_content
                + "\n\n"
                + "# ===== Candidate patch snippet (proposal: "
                + proposal.proposal_id
                + ") =====\n"
                + proposal.patch_content
                + "\n# ===== End of candidate patch snippet =====\n"
            )
        elif proposal.patch_type == "unified_diff":
            # TODO: 未来可以实现真正的 diff 应用，目前先返回失败占位
            msg = "unified_diff application not implemented yet."
            log = PatchApplyLog(
                proposal_id=proposal.proposal_id,
                target_path=str(target_rel),
                backup_path="",
                applied_at=_now_iso(),
                dry_run=dry_run,
                status="failed",
                message=msg,
                meta={},
            )
            self._record_apply_log(log)
            return log
        else:
            msg = f"Unknown patch_type: {proposal.patch_type}"
            log = PatchApplyLog(
                proposal_id=proposal.proposal_id,
                target_path=str(target_rel),
                backup_path="",
                applied_at=_now_iso(),
                dry_run=dry_run,
                status="failed",
                message=msg,
                meta={},
            )
            self._record_apply_log(log)
            return log

        # 如果是 dry_run，只返回 log，不写文件
        if dry_run:
            msg = "Dry run only. No file was modified."
            log = PatchApplyLog(
                proposal_id=proposal.proposal_id,
                target_path=str(target_rel),
                backup_path="",
                applied_at=_now_iso(),
                dry_run=True,
                status="dry_run",
                message=msg,
                meta={"new_content_preview": new_content[:400]},
            )
            self._record_apply_log(log)
            return log

        # 真正写入：先备份，再写文件
        backup_rel, backup_abs = self._backup_file(target_rel, original_content)

        try:
            target_abs.parent.mkdir(parents=True, exist_ok=True)
            target_abs.write_text(new_content, encoding="utf-8")
        except Exception as e:
            msg = f"Failed to write patched file: {e!r}"
            log = PatchApplyLog(
                proposal_id=proposal.proposal_id,
                target_path=str(target_rel),
                backup_path=str(backup_rel),
                applied_at=_now_iso(),
                dry_run=False,
                status="failed",
                message=msg,
                meta={},
            )
            self._record_apply_log(log)
            return log

        msg = "Patch applied successfully."
        log = PatchApplyLog(
            proposal_id=proposal.proposal_id,
            target_path=str(target_rel),
            backup_path=str(backup_rel),
            applied_at=_now_iso(),
            dry_run=False,
            status="applied",
            message=msg,
            meta={},
        )
        self._record_apply_log(log)
        return log

    # ----------------------------
    # 回滚：按 proposal_id
    # ----------------------------
    def rollback_by_proposal(self, proposal_id: str) -> PatchApplyLog:
        """
        根据 proposal_id 查找最近一次 apply 日志，然后用对应备份回滚。
        """
        log_path = self._applied_dir / f"{proposal_id}.jsonl"
        if not log_path.exists():
            log = PatchApplyLog(
                proposal_id=proposal_id,
                target_path="",
                backup_path="",
                applied_at=_now_iso(),
                dry_run=False,
                status="failed",
                message="No apply log found for this proposal.",
                meta={},
            )
            self._record_apply_log(log)
            return log

        # 使用最后一行作为最近一次 apply
        last_line = None
        with log_path.open("r", encoding="utf-8") as f:
            for line in f:
                last_line = line.strip()
        if not last_line:
            log = PatchApplyLog(
                proposal_id=proposal_id,
                target_path="",
                backup_path="",
                applied_at=_now_iso(),
                dry_run=False,
                status="failed",
                message="Apply log file is empty.",
                meta={},
            )
            self._record_apply_log(log)
            return log

        data = json.loads(last_line)
        target_rel = Path(data["target_path"])
        backup_rel = Path(data["backup_path"])

        target_abs = self.repo_root / target_rel
        backup_abs = self.repo_root / backup_rel

        if not backup_abs.exists():
            log = PatchApplyLog(
                proposal_id=proposal_id,
                target_path=str(target_rel),
                backup_path=str(backup_rel),
                applied_at=_now_iso(),
                dry_run=False,
                status="failed",
                message="Backup file not found.",
                meta={},
            )
            self._record_apply_log(log)
            return log

        # 回滚：用备份覆盖当前文件
        try:
            content = backup_abs.read_text(encoding="utf-8")
            target_abs.parent.mkdir(parents=True, exist_ok=True)
            target_abs.write_text(content, encoding="utf-8")
        except Exception as e:
            log = PatchApplyLog(
                proposal_id=proposal_id,
                target_path=str(target_rel),
                backup_path=str(backup_rel),
                applied_at=_now_iso(),
                dry_run=False,
                status="failed",
                message=f"Failed to rollback: {e!r}",
                meta={},
            )
            self._record_apply_log(log)
            return log

        log = PatchApplyLog(
            proposal_id=proposal_id,
            target_path=str(target_rel),
            backup_path=str(backup_rel),
            applied_at=_now_iso(),
            dry_run=False,
            status="applied",
            message="Rollback succeeded.",
            meta={"rollback": True},
        )
        self._record_apply_log(log)
        return log

    # ----------------------------
    # 内部：读 Proposal
    # ----------------------------
    def _load_proposal_by_id(self, proposal_id: str) -> Optional[PatchProposal]:
        path = self._proposals_dir / f"{proposal_id}.json"
        if not path.exists():
            return None
        with path.open("r", encoding="utf-8") as f:
            data = json.load(f)

        # 手动重建 dataclass
        target = PatchTarget(**data["target"])
        test_plan = PatchTestPlan(
            **data.get("test_plan", {}),
        )
        proposal = PatchProposal(
            proposal_id=data["proposal_id"],
            created_at=data["created_at"],
            created_by=data["created_by"],
            target=target,
            rationale=data["rationale"],
            patch_type=data["patch_type"],
            patch_content=data["patch_content"],
            test_plan=test_plan,
            meta=data.get("meta", {}),
        )
        return proposal

    # ----------------------------
    # 内部：备份和日志
    # ----------------------------
    def _backup_file(self, target_rel: Path, original_content: str) -> tuple[Path, Path]:
        ts = datetime.utcnow().strftime("%Y%m%dT%H%M%S")
        backup_rel = Path("backups") / f"{target_rel.as_posix().replace('/', '_')}.{ts}.bak"
        backup_abs = self._backups_dir / backup_rel
        backup_abs.parent.mkdir(parents=True, exist_ok=True)
        backup_abs.write_text(original_content, encoding="utf-8")
        return backup_rel, backup_abs

    def _record_apply_log(self, log: PatchApplyLog) -> None:
        """
        写入：
          - data/patches/applied/{proposal_id}.jsonl
          - TraceEvent (SELF_PROGRAMMING)
        """
        # JSONL
        path = self._applied_dir / f"{log.proposal_id}.jsonl"
        tmp = path.with_suffix(".tmp")
        with tmp.open("a", encoding="utf-8") as f:
            f.write(json.dumps(log.to_dict(), ensure_ascii=False) + "\n")
        if path.exists():
            # 追加写时，用简单方式合并：先读旧，再写新（这里为了简单，直接替换 tmp 为 path）
            pass
        tmp.replace(path)

        # Trace
        self.trace_recorder.record_event(
            new_trace_event(
                trace_id="self-programming",
                event_type=TraceEventType.SELF_PROGRAMMING,
                payload={
                    "patch_apply_log": log.to_dict(),
                },
            )
        )

# scripts/apply_patch.py
"""
按 proposal_id 应用 Patch。

用法（仓库根目录）：
  # 只预览（dry-run），不写文件：
  python -m scripts.apply_patch --proposal-id <ID>

  # 真正应用：
  python -m scripts.apply_patch --proposal-id <ID> --apply
"""

from __future__ import annotations

import argparse
from pathlib import Path

from quentin.traces.core import JsonlTraceRecorder
from quentin.self_programming.apply import PatchApplier


def main() -> None:
    parser = argparse.ArgumentParser(description="Apply a patch proposal safely.")
    parser.add_argument("--proposal-id", required=True, help="Patch proposal_id to apply.")
    parser.add_argument(
        "--repo-root",
        type=str,
        default=".",
        help="Repository root (default: current directory).",
    )
    parser.add_argument(
        "--apply",
        action="store_true",
        help="Actually modify files. If omitted, only do a dry-run.",
    )
    args = parser.parse_args()

    repo_root = Path(args.repo_root).resolve()
    recorder = JsonlTraceRecorder(repo_root=repo_root)
    applier = PatchApplier(repo_root=repo_root, trace_recorder=recorder)

    log = applier.apply_by_id(args.proposal_id, dry_run=not args.apply)

    print("status     :", log.status)
    print("message    :", log.message)
    print("proposal   :", log.proposal_id)
    print("target     :", log.target_path)
    print("backup     :", log.backup_path)
    print("dry_run    :", log.dry_run)
    print("applied_at :", log.applied_at)


if __name__ == "__main__":
    main()
