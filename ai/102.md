# quentin/diagnostics/dashboard.py
from __future__ import annotations

from dataclasses import dataclass, asdict, is_dataclass
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List, Optional
import json
import logging

logger = logging.getLogger(__name__)


@dataclass
class DashboardSnapshot:
    """
    系统状态快照，用于“仪表盘”展示。
    """
    timestamp: str
    blueprint_summary: Dict[str, Any]
    immune_summary: Dict[str, Any]
    patch_summary: Dict[str, Any]
    notes: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


def _get_patch_status(obj: Any) -> Optional[str]:
    """
    兼容：
    - PatchTestReport dataclass
    - dict（from to_dict()）
    - 其它含 status 属性的对象
    """
    if isinstance(obj, dict):
        return obj.get("status")
    if is_dataclass(obj):
        return getattr(obj, "status", None)
    return getattr(obj, "status", None)


def build_dashboard_snapshot(
    *,
    blueprint_report: Optional[Dict[str, Any]],
    immune_result: Optional[Dict[str, Any]],
    patch_reports: List[Any],
    notes: Optional[Dict[str, Any]] = None,
) -> DashboardSnapshot:
    """
    根据当前一轮的蓝皮书报告 / 免疫结果 / patch 测试结果，构造 DashboardSnapshot。
    """
    ts = datetime.utcnow().isoformat()

    # 1) 蓝皮书部分
    if blueprint_report is None:
        bp_summary = {
            "gap_count": 0,
            "missing_files": 0,
            "stub_implementations": 0,
            "spec_version": "unknown",
        }
    else:
        summary = blueprint_report.get("summary") or {}
        bp_summary = {
            "gap_count": summary.get("gap_count", 0),
            "missing_files": summary.get("missing_files", 0),
            "stub_implementations": summary.get("stub_implementations", 0),
            "spec_version": blueprint_report.get("spec_version", "unknown"),
        }

    # 2) 免疫循环部分
    if immune_result is None:
        im_summary = {
            "cases_in_last_scan": 0,
        }
    else:
        im_summary = {
            "cases_in_last_scan": immune_result.get("count", 0),
        }

    # 3) patch 测试部分
    num_patches = len(patch_reports)
    num_passed = sum(1 for r in patch_reports if _get_patch_status(r) == "passed")
    num_failed = sum(1 for r in patch_reports if _get_patch_status(r) == "failed")
    num_error = sum(1 for r in patch_reports if _get_patch_status(r) == "error")

    patch_summary = {
        "num_patches": num_patches,
        "num_passed": num_passed,
        "num_failed": num_failed,
        "num_error": num_error,
    }

    snapshot = DashboardSnapshot(
        timestamp=ts,
        blueprint_summary=bp_summary,
        immune_summary=im_summary,
        patch_summary=patch_summary,
        notes=notes or {},
    )
    return snapshot


def write_dashboard_snapshot(
    *,
    repo_root: str | Path,
    snapshot: DashboardSnapshot,
    filename: str = "data/dashboard/latest_snapshot.json",
) -> Path:
    """
    把 DashboardSnapshot 写成 JSON 文件。

    默认路径：<repo_root>/data/dashboard/latest_snapshot.json
    """
    repo_root = Path(repo_root).resolve()
    out_path = repo_root / filename
    out_path.parent.mkdir(parents=True, exist_ok=True)

    text = json.dumps(snapshot.to_dict(), ensure_ascii=False, indent=2)
    out_path.write_text(text, encoding="utf-8")

    logger.info("Dashboard snapshot written to %s", out_path)
    return out_path

def setup_logging(level: str = "INFO") -> None:
    logging.basicConfig(
        level=getattr(logging, level.upper(), logging.INFO),

def build_runtime(repo_root: Path) -> MetaAgentRuntime:
    """
    使用本地 Ollama 模型构建一个最简 MetaAgentRuntime。
    - 用 qwen3-vl:latest 作为主模型
    - 其他模型作为 council 成员
    """
    # 主模型（规划 / 反思 / aggregator）
    qwen_main = OllamaLLM(model="qwen3-vl:latest")

    # 其他成员模型（可按需精简/扩展）
    deepseek_small = OllamaLLM(model="deepseek-r1:1.5b")
    qwen_vision = OllamaLLM(model="qwen2.5vl:latest")
    uncensored = OllamaLLM(model="my-deepseek-uncensored:latest")

    members = [
        ("qwen_main", qwen_main),
        ("deepseek_small", deepseek_small),
        ("qwen_vision", qwen_vision),
        ("uncensored", uncensored),
    ]

    council = MultiModelCouncil(
        members=members,
        aggregator=qwen_main,
    )

    orchestrator = SimpleOrchestrator(
        llm=qwen_main,
        council=council,
    )

    runtime = MetaAgentRuntime(
        repo_root=str(repo_root),
        orchestrator=orchestrator,
        council=council,
    )
    return runtime

三、落地 SleepLoop：记忆巩固 + 调用 SkillEvolutionLoop
接下来是 “睡眠循环（Sleep Loop）” 的落地骨架。它做到：

从 trace 列表里挑出成功的任务

做一个简单的 STS/MTS/LTS 候选统计（真正的 MemoryFabric 合并你以后慢慢接）

调用 SkillEvolutionLoop.run_once() 把成功 Trace + issues（来自 ImmuneLoop / Blueprint / 其它）丢进去

# quentin/loops/sleep.py
from __future__ import annotations

from dataclasses import asdict, is_dataclass
from typing import Any, Dict, List, Optional
import logging

from quentin.evolution.skill_loop import SkillEvolutionLoop

logger = logging.getLogger(__name__)


def _trace_to_dict(t: Any) -> Dict[str, Any]:
    """
    兼容：
    - ExecutionTrace dataclass
    - dict
    - 其它对象（作为 raw 包一层）
    """
    if isinstance(t, dict):
        return t
    if is_dataclass(t):
        return asdict(t)
    return {"raw": t}


class SleepLoop:
    """
    睡眠循环（Sleep / Consolidation Loop）v1 实现。

    职责：
      1. 聚合最近一段时间的 ExecutionTrace
      2. 统计成功 / 失败等概况
      3. 为 MemoryFabric 准备“巩固候选”（先以摘要形式给出）
      4. 调用 SkillEvolutionLoop（成功 Trace + issues）做技能/代码进化

    这里刻意保持“研究友好”：
      - 不强绑定任何具体 MemoryFabric API，只是约定性地调用 consolidate_from_traces()
      - 所有输入输出都是结构化 dict，方便以后接不同算法
    """

    def __init__(
        self,
        *,
        memory_fabric: Optional[Any] = None,
        skill_evolution_loop: Optional[SkillEvolutionLoop] = None,
        max_traces: int = 500,
    ) -> None:
        self.memory_fabric = memory_fabric
        self.skill_evolution_loop = skill_evolution_loop
        self.max_traces = max_traces

    def run_once(
        self,
        *,
        recent_traces: List[Any],
        issues_from_immune: Optional[List[Dict[str, Any]]] = None,
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        由 MetaAgentRuntime 调用：

        - recent_traces: 最近一段时间的 ExecutionTrace 列表（新在前或新在后均可）
        - issues_from_immune: 来自 ImmuneLoop 的问题列表（可为空）
        - extra_context: 额外信息（如运行批次 ID 等）

        返回：
        {
          "trace_stats": {...},
          "memory_consolidation": {...},
          "skill_evolution": {...},
        }
        """
        ctx = extra_context or {}
        issues = issues_from_immune or []

        # 1) 预处理 trace（截断到 max_traces）
        if len(recent_traces) > self.max_traces:
            # 假设 recent_traces 是按时间排序的：我们保留最新的 max_traces
            recent_traces = recent_traces[-self.max_traces :]

        trace_dicts = [_trace_to_dict(t) for t in recent_traces]

        # 2) 统计成功 / 失败 / error 概况
        stats = self._summarize_traces(trace_dicts)

        # 3) MemoryFabric 巩固（如果有）
        mem_result: Dict[str, Any] = {}
        if self.memory_fabric is not None and hasattr(self.memory_fabric, "consolidate_from_traces"):
            try:
                mem_result = self.memory_fabric.consolidate_from_traces(
                    trace_dicts,
                    context={"source": "sleep_loop", **ctx},
                )
            except Exception:
                logger.exception("SleepLoop: memory_fabric.consolidate_from_traces failed")
                mem_result = {"status": "error"}

        # 4) Skill & Code Evolution（如果有）
        se_result: Dict[str, Any] = {}
        if self.skill_evolution_loop is not None:
            successful = [t for t in trace_dicts if t.get("status") == "success"]
            try:
                se_result = self.skill_evolution_loop.run_once(
                    successful_traces=successful,
                    issues=issues,
                    test_suites=["unit", "static_analysis"],
                    extra_context={"source": "sleep_loop", **ctx},
                )
            except Exception:
                logger.exception("SleepLoop: skill_evolution_loop.run_once failed")
                se_result = {"status": "error"}

        return {
            "trace_stats": stats,
            "memory_consolidation": mem_result,
            "skill_evolution": se_result,
        }

    # --------------------------------------------------
    # 内部：Trace 概况统计
    # --------------------------------------------------
    def _summarize_traces(self, traces: List[Dict[str, Any]]) -> Dict[str, Any]:
        total = len(traces)
        success = sum(1 for t in traces if t.get("status") == "success")
        failed = sum(1 for t in traces if t.get("status") == "failed")
        error = sum(1 for t in traces if t.get("status") == "error")
        unknown = total - success - failed - error

        domains: Dict[str, int] = {}
        for t in traces:
            d = t.get("domain", "unknown")
            domains[d] = domains.get(d, 0) + 1

        return {
            "total_traces": total,
            "success": success,
            "failed": failed,
            "error": error,
            "unknown": unknown,
            "by_domain": domains,
        }

from __future__ import annotations

from dataclasses import dataclass, asdict, is_dataclass
from typing import Any, Dict, List, Optional
import logging
import uuid

logger = logging.getLogger(__name__)


def _trace_to_dict(t: Any) -> Dict[str, Any]:
    if isinstance(t, dict):
        return t
    if is_dataclass(t):
        return asdict(t)
    return {"raw": t}


@dataclass
class ImmuneCase:
    """
    免疫系统识别出的一个异常案例。

    - trace_id: 关联的 ExecutionTrace ID（如果有的话）
    - reason: 简短文本，说明异常原因
    - severity: "low" | "medium" | "high" | "critical"
    - trace_snapshot: 直接把 trace 的关键字段打包，方便后续分析
    - meta: 任意扩展信息（如发生时间、domain、error_type 等）
    """
    case_id: str
    trace_id: Optional[str]
    reason: str
    severity: str
    trace_snapshot: Dict[str, Any]
    meta: Dict[str, Any]

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class ImmuneLoop:
    """
    免疫循环（Immune Loop）v1 实现。

    目标：
      - 从近期 Trace 中发现异常（error / failed / cost 过高等）
      - 生成结构化 ImmuneCase 列表
      - 可选：转成自编程引擎可用的 issue 列表

    v1 的检测策略很简单：
      - status == "error" → severity = "critical"
      - status == "failed" → severity = "medium"
      - 如果 top_level_metrics.cost > cost_threshold → severity 至少 "medium"
    """

    def __init__(
        self,
        *,
        cost_threshold: float = 1e4,
    ) -> None:
        self.cost_threshold = cost_threshold

    def run_once(
        self,
        *,
        recent_traces: List[Any],
        extra_context: Optional[Dict[str, Any]] = None,
    ) -> Dict[str, Any]:
        """
        由 MetaAgentRuntime 调用。

        返回：
        {
          "count": int,
          "cases": [ImmuneCase.as_dict(), ...],
        }
        """
        ctx = extra_context or {}
        trace_dicts = [_trace_to_dict(t) for t in recent_traces]

        cases: List[ImmuneCase] = []
        for t in trace_dicts:
            status = t.get("status")
            if status in ("success", None):
                # 成功或未知，先跳过
                continue

            trace_id = t.get("trace_id") or t.get("id") or None
            domain = t.get("domain", "unknown")
            metrics = t.get("top_level_metrics") or {}

            severity = "medium"
            reason_parts = []

            if status == "error":
                severity = "critical"
                reason_parts.append("trace_status=error")
            elif status == "failed":
                severity = "medium"
                reason_parts.append("trace_status=failed")
            else:
                severity = "low"
                reason_parts.append(f"trace_status={status}")

            cost = metrics.get("cost") or metrics.get("token_cost") or 0.0
            if isinstance(cost, (int, float)) and cost > self.cost_threshold:
                reason_parts.append(f"cost>{self.cost_threshold}")
                if severity != "critical":
                    severity = "high"

            reason = ", ".join(reason_parts) if reason_parts else "abnormal_trace"

            case = ImmuneCase(
                case_id=f"immune::{uuid.uuid4().hex}",
                trace_id=trace_id,
                reason=reason,
                severity=severity,
                trace_snapshot=t,
                meta={"domain": domain, "context": ctx},
            )
            cases.append(case)

        return {
            "count": len(cases),
            "cases": [c.to_dict() for c in cases],
        }

    # ---------------------------------------------
    # ImmuneCase → issue（给自编程引擎用）
    # ---------------------------------------------
    def cases_to_issues(self, cases: List[ImmuneCase]) -> List[Dict[str, Any]]:
        """
        将 ImmuneCase 转换为 SelfProgrammingEngine 可消费的 issue 结构。

        这里默认粒度是“模块级”：
          - target_type: "runtime_component"
          - identifier: trace_snapshot.domain
          - 我们暂时无法精确到具体文件/函数，但可以作为高层 issue 交给 ResearchLayer。
        """
        issues: List[Dict[str, Any]] = []
        for c in cases:
            domain = c.meta.get("domain", "unknown")

            target = {
                "target_type": "runtime_component",
                "identifier": domain,
                "file_path": None,
                "symbol_name": None,
                "meta": {
                    "trace_id": c.trace_id,
                    "case_id": c.case_id,
                    "reason": c.reason,
                },
            }

            issue = {
                "issue_id": c.case_id,
                "source": "immune_loop",
                "target": target,
                "reason": c.reason,
                "metrics": {
                    "severity": c.severity,
                },
            }
            issues.append(issue)

        return issues
