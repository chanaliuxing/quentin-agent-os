# quentin/skills/manager.py
from __future__ import annotations

from dataclasses import dataclass, asdict
from datetime import datetime
from typing import Any, Dict, List, Optional

import json
import logging

from quentin.skills.registry import FileBackedSkillRegistry
from quentin.skills.models import SkillDefinition
from quentin.llm.multi_council import MultiAICouncil, CouncilDecision
from quentin.llm.types import LLMConfig

logger = logging.getLogger(__name__)


@dataclass
class SkillRunResult:
    success: bool
    output: Any
    raw_answer: str
    used_skill_id: str
    used_model: Optional[str]
    error: Optional[str] = None
    started_at: str = datetime.utcnow().isoformat()
    finished_at: Optional[str] = None

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


class SkillManager:
    """
    运行时技能管理器：

    - 从 SkillRegistry 里查找“适合某个任务/子任务”的技能（按 domain + stats 简单排序）
    - 用 MultiAICouncil 通过本地模型执行技能（基于代码/伪代码 + 输入）
    - 记录一个最小的运行结果（成功/失败 + 输出）
    """

    def __init__(
        self,
        *,
        registry: Optional[FileBackedSkillRegistry] = None,
        council: Optional[MultiAICouncil] = None,
    ) -> None:
        self.registry = registry or FileBackedSkillRegistry()

        if council is None:
            default_members = [
                LLMConfig(
                    provider="ollama",
                    model="qwen3-vl:latest",
                    role="skill_executor",
                ),
                LLMConfig(
                    provider="ollama",
                    model="deepseek-r1:1.5b",
                    role="skill_checker",
                ),
            ]
            council = MultiAICouncil(members=default_members)

        self.council = council

    # ---------- 选技能 ----------

    def suggest_for_task(
        self,
        *,
        goal: str,
        domain: str,
        top_k: int = 3,
    ) -> List[SkillDefinition]:
        """
        非智能选择：按 domain 匹配 + 成功率排序。

        - 先选 domain 完全匹配的技能
        - 不够的话再补 general / cross 域技能
        """
        skills = self.registry.list_skills()
        if not skills:
            return []

        def score(s: SkillDefinition) -> float:
            stats = s.stats
            if stats.call_count <= 0:
                return 0.0
            return stats.success_count / max(stats.call_count, 1)

        domain_exact: List[SkillDefinition] = []
        domain_fallback: List[SkillDefinition] = []

        for s in skills:
            if s.domain == domain:
                domain_exact.append(s)
            elif s.domain in ("general", "cross"):
                domain_fallback.append(s)

        domain_exact.sort(key=score, reverse=True)
        domain_fallback.sort(key=score, reverse=True)

        merged: List[SkillDefinition] = domain_exact + domain_fallback
        return merged[:top_k]

    # ---------- 执行技能（通过多 AI） ----------

    def run_skill(
        self,
        skill: SkillDefinition,
        *,
        input_payload: Dict[str, Any],
        expect_json: bool = True,
    ) -> SkillRunResult:
        """
        v1 不直接执行 Python 代码，而是通过多模型协作来“解释执行” skill.code：

        - 给出：技能描述 + 伪代码/代码 + 输入 JSON
        - Council 负责：
            * skill_executor 先给出执行结果建议
            * skill_checker 检查合理性并修正
        - 返回：raw_answer + 尝试解析后的 output
        """
        started = datetime.utcnow().isoformat()

        question = (
            "You are executing a SKILL for a self-improving AI system.\n"
            "The skill is described below. Follow its intent and pseudo-code.\n"
            "You MUST focus on producing the final result of applying this skill "
            "to the given JSON input.\n\n"
            "If expect_json=true, you MUST output ONLY a single valid JSON object.\n"
        )

        context_parts = [
            f"SKILL_NAME: {skill.name}",
            f"DOMAIN: {skill.domain}",
            "DESCRIPTION:",
            skill.description,
            "",
            "PSEUDO_CODE_OR_IMPLEMENTATION:",
            skill.code[:4000],  # 防止太长
            "",
            "INPUT_JSON:",
            json.dumps(input_payload, ensure_ascii=False, indent=2),
        ]
        context = "\n".join(context_parts)

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="skill_execution",
        )

        raw = decision.final_answer or ""
        used_model = decision.chosen_model

        success = True
        parsed: Any = raw
        error: Optional[str] = None

        if expect_json:
            try:
                # 尝试从文本里提取 JSON（宽松一点，可以先找第一个 { 开头）
                json_str = raw.strip()
                first = json_str.find("{")
                last = json_str.rfind("}")
                if first != -1 and last != -1 and last > first:
                    json_str = json_str[first : last + 1]
                parsed = json.loads(json_str)
            except Exception as e:
                success = False
                error = f"Failed to parse JSON output: {e}"

        finished = datetime.utcnow().isoformat()

        return SkillRunResult(
            success=success,
            output=parsed,
            raw_answer=raw,
            used_skill_id=skill.skill_id,
            used_model=used_model,
            error=error,
            started_at=started,
            finished_at=finished,
        )
这里故意 不直接执行字符串代码，而是用「多 AI 解释执行」，符合你「多 AI 开会避免单一 LLM 幻觉」的方向。
以后你想改成真 Python 执行，也可以在这个接口里替换实现。

4. 把技能信息喂给 Planner（上下文增强）
在你构建 planner 上下文的地方（通常是调用 self.council.deliberate 前），把技能摘要拼进去。举个补丁式写法：

假设你原来有类似代码：

        planning_question = "You are the planner for a self-improving agent..."
        planning_context = self._build_planning_context(spec, memory_snapshot)

        plan_decision = self.council.deliberate(
            question=planning_question,
            context=planning_context,
            mode="planning",
        )
可以改成：

        planning_question = "You are the planner for a self-improving agent..."

        # 增强上下文：加入可用技能摘要
        skills_section = ""
        if suggested_skills:
            lines = ["Available SKILLS you may reuse instead of re-inventing steps:"]
            for s in suggested_skills:
                lines.append(
                    f"- {s.name} (id={s.skill_id}, domain={s.domain}, tags={','.join(s.tags)})"
                )
                lines.append(f"  desc: {s.description[:200]}...")
            skills_section = "\n".join(lines)

        base_context = self._build_planning_context(spec, memory_snapshot)
        if skills_section:
            planning_context = base_context + "\n\n" + skills_section
        else:
            planning_context = base_context

        plan_decision = self.council.deliberate(
            question=planning_question,
            context=planning_context,
            mode="planning",
        )
这样 Planner 在做 plan 的时候已经知道「有哪些技能可用」，可以自己选择在步骤里写：

“Step 1: call skill <skill_name> with input {...}”

后面你再往执行器里加对这种步骤的解析 & 真正 run_skill 即可。

def cmd_skill_run(args: argparse.Namespace) -> None:
    registry = FileBackedSkillRegistry()
    skill = registry.get_skill(args.skill_id)
    if not skill:
        print(f"Skill not found: {args.skill_id}")
        return

    try:
        input_payload = json.loads(args.input_json)
    except Exception as e:
        print(f"Failed to parse input-json: {e}")
        return

    manager = SkillManager()
    res = manager.run_skill(
        skill,
        input_payload=input_payload,
        expect_json=not args.text_output,
    )

    print(f"SkillRunResult for skill_id={res.used_skill_id}")
    print(f"- success : {res.success}")
    print(f"- used_model : {res.used_model}")
    if res.error:
        print(f"- error   : {res.error}")

    print("\n=== RAW ANSWER ===\n")
    print(res.raw_answer)

    if res.success:
        print("\n=== PARSED OUTPUT ===\n")
        print(json.dumps(res.output, ensure_ascii=False, indent=2))
