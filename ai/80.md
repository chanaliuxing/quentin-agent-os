# quentin/core/cognitive_loop.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from typing import Any, Dict, List, Optional, Callable
from datetime import datetime
from pathlib import Path
import logging
import uuid

from quentin.llm.router import ModelRouter
from quentin.llm.client import call_llm
from quentin.goals.done_definition import (
    GoalDoneManager,
    GoalSpec,
    DoneCheckContext,
    DoneCheckResult,
)
from quentin.core.traces import ExecutionTrace, ExecutionTraceStore
from quentin.memory.fabric import MemoryFabric  # 假设已有 v1，实现可先是占位

logger = logging.getLogger(__name__)


# ---- 类型别名：Planner / Executor 接口 ----

PlannerFn = Callable[["CognitiveTask", GoalSpec], str]
"""
PlannerFn:
  输入: task + goal
  输出: plan_summary (自然语言，描述接下来要做什么)
"""

ExecutorFn = Callable[["CognitiveTask", GoalSpec, str], Dict[str, Any]]
"""
ExecutorFn:
  输入: task + goal + plan_summary
  输出: dict:
    {
      "final_output": str,        # 给上层/用户的最终输出
      "actions_summary": str,     # 执行过程摘要（日志/工具调用概要）
      "raw": Any                  # 可选: 原始结果结构
    }
"""


# ---- 数据结构：任务与结果 ----

@dataclass
class CognitiveTask:
    """
    一次认知任务的输入：

    - task_id          : 唯一 ID（若未提供，系统自动生成）
    - input_text       : 自然语言任务描述（或 prompt）
    - domain           : 领域标签（core/desktop/legal/trading/...）
    - environment      : sandbox/staging/prod
    - goal_title       : 可选，若为空则用 input_text 截断生成
    - goal_description : 可选，若为空则使用 input_text
    - success_criteria : 成功标准列表
    - hard_constraints : 硬约束（不能违反）
    - soft_preferences : 软偏好
    - metadata         : 任意附加信息（调用方可用）
    """

    input_text: str
    task_id: str = field(default_factory=lambda: str(uuid.uuid4()))
    domain: str = "core"
    environment: str = "sandbox"
    goal_title: Optional[str] = None
    goal_description: Optional[str] = None
    success_criteria: List[str] = field(default_factory=list)
    hard_constraints: List[str] = field(default_factory=list)
    soft_preferences: List[str] = field(default_factory=list)
    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_goal_spec(self) -> Dict[str, Any]:
        """
        生成用于创建 GoalSpec 的参数字典。
        """
        title = self.goal_title or self._auto_title()
        desc = self.goal_description or self.input_text
        return {
            "title": title,
            "description": desc,
            "domain": self.domain,
            "environment": self.environment,
            "success_criteria": self.success_criteria,
            "hard_constraints": self.hard_constraints,
            "soft_preferences": self.soft_preferences,
            "created_by": "agent",
            "parent_task_id": self.task_id,
            "metadata": self.metadata,
        }

    def _auto_title(self, max_len: int = 80) -> str:
        s = (self.goal_title or "").strip()
        if not s:
            s = (self.input_text or "").strip()
        if len(s) > max_len:
            s = s[: max_len - 3] + "..."
        return s or "Unnamed Cognitive Task"


@dataclass
class CognitiveResult:
    """
    一次认知任务的完整结果快照：

    - task_id              : 对应任务 ID
    - goal_id              : 对应 GoalSpec ID
    - final_output         : 给上层/用户的最终输出
    - plan_summary         : Planner 生成的计划摘要
    - actions_summary      : Executor 执行过程摘要
    - done_result          : DoneCouncil 的判断结果
    - done                : 是否完成
    - done_confidence      : 完成置信度
    - raw_executor_result  : 执行器的原始结果结构（可选）
    - created_at           : 结果时间戳
    """

    task_id: str
    goal_id: str
    final_output: str
    plan_summary: str
    actions_summary: str
    done_result: DoneCheckResult
    done: bool
    done_confidence: float
    raw_executor_result: Any = None
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> Dict[str, Any]:
        return {
            "task_id": self.task_id,
            "goal_id": self.goal_id,
            "final_output": self.final_output,
            "plan_summary": self.plan_summary,
            "actions_summary": self.actions_summary,
            "done": self.done,
            "done_confidence": self.done_confidence,
            "done_result": self.done_result.to_dict(),
            "created_at": self.created_at,
        }


# ---- CognitiveLoop 主体 ----

class CognitiveLoop:
    """
    CognitiveLoop v1:

    流程：
        1. 根据 CognitiveTask 生成 GoalSpec（写入 GoalStore）
        2. 调用 Planner 生成 plan_summary（或使用默认单步计划）
        3. 调用 Executor 执行（或使用默认 LLM 单步执行）
        4. 构造 DoneCheckContext，由 DoneCouncil 评估“是否完成”
        5. 写 ExecutionTrace（blueprint_capability_id="cognitive_loop"）
        6. 写入 MemoryFabric（短期记忆），供 SleepLoop 后续巩固
        7. 返回 CognitiveResult

    未来扩展：
        - 若未完成且信心高，可由 Critic 触发重新规划/自编程
        - 多轮认知循环组成更长的任务生命周期
    """

    def __init__(
        self,
        *,
        repo_root: str | Path = ".",
        planner: Optional[PlannerFn] = None,
        executor: Optional[ExecutorFn] = None,
        goal_manager: Optional[GoalDoneManager] = None,
        memory_fabric: Optional[MemoryFabric] = None,
        traces_store: Optional[ExecutionTraceStore] = None,
        model_router: Optional[ModelRouter] = None,
    ) -> None:
        self.repo_root = Path(repo_root).resolve()
        self.model_router = model_router or ModelRouter()
        self.goal_manager = goal_manager or GoalDoneManager(domain="core")
        self.memory_fabric = memory_fabric
        self.traces_store = traces_store or ExecutionTraceStore()

        # 可选自定义 Planner / Executor；若为空则用 LLM 默认实现
        self._planner = planner
        self._executor = executor

    # ---- 对外统一入口 ----

    def run_task(self, task: CognitiveTask) -> CognitiveResult:
        start_ts = datetime.utcnow().isoformat()

        # 1) GoalSpec
        goal = self._ensure_goal(task)

        # 2) Planner
        plan_summary = self._run_planner(task, goal)

        # 3) Executor
        exec_res = self._run_executor(task, goal, plan_summary)
        final_output: str = str(exec_res.get("final_output") or "")
        actions_summary: str = str(exec_res.get("actions_summary") or "")
        raw_executor_result = exec_res.get("raw")

        # 4) DoneCouncil
        done_ctx = DoneCheckContext(
            task_id=task.task_id,
            plan_summary=plan_summary,
            actions_summary=actions_summary,
            final_output=final_output,
            metrics={},       # 以后可以加入 error_count/latency 等
            logs_excerpt="",  # 以后可以挂部分日志
        )
        done_result = self.goal_manager.check_done_and_record(goal=goal, context=done_ctx)

        # 5) 写 Trace
        self._write_trace(
            task=task,
            goal=goal,
            plan_summary=plan_summary,
            actions_summary=actions_summary,
            final_output=final_output,
            done_result=done_result,
            start_ts=start_ts,
        )

        # 6) 写 Memory 短期记忆
        self._write_memory(
            task=task,
            goal=goal,
            plan_summary=plan_summary,
            actions_summary=actions_summary,
            final_output=final_output,
            done_result=done_result,
        )

        # 7) 返回结果
        result = CognitiveResult(
            task_id=task.task_id,
            goal_id=goal.goal_id,
            final_output=final_output,
            plan_summary=plan_summary,
            actions_summary=actions_summary,
            done_result=done_result,
            done=done_result.is_done,
            done_confidence=done_result.confidence,
            raw_executor_result=raw_executor_result,
        )
        return result

    # ---- 内部：Goal ----

    def _ensure_goal(self, task: CognitiveTask) -> GoalSpec:
        params = task.to_goal_spec()
        goal = self.goal_manager.create_and_save_goal(**params)
        return goal

    # ---- 内部：Planner ----

    def _run_planner(self, task: CognitiveTask, goal: GoalSpec) -> str:
        if self._planner is not None:
            try:
                return self._planner(task, goal)
            except Exception:
                logger.exception("CognitiveLoop._run_planner: custom planner failed, fallback to default.")

        # 默认 Planner：用 LLM 给一个简短计划
        cfg = self.model_router.route(mode="planner", task_type=task.domain)
        system = (
            "You are a PLANNER agent in a multi-step AI system.\n"
            "Given a GOAL and TASK DESCRIPTION, produce a concise plan of 3-7 steps.\n"
            "Plan should be high-level but actionable. Output as plain text bullet list.\n"
        )
        user = (
            "=== GOAL ===\n"
            f"{goal.title}\n\n"
            f"{goal.description}\n\n"
            "=== TASK INPUT ===\n"
            f"{task.input_text}\n"
        )
        plan = call_llm(cfg, user_prompt=user, system_prompt=system)
        return plan.strip()

    # ---- 内部：Executor ----

    def _run_executor(self, task: CognitiveTask, goal: GoalSpec, plan_summary: str) -> Dict[str, Any]:
        if self._executor is not None:
            try:
                return self._executor(task, goal, plan_summary)
            except Exception:
                logger.exception("CognitiveLoop._run_executor: custom executor failed, fallback to default.")

        # 默认 Executor：单步 LLM 回答
        cfg = self.model_router.route(mode="executor", task_type=task.domain)
        system = (
            "You are an EXECUTION agent.\n"
            "You are given a GOAL, TASK INPUT and a PLAN SUMMARY.\n"
            "You must directly execute the plan and produce the best possible FINAL ANSWER.\n"
            "Return only the final answer content; no meta-comments.\n"
        )
        user = (
            "=== GOAL ===\n"
            f"{goal.title}\n\n"
            f"{goal.description}\n\n"
            "=== PLAN SUMMARY ===\n"
            f"{plan_summary}\n\n"
            "=== TASK INPUT ===\n"
            f"{task.input_text}\n"
        )
        final = call_llm(cfg, user_prompt=user, system_prompt=system)
        return {
            "final_output": final.strip(),
            "actions_summary": "single-step LLM execution based on provided plan",
            "raw": {"final": final},
        }

    # ---- 内部：Trace ----

    def _write_trace(
        self,
        *,
        task: CognitiveTask,
        goal: GoalSpec,
        plan_summary: str,
        actions_summary: str,
        final_output: str,
        done_result: DoneCheckResult,
        start_ts: str,
    ) -> None:
        try:
            trace = ExecutionTrace(
                trace_id=str(uuid.uuid4()),
                task_id=task.task_id,
                component_id="cognitive_loop.v1",
                blueprint_capability_id="cognitive_loop",  # 对应 spec_v3.json 里的 L1.cognitive_loop
                target_module="quentin/core/cognitive_loop.py",
                target_symbol="CognitiveLoop.run_task",
                success=True,
                error_type=None,
                error_message=None,
                start_time=start_ts,
                end_time=datetime.utcnow().isoformat(),
                metrics={
                    "done_flag": 1.0 if done_result.is_done else 0.0,
                    "done_confidence": done_result.confidence,
                },
                tags={
                    "domain": task.domain,
                    "environment": task.environment,
                    "goal_id": goal.goal_id,
                },
            )
            self.traces_store.append(trace)
        except Exception:
            logger.exception("CognitiveLoop._write_trace: failed to append trace")

    # ---- 内部：Memory ----

    def _write_memory(
        self,
        *,
        task: CognitiveTask,
        goal: GoalSpec,
        plan_summary: str,
        actions_summary: str,
        final_output: str,
        done_result: DoneCheckResult,
    ) -> None:
        if self.memory_fabric is None:
            return

        episode = {
            "task_id": task.task_id,
            "goal_id": goal.goal_id,
            "domain": task.domain,
            "environment": task.environment,
            "input_text": task.input_text,
            "plan_summary": plan_summary,
            "actions_summary": actions_summary,
            "final_output": final_output,
            "done": done_result.is_done,
            "done_confidence": done_result.confidence,
            "reasons": done_result.reasons,
            "violated_constraints": done_result.violated_constraints,
            "improvement_suggestions": done_result.improvement_suggestions,
            "timestamp": datetime.utcnow().isoformat(),
        }

        # 兼容不同 MemoryFabric v1 实现，尽量不强绑接口
        try:
            if hasattr(self.memory_fabric, "store_short_term"):
                # 假设 v1 接口: store_short_term(content: Dict[str, Any])
                self.memory_fabric.store_short_term(episode)
            elif hasattr(self.memory_fabric, "add_episode"):
                # 或类似: add_episode(tier, content)
                self.memory_fabric.add_episode("STS", episode)
            else:
                # 不认识的实现暂时忽略（或者可以 log warning）
                logger.debug("CognitiveLoop._write_memory: MemoryFabric has no known STS method.")
        except Exception:
            logger.exception("CognitiveLoop._write_memory: failed to store STS episode")
二、现在这步和蓝皮书的对齐情况
和你蓝皮书里的 L1.cognitive_loop & L2.goal_done_definition 的能力对应关系：

✅ GoalSpec 创建：

每次任务开始，CognitiveTask.to_goal_spec() → GoalDoneManager.create_and_save_goal

目标里有 domain/environment/success_criteria/constraints。

✅ Planner / Executor / Critic（初步）：

Planner：可插拔 PlannerFn，默认用 LLM 做 3–7 步计划；

Executor：可插拔 ExecutorFn，默认单步 LLM 执行；

Critic/反思：目前由 DoneCouncil 的 improvement_suggestions 承担第一版角色（之后可以接入单独 Critic 模块）。

✅ DoneCouncil 多 AI 模式集成：

GoalDoneManager.check_done_and_record 在 CognitiveLoop 里被调用；

结果写入 Research 实验 & ExecutionTrace。

✅ Trace & Memory 接通：

ExecutionTrace 使用 blueprint_capability_id="cognitive_loop"，和 spec_v3.json 里的 item 对齐；

MemoryFabric 被调用写入 STS 级 episodic 记忆，为 SleepLoop 后续巩固准备数据。

✅ 扩展点：

Planner / Executor 都是可注入的函数接口，Research Layer 以后可以挂不同 variant 做 A/B 测试；

DoneCouncil 已经记录到 ResearchManager 中，可对不同 council configuration 做对比。
