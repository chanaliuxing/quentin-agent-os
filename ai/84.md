    def _run_planner(self, goal: Goal, context: Dict[str, Any]) -> "PlannerResult":
        """
        Planner v1.1：
        - 先根据 goal 描述，从 SkillManager 里推荐相关技能
        - 把技能列表（JSON）放到 system prompt 里
        - 要求 Planner 在输出计划时，尽量引用 skill_id 作为可调用的“技能钩子”
        """
        # 1) 基本路由
        domain = getattr(goal, "domain", None) or "core"
        cfg = self.model_router.route(mode="planner", task_type=domain)

        # 2) 组装任务文本（简单版本）
        task_text = goal.description or goal.title or goal.raw_text or ""
        if not task_text:
            task_text = f"Goal: {goal.id}"

        # 3) 从 SkillManager 获取相关技能
        skill_suggestions: List[Dict[str, Any]] = []
        try:
            if self.skill_manager is not None:
                skill_suggestions = self.skill_manager.suggest_for_task(
                    task_text=task_text,
                    domain=domain,
                    top_k=5,
                    include_candidates=True,  # 候选技能也可以给 Planner 当参考
                )
        except Exception:
            logger.exception("CognitiveLoop._run_planner: failed to get skill suggestions")

        skills_json = json.dumps(skill_suggestions, ensure_ascii=False, indent=2)

        # 4) 构造 prompts
        system_prompt = (
            "You are the PLANNING AGENT in a multi-stage self-improving system.\n"
            "Your job is to decompose the given GOAL into a small sequence of executable steps.\n"
            "\n"
            "You are given a list of KNOWN SKILLS that the system may be able to call.\n"
            "Each skill has:\n"
            "- skill_id: unique id\n"
            "- title, description, domain\n"
            "- implemented: whether there is already code behind it\n"
            "\n"
            "You MUST:\n"
            "1) Use the skills when they are clearly helpful;\n"
            "2) For each step, optionally attach a 'skill_id' from the list if relevant;\n"
            "3) Output a strict JSON object with the following structure:\n"
            "{\n"
            "  \"steps\": [\n"
            "    {\n"
            "      \"step_id\": \"s1\",\n"
            "      \"description\": \"What to do in this step (natural language)\",\n"
            "      \"skill_id\": \"<optional skill_id from skills list or null>\",\n"
            "      \"notes\": \"optional clarifications for Executor\"\n"
            "    },\n"
            "    ...\n"
            "  ]\n"
            "}\n"
            "Do NOT include any extra commentary outside the JSON.\n"
        )

        user_prompt = {
            "goal": {
                "id": getattr(goal, "id", None),
                "title": getattr(goal, "title", None),
                "description": goal.description,
                "domain": domain,
                "metadata": getattr(goal, "metadata", {}),
            },
            "context": context or {},
            "known_skills": skill_suggestions,
        }
        user_prompt_text = json.dumps(user_prompt, ensure_ascii=False, indent=2)

        # 5) 调用 LLM
        raw = call_llm(cfg, user_prompt=user_prompt_text, system_prompt=system_prompt).strip()

        # 6) 解析 JSON（容错：截取首尾中括号/大括号）
        plan_obj: Dict[str, Any]
        try:
            # 容错处理：有些模型会在前后加废话，我们尝试找第一个 '{' 和对应的 '}'
            first = raw.find("{")
            last = raw.rfind("}")
            if first != -1 and last != -1 and last > first:
                raw_json = raw[first : last + 1]
            else:
                raw_json = raw
            plan_obj = json.loads(raw_json)
        except Exception:
            logger.exception("CognitiveLoop._run_planner: failed to parse planner JSON: %r", raw)
            # 保底：构造一个只有一个大步骤的 plan
            plan_obj = {
                "steps": [
                    {
                        "step_id": "s1",
                        "description": raw,
                        "skill_id": None,
                        "notes": "planner_output_unparsed",
                    }
                ]
            }

        # 7) 组装 PlannerResult（你原来怎么做就怎么接，这里给一个示例）
        steps = []
        for s in plan_obj.get("steps", []):
            steps.append(
                {
                    "step_id": s.get("step_id") or f"s{len(steps)+1}",
                    "description": s.get("description") or "",
                    "skill_id": s.get("skill_id"),
                    "notes": s.get("notes"),
                }
            )

        result = PlannerResult(
            goal_id=getattr(goal, "id", None),
            raw_output=raw,
            steps=steps,
            used_skills=skill_suggestions,
        )
        return result

四、效果：系统现在具备什么能力？
从蓝皮书的角度，这一步完成的是：

「从成功 Trace 中抽取技能 → 存入 SkillRegistry → 在未来的计划和执行中优先复用这些技能」，
我们已经打通了前两步 + 计划阶段的引用。

现在的完整链路是：

CognitiveLoop 执行任务 → 写 Trace → SleepLoop 读 Trace；

SleepLoop 从成功 Trace 抽出技能候选 → 写 data/skills.jsonl；

你（或 Cron）跑 qctl skills-sync → 合并进 skills_registry.jsonl；

之后任何一次 Planner 运行：

先用 SkillManager.suggest_for_task(...) 找相关技能；

把技能列表给 LLM planner；

LLM 在计划 JSON 里附上 skill_id；

Executor 执行计划时，可以看到每一步建议用哪个 skill（下一步再挂真实代码）。
