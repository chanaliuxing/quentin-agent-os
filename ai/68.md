# quentin/selfprog/models.py
from __future__ import annotations

from dataclasses import dataclass, asdict, field
from datetime import datetime
from typing import Any, Dict, Optional


@dataclass
class PatchEvalReport:
    """
    对一个补丁的静态评估结果（不直接运行）：

    - ok         : 是否通过最小要求
    - score      : 0.0 ~ 1.0 的粗略评分
    - risk_level : low / medium / high / unknown
    - issues     : 严重问题列表
    - warnings   : 警告（非致命）
    """
    ok: bool
    score: float
    risk_level: str
    issues: list[str] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)


@dataclass
class Patch:
    """
    自编程引擎所管理的“补丁”最小数据结构。

    v1 不直接执行，只作为：
      - LLM 生成的修改建议（代码 / diff）
      - 测试草稿
      - 风险等级与评估报告
    """

    patch_id: str

    # 作用目标：repo 内部的路径（相对路径）
    target_path: str                     # 如 "quentin/core/cognitive_loop.py"
    target_symbol: Optional[str] = None  # 如 "CognitiveLoop.run_task"

    change_type: str = "refactor"        # "bugfix" / "refactor" / "optimize"

    intent: str = ""                     # 这个 patch 想解决什么问题
    patch_text: str = ""                 # 代码 diff 或完整替换代码
    tests: str = ""                      # 单元测试 / 回归测试草稿（Python 或伪代码）

    created_by: str = "selfprog"         # "selfprog" / "human"
    created_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at: str = field(default_factory=lambda: datetime.utcnow().isoformat())

    # 评估与状态
    status: str = "proposed"             # "proposed"/"evaluated"/"approved"/"rejected"/"applied"
    risk_level: str = "unknown"          # "low"/"medium"/"high"/"unknown"

    eval_report: Optional[PatchEvalReport] = None

    metadata: Dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        # eval_report 单独处理
        if self.eval_report is not None:
            d["eval_report"] = self.eval_report.to_dict()
        return d

    @classmethod
    def from_dict(cls, d: Dict[str, Any]) -> "Patch":
        eval_data = d.get("eval_report")
        if eval_data:
            eval_report = PatchEvalReport(
                ok=eval_data.get("ok", False),
                score=float(eval_data.get("score", 0.0)),
                risk_level=eval_data.get("risk_level", "unknown"),
                issues=list(eval_data.get("issues", []) or []),
                warnings=list(eval_data.get("warnings", []) or []),
            )
        else:
            eval_report = None

        return cls(
            patch_id=d["patch_id"],
            target_path=d.get("target_path", ""),
            target_symbol=d.get("target_symbol"),
            change_type=d.get("change_type", "refactor"),
            intent=d.get("intent", ""),
            patch_text=d.get("patch_text", ""),
            tests=d.get("tests", ""),
            created_by=d.get("created_by", "selfprog"),
            created_at=d.get("created_at"),
            updated_at=d.get("updated_at"),
            status=d.get("status", "proposed"),
            risk_level=d.get("risk_level", "unknown"),
            eval_report=eval_report,
            metadata=d.get("metadata", {}) or {},
        )

# quentin/selfprog/registry.py
from __future__ import annotations

from pathlib import Path
from typing import Dict, List, Optional, Union
import json
import logging

from .models import Patch

logger = logging.getLogger(__name__)


class FileBackedPatchRegistry:
    """
    简单的文件型 Patch Registry。

    JSON 结构：
      {
        "patches": {
          "<patch_id>": { ...Patch... },
          ...
        }
      }
    """

    def __init__(self, path: Union[str, Path] = "data/patches.json") -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._loaded = False
        self._patches: Dict[str, Patch] = {}

    # ---------- 内部 ----------

    def _load(self) -> None:
        if self._loaded:
            return
        if self.path.exists():
            try:
                raw = json.loads(self.path.read_text(encoding="utf-8") or "{}")
                patches_raw = raw.get("patches", {}) or {}
                self._patches = {
                    pid: Patch.from_dict(pd) for pid, pd in patches_raw.items()
                }
            except Exception:
                logger.exception("FileBackedPatchRegistry: failed to load patches; using empty.")
                self._patches = {}
        else:
            self._patches = {}
        self._loaded = True

    def _save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        raw = {
            "patches": {pid: p.to_dict() for pid, p in self._patches.items()},
        }
        self.path.write_text(
            json.dumps(raw, ensure_ascii=False, indent=2),
            encoding="utf-8",
        )

    # ---------- CRUD API ----------

    def upsert_patch(self, patch: Patch) -> None:
        self._load()
        self._patches[patch.patch_id] = patch
        self._save()

    def get_patch(self, patch_id: str) -> Optional[Patch]:
        self._load()
        return self._patches.get(patch_id)

    def list_patches(
        self,
        *,
        status: Optional[str] = None,
        target_path: Optional[str] = None,
    ) -> List[Patch]:
        self._load()
        patches = list(self._patches.values())
        if status:
            patches = [p for p in patches if p.status == status]
        if target_path:
            patches = [p for p in patches if p.target_path == target_path]
        # 最近更新排前面
        patches.sort(key=lambda p: p.updated_at, reverse=True)
        return patches

# quentin/selfprog/engine.py
from __future__ import annotations

import json
import logging
import uuid
from dataclasses import asdict
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, Optional

from quentin.llm.multi_council import MultiAICouncil, CouncilDecision
from quentin.llm.types import LLMConfig
from quentin.research.experiments import ResearchManager, Experiment
from .models import Patch, PatchEvalReport
from .registry import FileBackedPatchRegistry

logger = logging.getLogger(__name__)


class SelfProgrammingEngine:
    """
    自编程引擎 v1：

    - 不直接改源码，只生成补丁（Patch）+ 静态评估结果
    - 补丁存到 FileBackedPatchRegistry
    - 评估结果写入 ResearchManager

    后续可以让：
      - ImmuneLoop / GapTracker 在发现问题时调用 propose_patch(...)
      - 人类或多 AI “审批层” 决定是否应用 patch
    """

    def __init__(
        self,
        *,
        council: Optional[MultiAICouncil] = None,
        registry: Optional[FileBackedPatchRegistry] = None,
        research_manager: Optional[ResearchManager] = None,
        repo_root: Optional[Path] = None,
    ) -> None:
        self.registry = registry or FileBackedPatchRegistry()
        self.research_manager = research_manager or ResearchManager()
        self.repo_root = repo_root or Path(".")

        if council is None:
            members = [
                LLMConfig(
                    provider="ollama",
                    model="qwen3-vl:latest",
                    role="patch_designer",
                ),
                LLMConfig(
                    provider="ollama",
                    model="deepseek-r1:1.5b",
                    role="patch_reviewer",
                ),
            ]
            council = MultiAICouncil(members=members)
        self.council = council

    # ---------- 安全边界检查 ----------

    def _is_target_allowed(self, target_path: str) -> tuple[bool, str]:
        """
        v1 简单安全策略：

        - 禁止自编程引擎修改 kernel/ 相关文件
        - 推荐把可自动修改代码放在 mutable/ 目录下
        """
        if "kernel/" in target_path or "quentin/kernel" in target_path:
            return False, "target_path points to kernel, which is immutable for self-programming."

        # 这里不强制要求必须在 mutable/ 下，但给一个建议
        if not target_path.startswith("mutable/"):
            return True, "target_path is outside kernel; recommended to keep auto-updatable code in mutable/."

        return True, "target_path is within mutable/."

    # ---------- 提出补丁 ----------

    def propose_patch(
        self,
        *,
        target_path: str,
        problem_description: str,
        change_type: str = "refactor",
        target_symbol: Optional[str] = None,
        extra_context: Optional[str] = None,
    ) -> Patch:
        """
        入口：给一个代码文件 + 问题描述，让多模型设计一个 Patch（不直接改代码）。

        - target_path 必须是 repo 相对路径
        - problem_description 用自然语言描述“什么地方不够好 / 想优化什么”
        - change_type: "bugfix" / "refactor" / "optimize" / ...
        """
        ok, reason = self._is_target_allowed(target_path)

        current_code_snippet = self._read_file_snippet(target_path, target_symbol)

        context_parts = [
            f"TARGET_PATH: {target_path}",
            f"CHANGE_TYPE: {change_type}",
            f"SAFETY_NOTE: {reason}",
            "",
            "PROBLEM_DESCRIPTION:",
            problem_description,
            "",
        ]
        if extra_context:
            context_parts.append("EXTRA_CONTEXT:")
            context_parts.append(extra_context)
            context_parts.append("")

        context_parts.append("CURRENT_CODE_SNIPPET:")
        context_parts.append(current_code_snippet or "<file not found or empty>")

        context = "\n".join(context_parts)

        question = (
            "You are the SELF-PROGRAMMING ENGINE of a self-improving agent.\n"
            "Your job is to propose ONE code patch that improves the TARGET module.\n\n"
            "You MUST return ONLY ONE JSON object, no extra text.\n\n"
            "JSON SCHEMA (informal):\n"
            "{\n"
            "  \"intent\": \"what this patch aims to fix or improve\",\n"
            "  \"patch_text\": \"the code change: either a unified diff or full replacement of the relevant function(s)\",\n"
            "  \"tests\": \"test code or test plan to validate the patch\",\n"
            "  \"risk_level\": \"low | medium | high\"\n"
            "}\n"
        )

        decision: CouncilDecision = self.council.deliberate(
            question=question,
            context=context,
            mode="self_programming_propose",
        )
        answer = decision.final_answer or "{}"

        patch_info = self._parse_patch_json(answer)

        patch_id = str(uuid.uuid4())
        now = datetime.utcnow().isoformat()

        patch = Patch(
            patch_id=patch_id,
            target_path=target_path,
            target_symbol=target_symbol,
            change_type=change_type,
            intent=patch_info.get("intent", problem_description).strip(),
            patch_text=patch_info.get("patch_text", "").strip(),
            tests=patch_info.get("tests", "").strip(),
            created_by="selfprog",
            created_at=now,
            updated_at=now,
            status="proposed" if ok else "blocked",
            risk_level=patch_info.get("risk_level", "unknown"),
            metadata={
                "safety_check_ok": ok,
                "safety_reason": reason,
                "source_mode": "self_programming_propose",
                "raw_answer": answer,
            },
        )

        # 若安全检查不通过，标记高风险
        if not ok:
            patch.risk_level = "high"
            patch.metadata["blocked_by_safety_kernel"] = True

        self.registry.upsert_patch(patch)
        return patch

    # ---------- 评估补丁 ----------

    def evaluate_patch(self, patch_id: str) -> Optional[Patch]:
        """
        对某个 Patch 做静态评估：

        - 检查字段完整度（patch_text/tests）
        - 用 Council 做一次“静态 code review”，给出风险等级与建议
        - 更新 Patch.status / risk_level / eval_report
        - 写入 ResearchManager
        """
        patch = self.registry.get_patch(patch_id)
        if not patch:
            logger.warning("SelfProgrammingEngine: patch not found: %s", patch_id)
            return None

        heuristics_report = self._heuristic_eval(patch)
        llm_report = self._llm_eval(patch)

        # 合并结果（简单策略：score 取平均，risk_level 取风险较高的）
        combined = self._combine_eval(heuristics_report, llm_report)
        patch.eval_report = combined
        patch.risk_level = combined.risk_level
        patch.status = "evaluated"
        patch.updated_at = datetime.utcnow().isoformat()

        self.registry.upsert_patch(patch)
        self._record_patch_eval_experiment(patch, combined)

        return patch

    # ---------- 内部：文件读取 ----------

    def _read_file_snippet(
        self,
        target_path: str,
        target_symbol: Optional[str],
        max_chars: int = 4000,
    ) -> str:
        try:
            full_path = (self.repo_root / target_path).resolve()
            if not full_path.exists():
                return ""
            text = full_path.read_text(encoding="utf-8")
            # v1 简单截断；以后可以根据 symbol 做 AST 切片
            return text[:max_chars]
        except Exception:
            logger.exception("SelfProgrammingEngine: failed to read file snippet.")
            return ""

    # ---------- 内部：JSON 解析 ----------

    def _parse_patch_json(self, text: str) -> Dict[str, Any]:
        try:
            s = text.strip()
            first = s.find("{")
            last = s.rfind("}")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)
            if not isinstance(data, dict):
                return {}
            return {
                "intent": str(data.get("intent", "")),
                "patch_text": str(data.get("patch_text", "")),
                "tests": str(data.get("tests", "")),
                "risk_level": str(data.get("risk_level", "unknown")),
            }
        except Exception:
            logger.exception("SelfProgrammingEngine: failed to parse patch JSON: %r", text)
            return {}

    # ---------- 内部：启发式评估 ----------

    def _heuristic_eval(self, patch: Patch) -> PatchEvalReport:
        issues: list[str] = []
        warnings: list[str] = []
        score = 1.0

        if not patch.patch_text.strip():
            issues.append("Empty patch_text (no code change provided).")
            score -= 0.5

        if not patch.tests.strip():
            warnings.append("No tests provided; cannot ensure behavior is preserved.")
            score -= 0.2

        if "blocked_by_safety_kernel" in patch.metadata:
            issues.append("Patch is blocked by safety kernel for targeting immutable kernel code.")
            score -= 0.3

        if score < 0:
            score = 0.0

        ok = score >= 0.5 and not issues
        risk_level = "low"
        if not ok or "blocked_by_safety_kernel" in patch.metadata:
            risk_level = "high"
        elif score < 0.7:
            risk_level = "medium"

        return PatchEvalReport(
            ok=ok,
            score=score,
            risk_level=risk_level,
            issues=issues,
            warnings=warnings,
        )

    # ---------- 内部：LLM 静态审查 ----------

    def _llm_eval(self, patch: Patch) -> PatchEvalReport:
        # 若安全核已 block，就不再花 Token 做审查
        if "blocked_by_safety_kernel" in patch.metadata:
            return PatchEvalReport(
                ok=False,
                score=0.0,
                risk_level="high",
                issues=["Patch blocked by safety kernel; skip LLM evaluation."],
                warnings=[],
            )

        context_parts = [
            f"TARGET_PATH: {patch.target_path}",
            f"TARGET_SYMBOL: {patch.target_symbol or '<unknown>'}",
            f"CHANGE_TYPE: {patch.change_type}",
            "",
            "PATCH_INTENT:",
            patch.intent,
            "",
            "PATCH_TEXT:",
            patch.patch_text[:4000],
            "",
            "TEST_PLAN_OR_CODE:",
            patch.tests[:2000],
        ]
        context = "\n".join(context_parts)

        question = (
            "You are a senior code reviewer in charge of SAFE SELF-PROGRAMMING.\n"
            "You must review the given patch and test plan.\n"
            "Assess:\n"
            "- potential risk of breaking behavior\n"
            "- clarity and completeness of tests\n"
            "- overall quality of the change\n\n"
            "Return ONLY ONE JSON object with the following fields:\n"
            "{\n"
            "  \"ok\": true/false,\n"
            "  \"score\": 0.0~1.0,\n"
            "  \"risk_level\": \"low\"|\"medium\"|\"high\",\n"
            "  \"issues\": [\"...\"],\n"
            "  \"warnings\": [\"...\"]\n"
            "}\n"
        )

        try:
            decision: CouncilDecision = self.council.deliberate(
                question=question,
                context=context,
                mode="self_programming_eval",
            )
            ans = decision.final_answer or "{}"
            s = ans.strip()
            first = s.find("{")
            last = s.rfind("}")
            if first != -1 and last != -1 and last > first:
                s = s[first : last + 1]
            data = json.loads(s)

            ok = bool(data.get("ok", False))
            score = float(data.get("score", 0.0))
            risk_level = str(data.get("risk_level", "unknown"))
            issues = list(data.get("issues", []) or [])
            warnings = list(data.get("warnings", []) or [])
            if score < 0:
                score = 0.0
            if score > 1:
                score = 1.0

            return PatchEvalReport(
                ok=ok,
                score=score,
                risk_level=risk_level,
                issues=issues,
                warnings=warnings,
            )
        except Exception:
            logger.exception("SelfProgrammingEngine: LLM eval failed.")
            return PatchEvalReport(
                ok=False,
                score=0.0,
                risk_level="unknown",
                issues=["LLM evaluation failed."],
                warnings=[],
            )

    # ---------- 内部：合并评估结果 ----------

    def _combine_eval(
        self,
        h: PatchEvalReport,
        l: PatchEvalReport,
    ) -> PatchEvalReport:
        score = (h.score + l.score) / 2.0

        # 风险等级取“更危险”的那个
        order = {"low": 0, "medium": 1, "high": 2, "unknown": 1}
        risk_level = h.risk_level
        if order.get(l.risk_level, 1) > order.get(risk_level, 1):
            risk_level = l.risk_level

        issues = list(h.issues) + list(l.issues)
        warnings = list(h.warnings) + list(l.warnings)

        ok = score >= 0.5 and not issues

        return PatchEvalReport(
            ok=ok,
            score=score,
            risk_level=risk_level,
            issues=issues,
            warnings=warnings,
        )

    # ---------- 内部：记录 Research 实验 ----------

    def _record_patch_eval_experiment(
        self,
        patch: Patch,
        report: PatchEvalReport,
    ) -> None:
        rm = self.research_manager

        capability = f"selfprog:{patch.target_path}"
        requirement_id = "patch_safety_and_quality"

        exps = rm.list_experiments()
        exp: Optional[Experiment] = None
        for e in exps:
            if (
                e.capability == capability
                and e.requirement_id == requirement_id
                and e.purpose == "self_programming"
            ):
                exp = e
                break

        if exp is None:
            exp = rm.create_experiment(
                name=f"SelfProgramming-{patch.target_path}",
                purpose="self_programming",
                capability=capability,
                requirement_id=requirement_id,
                tags=["selfprog", "patch"],
            )

        metrics = {
            "score": report.score,
            "ok": int(report.ok),
        }
        context = {
            "patch_id": patch.patch_id,
            "target_path": patch.target_path,
            "target_symbol": patch.target_symbol,
            "status": patch.status,
            "risk_level": report.risk_level,
        }

        rm.record_run(
            experiment_id=exp.experiment_id,
            variant_id="baseline_v1",
            metrics=metrics,
            context=context,
        )

def cmd_patch_list(args: argparse.Namespace) -> None:
    registry = FileBackedPatchRegistry()
    patches = registry.list_patches(status=args.status)
    if not patches:
        print("No patches found.")
        return

    print(f"Found {len(patches)} patches:")
    for p in patches:
        print(
            f"- {p.patch_id} | {p.status} | risk={p.risk_level} | "
            f"{p.target_path}::{p.target_symbol or ''}"
        )


def cmd_patch_show(args: argparse.Namespace) -> None:
    registry = FileBackedPatchRegistry()
    patch = registry.get_patch(args.patch_id)

        return

    print(f"- target_path : {patch.target_path}")
    print(f"- target_sym  : {patch.target_symbol}")
    print(f"- change_type : {patch.change_type}")
    print(f"- status      : {patch.status}")
    print(f"- risk_level  : {patch.risk_level}")
    print(f"- created_by  : {patch.created_by}")
    print(f"- created_at  : {patch.created_at}")
    print(f"- updated_at  : {patch.updated_at}")
    print(f"- intent      : {patch.intent}\n")

    print("=== PATCH TEXT (truncated) ===")
    print(patch.patch_text[:4000])

    print("\n=== TESTS (truncated) ===")
    print(patch.tests[:2000])

    if patch.eval_report:
        er = patch.eval_report
        print("\n=== EVAL REPORT ===")
        print(f"- ok         : {er.ok}")
        print(f"- score      : {er.score}")
        print(f"- risk_level : {er.risk_level}")

            print("- issues:")
            for i in er.issues:
                print(f"  * {i}")
        if er.warnings:
            print("- warnings:")
            for w in er.warnings:
                print(f"  * {w}")
